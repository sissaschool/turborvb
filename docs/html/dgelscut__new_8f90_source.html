16!> @brief Cut-off procedure for stochastic reconfiguration matrix using eigenvalue analysis
17!> @details This subroutine performs a cut-off procedure on the stochastic reconfiguration (SR) matrix
18!> by eliminating parameters with eigenvalues below a threshold epsdgel. The algorithm diagonalizes
19!> the SR matrix and iteratively removes parameters with the smallest eigenvalues until all remaining
20!> eigenvalues are above the threshold. This helps stabilize the optimization process by removing
21!> ill-conditioned parameters.
22!> @param[in,out] sov SR matrix (input), transformed matrix (output)
23!> @param[in] nmat Dimension of the SR matrix
24!> @param[in] npm Leading dimension of matrix sov
25!> @param[in,out] psip Work array for eigenvalue storage and temporary calculations
26!> @param[in] epsdgel Eigenvalue threshold for parameter elimination
27!> @param[in,out] ipsip Parameter status array (1=active, 0=eliminated)
28!> @param[in] np Number of parameters
29!> @param[out] info Information flag from eigenvalue solver
30!> @param[in] scalpar Parameter scaling factors
31!> @param[in] lwork Size of work array for eigenvalue solver
32!> @param[in] epsmach Machine precision
33!> @param[in] indexpar Parameter index mapping
34!> @param[in] rank Current processor rank
35!> @param[in] nprocopt Number of processors for optimization
36!> @param[in] rankopt Rank for optimization
37!> @param[in] commopt_mpi MPI communicator for optimization
38subroutine dgelscut(sov, nmat, npm, psip, epsdgel, ipsip, np, info &
39 &, scalpar, lwork, epsmach, indexpar, rank, nprocopt, rankopt, commopt_mpi)
40 use allio, only: nproc_diag
41 implicit none
42 integer ip, nmat, i, j, npm, n1, n2, n3, n4, n5, info, rank, ipsip(*) &
43 &, ind, lwork, iscra, k, ierr, imax, imaxg, indi, indj, np &
44 &, indexpar(*), indscali, nprocopt, rankopt, commopt_mpi
45 real*8 sov(npm, npm, *), psip(*), epsdgel, cost, epsmach, ratoff &
46 &, costmax, costmaxg, scalpar(*), eigmin, eigs, eigb, eigs_min, eigb_min, maxsr
47
48 ! epsmach=1d-12
49 n1 = nmat + 1
50 n2 = n1 + nmat
51 n3 = n2 + nmat
52 n4 = n3 + nmat
53 n5 = n4 + nmat
54 ! preconditioning the matrix
55 maxsr = 0.d0
56 do i = 1, nmat
57 costmax = abs(sov(i, i, 1)*scalpar(i))
58 if (costmax .gt. maxsr) maxsr = costmax
59 end do
60 indj = 0
61 do i = 1, nmat
62 if (abs(sov(i, i, 1)*scalpar(i)) .gt. epsmach*maxsr) then
63 psip(n1 + i - 1) = 1.d0/dsqrt(dabs(sov(i, i, 1)))
64 else
65 ! write(6,*) ' Warning eliminated =',i,sov(i,i,1)
66 psip(n1 + i - 1) = 0.d0
67 ipsip(i) = 0
68 indj = indj + 1
69 end if
70 end do
71
72 indi = 0
73 do i = 1, nmat
74 if (ipsip(i) .eq. 1) then
75 indi = indi + 1
76 indj = 0
77 do j = 1, nmat
78 if (ipsip(j) .eq. 1) then
79 indj = indj + 1
80 sov(indi, indj, 2) = sov(i, j, 1)*psip(n1 + i - 1)*psip(n1 + j - 1)
81 end if
82 end do
83 end if
84 end do
85
86 ! first step diagonalize the matrix
87 if (indi .gt. 1) then
88 if (nproc_diag .gt. 1) then
89 call dsyev_my('N', 'L', indi, sov(1, 1, 2), npm, psip, info &
90 &, nprocopt, rankopt, commopt_mpi)
91 else
92 call dsyev('N', 'L', indi, sov(1, 1, 2), npm, psip, psip(n2), lwork, info)
93 end if
94 if (rank .eq. 0) write (6, *) ' Lowest/Max eigenvalues SR mat ', psip(1), psip(indi)
95
96 eigs = psip(1)
97 eigb = psip(2)
98 eigs_min = psip(1)
99 eigb_min = psip(2)
100
101 eigmin = psip(1)
102
103 else
104 if (rank .eq. 0) write (6, *) ' Warning no par to eliminate =', indi
105 eigmin = 1.d0
106
107 end if
108
109 if (eigmin .lt. epsdgel .and. indi .gt. 1) then
110 do while (eigmin .lt. epsdgel)
111
112 ! definition matrix for the chosen raw
113 indi = 0
114 do i = 1, nmat
115 if (ipsip(i) .eq. 1) then
116 indi = indi + 1
117 indj = 0
118 do j = 1, nmat
119 if (ipsip(j) .eq. 1) then
120 indj = indj + 1
121 sov(indi, indj, 2) = sov(i, j, 1)*psip(n1 + i - 1)*psip(n1 + j - 1)
122 end if
123 end do
124 end if
125 end do
126 do j = 1, indi
127 sov(j, j, 2) = 1.d0
128 end do
129
130 if (nproc_diag .gt. 1) then
131 call dsyev_my('V', 'L', indi, sov(1, 1, 2), npm, psip, info &
132 &, nprocopt, rankopt, commopt_mpi)
133 else
134 call dsyev('V', 'L', indi, sov(1, 1, 2), npm, psip, psip(n2), lwork, info)
135 end if
136
137 eigs_min = psip(1)
138 eigb_min = psip(2)
139
140 ! write(6,*) ' before findzero '
141
142 call findzero(eigs_min, eigb_min, npm, indi, sov(1, 1, 2), psip &
143 &, psip(n2), psip(n3), psip(n4), psip(n5))
144
145 ! write(6,*) ' after findzero '
146 costmax = eigs_min
147 imax = 0
148 indj = 0
149 ! write(6,*) ' Eigenvalues inside new '
150
151 ! In principle we can eliminate all parameters < epsdgel
152 ! with the constraint that they are independent
153 do k = 1, nmat
154 if (ipsip(k) .eq. 1) then
155 indj = indj + 1
156 if (psip(n5 + indj - 1) .ge. costmax) then
157 imax = k
158 costmax = psip(n5 + indj - 1)
159 end if
160 ! write(6,*) indj,psip(n5+indj-1)
161 end if
162 end do
163
164 indscali = indexpar(imax)
165
166 if (rank .eq. 0) then
167 if (indscali .eq. 0) then
168 write (6, *) 'eliminated collective par', imax
169 else
170 write (6, *) 'eliminated normal par', indscali
171 end if
172 end if
173
174 if (imax .eq. 0) then
175 if (rank .eq. 0) write (6, *) ' Some error in findzero ', eigs_min, eigb_min
176#ifdef PARALLEL
177 call mpi_finalize(ierr)
178#endif
179 stop
180 else
181 ipsip(imax) = 0
182 eigmin = costmax
183 end if
184
185 end do
186
187 if (rank .eq. 0) write (6, *) ' New Lowest eigenval ', eigmin
188 end if
189 if (rank .eq. 0) write (6, *) ' Fixed parameters '
190 ind = 0
191 do i = 1, nmat
192 ipsip(i + np) = ipsip(i)
193 if (ipsip(i) .eq. 0) then
194 ind = ind + 1
195 indscali = indexpar(i)
196 if (rank .eq. 0) then
197 if (indscali .eq. 0) then
198 write (6, *) ind, i
199 else
200 write (6, *) ind, indscali
201 end if
202 end if
203 end if
204 ! only the regul
205 if (scalpar(i) .gt. 0.d0 .and. ipsip(i) .ne. 0) then
206 ipsip(i + np) = 2
207 end if
208 end do
209
210 return
211end
