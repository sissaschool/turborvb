16!> @brief Matrix increment utilities for quantum Monte Carlo calculations
17!>
18!> This module provides subroutines for incrementing matrix elements in quantum
19!> Monte Carlo calculations, supporting both real and complex matrices.
20!> It handles different matrix types including standard matrices and
21!> Pfaffian matrices for paired and unpaired electrons.
22!>
23!> The module includes:
24!> - Real matrix increment with symmetry handling
25!> - Complex matrix increment with Hermitian/transpose symmetry
26!> - Support for Pfaffian matrices with up/down electron blocks
27!> - Proper handling of paired and unpaired electron configurations
28!>
29!> @author TurboRVB group

33!> @brief Increment real matrix elements with symmetry handling
34!>
35!> This subroutine increments matrix elements in a real matrix, handling
36!> different matrix types and symmetry conditions. It supports both
37!> standard matrices and Pfaffian matrices for quantum Monte Carlo
38!> calculations.
39!>
40!> @param[in,out] amat Matrix to be updated (ndim, *)
41!> @param[in] ndim Leading dimension of the matrix
42!> @param[in] ind Linear index of the matrix element to update
43!> @param[in] value Value to add to the matrix element
44!> @param[in] symmagp Flag for symmetric matrix updates
45!> @param[in] ipf Flag indicating matrix type:
46!> - ipf = 1: Standard matrix
47!> - ipf ≠ 1: Pfaffian matrix
48!>
49!> @details
50!> The subroutine performs the following operations:
51!> 1. Converts linear index to (i,j) matrix coordinates
52!> 2. For standard matrices (ipf = 1):
53!> - Adds value to amat(i,j)
54!> - If symmagp is true and i ≠ j, adds value to amat(j,i)
55!> 3. For Pfaffian matrices (ipf ≠ 1):
56!> - Adds value to amat(i,j) and subtracts from amat(j,i)
57!> - Handles up/down electron blocks with proper symmetry
58!> - Manages paired and unpaired electron configurations
59!> - Applies symmetry conditions based on ion types
60!>
61!> @note The subroutine handles antisymmetry of Pfaffian matrices
62!> @note Up/down electron blocks are handled separately for paired electrons
63!> @note The subroutine respects ion type constraints (kiontot)
64!> @note Used in wave function optimization and gradient calculations
65subroutine upsimp(amat, ndim, ind, value, symmagp, ipf)
66 use allio, only: nelorb_at, pfaffup, kiontot
67 implicit none
68 integer ndim, ndimh, ind, i, j, ipf
69 real*8 amat(ndim, *), value
70 logical symmagp
71 if (ipf .eq. 1) then
72 j = (ind - 1)/ndim + 1
73 i = ind - ndim*(j - 1)
74 amat(i, j) = amat(i, j) + value
75 if (j .ne. i .and. j .le. ndim .and. symmagp) amat(j, i) = amat(j, i) + value
76 else
77 ndimh = nelorb_at/2
78 j = (ind - 1)/ndim + 1
79 i = ind - ndim*(j - 1)
80 amat(i, j) = amat(i, j) + value
81 if (j .le. ndim) amat(j, i) = amat(j, i) - value
82 if (symmagp .and. kiontot(i) .ne. 0 .and. kiontot(j) .ne. 0) then
83 if (j .le. ndimh .and. i .le. ndimh .and. .not. pfaffup) then
84 amat(i + ndimh, j + ndimh) = amat(i + ndimh, j + ndimh) + value
85 amat(j + ndimh, i + ndimh) = amat(j + ndimh, i + ndimh) - value
86 elseif (j .gt. ndimh .and. i .le. ndimh) then
87 amat(j - ndimh, i + ndimh) = amat(j - ndimh, i + ndimh) + value
88 amat(i + ndimh, j - ndimh) = amat(i + ndimh, j - ndimh) - value
89 ! elseif(j.le.ndimh.and.i.gt.ndimh) then
90 ! amat(j+ndimh, i-ndimh)=amat(j+ndimh, i-ndimh)+value
91 ! amat(i-ndimh, j+ndimh)=amat(i-ndimh, j+ndimh)-value
92 end if
93 end if
94 end if
95 return
96end

98!> @brief Increment complex matrix elements with symmetry handling
99!>
100!> This subroutine increments matrix elements in a complex matrix, handling
101!> different matrix types and symmetry conditions. It supports both
102!> standard matrices and Pfaffian matrices with proper handling of
103!> Hermitian and transpose symmetries.
104!>
105!> @param[in,out] amat Complex matrix to be updated (ndim, *)
106!> @param[in] ndim Leading dimension of the matrix
107!> @param[in] ind Linear index of the matrix element to update
108!> @param[in] value Array containing complex value to add to the matrix element
109!> @param[in] symmagp Flag for symmetric matrix updates
110!> @param[in] ipf Flag indicating matrix type:
111!> - ipf = 1: Standard matrix
112!> - ipf ≠ 1: Pfaffian matrix
113!>
114!> @details
115!> The subroutine performs the following operations:
116!> 1. Converts linear index to (i,j) matrix coordinates
117!> 2. For standard matrices (ipf = 1):
118!> - For diagonal elements with Hermitian symmetry, adds real part only
119!> - For off-diagonal elements, adds value(1) to amat(i,j)
120!> - If symmagp is true, applies Hermitian or transpose symmetry
121!> 3. For Pfaffian matrices (ipf ≠ 1):
122!> - Adds value(1) to amat(i,j) and subtracts from amat(j,i)
123!> - Handles up/down electron blocks with proper symmetry
124!> - Applies Hermitian or transpose symmetry based on yes_hermite flag
125!> - Manages paired and unpaired electron configurations
126!>
127!> @note For Hermitian matrices (yes_hermite = .true.), uses conjugate symmetry
128!> @note For non-Hermitian matrices, uses transpose symmetry
129!> @note Diagonal elements in Hermitian matrices are constrained to be real
130!> @note The subroutine handles antisymmetry of Pfaffian matrices
131!> @note Used in complex wave function optimization and gradient calculations
132subroutine upsimp_complex(amat, ndim, ind, value, symmagp, ipf)
133 use allio, only: yes_hermite, nelorb_at, pfaffup, kiontot
134 implicit none
135 integer ndim, ndimh, ind, ipf, i, j
136 complex*16 amat(ndim, *), value(*)
137 logical symmagp
138 if (ipf .eq. 1) then
139 j = (ind - 1)/ndim + 1
140 i = ind - ndim*(j - 1)
141 if (i .eq. j .and. yes_hermite .and. symmagp .and. j .le. nelorb_at) then
142 amat(i, i) = amat(i, i) + real(value(1))
143 else
144 amat(i, j) = amat(i, j) + value(1)
145 end if
146 if (j .ne. i .and. j .le. nelorb_at .and. i .le. nelorb_at .and. symmagp) then
147 if (yes_hermite) then
148 amat(j, i) = amat(j, i) + dconjg(value(1))
149 else
150 amat(j, i) = amat(j, i) + value(1)
151 end if
152 end if
153 else
154 ndimh = nelorb_at/2
155 j = (ind - 1)/ndim + 1
156 i = ind - ndim*(j - 1)
157 amat(i, j) = amat(i, j) + value(1)
158 !if(j.le.2*nelorb_at.and.i.le.2*nelorb_at)
159 if (j .le. ndim) amat(j, i) = amat(j, i) - value(1) ! it has to be replicated even if it is unpaired
160
161 if (symmagp .and. kiontot(j) .ne. 0 .and. kiontot(i) .ne. 0) then
162 if (yes_hermite) then
163 if (j .le. ndimh .and. i .le. ndimh .and. .not. pfaffup) then
164 amat(i + ndimh, j + ndimh) = amat(i + ndimh, j + ndimh) + conjg(value(1))
165 amat(j + ndimh, i + ndimh) = amat(j + ndimh, i + ndimh) - conjg(value(1))
166 elseif (j .gt. ndimh .and. i .le. ndimh) then
167 amat(j - ndimh, i + ndimh) = amat(j - ndimh, i + ndimh) + conjg(value(1))
168 amat(i + ndimh, j - ndimh) = amat(i + ndimh, j - ndimh) - conjg(value(1))
169 ! elseif(j.le.ndimh.and.i.gt.ndimh) then
170 ! amat(j+ndimh, i-ndimh)=conjg(value(1))
171 ! amat(i-ndimh, j+ndimh)=-conjg(value(1))
172 end if
173 else
174 if (j .le. ndimh .and. i .le. ndimh .and. .not. pfaffup) then
175 amat(i + ndimh, j + ndimh) = amat(i + ndimh, j + ndimh) + value(1)
176 amat(j + ndimh, i + ndimh) = amat(j + ndimh, i + ndimh) - value(1)
177 elseif (j .gt. ndimh .and. i .le. ndimh) then
178 amat(j - ndimh, i + ndimh) = amat(j - ndimh, i + ndimh) + value(1)
179 amat(i + ndimh, j - ndimh) = amat(i + ndimh, j - ndimh) - value(1)
180 ! elseif(j.le.ndimh.and.i.gt.ndimh) then
181 ! amat(j+ndimh, i-ndimh)=value(1)
182 ! amat(i-ndimh, j+ndimh)=-value(1)
183 end if
184 end if
185 end if
186
187 end if
188 return
189end
