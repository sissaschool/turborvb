140 !> @brief Get atomic number from element symbol
141 !>
142 !> @param symbol Chemical symbol of the element (2 characters)
143 !> @return Atomic number if found, -1 if not found
144 function get_number(symbol)
145 implicit none
146 character(2), intent(in) :: symbol
147 integer i, get_number
148
149 get_number = -1
150 do i = 1, 103
151 if (trim(symbol) == atomsnames(i)) get_number = i
152 end do
153 return
154 end function get_number
