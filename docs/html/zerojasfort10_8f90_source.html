16!> @brief Jastrow factor zeroing program for TurboRVB
17!> @details This program reads a TurboRVB wave function (fort.10) and
18!> selectively zeros Jastrow factors while preserving the wave function
19!> structure. It can zero both charge and spin-dependent Jastrow factors,
20!> with an option to preserve one-body terms. The program outputs a new
21!> fort.10_new file with the modified Jastrow parameters.
22!>
23!> The program supports two modes:
24!> - Normal mode: zeros all Jastrow factors
25!> - leave1body mode: preserves one-body Jastrow terms
26!>
27!> @author TurboRVB group
28!> @date 2022

66 if (inputyes .eq. 1) then

68 do i = 1, ipj*nelorbj_c

70 if (.not. leave1body .or. (.not. orbcostn(i) .and. .not. orbcostn(j))) then
71 jasmat_c((j - 1)*ipj*nelorbj_c + i) = 0.d0
72 end if
73 end do
74 end do
75 else
76 do i = 1, ipj*nelorbj
77 do j = 1, ipj*nelorbj
78 if (.not. leave1body .or. (.not. orbcostn(i) .and. .not. orbcostn(j))) then
79 jasmat((j - 1)*ipj*nelorbj + i) = 0.d0
80 end if
81 end do
82 end do
83 end if
84 end if
85
86 if (iessz) then
87 if (contractionj .ne. 0) then
88 do i = 1, nelorbj_c
89 do j = 1, nelorbj_c
90 if (.not. leave1body .or. (.not. orbcostn(i) .and. .not. orbcostn(j))) then
91 jasmatsz_c((j - 1)*nelorbj_c + i) = 0.d0
92 end if
93 end do
94 end do
95 else
96 do i = 1, nelorbj
97 do j = 1, nelorbj
98 if (.not. leave1body .or. (.not. orbcostn(i) .and. .not. orbcostn(j))) then
99 jasmatsz((j - 1)*nelorbj + i) = 0.d0
100 end if
101 end do
102 end do
103 end if
104 end if
105
106 open (unit=10, file='fort.10_new', status='unknown' &
107 &, form='formatted')
108
109 call write_fort10(10)
110
111 close (10)
112
113end
