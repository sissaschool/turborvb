99 COMPLEX*16 FUNCTION zdotc_(N,ZX,INCX,ZY,INCY)
100*
101* -- Reference BLAS level1 routine --
102* -- Reference BLAS is a software package provided by Univ. of Tennessee, --
103* -- Univ. of California Berkeley, Univ. of Colorado Denver and NAG Ltd..--
104*
105* .. Scalar Arguments ..
106 INTEGER incx,incy,n
107* ..
108* .. Array Arguments ..
109 COMPLEX*16 zx(*),zy(*)
110* ..
111*
112* =====================================================================
113*
114* .. Local Scalars ..
115 COMPLEX*16 ztemp
116 INTEGER i,ix,iy
117* ..
118* .. Intrinsic Functions ..
119 INTRINSIC dconjg
120* ..
121
122#ifndef _FIXBUG_ZDOTC
123 COMPLEX*16, EXTERNAL :: zdotc
124 zdotc_ = zdotc(n,zx,incx,zy,incy)
125#else
126 ztemp = (0.0d0,0.0d0)
127 zdotc_ = (0.0d0,0.0d0)
128 IF (n.LE.0) RETURN
129 IF (incx.EQ.1 .AND. incy.EQ.1) THEN
130*
131* code for both increments equal to 1
132*
133 DO i = 1,n
134 ztemp = ztemp + dconjg(zx(i))*zy(i)
135 END DO
136 ELSE
137*
138* code for unequal increments or equal increments
139* not equal to 1
140*
141 ix = 1
142 iy = 1
143 IF (incx.LT.0) ix = (-n+1)*incx + 1
144 IF (incy.LT.0) iy = (-n+1)*incy + 1
145 DO i = 1,n
146 ztemp = ztemp + dconjg(zx(ix))*zy(iy)
147 ix = ix + incx
148 iy = iy + incy
149 END DO
150 END IF
151 zdotc_ = ztemp
152#endif
153 RETURN
154*
155* End of ZDOTC
156*
157 END
