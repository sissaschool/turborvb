16program test_zgetrfi
17
18#if defined(_OFFLOAD) && defined(_CUSOLVER)
19 use allio, only: handle, dev_zgetrf_workspace&
20 &, dev_zgetri_workspace, dev_info
21#endif
22
23 implicit none
24
25 complex*16, allocatable, dimension(:, :) :: A, C, C_ans, dummy
26 complex*16, allocatable, dimension(:) :: temp, work
27 real*8, allocatable, dimension(:, :) :: helper_r, helper_c
28 integer*4, allocatable, dimension(:) :: ipiv
29 complex*16 :: one = 1.d0, zero = 0.d0
30 integer :: s, gen, info, ii, jj, kk, lwork
31
32#if defined(_OFFLOAD) && defined(_CUSOLVER)
33 integer :: lworkspace, stat
34#endif
35
36 ! gen = 1 : Generate matrices
37 ! gen = 0 : Compare matrices
38
39 read (*, *) s, gen
40
41#if defined(_OFFLOAD) && defined(_CUSOLVER)
42 lworkspace = 1
43#ifdef RISC
44 call cusolver_handle_init_(handle)
45#else
46 call cusolver_handle_init(handle)
47#endif
48
49 allocate (dummy(s, s))
50!$omp target data map(alloc:dummy)
51
52#ifdef RISC
53 call cusolver_zgetrf_buffersize_(handle, stat, s, s&
54 &, dummy, s, lworkspace)
55#else
56 call cusolver_zgetrf_buffersize(handle, stat, s, s&
57 &, dummy, s, lworkspace)
58#endif
59
60!$omp end target data
61 deallocate (dummy)
62
63 allocate (dev_zgetri_workspace(s, s))
64 allocate (dev_zgetrf_workspace(lworkspace))
65
66!$omp target data map(alloc:dev_zgetrf_workspace, dev_zgetri_workspace&
67!$omp &, dev_Info)
68#endif
69
70 allocate (a(s, s))
71 allocate (ipiv(s))
72 allocate (temp(s))
73 allocate (c(s, s))
74 allocate (c_ans(s, s))
75 allocate (helper_r(s, s))
76 allocate (helper_c(s, s))
77
78 if (gen .eq. 1) then
79 call random_number(helper_r)
80 call random_number(helper_c)
81 a = cmplx(helper_r, helper_c)
82 open (unit=10, form="unformatted", file="A", action="write")
83 write (10) a
84 close (10)
85 print *, "GENERATED"
86 stop 0
87 else
88 open (unit=10, form="unformatted", file="A", action="read")
89 read (10) a
90 close (10)
91 end if
92
93 c = a
94
95 call zgetrf(s, s&
96 &, c, s&
97 &, ipiv&
98 &, info)
99
100 if (info .ne. 0) then
101 print *, "ERROR non-zero exit status from zgetrf", info
102 stop 0
103 end if
104
105#if defined(_OFFLOAD) && defined(_CUSOLVER) && defined(_X_)
106!$omp target data map(tofrom:C, ipiv)
107#endif
108 lwork = -1
109 allocate (work(1))
110 call zgetri(s&
111 &, c, s&
112 &, ipiv&
113 &, work, lwork&
114 &, info)
115 if (info .ne. 0) then
116 print *, "ERROR non-zero exit status from zgetri work size assesment"&
117 &, info
118 stop 0
119 end if
120 lwork = int(work(1))
121 deallocate (work)
122 allocate (work(lwork))
123 call zgetri(s&
124 &, c, s&
125 &, ipiv&
126 &, work, lwork&
127 &, info)
128 deallocate (work)
129#if defined(_OFFLOAD) && defined(_CUSOLVER) && defined(_X_)
130!$omp end target data
131#endif
132
133 if (info .ne. 0) then
134 print *, "ERROR non-zero exit status from zgetri_ wrapper", info
135 stop 0
136 end if
137
138 ! Multiply lower and upper matrix
139 !C_ans = 0
140 !do ii = 1, s
141 ! do jj = 1, s
142 ! do kk = 1, s
143 ! C_ans(ii, jj) = C_ans(ii, jj) + A(ii, kk) * C(kk, jj)
144 ! end do
145 ! end do
146 !end do
147 call zgemm("N", "N", s, s, s, one, a, s, c, s, zero, c_ans, s)
148
149 print *, c_ans
150 ! Remove one form the diagonal
151 c = c_ans
152 do ii = 1, s
153 c(ii, ii) = c_ans(ii, ii) - 1
154 end do
155
156 if (maxval(abs(c)) > 1.0d-6) then
157 print *, "ERROR"
158 else
159 print *, "OK"
160 end if
161
162#if defined(_OFFLOAD) && defined(_CUSOLVER)
163!$omp end target data
164#ifdef RISC
165 call cusolver_handle_destroy_(handle)
166#else
167 call cusolver_handle_destroy(handle)
168#endif
169#endif
170
171 deallocate (a, ipiv, temp, c, c_ans, helper_r, helper_c)
172
173#if defined(_OFFLOAD) && defined(_CUSOLVER)
174 deallocate (dev_zgetrf_workspace, dev_zgetri_workspace)
175#endif
176
177end program test_zgetrfi
