106 !> @brief Sets up matrices for exact diagonalization
107 !> @details This subroutine initializes the basis set, sets up integration
108 !> grids, and prepares overlap matrices for exact diagonalization
109 !> calculations. It handles parameter selection and weighted metrics.
110 !> @param[in] weights Weight factors for integration points
111 subroutine matrix_setup(weights)
112
113 implicit none
114 integer :: i, j
115 real(8) :: p(3), norm
116 real(8), intent(in) :: weights(*)
117
118 ! count the number of basis set elements
119 j = 0
120 do i = 1, param_number
121 if (sel_ed(i)) then
122 j = j + 1
123 if (mod(i, 3) .ne. 1) then
124 write (6, *) 'you are not allowed to change the power or the gaussian exponent by ED'
125 write (6, *) 'please change the sel_edection options'
126 stop
127 end if
128 end if
129 end do
130 bs_number = j
131
132 allocate (bs_coeff(bs_number), point2param(bs_number))
133
134 ! initialize the linear combination coefficients
135 j = 0
136 do i = 1, param_number
137 if (sel_ed(i)) then
138 j = j + 1
139 bs_coeff(j) = param0_ed(i)
140 point2param(j) = i
141 end if
142 end do
143
144 ! set integration grid dx for the interval [0,points_ed(points_number)]
145 ! DEFINE THE WEIGHTED METRICS
146 allocate (dx(points_number))
147
148 ! old integration
149 ! dx(1)=(points_ed(1,2)-points_ed(1,1))/2.d0+points_ed(1,1)
150 ! dx(1)=(points_ed(1,2)-points_ed(1,1))/2.d0
151 ! dx(1)=points_ed(1,2)/2.d0
152 ! dx(1)=0.d0
153 ! dx(2)=0.d0
154 ! dx(points_number)=(points_ed(1,points_number)-points_ed(1,points_number-1))/2.d0
155 ! do i=2,points_number-1
156 ! dx(i)=0.5d0*(points_ed(1,i+1)-points_ed(1,i-1))
157 ! enddo
158
159 do i = 1, points_number
160 dx(i) = points_ed(1, i)
161 end do
162
163 do i = 1, points_number
164 dx(i) = dx(i)*weights(i)
165 end do
166
167 ! for consistency the same as vecsigma in amoeba
168 ! pin the value at the origin
169 ! dx(1)=1.d0/abs(points_ed(1,1)-points_ed(1,2))/dble(points_number)/1000.d0
170 ! do i=2,points_number-1
171 ! dx(i)=1.d0/abs(points_ed(1,i)-points_ed(1,i+1))/dble(points_number)
172 ! enddo
173 ! dx(points_number)=dx(points_number-1)
174
175 ! norm=1.d0
176 ! do i=1,points_number
177 ! norm=norm+dx(i)
178 ! enddo
179 ! norm=norm*abs(points_ed(1,points_number)-points_ed(1,1))/dble(points_number)**2
180 ! do i=1,points_number
181 ! dx(i)=dx(i)*sigma/norm
182 ! dx(i)=1.d0/dx(i)**2
183 ! enddo
184
185 allocate (vecscratch(param_number/3))
186
187 ! reset the data to fit based on the fixed functions
188 ! new y_i = y_i - \sum_j alpha_j f_j (x_i)
189
190 i = 0
191 do while (i .lt. param_number)
192 i = i + 1
193 if (.not. sel_ed(i) .and. mod(i, 3) .eq. 1) then
194 p(1) = param0_ed(i)*norm_ed(1 + (i - 1)/3)
195 p(2) = param0_ed(i + 1)
196 p(3) = param0_ed(i + 2)
197
198 do j = 1, points_number
199 points_ed(2, j) = points_ed(2, j) - pseudofun(1, points_ed(1, j), p, vecscratch)
200 end do
201
202 i = i + 2
203
204 end if
205 end do
206
207 allocate (overlap(bs_number, bs_number), proj(bs_number))
208
209 end subroutine matrix_setup

230 !> @brief Computes overlap matrix for exact diagonalization
231 !> @details This subroutine computes the overlap matrix between basis
232 !> functions and projection vectors for exact diagonalization
233 !> calculations.
234 subroutine overlap_matrix
235
236 implicit none
237 integer :: i, j
238 real(8) :: integral, p1(3), p2(3)
239
240 do i = 1, bs_number
241 p1(1) = norm_ed(1 + (point2param(i) - 1)/3)
242 p1(2) = param0_ed(point2param(i) + 1)
243 p1(3) = param0_ed(point2param(i) + 2)
244
245 do j = 1, i
246 p2(1) = norm_ed(1 + (point2param(j) - 1)/3)
247 p2(2) = param0_ed(point2param(j) + 1)
248 p2(3) = param0_ed(point2param(j) + 2)
249
250 ! write(*,*) i,j,p1,p2
251
252 call oned_integral(p1, p2, integral)
253
254 overlap(i, j) = integral
255
256 ! if(i.eq.1.and.j.eq.1) overlap(1,1)=0.5d0
257 ! if(i.eq.2.and.j.eq.1) overlap(2,1)=0.0d0
258 ! if(i.eq.2.and.j.eq.2) overlap(2,2)=1.5d0
259
260 ! if(i.eq.j) write(6,*) i,i,overlap(i,i)
261
262 if (i .ne. j) overlap(j, i) = overlap(i, j)
263
264 end do
265 end do
266
267 do i = 1, bs_number
268 p1(1) = norm_ed(1 + (point2param(i) - 1)/3)
269 p1(2) = param0_ed(point2param(i) + 1)
270 p1(3) = param0_ed(point2param(i) + 2)
271 call oned_integral2(p1, integral)
272 proj(i) = integral
273 ! if(i.eq.1) proj(1)=0.5d0
274 ! if(i.eq.2) proj(2)=1.5d0
275 end do
276
277 end subroutine overlap_matrix

279 !> @brief Computes one-dimensional integral between two functions
280 !> @details This subroutine computes the overlap integral between two
281 !> Gaussian functions with given parameters.
282 !> @param[in] p1 Parameters for first function (norm, power, exponent)
283 !> @param[in] p2 Parameters for second function (norm, power, exponent)
284 !> @param[out] integral Computed integral value
285 subroutine oned_integral(p1, p2, integral)
286
287 implicit none
288
289 real(8) :: integral, p1(3), p2(3)
290 integer :: i
291
292 integral = 0.d0
293 do i = 1, points_number
294 integral = integral + pseudofun(1, points_ed(1, i), p1, vecscratch)*pseudofun(1, points_ed(1, i), p2, vecscratch)*dx(i)
295 end do
296
297 end subroutine oned_integral

318 !> @brief Performs orthonormalization of basis set
319 !> @details This subroutine orthonormalizes the basis set using
320 !> either diagonal or Gram-Schmidt orthonormalization
321 !> methods for exact diagonalization calculations.
322 !> @param[in] diag_type Type of orthonormalization (1=diagonal, 2=Gram-Schmidt)
323 !> @param[in] eps Tolerance for orthonormalization
324 subroutine orthonormalization(diag_type, eps)
325 ! it uses either the cholesky factorization and diagonalization or
326 ! the Sorella and Cavazzoni algorithm for overlap matrix preconditioning
327 ! (still experimental in this particular case...)
328 ! for the moment cholesky recommended
329
330 implicit none
331
332 integer :: nelorb_c, n, lwork, mine, lda, info, i, j, neig
333 real(8) :: abstol, eps, condnumber, cost, dlamch, norm
334 real(8), dimension(:, :), allocatable :: umatl
335 real(8), dimension(:), allocatable :: work, eig, eigmat
336 integer, dimension(:), allocatable :: iwork, ifail
337 real(8), dimension(:, :), allocatable :: mat_in, overs
338 character(60) :: diag_type
339
340 nelorb_c = bs_number
341 lda = bs_number
342 abstol = 2.d0*dlamch('S')
343 n = nelorb_c
344 mine = 1
345
346 allocate (overs(lda, nelorb_c))
347
348 ! calculation normalized orbitals and eigenvectors of the overlap matrix
349 ! maximum accuracy
350 ! by stable pre-diagonalization of the matrix overs
351
352 do j = 1, nelorb_c
353 do i = 1, lda
354 overs(i, j) = overlap(i, j)
355 ! write(6,*) i,j,overs(i,j)
356 end do
357 end do
358
359 if (trim(diag_type) .eq. 'cholesky') then
360
361 ! allocate(umatl(nelorb_c,nelorb_c))
362 lwork = 3*n
363 allocate (work(lwork), iwork(n))
364
365 ! cholesky factorization
366 call dpotrf('U', n, overs, n, info)
367 if (info .ne. 0) write (6, *) 'problems in dpotrf'
368
369 ! compute 1-norm of the matrix overs
370 norm = 0.d0
371 do j = 1, n
372 norm = max(norm, sum(abs(overs(:, j))))
373 end do
374
375 call dpocon('U', n, overs, n, norm, condnumber, work, iwork, info)
376 if (info .ne. 0) write (6, *) 'problems in dpocon'
377
378 write (6, *) '1-norm of overlap and condition number', norm
379 write (6, *) 'condition number', condnumber
380
381 ! compute inverse
382 call dpotri('U', n, overs, n, info)
383 if (info .ne. 0) write (6, *) 'problems in dpotri'
384
385 ! full matrix
386 do i = 1, n
387 do j = 1, i - 1
388 overs(i, j) = overs(j, i)
389 end do
390 end do
391
392 ! check
393 ! call dgemm('N','N',n,n,n,1.d0,overs,n,overlap,n,0.d0,umatl,n)
394 ! do i=1,n
395 ! do j=1,n
396 ! write(6,*) i,j,umatl(i,j)
397 ! enddo
398 ! enddo
399
400 call dgemv('T', n, n, 1.d0, overs, n, proj, 1, 0.d0, bs_coeff, 1)
401
402 ! other way directly calling dpotrs after dpotrf
403 ! equivalent to the above
404 ! do i=1,n
405 ! bs_coeff(i)=proj(i)
406 ! enddo
407
408 ! call DPOTRS('U',n,1,overs,n,bs_coeff,n,info)
409
410 ! if(info.lt.0) write(6,*) 'problems on dpotrs'
411
412 deallocate (work, iwork)
413 ! deallocate(umatl)
414
415 elseif (trim(diag_type) .eq. 'sorella_cavazzoni') then
416 ! perform Sorella Cavazzoni algorithm
417
418 lwork = 30*n
419 allocate (umatl(nelorb_c, nelorb_c), eig(nelorb_c), eigmat(nelorb_c))
420 allocate (work(lwork), iwork(5*nelorb_c), ifail(nelorb_c))
421
422 ! diagonalize the overlap matrix
423 call dsyevx('V', 'A', 'L', n, overs, lda, 0.d0, 0.d0, 1, 1, abstol &
424 , neig, eig, umatl, nelorb_c, work, lwork, iwork, ifail, info)
425
426 write (6, *) ' Lowest/Max eigenvalue overlap mat =', eig(1), eig(n)
427 condnumber = abs(eig(1)/eig(n))
428 do i = 2, n
429 cost = abs(eig(i)/eig(n))
430 if (cost .lt. condnumber) condnumber = cost
431 end do
432 write (6, *) ' Condition number basis set =', condnumber
433
434 do i = 1, n
435 if (eig(i)/eig(n) .gt. abs(eps)) then ! the condition number criterium
436 eigmat(i) = dsqrt(1.d0/eig(i))
437 else
438 mine = mine + 1
439 eigmat(i) = 0.d0
440 end if
441 end do
442
443 if (info .gt. 0) write (6, *) ' info > 0 in dsyevx !!! ', info
444
445 ! we assume here that the garbage eigenvectors are the ones
446 ! close to zero eigenvalue.
447 do i = 1, info
448 if (eigmat(ifail(i)) .ne. 0.d0) then
449 if (ifail(i) .ge. mine) mine = ifail(i) + 1
450 eigmat(1:ifail(i)) = 0.d0
451 end if
452 end do
453
454 if (mine .ne. 1) write (6, *) ' disregarded coll. =', mine - 1
455
456 ! first transformation umatl
457 do i = 1, n
458 umatl(:, i) = umatl(:, i)*eigmat(i)
459 end do
460
461 ! second diagonalization
462 ! now compute the overlap matrix in the new basis
463
464 allocate (mat_in(nelorb_c, nelorb_c))
465
466 ! destroy previously computed overlap
467 call dgemm('N', 'N', n, n, n, 1.d0, overlap, lda, umatl, n, 0.d0, mat_in, n)
468 call dgemm('T', 'N', n, n, n, 1.d0, umatl, n, mat_in, lda, 0.d0, overlap, lda)
469
470 do i = 1, n
471 if (eigmat(i) .eq. 0) overlap(i, i) = -1000.d0 - dble(i)
472 end do
473
474 ! diagonalize again the newly rotated overlap matrix
475 call dsyevx('V', 'A', 'L', n, overlap, lda, 0.d0, 0.d0, 1, 1, abstol &
476 , neig, eig, mat_in, n, work, lwork, iwork, ifail, info)
477
478 ! check again eigenvalues
479 do i = 1, n
480 if (eig(i) .gt. eps) then
481 mat_in(:, i) = mat_in(:, i)/dsqrt(eig(i))
482 else
483 if (eig(i) .gt. -1.d0) write (6, *) ' Further singular eigenvalue ', i
484 mat_in(:, i) = 0.d0
485 end if
486 end do
487 overs(1:n, 1:n) = umatl(1:n, 1:n)
488
489 ! Final dgemm: total rotation matrix
490 call dgemm('N', 'N', n, n, n, 1.d0, overs, lda, mat_in, n, 0.d0, umatl, n)
491
492 deallocate (mat_in)
493
494 ! end second diagonalization
495
496 ! solution for the coefficients of the linear combination
497 call dgemv('T', n, n, 1.d0, umatl, lda, proj, 1, 0.d0, eigmat, 1)
498 call dgemv('N', n, n, 1.d0, umatl, lda, eigmat, 1, 0.d0, bs_coeff, 1)
499
500 ! write(6,*) bs_coeff
501
502 deallocate (umatl, eig, eigmat)
503 deallocate (work, iwork, ifail)
504
505 else
506
507 write (6, *) 'I do not know the diagonalization algorithm', trim(diag_type)
508 write (6, *) 'Choose between "cholesky" and "sorella_cavazzoni"'
509 stop
510
511 end if
512
513 deallocate (overs)
514
515 end subroutine orthonormalization
516
517end module exact_diagonalization
