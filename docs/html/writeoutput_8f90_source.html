16!> @brief Write final output and statistics for TurboRVB calculation
17!> @details This subroutine writes comprehensive output including:
18!> - Performance profiling information (timing statistics)
19!> - Acceptance rates and move statistics
20!> - Energy averages and uncertainties
21!> - Branching statistics for DMC calculations
22!> - Optimization convergence information
23!> - Various diagnostic information based on calculation type
24subroutine write_output
25 use allio
26 implicit none
27 real*8 cclock
28 integer i
29 logical tst
30 write (6, *) ' All files written correctly ...'
31
32 if (kaverage) then
33 sumdiff = sumdiff/dble((nw/nk))/dble(i_main - iend)
34 else
35 sumdiff = sumdiff/dble(nw)/dble(i_main - iend)
36 end if
37 if (rata .gt. 0) rata = nacc/rata
38 if (signflip .ne. 0. .and. tbra .ne. 0.) then
39 signflip = signflip/(dble(i_main - iend)*dble(nw)*tbra)
40 write (6, *) ' Sign flips per unit time =', signflip
41 end if
42
43 ! time=cclock()-time
44 write (6, *) ' ##### TurboRVB profiling (sec.) ##### '
45 write (6, *) ' Time initialization =', timeinit
46 write (6, *) ' Total time with no initialization =', time
47 write (6, *) ' Total time with no measures ', timemc
48 write (6, *) ' Time measures =', time_meas
49 write (6, *) ' Time main =', time_main
50 if (itestr .eq. -5) then
51 write (6, *) ' Time optimization part =', timeopt
52 end if
53 if (iesbra) write (6, *) ' Time branching =', time_branch
54 write (6, *) ' Tracing the qmc update move '
55 write (6, *) ' Time ratiovar=', time_ratiovar
56 write (6, *) ' Time uptabtot=', time_uptabtot
57 write (6, *) ' Tracing the main routines '
58 write (6, *) ' Time uptabpip in uptabtot=', timepip
59 write (6, *) ' Time upnewwf in ratiovar/uptabtot=', timewf
60 write (6, *) ' Time upscratch =', timescra
61 write (6, *) ' accept. rate off diagonal moves =', rata
62 if (itest .eq. 2) then
63 rata = timemc/(time_meas + timeopt)
64 write (6, *) ' Optimal nbra suggested =', max(1, nint(nbra/rata))
65 end if
66
67 write (6, *) ' Average time for 1000 generations ', time/(i_main - iend)*1000.d0
68 psiav = psiav/counttot
69 psisav = psisav/counttot
70 countreg = countreg/counttot
71
72 if (psisav .gt. psiav**2) then
73 psisav = dsqrt(psisav - psiav**2)
74 else
75 psisav = 0.d0
76 end if
77
78 if (hopfraction .ne. 0.d0 .and. abs(parcut) .ne. 100.d0 .and. nacc .gt. 0) then
79 write (6, *) ' Fraction large hopping moves =' &
80 &, (nion - 1)*acclarge/nacc
81 end if
82 if (epscuttype .eq. -100 .and. nacc .gt. 0) then
83 write (6, *) ' Fraction large lattice a moves =' &
84 &, acclarge/nacc
85 end if
86 if (npsa .gt. 0 .and. itest .eq. 1) then
87 write (6, *) '# pseudo off diag moves per generation per walker =', &
88 &naccpseudo/(dble(i_main - iend)*dble(nw))
89 inquire (8, opened=tst)
90 if (tst) close (8)
91 end if
92 if (epscuttype .gt. 0) then
93 write (6, *) '# non trivial accepted/tried =', &
94 &nontr/(dble(i_main - iend)*dble(nw)*dble(nbra))
95 end if
96 if (epstldmc .ne. 0.d0 .and. nacc .gt. 0) then
97 write (6, *) ' Number of rejected moves inside dmc cutoff ', nint(countcut)
98 write (6, *) ' Corresponding to a fraction of ', countcut/nacc
99 end if
100
101 if (iesbra) write (6, *) 'Av. num. of survived walkers/ # walkers&
102 & in the branching', sumdiff
103 if (itest .eq. 2) write (6, *) &
104 &'Average inverse A wf =', psiav, '+/-', psisav
105 if (itest .ne. 2) then
106 if (mod(typereg, 2) .eq. 0) then
107 write (6, *) ' Average log det =', psiav, '+/-', psisav
108 else
109 write (6, *) ' Average log Psi =', psiav, '+/-', psisav
110 end if
111 if (yesnleft) then
112 write (6, *) ' Final value of av.energy =', -lambda*ris(2)
113 end if
114 write (6, *) ' Average time x branching =', tave_cyrus/tcount_cyrus/ris(2)
115
116 end if
117 if (change_parr .and. itestr .eq. -5) then
118 write (6, *) ' Number of times with parr <=', parr, ' amounts to ', iesconv
119 write (6, *) ' Warning you can safely average with the previous', iesconv, ' Optimization steps'
120 end if
121
122 if (parcutg .ne. 0) then
123 write (6, *) ' Fraction nodal surface/vol = ', countreg
124 end if
125 if (change_tstep .and. itest .eq. 2) then
126 write (6, *) ' Final tstep found ', tstep
127 end if
128#ifdef _TIME
129 if (yes_crystalj) then
130 write (6, *) ' Fraction zero wfs (det & J) = ', count_zerowf/count_allwf
131 else
132 write (6, *) ' Fraction zero wfs (det only) = ', count_zerowf/count_allwf
133 end if
134 do i = 1, 11
135 write (6, *) ' Time task direct/adjoint # ', i, timings(i), timingsb(i)
136 end do
137 write (6, *) ' Total time compute_fast/compute_fast_b ', sum(timings(1:11)), sum(timingsb(1:11))
138#endif
139end subroutine write_output
