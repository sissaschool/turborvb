16subroutine pseudoset(jel, kel, ivic, prefactor, pseudolocal &
17 &, nel, dist, rion, wpseudo, ncore, skip &
18 &, indtm, iopt, angle, psip, Lbox, mindist, iflagpseudo, vec)
19 use allio, only: rcutoff, versor, nintpseudo, kindion, pshell, nparpshell &
20 &, parshell, lmax, wintpseudo, jpseudo, enforce_detailb, norm_metric
21 use cell, only: cellscale, s2r, car2cry, lmin, applypbc, yes_tilted, metric
22 implicit none
23 integer jel, core, ncore, lang, jj, kk, nel, skip, indtm, npseudo &
24 &, iflagpseudo, kimages, maximages
25 real*8 ivic(3, *), prefactor(skip, *), dist(*), rion(3, *), distmin &
26 &, wpseudo(*), pseudofun, pseudolocal, scal, legfun &
27 &, rver(3), r(3), rsav(3), angle(3, 3), psip(*), Lbox, mindist, vec(3)
28
29 ! input kel,rion,dist
30 ! output prefactor (passed ) ,legendre,wpseudo, ivic (passed)
31 ! pseudolocal (passed)
32 logical iopt
33 real*8 kel(3)
34
35 if (iflagpseudo .ne. 0 .or. nintpseudo .le. 0) return
36
37 ! if iopt.eq.true generate a new rotation matrix
38 if (iopt) then
39 if (enforce_detailb) then
40 call fillmatrix_vec(angle, vec)
41 else
42 call fillmatrix(angle)
43 end if
44 end if
45
46 do jj = 1, skip
47 do kk = 1, 3
48 ivic(kk, jj) = 0.d0
49 end do
50 prefactor(jj, jel) = 0.d0
51 end do
52 pseudolocal = 0.d0
53
54 npseudo = 0
55 do core = 1, ncore
56 distmin = dist(kindion(core))
57
58 if (distmin .le. rcutoff(core)) then
59
60 if (distmin .le. mindist) distmin = mindist
61
62 do kk = 1, 3
63 rsav(kk) = kel(kk) - rion(kk, kindion(core))
64 end do
65 if (lbox .gt. 0.d0) then
66 ! Find the image ion rion(:,kindion(core))-rdiff closest to kel
67 ! NB I am not using PBC on electron positions
68 ! here so it works also for APBC or whatever boundary condition.
69 ! call ApplyPBC(rsav,1) ! the output will be rsav-rion(:,..)+rdiff
70 rsav(:) = car2cry(:, 1)*rsav(1) + car2cry(:, 2)*rsav(2) + car2cry(:, 3)*rsav(3)
71 rsav(1:3) = rsav(1:3) - cellscale(1:3)*anint(rsav(1:3)/cellscale(1:3))
72 if (rcutoff(core) .gt. lmin/2.d0) then
73 maximages = 4
74 else
75 maximages = 1
76 end if
77 else
78 maximages = 1
79 end if
80
81 do kimages = 1, maximages
82 ! redefine all for the images
83 r = rsav
84 ! if(kimages.eq.1) then
85 ! d=dsav
86 if (kimages .eq. 2) then ! below PBC is certain
87 ! first image over x
88 if (r(1) .gt. 0.d0) then
89 r(1) = rsav(1) - cellscale(1)
90 else
91 r(1) = rsav(1) + cellscale(1)
92 end if
93 distmin = norm_metric(r, metric)
94 elseif (kimages .eq. 3) then
95 ! second image over y
96 if (r(2) .gt. 0.d0) then
97 r(2) = rsav(2) - cellscale(2)
98 else
99 r(2) = rsav(2) + cellscale(2)
100 end if
101 distmin = norm_metric(r, metric)
102 elseif (kimages .eq. 4) then
103 ! third image over z
104 if (r(3) .gt. 0.d0) then
105 r(3) = rsav(3) - cellscale(3)
106 else
107 r(3) = rsav(3) + cellscale(3)
108 end if
109 distmin = norm_metric(r, metric)
110 end if
111
112 if (lbox .gt. 0.d0) then
113 ! go back in Cartesian Coordinates
114 r(1:3) = r(1:3)/cellscale(1:3)
115 r(:) = s2r(:, 1)*r(1) + s2r(:, 2)*r(2) + s2r(:, 3)*r(3)
116 end if
117
118 if (distmin .le. rcutoff(core)) then
119
120 indtm = indtm + nintpseudo
121
122 npseudo = npseudo + 1
123
124 if (npseudo .gt. skip/nintpseudo) then
125 write (6, *) ' ERROR nuclei too close for pseudo !!! '
126 write (6, *) 'increase max # of nuclei, npsamax>', skip/nintpseudo
127 iflagpseudo = 1
128 return
129 end if
130
131 do lang = 1, pshell(core) - 1
132 wpseudo(lang) = pseudofun(nparpshell(lang, core) &
133 &, distmin, parshell(1, jpseudo(lang, core)), psip)
134 end do
135 pseudolocal = pseudolocal &
136 & + pseudofun(nparpshell(pshell(core), core) &
137 &, distmin, parshell(1, jpseudo(pshell(core), core)), psip)
138
139 do jj = (npseudo - 1)*nintpseudo + 1, npseudo*nintpseudo
140
141 ! call rotation(angle
142 ! & ,versor(1,jj-(npseudo-1)*nintpseudo),rver)
143 call dgemv('N', 3, 3, 1.d0, angle, 3 &
144 &, versor(1, jj - (npseudo - 1)*nintpseudo), 1, 0.d0, rver, 1)
145
146 scal = 0.d0
147 do kk = 1, 3
148 scal = scal + rver(kk)*r(kk)
149 ! This has to be referred to the chosen Image.
150 ! rver(kk)=distmin*rver(kk)+rion(kk,kindion(core))-rdiff(kk)
151 ! ivic(kk,jj)=rver(kk)-kel(kk)
152 ! N.B. rion-rdiff-kel=-r by the above definition
153
154 ! ivic is the versor to be added to kel to have the new
155 ! mesh electronic point in the pseudo integration.
156 ivic(kk, jj) = rver(kk)*distmin - r(kk)
157 end do
158
159 scal = scal/distmin
160
161 do lang = 1, pshell(core) - 1
162 prefactor(jj, jel) = prefactor(jj, jel) + &
163 & (2.d0*lang - 1.d0)*wpseudo(lang)*legfun(lang - 1, scal)
164 end do
165 prefactor(jj, jel) = prefactor(jj, jel) &
166 & *2.d0*wintpseudo(jj - (npseudo - 1)*nintpseudo)
167
168 end do
169
170 end if ! endif check distance images
171
172 end do ! enddo kimages
173
174 end if ! endif check distance
175
176 end do ! enddo core
177
178 return
179end subroutine pseudoset
