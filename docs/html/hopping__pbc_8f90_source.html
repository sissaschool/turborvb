16!> @brief Perform hopping moves for quantum Monte Carlo with periodic boundary conditions
17!> @details This subroutine implements a hopping algorithm for quantum Monte Carlo simulations
18!> where particles can move between different ion positions. With probability |tjasder|,
19!> the particle performs a hopping move to a randomly selected ion position (different from
20!> the current minimum position). With probability 1-|tjasder|, the particle performs a
21!> regular displacement move. The hopping move satisfies detailed balance by considering
22!> both forward and backward hopping directions with equal probability. The subroutine
23!> also handles periodic boundary conditions for the simulation box.
24!> @param[in] nion Number of ions in the system
25!> @param[in] jn Index of the coordinate being updated (1, 2, or 3 for x, y, z)
26!> @param[in,out] dstep Displacement step size (set to 0 for hopping moves)
27!> @param[in,out] rcart Cartesian coordinates of the particle
28!> @param[in] rion Cartesian coordinates of all ions
29!> @param[in] imin Index of the current minimum ion position
30!> @param[out] itry Index of the target ion for hopping
31!> @param[in] tjasder Jastrow derivative parameter controlling hopping probability
32!> @param[in] LBox Simulation box length for periodic boundary conditions
33subroutine hopping(nion, jn, dstep, rcart &
34 &, rion, imin, itry, tjasder, LBox)
35 use cell
36 implicit none
37 integer nion, jn, i, itry, imin
38 real(8) rcart(3), LBox, scart(3)
39 real(8) rion(3, *), dstep, drand1, tjasder, zeta
40
41 ! Make an hopping move with probability 50%
42 ! In the remaining case make the usual move around the atom
43
44 zeta = drand1()
45
46 if (zeta .lt. abs(tjasder) .and. nion .gt. 1) then
47
48 ! Generate a random itry ne imin
49
50 ! imin=nion*drand1()+1
51
52 itry = (nion - 1)*drand1() + 1
53 if (itry .ge. imin) itry = itry + 1
54
55 ! In order to satisfy detailed balance both the
56 ! hopping directions have to be taken into account.
57 if (drand1() .gt. 0.5d0) then
58 do i = 1, 3
59 rcart(i) = rcart(i) + rion(i, itry) - rion(i, imin)
60 end do
61 else
62 do i = 1, 3
63 rcart(i) = rcart(i) - rion(i, itry) + rion(i, imin)
64 end do
65 end if
66 dstep = 0.d0
67 else
68 rcart(jn) = rcart(jn) + dstep
69 end if
70
71 !********** Periodic Boundary Conditions **********
72 ! if(LBox.gt.0) call ApplyPBC(rcart,1)
73 !********** Periodic Boundary Conditions **********
74
75 return
76end
