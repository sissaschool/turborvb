19!> @brief Write and handle error messages for parallel and serial execution
20!> @details This subroutine writes an error message to output. If ierr <= 0, it does nothing.
21!> If ierr > 0, it prints the error and stops execution. In parallel execution,
22!> it also writes to a crash file and aborts MPI if available.
23!> @param[in] calling_routine Name of the calling routine
24!> @param[in] message Error message to output
25!> @param[in] ierr Error flag (if > 0, triggers stop)
26subroutine errore(calling_routine, message, ierr)
27 !----------------------------------------------------------------------------
28 !
29 ! ... This is a simple routine which writes an error message to output:
30 ! ... if ierr <= 0 it does nothing,
31 ! ... if ierr > 0 it stops.
32 !
33 ! ... **** Important note for parallel execution ***
34 !
35 ! ... in parallel execution unit 6 is written only by the first node;
36 ! ... all other nodes have unit 6 redirected to nothing (/dev/null).
37 ! ... As a consequence an error not occurring on the first node
38 ! ... will be invisible. For T3E and ORIGIN machines, this problem
39 ! ... is solved by writing an error message to unit * instead of 6.
40 ! ... Whenever possible (IBM SP machines), we write to the standard
41 ! ... error, unit 0 (the message will appear in the error files
42 ! ... produced by loadleveler).
43 !
44 use io_global, only: stdout
45 use io_files, only: crashunit, crash_file
46 use parallel_include
47 !
48 implicit none
49 !
50 character(LEN=*), intent(IN) :: calling_routine, message
51 ! the name of the calling calling_routinee
52 ! the output messagee
53 integer, intent(IN) :: ierr
54 ! the error flag
55 integer :: mpime, mpierr
56 ! the task id
57 !
58 logical :: exists
59 !
60 !
61 if (ierr <= 0) return
62 !
63 ! ... the error message is written un the "*" unit
64 !
65 write (unit=*, fmt='(/,1X,78("%"))')
66 write (unit=*, &
67 fmt='(5X,"from ",A," : error #",I10)') calling_routine, ierr
68 write (unit=*, fmt='(5X,A)') message
69 write (unit=*, fmt='(1X,78("%"),/)')
70 !
71#if defined (__PARA) && defined (__AIX)
72 !
73 ! ... in the case of ibm machines it is also written on the "0" unit
74 ! ... which is automatically connected to stderr
75 !
76 write (unit=0, fmt='(/,1X,78("%"))')
77 write (unit=0, &
78 fmt='(5X,"from ",A," : error #",I10)') calling_routine, ierr
79 write (unit=0, fmt='(5X,A)') message
80 write (unit=0, fmt='(1X,78("%"),/)')
81 !
82#endif
83 !
84 write (*, '(" stopping ...")')
85 !
86 call flush_unit(stdout)
87 !
88#if defined (__PARA) && defined (__MPI)
89 !
90 mpime = 0
91 !
92 call mpi_comm_rank(mpi_comm_world, mpime, mpierr)
93 !
94 ! .. write the message to a file and close it before exiting
95 ! .. this will prevent loss of information on systems that
96 ! .. do not flush the open streams
97 ! .. added by C.C.
98 !
99 open (unit=crashunit, file=crash_file, &
100 position='APPEND', status='UNKNOWN')
101 !
102 write (unit=crashunit, fmt='(/,1X,78("%"))')
103 write (unit=crashunit, fmt='(5X,"task #",I10)') mpime
104 write (unit=crashunit, &
105 fmt='(5X,"from ",A," : error #",I10)') calling_routine, ierr
106 write (unit=crashunit, fmt='(5X,A)') message
107 write (unit=crashunit, fmt='(1X,78("%"),/)')
108 !
109 close (unit=crashunit)
110 !
111 ! ... try to exit in a smooth way
112 !
113 call mpi_abort(mpi_comm_world, mpierr)
114 !
115 call mpi_finalize(mpierr)
116 !
117#endif
118 !
119 stop 2
120 !
121 return
122 !
123end subroutine errore

156!> @details This module provides utilities for tracking the call stack, reporting

174 !> @brief Initialize the error handler with the first routine name
175 !> @param[in] routine_name Name of the first routine in the call stack
176 subroutine init_error(routine_name)
177 implicit none
178 character(len=*), intent(in) :: routine_name
179
180 allocate (routine_chain)
181
182 routine_chain%routine_name = routine_name
183 nullify (routine_chain%previous_link)
184
185 return
186 end subroutine init_error
187
188 !> @brief Add a routine name to the call stack
189 !> @param[in] routine_name Name of the routine to add
190 subroutine add_name(routine_name)
191 implicit none
192 character(len=*), intent(in) :: routine_name
193 type(chain), pointer :: new_link
194
195 allocate (new_link)
196 new_link%routine_name = routine_name
197 new_link%previous_link => routine_chain
198 routine_chain => new_link
199
200 return
201 end subroutine add_name

203 !> @brief Remove the most recent routine name from the call stack
204 subroutine chop_name
205 implicit none
206 type(chain), pointer :: chopped_chain
207
208 chopped_chain => routine_chain%previous_link

215 !> @brief Recursively print the call stack for error tracing
216 !> @param[in] error_code Error code to determine stop or return
217 recursive subroutine trace_back(error_code)
218
219 implicit none
220 integer :: error_code
221
222 write (unit=*, fmt=*) " Called by ", routine_chain%routine_name
223 if (.not. associated(routine_chain%previous_link)) then
224 write (unit=*, fmt=*) &
225 " +++ +++ +++ +++ +++ +++ +++ +++ +++ +++ +++ +++ +++ +++ +++"
226 write (unit=*, fmt=*) " "
227 if (error_code > 0) then
228 stop
229 else
230 return
231 end if
232 end if
233
234 routine_chain => routine_chain%previous_link
235 call trace_back(error_code)
236
237 end subroutine trace_back

239 !> @brief Report a memory error or warning with traceback
240 !> @param[in] message Error or warning message
241 !> @param[in] error_code (Optional) Error code (default 1: fatal, -1: warning)
242 subroutine error_mem(message, error_code)
243 character(len=*), intent(in) :: message
244 integer, intent(in), optional :: error_code
245 integer :: action_code
246 type(chain), pointer :: save_chain
247
248 if (present(error_code)) then
249 action_code = error_code
250 else
251 action_code = 1
252 end if
253
254 if (action_code /= 0) then
255 write (unit=*, fmt=*) " "
256 write (unit=*, fmt=*) &
257 " +++ +++ +++ +++ +++ +++ +++ +++ +++ +++ +++ +++ +++ +++ +++"
258
259 if (action_code > 0) then
260 write (unit=*, fmt=*) " Fatal error in routine `", &
261 trim(routine_chain%routine_name), "': ", message
262 else
263 write (unit=*, fmt=*) " Warning from routine `", &
264 trim(routine_chain%routine_name), "': ", message
265 save_chain => routine_chain
266 end if
267 write (unit=*, fmt=*) &
268 " +++ +++ +++ +++ +++ +++ +++ +++ +++ +++ +++ +++ +++ +++ +++"
269 routine_chain => routine_chain%previous_link
270 call trace_back(action_code)
271 routine_chain => save_chain
272 end if
273
274 return
275 end subroutine error_mem
276
277 !> @brief Issue a warning message with traceback
278 !> @param[in] message Warning message
279 subroutine warning(message)
280 character(len=*), intent(in) :: message
281 call error_mem(message, -1)
282 return
283 end subroutine warning
