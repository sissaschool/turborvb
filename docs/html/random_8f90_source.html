16!> @brief Random configuration selection for quantum Monte Carlo moves
17!>
18!> This subroutine implements a random configuration selection algorithm
19!> for quantum Monte Carlo calculations, particularly useful for
20!> non-diagonal moves in variational Monte Carlo and diffusion Monte Carlo.
21!> It uses a cumulative probability table to select configurations
22!> based on their transition probabilities.
23!>
24!> @param[in] L Number of particles or components per configuration
25!> @param[in] Lz Total number of possible configurations
26!> @param[in] ztry Random number in [diag, wtot] for configuration selection
27!> @param[in] table Probability table containing transition weights (ipc, *)
28!> @param[in] diag Diagonal element weight (lower bound for ztry)
29!> @param[out] iout Selected particle index within the configuration
30!> @param[out] indvic Selected configuration index
31!> @param[out] sign Sign of the selected transition (+1 or -1)
32!> @param[in] npow Power parameter for transition weight modification
33!> @param[in] gamma Gamma parameter for transition weight modification
34!> @param[in] istart Starting index for configuration selection
35!> @param[in] indtm Array of maximum indices for each particle
36!> @param[in] ipc Leading dimension of table array
37!>
38!> @details
39!> The subroutine performs the following operations:
40!> 1. Initializes cumulative probability with diagonal element
41!> 2. Iterates through configurations until cumulative probability ≥ ztry:
42!> - Calculates configuration and particle indices
43!> - Applies power and gamma modifications to transition weights
44!> - Accumulates modified weights in cumulative probability
45!> 3. Handles edge cases and roundoff errors:
46!> - Adjusts index if final configuration has zero weight
47!> - Ensures valid configuration selection
48!> 4. Determines sign based on transition weight sign
49!> 5. Returns selected configuration and particle indices
50!>
51!> @note The algorithm assumes diag < ztry ≤ wtot (non-diagonal move)
52!> @note Configuration index is calculated as: index = (i-1)/L + 1
53!> @note Particle index is calculated as: ipart = i - (index-1)*L
54!> @note Negative indvic indicates unrecoverable error (zero weight)
55!> @note The subroutine handles both positive and negative transition weights
56!> @note Power and gamma parameters modify transition weights for optimization
57subroutine random(L, Lz, ztry, table, diag, iout, indvic, sign &
58 &, npow, gamma, istart, indtm, ipc)
59 implicit none
60 integer ipc, i, L, Lz, indvic, iout, index, ipart, istart, indtm(*)
61 real*8 diag, table(ipc, *), try, ztry, sign, gamma, cost, npow
62
63 ! choose a random configuration given the table,wtot
64 ! it is assumed that diag < ztry <= wtot ,i.e. non diagonal move
65
66 try = diag
67 i = 0
68 do while (ztry .ge. try .and. i .lt. lz)
69 i = i + 1
70 index = (i - 1)/l + 1
71 ipart = i - (index - 1)*l
72 if (npow .ne. 0.d0 .and. table(1, i) .gt. 0.d0 .and. &
73 (index .ge. istart .and. index .le. indtm(ipart))) then
74 try = try + table(1, i)*(1.d0 - npow*(1.d0 + gamma))
75 elseif (table(1, i) .gt. 0.d0) then
76 try = try + table(1, i)
77 else
78 cost = -gamma*table(1, i)
79 try = try + cost
80 end if
81 end do
82
83 ! It is possible by chance and roundoff that the final i=Lz and table(Lz)=0
84 ! if for roundoff i=Lz set i the maximum with table(i)=/0
85 if (i .eq. lz) then
86 do while ((table(1, i) .eq. 0.d0 .or. (table(1, i) .lt. 0 .and. gamma .eq. 0)&
87 &.or. (npow*(1.d0 + gamma) .eq. 1.d0 .and. index .ge. istart)) .and. i .gt. 1)
88 i = i - 1
89 index = (i - 1)/l + 1
90 end do
91 end if
92
93 ! if(i.eq.0) i=1
94 if (table(1, i) .lt. 0.) then
95 sign = -1.d0
96 else
97 sign = 1.d0
98 end if
99
100 indvic = index
101 iout = ipart
102 ! indvic=(i-1)/L+1
103 ! iout=i-(indvic-1)*L
104
105 if (table(1, i) .eq. 0.d0) indvic = -indvic ! Unrecoverable error
106
107 return
108end
