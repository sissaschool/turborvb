42 ! computes the vectors x(ndim)=(a(1)*n(1),..,a(ndim)*n(ndim))
43 ! where n(i) are integers and x(1)**2+..+x(ndim)**2.le.cut**2
44 ! the vectors x(i) are stored in rkcomp in
45 ! the order given by the values of their norms.
46 ! also nshlls gives the number of
47 ! different values of the norms ( the relative square norms
48 ! differ by less than 1.e-5) and rknorm(i) gives
49 ! these nshlls norms and kmult(i) is the last vector
50 ! whose norm is given by rknorm(i). hence the total
51 ! number of vectors of magnitude less then or equal to cut is kmult(nshlls).
52 !
53
54 kmult(0) = 0
55 rknorm(0) = 0.d0
56 c2 = cut**2
57 npts = 1
58 do l = 1, ndim
59 nkspan(l) = (0.00001 + abs(cut/a(l)))
60 ! range of search is (-nkspan(l),+nkspan(l))
61 icount(l) = -nkspan(l)
62 npts = (2*nkspan(l) + 1)*npts
63 end do

65 nvects = 0
66 do 3 i = 1, npts

68 ! nzero will be the first nonzero entry in icount

70 do 4 l = 1, ndim

72 x(l) = icount(l)*a(l)
73 rsq = rsq + x(l)**2
74 if (rsq .gt. c2) go to 30
754 continue
76 ! we only take half of the vectors and exclude the one at the origin
77 if (nzero .le. 0) go to 30
