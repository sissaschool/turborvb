16!> @brief Orbital plotting program for TurboRVB
17!> @details This program generates 3D plots of molecular orbitals, charge
18!> density, and spin density from TurboRVB wave functions. It reads a
19!> fort.10 file and outputs data files suitable for visualization
20!> software like XCrySDen. The program supports both open and periodic
21!> boundary conditions and can handle complex wave functions.

92 end do

94 if (.not. iespbc) then
95 write (*, *) ' Choose box size (x,y,z) '
96 read (*, *) cell_loc(:)
97 write (*, *) cell_loc(:)
98 cellscale(1:3) = cell_loc(1:3)
99 do i = 1, nion
100 rion(1:3, i) = rion(1:3, i) - center(1:3) + cell_loc(1:3)/2.d0
101 end do
102 ! for open system the center of the box is the (0,0,0) point
103 origin = 0.d0
104 else
105 !call ApplyPBC(rion,nion)
106 !do i=1,nion
107 !rion(1:3,i)=rion(1:3,i)+cellscale(1:3)/2.d0
108 !enddo
109 cell_loc(1:3) = cellscale(1:3)
110 write (*, *) ' Choose shift reference unit cell PBC : '
111 read (*, *) origin(:)
112 origin(1:3) = origin(1:3)*cellscale(1:3)
113 end if
114
115 write (*, *) ' Choose number of mesh points (x,y,z) : '
116 read (*, *) mesh(:)
117 write (*, *) mesh(:)
118
119 ! To match with output plot in xcrysden
120 if (iespbc) mesh(:) = mesh(:) + 1
121

124 lq = index(question, ' ') - 1

126

128 chargeon = .false.
129 nochsp = .true.
130 lower = 1
131 upper = molecular

134 write (*, *) ' Please give a range between 1 and ', molecular
135 read (*, *) lower, upper
136 write (*, *) lower, upper
137 end if
138
139 if (question(1:lq) .eq. 'charge') then
140 write (*, *) ' Please give the lowest molecular orbital within 1 and ', molecular
141 read (*, *) lower
142 upper = -1
143 write (6, *) 'Number of fully occupied molecular orbital/total number occupied by up and down ?'
144 read (5, *) nfil, ntot
145
146 end if
147
148 if (question(1:lq) .eq. 'spin') then
149 write (*, *) ' Please give the lowest molecular orbital within 1 and ', molecular
150 read (*, *) lower
151 upper = -2
152 write (6, *) 'Number of fully occupied molecular orbital/total number occupied by up and down ?'
153 read (5, *) nfil, ntot
154
155 write (6, *) ' Momentum magnetization ? (unit 2pi/cellscale) '
156 read (5, *) kspin(1:3)
157 kspin(1:3) = 2.d0*pi*kspin(1:3)/cellscale(1:3)
158
159 write (6, *) ' K rescaled =', kspin(1:3)
160
161 end if
162
163 if (upper .eq. -1) then
164 chargeon = .true.
165 nochsp = .false.
166 if (symmagp) then
167 if (nelup - neldo .ne. 0) then
168 upper = molecular
169 else
170 upper = ntot
171 end if
172 else
173 if (nelup - neldo .ne. 0) then
174 upper = molecular
175 else
176 upper = 2*ntot
177 end if
178 end if
179 write (6, *) ' Warning computing charge density '
180 elseif (upper .eq. -2) then
181 spinon = .true.
182 nochsp = .false.
183 if (nelup - neldo .ne. 0) then
184 upper = molecular
185 else
186 upper = 2*ntot
187 end if
188 if (symmagp) call error("plot_orbitals", " Spin density is not possible for symmagp=.true.", 1, 0)
189 write (6, *) ' Warning computing spin density '
190 end if
191
192 allocate (imap_loc(molecular), control(molecular), psi(ipc))
193
194 control = .false.
195 indorb = 0
196 imol = 0
197 ind = 0
198 ind_mol = 0
199
200 do i = 1, nshell_c
201 if (ioptorb_c(i) .eq. 1000000) then
202 ind = ind + 1
203 if (ioccup_c(ind) .eq. 1) then
204 indorb = indorb + 1
205 imol = imol + 1
206 imap_loc(imol) = indorb
207 if (imol .ge. lower .and. imol .le. upper) then
208 control(imol) = .true.
209 ind_mol = ind_mol + 1
210 end if
211 end if
212 else
213 do j = 1, mult_c(i)
214 ind = ind + 1
215 if (ioccup_c(ind) .eq. 1) indorb = indorb + ipf
216 end do
217 end if
218 end do
219
220 chosen_mol = ind_mol
221 write (*, *) ' # of orbitals written =', chosen_mol
222
223 allocate (norm(molecular + 1))
224 norm = 0.d0
225
226 if (iespbc) then
227 step(:) = cell_loc(:)/dble(mesh(:) - 1)
228 else
229 step(:) = cell_loc(:)/dble(mesh(:))
230 end if
231 vol = step(1)*step(2)*step(3)
232
233 if (chargeon .or. spinon) then
234 allocate (datagrid(mesh(1), mesh(2), mesh(3)))
235 else
236 allocate (datagrid(ipc*mesh(1), mesh(2), mesh(3)))
237 if (ipf .eq. 2) allocate (datagrid_sum(ipc*mesh(1), mesh(2), mesh(3)), datagrid_prod(ipc*mesh(1), mesh(2), mesh(3)))
238 end if
239
240 allocate (buffer_winv(ipc*nelorb))
241 iflagnorm = 3
242
243 ! compute psi^2 normalization for each orbital
244 if (nochsp) then
245
246 do j = 1, molecular
247 norm(j) = 1.d0
248 end do
249
250 else
251 if (symmagp) then
252 norm = 0.d0
253 norm(1:nfil) = 2.d0
254
255 if (nfil .lt. ntot) then
256 costocc = dble(nel - 2*nfil - nelup + neldo)/(ntot - nfil)
257 norm(nfil + 1:ntot) = costocc
258 end if
259
260 if (nelup .gt. neldo) norm(molecular - nelup + neldo + 1:molecular) = 1.d0
261
262 else
263 norm(1:2*nfil) = 1.d0
264
265 if (nfil .lt. ntot) then
266 costocc = dble(nel - 2*nfil - nelup + neldo)/(2*ntot - 2*nfil)
267 norm(2*nfil + 1:2*ntot) = costocc
268 elseif (nelup .gt. neldo) then
269 norm(molecular - nelup + neldo + 1:molecular) = 1.d0
270 end if
271
272 end if
273 end if
274
275 datagrid = 0.d0
276 magorb = 0.d0
277
278 ind_mol = 0
279
280 psip = 0.d0
281
282 do j = 1, molecular
283
284 !!It does not do anything if ipf=1, otherwise it allows you to print
285 !!the up and the down part of the molecular orbitals
286 do up_down = 1, ipf
287 imu = (up_down - 1)*nelorbh + 1
288
289 if (control(j)) then
290 !
291 !
292 if (up_down .eq. 1) ind_mol = ind_mol + 1
293 do iii = 1, mesh(3)
294 do ii = 1, mesh(2)
295 do i = 1, mesh(1)
296 !
297 ! rpoint(1)=(i-1)*step(1)-origin(1)
298 ! rpoint(2)=(ii-1)*step(2)-origin(2)
299 ! rpoint(3)=(iii-1)*step(3)-origin(3)
300
301 rpoint(:) = (i - 1)*step(1)*at(:, 1) + (ii - 1)*step(2)*at(:, 2) + (iii - 1)*step(3)*at(:, 3)
302
303 if (ipc .eq. 2 .and. mod(j, 2) .eq. 1) then
304 call upnewwf(1, 0, 0, 1, nshell, ioptorb, ioccup, rpoint, 1, r, rmu, &
305 dupr, zetar, rion, psip, buffer_winv, nelorb, nion, kion, &
306 iflagnorm, cnorm, lbox, rmucos, rmusin, minimum_distance, &
307 indpar_tab, indorb_tab, indshell_tab, .false.)
308 else
309 call upnewwf(1, 0, 0, 1, nshell, ioptorb, ioccup, rpoint, 1, r, rmu, &
310 dupr, zetar, rion, psip, buffer_winv, nelorb, nion, kion, &
311 iflagnorm, cnorm, lbox, rmucos, rmusin, minimum_distance, &
312 indpar_tab, indorb_tab, indshell_tab, .true.)
313 end if
314
315 ! if(add_onebody2det) then
316 psilg = -scale_one_body
317 do jj = 1, nion
318 if (iespbc) then
319 rc(:) = rpoint(:) - rion(:, jj)
320 call cartesiantocrystal(rc, 1)
321 do kk = 1, 3
322 rc(kk) = costz(jj)*map(rc(kk), cellscale(kk))
323 end do
324 r0 = norm_metric(rc, metric)
325 else
326 rc(:) = (rpoint(:) - rion(:, jj))*costz(jj)
327 r0 = dsqrt(sum(rc(:)**2))
328 end if
329 psilg = psilg - jastrow_ei(r0, vj(pointvj(1, jj)), pointvj(2, jj))*costz3(jj)
330 end do
331
332 buffer_winv(1:ipc*nelorb) = buffer_winv(1:ipc*nelorb)*dexp(psilg)
333
334 ! endif
335
336 if (ipc .eq. 1) then
337 psi(1) = ddot(nelorbh, buffer_winv, 1, mu_c(imu, imap_loc(j)), 1)
338 else
339 psi_c = zdotu(nelorbh, buffer_winv, 1, mu_c(2*imu - 1, imap_loc(j)), 1)
340 psi(1) = real(psi_c)
341 psi(2) = aimag(psi_c)
342 end if
343
344 if (nochsp) then
345 datagrid(ipc*(i - 1) + 1:ipc*i, ii, iii) = psi(:)
346 elseif (chargeon) then
347 datagrid(i, ii, iii) = datagrid(i, ii, iii) + norm(j)*sum(psi(:)**2)
348 elseif (spinon) then
349 if (mod(j, 2) .eq. 0 .or. j .gt. 2*nfil) then
350 datagrid(i, ii, iii) = datagrid(i, ii, iii) + norm(j)*sum(psi(:)**2)
351 else
352 datagrid(i, ii, iii) = datagrid(i, ii, iii) - norm(j)*sum(psi(:)**2)
353 end if
354 end if
355 end do
356 end do
357 end do
358
359 if (spinon) then
360 if (iespbc) then
361 totmag = sum(abs(datagrid(1:mesh(1) - 1, 1:mesh(2) - 1, 1:mesh(3) - 1)))*vol
362 else
363 totmag = sum(abs(datagrid(:, :, :)))*vol
364 end if
365 write (6, *) ' Mag up to orbital =', ind_mol, imap_loc(j), totmag
366 else
367 if (ipf .eq. 1) then
368 write (6, *) ' Molecular =', j, ind_mol
369 else
370 if (up_down .eq. 1) then
371 write (6, *) ' Up pfaffian Molecular =', j, ind_mol
372 else
373 write (6, *) ' Down pfaffian Molecular =', j, ind_mol
374 end if
375 end if
376 end if
377 !
378 if (nochsp) then
379 if (ipf .eq. 1) then
380 datagrid = datagrid*norm(ind_mol)
381 call plot_3d_data(ipc, cell_loc, cellscale, nion, rion &
382 , atom_number, iespbc, mesh, origin, datagrid, j, 'orbital ')
383 datagrid = datagrid**2
384 call plot_3d_data(ipc, cell_loc, cellscale, nion, rion &
385 , atom_number, iespbc, mesh, origin, datagrid, j, 'orbsqrd ')
386 else
387 if (up_down .eq. 1) then
388 datagrid = datagrid*norm(ind_mol)
389 call plot_3d_data(ipc, cell_loc, cellscale, nion, rion &
390 , atom_number, iespbc, mesh, origin, datagrid, j, 'orbital_up ')
391 datagrid_prod = datagrid
392 datagrid = datagrid**2
393 datagrid_sum = datagrid
394 call plot_3d_data(ipc, cell_loc, cellscale, nion, rion &
395 , atom_number, iespbc, mesh, origin, datagrid, j, 'orbsqrd_up ')
396 else
397 datagrid = datagrid*norm(ind_mol)
398 datagrid_prod = datagrid_prod*datagrid
399 call plot_3d_data(ipc, cell_loc, cellscale, nion, rion &
400 , atom_number, iespbc, mesh, origin, datagrid, j, 'orbital_down ')
401 call plot_3d_data(ipc, cell_loc, cellscale, nion, rion &
402 , atom_number, iespbc, mesh, origin, datagrid_prod, j, 'prod_updown ')
403 datagrid = datagrid**2
404 datagrid_sum = datagrid_sum + datagrid
405 call plot_3d_data(ipc, cell_loc, cellscale, nion, rion &
406 , atom_number, iespbc, mesh, origin, datagrid, j, 'orbsqrd_down ')
407 call plot_3d_data(ipc, cell_loc, cellscale, nion, rion &
408 , atom_number, iespbc, mesh, origin, datagrid_sum, j, 'orbsqrd_sum ')
409 datagrid_sum = datagrid_sum - 2.0*datagrid
410 call plot_3d_data(ipc, cell_loc, cellscale, nion, rion &
411 , atom_number, iespbc, mesh, origin, datagrid_sum, j, 'orbsqrd_diff ')
412 end if
413 end if
414
415 elseif (j .eq. upper) then
416 if (spinon) then
417 if (iespbc) then
418 totmag = sum(abs(datagrid(1:mesh(1) - 1, 1:mesh(2) - 1, 1:mesh(3) - 1)))*vol
419 else
420 totmag = sum(abs(datagrid(:, :, :)))*vol
421 end if
422 totmagc = (0.d0, 0.d0)
423 do iii = 1, mesh(3) - 1
424 do ii = 1, mesh(2) - 1
425 do i = 1, mesh(1) - 1
426 rpoint(1) = (i - 1)*step(1) - origin(1)
427 rpoint(2) = (ii - 1)*step(2) - origin(2)
428 rpoint(3) = (iii - 1)*step(3) - origin(3)
429 sumval = sum(kspin(1:3)*rpoint(1:3))
430 totmagc = totmagc + exp(dcmplx(0.d0, sumval))*datagrid(i, ii, iii)*vol
431 end do
432 end do
433 end do
434
435 write (6, *) ' Total magnetization in the cell (Bohr)', totmag
436 write (6, *) ' Total square root structure factor at K (Bohr)', abs(totmagc)
437
438 elseif (chargeon) then
439 if (iespbc) then
440 totmag = sum(datagrid(1:mesh(1) - 1, 1:mesh(2) - 1, 1:mesh(3) - 1))*vol
441 else
442 totmag = sum(datagrid(:, :, :))*vol
443 end if
444 write (6, *) ' Total charge in the cell ', totmag
445 end if
446 if (chargeon) then
447 if (ipf .eq. 1) then
448 call plot_3d_data(1, cell_loc, cellscale, nion, rion &
449 , atom_number, iespbc, mesh, origin, datagrid, 0, 'charge ')
450 else
451 if (up_down .eq. 1) then
452 call plot_3d_data(1, cell_loc, cellscale, nion, rion &
453 , atom_number, iespbc, mesh, origin, datagrid, 0, 'charge_up ')
454 else
455 call plot_3d_data(1, cell_loc, cellscale, nion, rion &
456 , atom_number, iespbc, mesh, origin, datagrid, 0, 'charge_down ')
457 end if
458 end if
459 elseif (spinon) then
460 ! datagrid=datagrid/2.d0 Bohr magn. units.
461 if (ipf .eq. 1) then
462 call plot_3d_data(1, cell_loc, cellscale, nion, rion &
463 , atom_number, iespbc, mesh, origin, datagrid, 0, 'spin ')
464 else
465 if (up_down .eq. 1) then
466 call plot_3d_data(1, cell_loc, cellscale, nion, rion &
467 , atom_number, iespbc, mesh, origin, datagrid, 0, 'spin_up ')
468 else
469 call plot_3d_data(1, cell_loc, cellscale, nion, rion &
470 , atom_number, iespbc, mesh, origin, datagrid, 0, 'spin_down ')
471 end if
472 end if
473 end if !
474 end if ! endif nocshp
475 end if ! endif control
476 end do
477 !
478 !
479 end do
480
481 call deallocate_all
482 close (ufort10)
483
484 stop

486end program plot_orbitals
