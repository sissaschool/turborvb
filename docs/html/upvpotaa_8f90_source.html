16!> @brief Calculate ion-ion potential energy with Ewald summation
17!>
18!> This function computes the ion-ion electrostatic potential energy
19!> using either direct Coulomb interaction (for isolated systems) or
20!> Ewald summation (for periodic boundary conditions). The calculation
21!> includes both the real-space and self-interaction contributions.
22!>
23!> Parameters
24!> ----------
25!> ZETA : real*8 array, in
26!> Array of ion charges (NION).
27!> IOND : real*8 array, in
28!> Matrix of ion-ion distances (NION × NION).
29!> NION : integer, in
30!> Number of ions in the system.
31!> KAPPA : real*8, in
32!> Ewald parameter for splitting the Coulomb interaction.
33!> LBOX : real*8, in
34!> Box length for periodic boundary conditions.
35!> If LBOX ≤ 0: isolated system (direct Coulomb).
36!> If LBOX > 0: periodic system (Ewald summation).
37!>
38!> Returns
39!> -------
40!> real*8
41!> Total ion-ion potential energy including self-interaction correction.
42!>
43!> Notes
44!> -----
45!> - For isolated systems (LBOX ≤ 0): uses direct Coulomb interaction.
46!> - For periodic systems (LBOX > 0): uses Ewald summation with:
47!> * Real-space contribution using complementary error function.
48!> * Self-interaction correction to avoid double-counting.
49!> - Includes nearest-neighbor interactions for periodic systems.
50!> - Uses SIMD vectorization for performance optimization.
51!> - The self-interaction term is: -2*κ*ε₀/√π * Σ(zeta²).
52!>
53!> Algorithm
54!> ---------
55!> For isolated systems:
56!> E = Σᵢⱼ 2*zeta(i)*zeta(j)*veps(iond(i,j))
57!>
58!> For periodic systems:
59!> E_real = Σᵢⱼ 2*zeta(i)*zeta(j)*erfc(κ*rᵢⱼ)/rᵢⱼ
60!> E_self = -2*κ*ε₀/√π * Σ(zeta²)
61!> E_total = E_real + E_self
62!>
63!> Example
64!> -------
65!> Used in quantum Monte Carlo calculations for ion-ion interactions.
66function upvpotaa(zeta, iond, nion, kappa, LBox)
67 use dielectric
68 use allio, only: iond_cart, x_neigh, neigh, dist_shift, rank
69 implicit none
70 integer nel, i, j, ii, jj, nion
71 real(8) :: derfc, x_shift(3), cost_z
72 real(8) zeta(nion), iond(nion, nion), kappa, LBox, pot_aa, upvpotaa, eself1b
73 double precision, parameter :: PI = 3.14159265358979323846d0
74
75!> @brief Initialize potential energy
76 pot_aa = 0.d0
77
78 if (lbox .le. 0.d0) then
79!> @brief Isolated system: direct Coulomb interaction
80 do i = 1, nion
81 do j = i + 1, nion
82 if (zeta(i)*zeta(j) .ne. 0.d0) &
83 &pot_aa = pot_aa + 2.d0*zeta(i)*zeta(j)*veps(iond(i, j))
84 end do
85 end do
86!> @brief No self-interaction correction for isolated systems
87 eself1b = 0.d0
88 else
89!> @brief Periodic system: Ewald summation
90 do i = 1, nion
91 do j = i + 1, nion
92 cost_z = 2.d0*zeta(i)*zeta(j)
93 jj = nion*(j - 1) + i
94 if (cost_z .ne. 0.d0) then
95!> @brief Calculate real-space contribution with nearest neighbors
96#ifdef _SIMD
97!$omp simd
98#endif
99 do ii = 1, neigh
100 dist_shift(ii) = dsqrt((iond_cart(1, jj)&
101 & + x_neigh(ii, 1))**2 + (iond_cart(2, jj)&
102 & + x_neigh(ii, 2))**2 + (iond_cart(3, jj)&
103 & + x_neigh(ii, 3))**2)
104 dist_shift(ii) = cost_z*rep_erfc(dist_shift(ii), kappa)
105 end do
106 pot_aa = pot_aa + sum(dist_shift(1:neigh))
107 end if
108 end do
109 end do
110!> @brief Self-interaction correction for periodic systems
111 eself1b = -2*kappa*epsilon0/dsqrt(pi)*sum(zeta(1:nion)**2)
112
113 end if
114!> @brief Return total potential energy including self-interaction
115 upvpotaa = pot_aa + eself1b
116
117 return
118end
