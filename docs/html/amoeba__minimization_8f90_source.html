133 !> @brief Constructs initial simplex for amoeba minimization
134 !> @details This subroutine initializes the simplex for amoeba
135 !> minimization by setting up the initial parameter values
136 !> and computing the cost function for each vertex.
137 !> @param[in] delta Step size for simplex construction
138 !> @param[in] vecsigma Sigma vector for cost function
139 !> @param[in] weights Weight factors for data points
140 subroutine simplex_constructor(delta, vecsigma, weights)
141
142! to be called at the beginning to initialize the simplex correctly!!!!!
143 real(8) :: vecsigma(*), weights(*), delta, sigma, norm
144 integer :: i, j
145
146! define vecsigma
147! pin the value at the origin
148! vecsigma(1)=1.d0/abs(points(1,1)-points(1,2))/dble(points_number)/1000.d0
149! do i=2,points_number-1
150! vecsigma(i)=1.d0/abs(points(1,i)-points(1,i+1))/dble(points_number)
151! enddo
152! vecsigma(points_number)=vecsigma(points_number-1)
153
154! norm=1.d0
155! do i=1,points_number
156! norm=norm+vecsigma(i)
157! enddo
158! norm=norm*abs(points(1,points_number)-points(1,1))/dble(points_number)**2
159! do i=1,points_number
160! vecsigma(i)=vecsigma(i)*sigma/norm
161! enddo
162
163!use same matrics as in the diagonalization algorithm for consistency
164! dx(i)=1/vecsigma(i)**2
165! vecsigma(i)=1.d0/sqrt(dx(i))
166
167! old integration
168! vecsigma(1)=(points(1,2)-points(1,1))/2.d0+points(1,1)
169! vecsigma(points_number)=(points(1,points_number)-points(1,points_number-1))/2.d0
170! do i=2,points_number-1
171! vecsigma(i)=0.5d0*(points(1,i+1)-points(1,i-1))
172! enddo
173
174 do i = 1, points_number
175 vecsigma(i) = points(1, i)
176 end do
177 do i = 1, points_number
178! normalize vecsigma --> dx / Ltot
179 vecsigma(i) = vecsigma(i)*weights(i)/abs(points(1, points_number) - points(1, 1))
180 vecsigma(i) = 1.d0/sqrt(vecsigma(i))
181 end do
182
183! define the simplex
184 j = 0
185 do i = 1, param_number
186 if (sel(i)) j = j + 1
187 end do
188 simplex_dim = j
189
190 allocate (y(simplex_dim + 1), p(simplex_dim + 1, simplex_dim))
191 allocate (x(simplex_dim))
192 allocate (xscratch(param_number))
193 allocate (vecscratch(param_number/3))
194
195 j = 0
196 do i = 1, param_number
197 if (sel(i)) then
198 j = j + 1
199 p(1, j) = param0(i)
200 x(j) = p(1, j)
201 end if
202 end do
203 y(1) = chi(x, vecsigma)
204
205 do i = 1, simplex_dim
206 do j = 1, simplex_dim
207 if (j .ne. i) then
208 p(i + 1, j) = p(1, j)
209 else
210 p(i + 1, j) = p(1, j) + delta
211 end if
212 x(j) = p(i + 1, j)
213 end do
214 y(i + 1) = chi(x, vecsigma)
215 end do
216
217 write (6, *) 'initial chi square', y(1)
218
219 end subroutine simplex_constructor

221 !> @brief Computes chi-square cost function for minimization
222 !> @details This function computes the chi-square cost function
223 !> that measures the distance between fitted form and
224 !> data points for amoeba minimization.
225 !> @param[in] x Parameter array for evaluation
226 !> @param[in] vecsigma Sigma vector for normalization
227 !> @return chi Chi-square cost function value
228 function chi(x, vecsigma)
229
230! chi: cost function for minimization
231! chi "measures" the distance of the fitted form with respect to the datapoints
232! vecsigma: control parameter to tune the distance
233 real(8) :: chi
234 real(8) :: x(*)
235 real(8) :: vecsigma(*)
236 integer :: i, j
237
238 do i = 1, param_number
239 xscratch(i) = param0(i)
240 end do
241
242 j = 0
243 do i = 1, param_number
244 if (sel(i)) then
245 j = j + 1
246 xscratch(i) = x(j)
247 end if
248 end do
249
250 chi = 0.d0
251 do i = 1, points_number
252 chi = chi + ((pseudofun(param_number/3, points(1, i), xscratch, vecscratch) - points(2, i))/vecsigma(i))**2
253 end do
254! chi=chi/(points_number-simplex_dim)
255
256 end function chi

258 !> @brief Performs amoeba simplex minimization
259 !> @details This subroutine implements the Nelder-Mead simplex
260 !> algorithm for function minimization. It performs
261 !> reflection, expansion, contraction, and reduction
262 !> operations to find the minimum of the cost function.
263 !> @param[in,out] p Simplex parameter array
264 !> @param[in,out] y Function values at simplex vertices
265 !> @param[in] tolerance Convergence tolerance
266 !> @param[in] itmax Maximum number of iterations
267 subroutine amoeba(p, y, tolerance, itmax)
268! kernel of the minimization module
269
270 real(8) :: p(simplex_dim + 1, *), y(*)
271 integer :: itmax
272 real(8) :: tolerance, ftol
273
274 real(8), parameter :: alpha = 1.0
275 real(8), parameter :: beta = 0.5
276 real(8), parameter :: gamma = 2.0
277
278 integer :: mpts, ndim, mp, np
279 integer :: ilo, ihi, inhi, i, j
280
281 real(8), dimension(:), allocatable :: pr, prr, pbar
282 real(8) :: ypr, yprr, rtol
283
284 ndim = simplex_dim
285 np = ndim
286 mp = np + 1
287 mpts = np + 1 !mpts logical dimension of the simplex
288
289 allocate (pr(ndim), prr(ndim), pbar(ndim))
290
291! real(8) :: p(mp,np),y(mp)
292
293 ftol = tolerance
294 iter = 1
295
296 do while (iter .le. itmax)
297 ilo = 1
298 if (y(1) .gt. y(2)) then
299 ihi = 1
300 inhi = 2
301 else
302 ihi = 2
303 inhi = 1
304 end if
305 do i = 1, mpts
306 if (y(i) .lt. y(ilo)) ilo = i
307 if (y(i) .gt. y(ihi)) then
308 inhi = ihi
309 ihi = i
310 elseif (y(i) .gt. y(inhi)) then
311 if (i .ne. ihi) inhi = i
312 end if
313 end do
314
315 rtol = 2.d0*abs(y(ihi) - y(ilo))/(abs(y(ihi)) + abs(y(ilo)))
316 if (rtol .le. ftol) go to 10
317 do j = 1, ndim
318 pbar(j) = 0.d0
319 end do
320 do i = 1, mpts
321 if (i .ne. ihi) then
322 do j = 1, ndim
323 pbar(j) = pbar(j) + p(i, j)
324 end do
325 end if
326 end do
327 do j = 1, ndim
328 pbar(j) = pbar(j)/dble(ndim)
329 pr(j) = (1 + alpha)*pbar(j) - alpha*p(ihi, j)
330 end do
331 ypr = chi(pr, vecsigma)
332 if (ypr .le. y(ilo)) then
333 do j = 1, ndim
334 prr(j) = gamma*pr(j) + (1.d0 - gamma)*pbar(j)
335 end do
336 yprr = chi(prr, vecsigma)
337 if (yprr .lt. y(ilo)) then
338 do j = 1, ndim
339 p(ihi, j) = prr(j)
340 end do
341 y(ihi) = yprr
342 else
343 do j = 1, ndim
344 p(ihi, j) = pr(j)
345 end do
346 y(ihi) = ypr
347 end if
348 elseif (ypr .ge. y(inhi)) then
349 if (ypr .lt. y(ihi)) then
350 do j = 1, ndim
351 p(ihi, j) = pr(j)
352 end do
353 y(ihi) = ypr
354 end if
355 do j = 1, ndim
356 prr(j) = beta*p(ihi, j) + (1.d0 - beta)*pbar(j)
357 end do
358 yprr = chi(prr, vecsigma)
359 if (yprr .lt. y(ihi)) then
360 do j = 1, ndim
361 p(ihi, j) = prr(j)
362 end do
363 y(ihi) = yprr
364 else
365 do i = 1, mpts
366 if (i .ne. ilo) then
367 do j = 1, ndim
368 pr(j) = 0.5*(p(i, j) + p(ilo, j))
369 p(i, j) = pr(j)
370 end do
371 y(i) = chi(pr, vecsigma)
372 end if
373 end do
374 end if
375 else
376 do j = 1, ndim
377 p(ihi, j) = pr(j)
378 end do
379 y(ihi) = ypr
380 end if
381! write(*,*) iter
382! do j=1,ndim+1
383! write(*,*) y(j),(p(j,i),i=1,ndim)
384! enddo
385 iter = iter + 1
386 end do
387
388 write (*, *) 'WARNING: amoeba exceeding maximum iterations'
389
39010 deallocate (pr, prr, pbar)
391
392 return
393
394 end subroutine amoeba
