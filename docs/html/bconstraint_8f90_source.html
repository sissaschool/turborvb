17!> @brief Module containing constraint handling for wave function optimization
18!> @author TurboRVB group

22!> @brief Translates changes back to the original unconstrained representation
23!> @details This subroutine handles the transformation of parameter changes from a
24!> constrained optimization space back to the original unconstrained
25!> representation used in the wave function. It supports both real and
26!> complex wave functions, as well as different types of constraints.
27!>
28!> The subroutine performs the following operations:
29!> 1. Maps constrained parameter changes to their original positions
30!> 2. Handles symmetry constraints for AGP (Antisymmetrized Geminal Power) matrices
31!> 3. Updates either the derivative matrix or the parameter matrix based on yes_update flag
32!> 4. Supports both real (ipc=1) and complex (ipc=2) wave functions
33!> 5. Handles EAGP (Extended AGP) Pfaffian matrices for complex wave functions
34!>
35!> The algorithm uses a mapping array jbradet to translate between constrained
36!> and unconstrained parameter spaces, where positive values indicate direct
37!> mapping and negative values indicate sign-flipped mapping.
38!>
39!> @param[in] iessw Switch flag for constraint handling (0: no constraints, >0: apply constraints)
40!> @param[in,out] derl Derivative matrix or parameter matrix to be updated
41!> @param[in] nelorb Number of orbitals
42!> @param[in] n Number of constrained parameters
43!> @param[in] nozero Array of non-zero element indices
44!> @param[in,out] psip Temporary workspace array
45!> @param[in] econf Configuration energy array
46!> @param[in] nw Number of walkers
47!> @param[in] jbradet Mapping array from constrained to unconstrained parameters
48!> @param[in] symmagp Logical flag for symmetric AGP matrix
49!> @param[in] yes_update Logical flag to determine update mode
50!>
51!> @note If iessw = 0, the subroutine returns immediately without any operations
52!>
53!> @warning The subroutine modifies the derl array in-place based on the constraint mapping
54!>
55!> @par Algorithm Details:
56!> The subroutine implements a two-step process:
57!> 1. <b>Parameter Mapping:</b> Uses jbradet array to map constrained parameters back to original positions
58!> 2. <b>Matrix Update:</b> Updates either derivative matrix (upsim/upsimp) or parameter matrix based on yes_update
59!>
60!> @par Real vs Complex Wave Functions:
61!> - <b>Real (ipc=1):</b> Direct parameter mapping with sign handling
62!> - <b>Complex (ipc=2):</b> Complex parameter mapping with real/imaginary part handling
63!>
64!> @par EAGP Support:
65!> For complex wave functions, the subroutine also handles EAGP Pfaffian matrices,
66!> which require special antisymmetric structure maintenance.
67!>
68!> @see constants module for ipc, ipf, deps parameters
69!> @see cell module for cellscale, phase2pi parameters
70!> @see allio module for rank, yes_correct, nnozero_eagp, eagp_pfaff, ndiff parameters
71subroutine bconstraint(iessw, derl, nelorb, n, nozero &
72 &, psip, econf, nw, jbradet, symmagp, yes_update)
73 use constants, only: ipc, ipf, deps
74 use cell, only: cellscale, phase2pi
75 use allio, only: rank, yes_correct, nnozero_eagp, eagp_pfaff, ndiff
76 implicit none
77
78 !> @param[in] iessw Switch flag for constraint handling (0: no constraints, >0: apply constraints)
79 integer, intent(in) :: iessw
80
81 !> @param[in] nozero Array of non-zero element indices
82 integer, intent(in) :: nozero(*)
83
84 !> @param[in] nelorb Number of orbitals
85 integer, intent(in) :: nelorb
86
87 !> @param[in] n Number of constrained parameters
88 integer, intent(in) :: n
89
90 !> @param[in] nw Number of walkers
91 integer, intent(in) :: nw
92
93 !> @param[in] jbradet Mapping array from constrained to unconstrained parameters
94 integer, intent(in) :: jbradet(*)
95
96 !> @param[in] symmagp Logical flag for symmetric AGP matrix
97 logical, intent(in) :: symmagp
98
99 !> @param[in] yes_update Logical flag to determine update mode
100 logical, intent(in) :: yes_update
101
102 !> @param[in] econf Configuration energy array
103 real*8, intent(in) :: econf(nw, *)
104
105 !> @param[in,out] psip Temporary workspace array
106 real*8, intent(inout) :: psip(*)
107
108 !> @param[in,out] derl Derivative matrix or parameter matrix to be updated
109 real*8, intent(inout) :: derl(*)
110
111 ! Local variables
112 integer :: i, ix, iy, j, ind, iesswread
113
114 ! Early return if no constraints are applied
115 if (iessw .eq. 0) return
116
117 ! Handle real wave functions (ipc = 1)
118 if (ipc .eq. 1) then
119 ! Initialize workspace array to zero
120 call dscalzero(n + nnozero_eagp, 0.d0, psip, 1)
121
122 ! Map constrained parameters back to original positions
123 ! Positive jbradet values: direct mapping
124 ! Negative jbradet values: sign-flipped mapping
125 do i = 1, n + nnozero_eagp
126 j = jbradet(i)
127 if (j .gt. 0) then
128 psip(i) = econf(1, j)
129 elseif (j .lt. 0) then
130 psip(i) = -econf(1, -j)
131 end if
132 end do
133
134 ! Update mode: add changes to existing matrix
135 if (yes_update) then
136 ! Update determinant matrix elements
137 do i = 1, n
138 if (psip(i) .ne. 0.d0) &
139 & call upsimp(derl, nelorb, nozero(i), psip(i), symmagp, ipf)
140 end do
141
142 ! Update EAGP Pfaffian matrix elements (antisymmetric structure)
143 do i = 1, nnozero_eagp
144 ind = i + n
145 ! Convert linear index to 2D matrix indices
146 iy = (nozero(ind) - 1)/ndiff + 1
147 ix = nozero(ind) - (iy - 1)*ndiff
148 j = jbradet(ind)
149 if (j .gt. 0) then
150 eagp_pfaff(ix, iy) = eagp_pfaff(ix, iy) + psip(ind)
151 eagp_pfaff(iy, ix) = -eagp_pfaff(ix, iy) ! Antisymmetric
152 elseif (j .lt. 0) then
153 eagp_pfaff(ix, iy) = eagp_pfaff(ix, iy) - psip(ind)
154 eagp_pfaff(iy, ix) = -eagp_pfaff(ix, iy) ! Antisymmetric
155 end if
156 end do
157
158 ! Set mode: replace matrix elements
159 else
160 ! Set determinant matrix elements
161 do i = 1, n
162 if (abs(jbradet(i)) .ne. 0) &
163 & call upsim(derl, nelorb, nozero(i), psip(i), symmagp, ipf)
164 end do
165
166 ! Set EAGP Pfaffian matrix elements
167 do i = 1, nnozero_eagp
168 ind = i + n
169 iy = (nozero(ind) - 1)/ndiff + 1
170 ix = nozero(ind) - (iy - 1)*ndiff
171 j = jbradet(ind)
172 if (j .gt. 0) then
173 eagp_pfaff(ix, iy) = psip(ind)
174 eagp_pfaff(iy, ix) = -psip(ind) ! Antisymmetric
175 elseif (j .lt. 0) then
176 eagp_pfaff(ix, iy) = -psip(ind)
177 eagp_pfaff(iy, ix) = psip(ind) ! Antisymmetric
178 end if
179 end do
180 end if
181
182 ! Handle complex wave functions (ipc = 2)
183 else
184 ! Initialize workspace array to zero (complex: 2x larger)
185 call dscalzero(2*n + 2*nnozero_eagp, 0.d0, psip, 1)
186
187 ! Handle symmetric AGP with correction
188 if (symmagp .and. yes_correct) then
189 ! Map complex parameters with symmetry constraints
190 do i = 1, n + nnozero_eagp
191 j = jbradet(i)
192 if (j .gt. 0) then
193 ! Real and imaginary parts for symmetric case
194 psip(2*i - 1) = econf(1, 4*j - 3) ! Real part
195 psip(2*i) = econf(1, 4*j - 1) ! Imaginary part
196 elseif (j .lt. 0) then
197 psip(2*i - 1) = -econf(1, -4*j - 3) ! Sign-flipped real part
198 psip(2*i) = -econf(1, -4*j - 1) ! Sign-flipped imaginary part
199 end if
200 end do
201
202 ! Standard complex parameter mapping
203 else
204 do i = 1, n + nnozero_eagp
205 j = jbradet(i)
206 if (j .gt. 0) then
207 psip(2*i - 1) = econf(1, 2*j - 1) ! Real part
208 psip(2*i) = econf(1, 2*j) ! Imaginary part
209 elseif (j .lt. 0) then
210 psip(2*i - 1) = -econf(1, -2*j - 1) ! Sign-flipped real part
211 psip(2*i) = -econf(1, -2*j) ! Sign-flipped imaginary part
212 end if
213 end do
214 end if
215
216 ! Update mode for complex wave functions
217 if (yes_update) then
218 ! Update complex determinant matrix elements
219 do i = 1, n
220 if (sum(abs(psip(2*i - 1:2*i))) .ne. 0.d0) &
221 & call upsimp_complex(derl, nelorb, nozero(i), psip(2*i - 1), symmagp, ipf)
222 end do
223
224 ! Update complex EAGP Pfaffian matrix elements
225 do i = 1, nnozero_eagp
226 ind = i + n
227 iy = (nozero(ind) - 1)/ndiff + 1
228 ix = nozero(ind) - (iy - 1)*ndiff
229 j = jbradet(ind)
230 if (j .gt. 0) then
231 ! Add complex values maintaining antisymmetry
232 eagp_pfaff(2*ix - 1, iy) = eagp_pfaff(2*ix - 1, iy) + psip(2*ind - 1)
233 eagp_pfaff(2*ix, iy) = eagp_pfaff(2*ix, iy) + psip(2*ind)
234 eagp_pfaff(2*iy - 1, ix) = -eagp_pfaff(2*ix - 1, iy) ! Antisymmetric
235 eagp_pfaff(2*iy, ix) = -eagp_pfaff(2*ix, iy) ! Antisymmetric
236 elseif (j .lt. 0) then
237 ! Subtract complex values maintaining antisymmetry
238 eagp_pfaff(2*ix - 1, iy) = eagp_pfaff(2*ix - 1, iy) - psip(2*ind - 1)
239 eagp_pfaff(2*ix, iy) = eagp_pfaff(2*ix, iy) - psip(2*ind)
240 eagp_pfaff(2*iy - 1, ix) = -eagp_pfaff(2*ix - 1, iy) ! Antisymmetric
241 eagp_pfaff(2*iy, ix) = -eagp_pfaff(2*ix, iy) ! Antisymmetric
242 end if
243 end do
244
245 ! Set mode for complex wave functions
246 else
247 ! Set complex determinant matrix elements
248 do i = 1, n
249 if (abs(jbradet(i)) .ne. 0)&
250 & call upsim_complex(derl, nelorb, nozero(i), psip(2*i - 1), symmagp, ipf)
251 end do
252
253 ! Set complex EAGP Pfaffian matrix elements
254 do i = 1, nnozero_eagp
255 ind = i + n
256 iy = (nozero(ind) - 1)/ndiff + 1
257 ix = nozero(ind) - (iy - 1)*ndiff
258 j = jbradet(ind)
259 if (j .gt. 0) then
260 ! Set complex values maintaining antisymmetry
261 eagp_pfaff(2*ix - 1, iy) = psip(2*ind - 1)
262 eagp_pfaff(2*ix, iy) = psip(2*ind)
263 eagp_pfaff(2*iy - 1, ix) = -eagp_pfaff(2*ix - 1, iy) ! Antisymmetric
264 eagp_pfaff(2*iy, ix) = -eagp_pfaff(2*ix, iy) ! Antisymmetric
265 elseif (j .lt. 0) then
266 ! Set sign-flipped complex values maintaining antisymmetry
267 eagp_pfaff(2*ix - 1, iy) = -psip(2*ind - 1)
268 eagp_pfaff(2*ix, iy) = -psip(2*ind)
269 eagp_pfaff(2*iy - 1, ix) = -eagp_pfaff(2*ix - 1, iy) ! Antisymmetric
270 eagp_pfaff(2*iy, ix) = -eagp_pfaff(2*ix, iy) ! Antisymmetric
271 end if
272 end do
273 end if ! endif yesupdate
274
275 end if ! endif ipc=1
276
277 return
278end
