16/**
17 * @brief mpi-io module for parallel file operations
18 *
19 * this module provides a high-level interface for parallel file i/o operations
20 * using mpi-io. it includes a file object type and subroutines for opening,
21 * reading, writing, and closing files in parallel environments.
22 *
23 * @details
24 * the module provides the following functionality:
25 * - file object type with mpi-io metadata
26 * - file opening with specified access modes
27 * - file view creation for parallel data distribution
28 * - file pointer management and positioning
29 * - file closing and cleanup
30 * - error handling for mpi-io operations
31 *
32 * key features:
33 * - parallel file access with collective operations
34 * - custom file views for data distribution
35 * - automatic error handling and abort on failure
36 * - support for both serial and parallel builds
37 * - file pointer tracking and management
38 *
39 * @note
40 * - only available when parallel is defined
41 * - serial version provides dummy type for compatibility
42 * - all operations include error checking
43 * - file views are automatically committed and freed
44 *
45 * @author turborvb group
46 * @date 2022
47 */
48module mpiio
49 implicit none
50
51#ifdef PARALLEL
52
53 include 'mpif.h'

55 /**
56 * @brief file object type for mpi-io operations
57 *
58 * this derived type encapsulates all the information needed for
59 * mpi-io file operations, including file handle, communicator,
60 * and view information.
61 *
62 * @details
63 * the type contains:
64 * - name: file name string
65 * - fp: mpi file pointer
66 * - comm: mpi communicator for collective operations
67 * - nproc, rank: process count and rank information
68 * - view: mpi datatype for file view
69 * - array_of_blocklengths, array_of_displacements, array_of_types: view parameters
70 * - etype: elementary datatype
71 * - disp: file displacement for positioning
72 *
73 * @note
74 * - used by all mpi-io subroutines in this module
75 * - automatically manages mpi datatype lifecycle
76 * - supports custom file views for parallel access patterns
77 */
78 type file_obj
79 ! file name
80 character(len=200) name
81 ! file pointer
82 integer fp
83 ! communicator
84 integer comm
85 ! mpi setup
86 integer nproc, rank
87 ! file view
88 integer view
89 ! view setup
90 integer array_of_blocklengths(3)
91 integer(kind=MPI_ADDRESS_KIND) array_of_displacements(3)
92 integer array_of_types(3), etype
93 ! view upper lower boundary
94 !integer(kind=MPI_ADDRESS_KIND) lb, ub
95 ! first displacement
96 integer(kind=MPI_OFFSET_KIND) disp
97 end type file_obj
98
99contains
100 /**
101 * @brief Open a file for parallel i/o operations
102 *
103 * this subroutine opens a file for parallel i/o using mpi-io.
104 * it initializes the file object with the necessary metadata
105 * and opens the file with the specified access mode.
106 *
107 * @param[in] input_comm mpi communicator for collective operations
108 * @param[in] filename name of the file to open
109 * @param[in] amode access mode(e.g., mpi_mode_rdonly, mpi_mode_wronly, mpi_mode_create)
110 * @param[out] myfile file object to be initialized
111 *
112 * @details
113 * the subroutine:
114 * 1. stores the communicator and file name
115 * 2. gets process count and rank information
116 * 3. opens the file with mpi_file_open
117 * 4. initializes the view to mpi_datatype_null
118 * 5. reports success or failure
119 *
120 * @note
121 * - collective operation(all processes must call)
122 * - aborts the program on file open failure
123 * - prints status messages from rank 0
124 * - file view must be created separately if needed
125 *
126 * @see mpiio_file_close(), mpiio_file_create_view()
127 */
128 subroutine mpiio_file_open(input_comm, filename, amode, myfile)
129 implicit none
130
131 integer, intent(in) :: input_comm, amode
132 character(len=*), intent(in) :: filename
133 type(file_obj), intent(out) :: myfile
134 integer :: ierr
135
136 myfile%comm = input_comm
137 myfile%name = trim(filename)
138 call mpi_comm_size(myfile%comm, myfile%nproc, ierr)
139 call mpi_comm_rank(myfile%comm, myfile%rank, ierr)
140
141 call mpi_file_open(myfile%comm, trim(filename), amode, mpi_info_null, myfile%fp, ierr)
142 if (ierr .ne. mpi_success) then
143 if (myfile%rank .eq. 0) write (6, *) "ERROR: Fail to open file ", trim(filename), " with MPI-IO!"
144 call mpi_abort(myfile%comm, ierr)
145 end if
146 if (myfile%rank .eq. 0) write (6, *) "Successfully open file ", trim(myfile%name), " with MPI-IO!"
147
148 myfile%view = mpi_datatype_null
149 end subroutine mpiio_file_open
150
151 /**
152 * @brief get current file pointer position
153 *
154 * this subroutine retrieves the current position of the file pointer
155 * and stores it in the file object's disp field. This position is
156 * used by reset_file_view to restore the file view.
157 *
158 * @param[in] myfile File object containing the file handle
159 *
160 * @details
161 * The subroutine calls MPI_File_get_position to obtain the current
162 * file pointer position and stores it in myfile%disp for later use
163 * in reset_file_view.
164 *
165 * @note
166 * - Individual operation (each process can call independently)
167 * - Aborts the program on failure
168 * - Position is stored in myfile%disp
169 * - Used in conjunction with reset_file_view
170 *
171 * @see mpiio_file_reset_view()
172 */
173 subroutine mpiio_file_get_disp(myfile)
174 ! This subroutine update the file displacement myfile%disp used by reset_file_view
175 implicit none
176
177 type(file_obj), intent(in) :: myfile
178
179 integer :: ierr
180
181 call MPI_File_get_position(myfile%fp, myfile%disp, ierr)
182.ne. if (ierr MPI_SUCCESS) then
183.eq. if (myfile%rank 0) write (6, *) "ERROR: Fail to get file pointer position ", trim(myfile%name), " with MPI-IO!"
184 call MPI_Abort(myfile%comm, ierr)
185 end if
186 !write(6,*) "my disp", myfile%disp, ", rank ", myfile%rank
187 end subroutine mpiio_file_get_disp

189 /**
190 * @brief Create a custom file view for parallel data distribution
191 *
192 * This subroutine creates a custom MPI datatype for the file view
193 * that defines how data is distributed across processes. It sets up
194 * a structured view where each process has a specific region of the file.
195 *
196 * @param[in,out] myfile File object to be modified
197 * @param[in] size_of_data Size of data per process
198 * @param[in] datatype MPI datatype for the data elements
199 *
200 * @details
201 * The subroutine creates a structured file view with:
202 * - One element of the specified datatype per process
203 * - Displacement based on process rank and data size
204 * - Upper bound to define the view extent
205 * - Automatic commitment of the datatype
206 *
207 * The view structure:
208 * - Lower bound: 0
209 * - Data block: size_of_data elements starting at rank*record_size
210 * - Upper bound: nproc*record_size
211 *
212 * @note
213 * - Collective operation (all processes must call)
214 * - Aborts the program on datatype creation failure
215 * - Automatically commits the datatype
216 * - Previous view is automatically freed
217 *
218 * @see mpiio_file_reset_view(), mpiio_file_close()
219 */
220 subroutine mpiio_file_create_view(myfile, size_of_data, datatype)
221 implicit none
222
223 type(file_obj), intent(inout) :: myfile
224 integer, intent(in) :: size_of_data, datatype
225 integer :: ierr, unit_size
226
227 integer(kind=MPI_ADDRESS_KIND) record_size
228 integer(kind=MPI_OFFSET_KIND) disp_zero
229
230 disp_zero = 0
231 myfile%etype = datatype
232 call MPI_Type_size(myfile%etype, unit_size, ierr)
233 !write(6,*) "Ye here", size_of_data, unit_size
234 record_size = unit_size*size_of_data
235 !write(6,*) "Ye record_size", record_size, myfile%rank, myfile%nproc
236 myfile%array_of_blocklengths = (/1, size_of_data, 1/)
237 myfile%array_of_displacements = (/disp_zero, myfile%rank*record_size, record_size*myfile%nproc/)
238 myfile%array_of_types = (/MPI_LB, myfile%etype, MPI_UB/)
239
240 !write(6,*) "Ye array", myfile%array_of_displacements
241
242 call MPI_Type_create_struct(3, myfile%array_of_blocklengths, myfile%array_of_displacements, &
243 myfile%array_of_types, myfile%view, ierr)
244.ne. if (ierr MPI_SUCCESS) then
245 write (*, *) 'mpi_type_create_struct failed'
246 call MPI_Abort(myfile%comm, ierr)
247 end if
248
249 call MPI_Type_commit(myfile%view, ierr)
250.ne. if (ierr MPI_SUCCESS) then
251 write (*, *) 'mpi_type_myfile%commit failed'
252 call MPI_Abort(myfile%comm, ierr)
253 end if
254
255 !call MPI_Type_get_extent(myfile%view, myfile%lb, myfile%ub, ierr)
256.ne. !if (ierrMPI_SUCCESS) then
257 ! write (*,*) 'mpi_type_extent failed'
258 ! call MPI_Abort(myfile%comm, ierr)
259 !else
260 ! write(*,*) 'extent = ', myfile%lb, myfile%ub
261 !endif
262 end subroutine mpiio_file_create_view
263
264 /**
265 * @brief Set file size to zero (truncate file)
266 *
267 * This subroutine truncates the file to zero size, effectively
268 * clearing its contents. It also resets the file displacement
269 * to zero.
270 *
271 * @param[in,out] myfile File object containing the file handle
272 *
273 * @details
274 * The subroutine calls MPI_File_set_size with size zero to
275 * truncate the file and resets myfile%disp to zero.
276 *
277 * @note
278 * - Collective operation (all processes must call)
279 * - Aborts the program on failure
280 * - Useful for clearing file contents before writing
281 * - Resets file pointer position
282 *
283 * @see mpiio_file_open(), mpiio_file_reset_view()
284 */
285 subroutine mpiio_file_set_zero(myfile)
286 implicit none
287
288 type(file_obj), intent(inout) :: myfile
289 integer(kind=MPI_OFFSET_KIND) zero
290 integer :: ierr
291
292 zero = 0
293 call MPI_File_set_size(myfile%fp, zero, ierr)
294.ne. if (ierr MPI_SUCCESS) then
295 write (*, *) 'mpi_file_set_zero failed'
296 call MPI_Abort(myfile%comm, ierr)
297 end if
298 myfile%disp = zero
299 end subroutine mpiio_file_set_zero
300
301 /**
302 * @brief Reset file view to the stored displacement
303 *
304 * This subroutine sets the file view using the stored displacement
305 * in myfile%disp. It is typically called after repositioning the
306 * file pointer to restore the custom view.
307 *
308 * @param[in,out] myfile File object containing file handle and view
309 *
310 * @details
311 * The subroutine calls MPI_File_set_view with the stored displacement
312 * and the custom view datatype to restore the file view for parallel
313 * I/O operations.
314 *
315 * @note
316 * - Individual operation (each process can call independently)
317 * - Aborts the program on failure
318 * - Uses stored displacement from get_disp
319 * - Requires a valid view datatype
320 *
321 * @see mpiio_file_get_disp(), mpiio_file_create_view()
322 */
323 subroutine mpiio_file_reset_view(myfile)
324 implicit none
325
326 type(file_obj), intent(inout) :: myfile
327 integer :: ierr
328
329 call MPI_File_set_view(myfile%fp, myfile%disp, myfile%etype, myfile%view, &
330 'native', MPI_INFO_NULL, ierr)
331.ne. if (ierr MPI_SUCCESS) then
332 write (*, *) 'mpi_file_set_view failed'
333 call MPI_Abort(myfile%comm, ierr)
334 end if
335 end subroutine mpiio_file_reset_view
336
337 /**
338 * @brief Close file and cleanup MPI-IO resources
339 *
340 * This subroutine closes the file and frees all associated MPI-IO
341 * resources, including the custom view datatype.
342 *
343 * @param[in,out] myfile File object to be closed and cleaned up
344 *
345 * @details
346 * The subroutine:
347 * 1. Frees the custom view datatype
348 * 2. Closes the file
349 * 3. Reports success or failure
350 * 4. Aborts the program on failure
351 *
352 * @note
353 * - Collective operation (all processes must call)
354 * - Automatically frees the view datatype
355 * - Aborts the program on close failure
356 * - Prints status messages from rank 0
357 *
358 * @see mpiio_file_open(), mpiio_file_create_view()
359 */
360 subroutine mpiio_file_close(myfile)
361 implicit none
362
363 type(file_obj), intent(inout) :: myfile
364 integer :: ierr
365
366 call MPI_Type_free(myfile%view, ierr)
367 call MPI_File_close(myfile%fp, ierr)
368.ne. if (ierr MPI_SUCCESS) then
369 write (*, *) 'mpi_file_close failed'
370 call MPI_Abort(myfile%comm, ierr)
371 end if
372.eq. if (myfile%rank 0) write (*, *) "Successfully closed file ", trim(myfile%name), " with MPI-IO!"
373 end subroutine mpiio_file_close
