17subroutine upvpotdiag(rkel, nel, zeta, rion, iond, vpot, nion &
18 &, kappa, LBox, vpotreg, cutreg, costz, costz3,vpotsav_ee,yes_fn)
19
20 ! by E. Coccia (30/11/10)
21 use extpot, only : ext_pot, link_atom, mm_restr
22 ! by E. Coccia (4/2/11)
23 use van_der_waals, only : vdw
24 ! by E. Coccia (12/5/11)
25 use link_angle, only : mm_pot_theta, mm_pot_dihed, mm_pot_impr
26 ! by E. Coccia (7/12/11)
27 use cl_restr, only : restr_bond, restr_angle, restr_dihe, restr_dimp
28 use dielectric
29 use cell, only: x_neigh,neigh,dist_shift,distreg_shift
30 use allio, only: iond_cart,pot_aasunel
31 implicit none
32 integer nel,ii, jj, i, j, k, nion
33 integer*8 maxll,ll
34 real(8) rkel(3, *)
35 real(8) zeta(nion), vpot, rion(3, *), iond(nion, nion) &
36 &, cost, tmp, kappa, LBox, pot_ee, pot_ea, pot_aa, derfc, max, costz(*)&
37 &, distb, vpotreg(2, nel), potreg_ea, costz3(*), cutreg, Zreg2, distbreg
38 real(8), external :: ngivej
39 real(8) x_shift(3),cost_z
40 logical yes_fn
41 real(8) vpotsav_ee(nel,nel)
42
43 ! by E. Coccia (10/11/10)
44 real*8 ext_ene(nel)
45 real*8 sum_ext
46 ! by E. Coccia (12/1/11)
47 real*8 ext_ion(nion)
48 real*8 sum_ion
49 ! by E. Coccia (4/2/11)
50 real*8 sum_vdw
51
52 vpot = 0.d0
53 pot_aa = 0.d0
54 vpotreg = 0.d0
55 ! by E. Coccia
56 ext_ene = 0.d0
57 ext_ion = 0.d0
58
59 if(lbox.le.0.d0) then
60!$omp parallel do default(shared) private(k,i,Zreg2,distb) reduction(+:vpotreg)
61 do k = 1, nion
62 zreg2 = costz(k) * costz3(k)
63 do i = 1, nel
64 distb = ngivej(rion(1, k), rkel(1, i), lbox)
65 vpotreg(1, i) = vpotreg(1, i) - 2.d0 * zeta(k) *veps(distb)
66 ! pot_ea=pot_ea-2.d0*zeta(k)/distb
67 ! pot_ea=pot_ea-2.d0*zeta(k)/distb
68 ! potreg_ea=potreg_ea+Zreg2/distb-Zreg2/max(distb,cutreg)
69 vpotreg(2, i) = vpotreg(2, i) + (zreg2 - 2.d0 * zeta(k)) *veps(distb) - zreg2 * veps(max(distb, cutreg))
70 enddo
71 enddo
72
73!$omp parallel do default(shared) private(j,i,cost) reduction(+:vpotreg)
74 do i = 1, nel
75 do j = i+1, nel
76 cost =veps( ngivej(rkel(1, i), rkel(1, j), lbox))
77 ! pot_ee=pot_ee+cost
78 vpotreg(1, j) = vpotreg(1, j) + cost
79 vpotreg(2, j) = vpotreg(2, j) + cost
80 vpotreg(1, i) = vpotreg(1, i) + cost
81 vpotreg(2, i) = vpotreg(2, i) + cost
82 if(yes_fn) then
83 vpotsav_ee(i,j)=2.d0*cost
84 vpotsav_ee(j,i)=2.d0*cost
85 endif
86 enddo
87 enddo
88
89!$omp parallel do default(shared) private(i,j) reduction(+:pot_aa)
90 do i = 1, nion
91 do j = i + 1, nion
92 if(zeta(i) * zeta(j).ne.0.d0) pot_aa = pot_aa + 2.d0 * zeta(i) * zeta(j) * veps(iond(i, j))
93 enddo
94 enddo
95
96 ! ******** EWALD REAL PART *************
97 else
98
99 ! write(6,*) cutreg
100 ! write(6,*) costz(1:nion)
101 ! write(6,*) costz3(1:nion)
102 ! write(6,*) zeta(1:nion)
103
104!$omp parallel do default(shared) private(k,i,ii,Zreg2,x_shift,cost,dist_shift,distreg_shift) reduction(+:vpotreg)
105 do k = 1, nion
106 zreg2 = costz(k) * costz3(k) ! 2 Z_singular
107 if(zreg2 * cutreg.gt.0) then
108 do i = 1, nel
109 call dgivej(rion(1, k), rkel(1, i), lbox,x_shift)
110#ifdef _SIMD
111!$omp simd
112#endif
113 do ii=1,neigh
114 dist_shift(ii) = max(dsqrt((x_shift(1)+x_neigh(ii,1))**2+&
115 & (x_shift(2)+x_neigh(ii,2))**2+(x_shift(3)+x_neigh(ii,3))**2),1d-9)
116 distreg_shift(ii) = max(dist_shift(ii), cutreg)
117! cost = derfc(kappa * distb) / distb
118 dist_shift(ii)=rep_erfc(dist_shift(ii),kappa)
119 ! pot_ea=pot_ea-2.d0*zeta(k)*cost
120 distreg_shift(ii)=rep_erfc(distreg_shift(ii),kappa)
121 enddo
122 do ii=1,neigh
123 vpotreg(1, i) = vpotreg(1, i) - 2.d0 * zeta(k) * dist_shift(ii)
124 ! potreg_ea=potreg_ea+Zreg2*(cost-derfc(kappa*distbreg)/distbreg)
125 vpotreg(2, i) = vpotreg(2, i) + (zreg2 - 2.d0 * zeta(k)) * dist_shift(ii)&
126! & - Zreg2 * derfc(kappa * distbreg) / distbreg
127 & - zreg2 * distreg_shift(ii)
128
129 enddo
130 enddo
131 else
132 do i = 1, nel
133 call dgivej(rion(1, k), rkel(1, i), lbox,x_shift)
134#ifdef _SIMD
135!$omp simd
136#endif
137 do ii=1,neigh
138 dist_shift(ii) = max(dsqrt((x_shift(1)+x_neigh(ii,1))**2+&
139 & (x_shift(2)+x_neigh(ii,2))**2+(x_shift(3)+x_neigh(ii,3))**2),1d-9)
140! distb = ngivej(rion(1, k), rkel(1, i), LBox)
141! cost = -2.d0 * zeta(k) * derfc(kappa * distb) / distb
142 dist_shift(ii)=-2.d0*zeta(k)*rep_erfc(dist_shift(ii),kappa)
143 ! pot_ea=pot_ea+cost
144 enddo
145 cost=sum(dist_shift(1:neigh))
146 vpotreg(1, i) = vpotreg(1, i) + cost
147 vpotreg(2, i) = vpotreg(2, i) + cost
148 enddo
149 endif
150 enddo
151
152 ! potreg_ea=pot_ea+potreg_ea
153 maxll=nel
154 maxll=(maxll*(nel-1))/2
155!$omp parallel do default(shared) private(ll,j,i,ii,cost,dist_shift,x_shift) reduction(+:vpotreg)
156 do ll=1,maxll
157 call find_j1j2(nel,ll,i,j)
158! do i = 1, nel
159! do j = i+1, nel
160! if(j.ne.i) then
161! tmp = ngivej(rkel(1, i), rkel(1, j), LBox)
162 call dgivej(rkel(1, i), rkel(1, j), lbox,x_shift)
163! cost = derfc(tmp * kappa) / tmp
164#ifdef _SIMD
165!$omp simd
166#endif
167 do ii=1,neigh
168 dist_shift(ii) = max(dsqrt((x_shift(1)+x_neigh(ii,1))**2+&
169 & (x_shift(2)+x_neigh(ii,2))**2+(x_shift(3)+x_neigh(ii,3))**2),1d-9)
170
171 dist_shift(ii)=rep_erfc(dist_shift(ii),kappa)
172 ! pot_ee=pot_ee+cost
173! vpotreg(1, j) = vpotreg(1, j) + cost
174! vpotreg(2, j) = vpotreg(2, j) + cost
175 enddo
176 cost=sum(dist_shift(1:neigh))
177 vpotreg(1,j)=vpotreg(1,j)+cost
178 vpotreg(2,j)=vpotreg(2,j)+cost
179 vpotreg(1,i)=vpotreg(1,i)+cost
180 vpotreg(2,i)=vpotreg(2,i)+cost
181 if(yes_fn) then
182 vpotsav_ee(i,j)=2.d0*cost
183 vpotsav_ee(j,i)=2.d0*cost
184 endif
185! endif
186! enddo
187 enddo
188 maxll=nion
189 maxll=(maxll*(nion-1))/2
190!$omp parallel do default(shared) private(ll,j,i,ii,jj,cost_z,dist_shift) reduction(+:pot_aa)
191 do ll=1,maxll
192 call find_j1j2(nion,ll,i,j)
193! do i = 1, nion
194! do j = i + 1, nion
195 jj=nion*(j-1)+i
196 cost_z=2.d0*zeta(i)*zeta(j)
197 if(cost_z.ne.0.d0) then
198#ifdef _SIMD
199!$omp simd
200#endif
201 do ii=1,neigh
202 dist_shift(ii)=dsqrt((iond_cart(1,jj)+x_neigh(ii,1))**2+(iond_cart(2,jj)+x_neigh(ii,2))**2+(iond_cart(3,jj)+x_neigh(ii,3))**2)
203 dist_shift(ii)=cost_z*rep_erfc(dist_shift(ii),kappa)
204! pot_aa = pot_aa + cost_z*rep_erfc(dist_shift(ii),kappa)
205 enddo
206 pot_aa=pot_aa+sum(dist_shift(1:neigh))
207 endif
208 enddo
209! enddo
210 endif
211
212 pot_aasunel=pot_aa/nel
213 vpot = 0.d0
214 do i = 1, nel
215 vpotreg(1:2, i) = pot_aa / nel + vpotreg(1:2, i)
216 vpot = vpot + vpotreg(1, i)
217 enddo
218
219 ! by E. Coccia (10/11/10): add an external potential (electronic)
220 if (ext_pot) then
221 call extpot_ene(rkel, nel, ext_ene, sum_ext)
222 do i = 1, nel
223 vpotreg(:, i) = vpotreg(:, i) + 2.d0 * ext_ene(i)
224 enddo
225 vpot = vpot + 2.d0 * sum_ext
226 ! by E. Coccia (12/1/11): add an external potential (nuclear)
227 call ion_ene(rion, zeta, nion, ext_ion, sum_ion)
228 do i = 1, nel
229 vpotreg(:, i) = vpotreg(:, i) + 2.d0 * sum_ion / real(nel)
230 enddo
231 vpot = vpot + 2.d0 * sum_ion
232 ! by E. Coccia (4/2/11): add the vdw term
233 if (vdw) then
234 call vdw_ene(rion, nion, sum_vdw)
235 do i = 1, nel
236 vpotreg(:, i) = vpotreg(:, i) + 2.d0 * sum_vdw / real(nel)
237 enddo
238 vpot = vpot + 2.d0 * sum_vdw
239 endif
240 ! by E. Coccia (11/5/11): add the mm angular term in case of QMC/MM boundary
241 if (link_atom) then
242 call mm_angle()
243 do i = 1, nel
244 vpotreg(:, i) = vpotreg(:, i) + 2.d0 * mm_pot_theta / real(nel)
245 enddo
246 vpot = vpot + 2.d0 * mm_pot_theta
247 call mm_dihed()
248 do i = 1, nel
249 vpotreg(:, i) = vpotreg(:, i) + 2.d0 * mm_pot_dihed / real(nel)
250 enddo
251 vpot = vpot + 2.d0 * mm_pot_dihed
252 call mm_improper()
253 do i = 1, nel
254 vpotreg(:, i) = vpotreg(:, i) + 2.d0 * mm_pot_impr / real(nel)
255 enddo
256 vpot = vpot + 2.d0 * mm_pot_impr
257 endif
258 endif
259
260 ! by E. Coccia (7/12/11): classical restraints
261 if (mm_restr) then
262 call r_bond()
263 do i = 1, nel
264 vpotreg(:, i) = vpotreg(:, i) + 2.d0 * restr_bond / real(nel)
265 enddo
266 vpot = vpot + 2.d0 * restr_bond
267 call r_angle()
268 do i = 1, nel
269 vpotreg(:, i) = vpotreg(:, i) + 2.d0 * restr_angle / real(nel)
270 enddo
271 vpot = vpot + 2.d0 * restr_angle
272 call r_dihe()
273 do i = 1, nel
274 vpotreg(:, i) = vpotreg(:, i) + 2.d0 * restr_dihe / real(nel)
275 enddo
276 vpot = vpot + 2.d0 * restr_dihe
277 call r_dimp()
278 do i = 1, nel
279 vpotreg(:, i) = vpotreg(:, i) + 2.d0 * restr_dimp / real(nel)
280 enddo
281 vpot = vpot + 2.d0 * restr_dimp
282
283 endif
284
285 return
286
287END subroutine upvpotdiag
