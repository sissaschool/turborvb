16!> @brief Jastrow factor support vector update for DMC and LR-DMC
17!>
18!> This subroutine updates the support vector jastrowall_ee used in
19!> Diffusion Monte Carlo (DMC) and Lattice Regularized DMC (LR-DMC)
20!> calculations. The updated vector is used by the fast update routine
21!> "uptabtot_new.f90" to speed up Jastrow factor calculations.
22!>
23!> The subroutine handles both 3/4-body Jastrow factors and wave function
24!> components, with support for spin-dependent calculations and OpenMP
25!> offloading for GPU acceleration.
26!>
27!> @param[in] yesfn Flag indicating whether to update gradients and Laplacians
28!> @param[in,out] jastrowall_ee Support vector for Jastrow factor calculations
29!> @param[in] winvjbar Jastrow factor basis coefficients for up electrons
30!> @param[in] winvjbarsz Jastrow factor basis coefficients for spin-dependent terms
31!> @param[in] winvj Jastrow factor basis vectors
32!> @param[in,out] psip Working array for matrix operations
33!>
34!> @details
35!> The subroutine performs the following operations:
36!> 1. Initializes working array psip to zero
37!> 2. If yesfn is true, updates gradients and Laplacians:
38!> - Processes 3/4-body terms (k = 1 to ip4)
39!> - Processes lattice terms (k = 1 to indt)
40!> - Uses matrix multiplication to compute Jastrow contributions
41!> - Handles both up and down electron components
42!> - Updates spin-dependent terms if iessz is true
43!> 3. Updates wave function components:
44!> - Computes Jastrow factor contributions to wave function
45!> - Handles both up and down electron components
46!> - Updates spin-dependent terms if iessz is true
47!>
48!> @note The 2-body part of jastrowall_ee is computed in "compute_fast.f90"
49!> @note The subroutine supports OpenMP offloading for GPU acceleration
50!> @note Handles both paired (ipj=1) and unpaired (ipj=2) electron configurations
51!> @note Spin-dependent calculations are performed when iessz is true
52!> @note Used in DMC and LR-DMC calculations for efficient Jastrow updates
53subroutine save_jall(yesfn, jastrowall_ee, winvjbar, winvjbarsz, winvj, psip)
54
55 use constants, only: ip4, ipj
56 use allio, only: nel, nelup, neldo, indt, indt4j, nelorbj, nelorbjh, iessz
57
58 implicit none
59 ! input
60 real(8), intent(inout) :: jastrowall_ee(nelup + neldo, nelup + neldo, 0:indt4j), psip(nel, nel)
61 real(8), intent(in) :: winvj(max(nelorbjh, 1), 0:indt4j, *), winvjbar(*), winvjbarsz(*)
62 integer :: nelused
63 ! local
64 integer :: i, k, nelorbj5
65 logical yesfn
66
67 if (nelorbj .le. 0) return
68
69 nelorbj5 = nelorbj*(indt4j + 1)
70 nelused = nelup + mod(ipj, 2)*neldo
71 !!!!!!!!!!
72 ! 3/4body
73 !!!!!!!!!!
74
75 ! in the following I assume that indt4j.ge.indt+ip4,
76 ! therefore the basis vector winvj must be already
77 ! completely filled.
78
79 ! clearing working space
80 psip = 0.d0
81#ifdef _OFFLOAD
82!$omp target update to (psip)
83#endif
84 if (yesfn) then
85 ! update gradients and laplacians
86 do k = 1, ip4
87 call dgemm_('T', 'N', nelused, nel, nelorbjh, 1.d0, winvj(1, indt + k, 1) &
88 &, nelorbj5, winvjbar, ipj*nelorbjh, 0.d0, psip, nel)
89 if (ipj == 2) then
90 call dgemm_('T', 'N', neldo, nel, nelorbjh, 1.d0, winvj(1, indt + k, 1 + nelup) &
91 &, nelorbj5, winvjbar(1 + nelorbjh), 2*nelorbjh, 0.d0, psip(1 + nelup, 1), nel)
92 end if
93#ifdef _OFFLOAD
94!$omp target update from(psip)
95#endif
96 call upjastrowall(nel, jastrowall_ee(1, 1, indt + k), psip)
97 if (iessz) then
98 call dgemm('T', 'N', nel, nel, nelorbjh, 1.d0, winvj(1, indt + k, 1) &
99 &, nelorbj5, winvjbarsz, nelorbjh, 0.d0, psip, nel)
100 call upjastrowall_sz(nel, nelup, jastrowall_ee(1, 1, indt + k), psip)
101 end if
102 end do
103 ! finally the 1:indt positions in the LRDMC lattice
104 do k = 1, indt
105 call dgemm_('T', 'N', nelused, nel, nelorbjh, 1.d0, winvj(1, k, 1) &
106 &, nelorbj5, winvjbar, ipj*nelorbjh, 0.d0, psip, nel)
107 if (ipj == 2) then
108 call dgemm_('T', 'N', neldo, nel, nelorbjh, 1.d0, winvj(1, k, 1 + nelup) &
109 &, nelorbj5, winvjbar(1 + nelorbjh), 2*nelorbjh, 0.d0, psip(1 + nelup, 1), nel)
110 end if
111
112#ifdef _OFFLOAD
113!$omp target update from(psip)
114#endif
115 call upjastrowallfat(nel, jastrowall_ee(1, 1, k), psip)
116 if (iessz) then
117 call dgemm('T', 'N', nel, nel, nelorbjh, 1.d0, winvj(1, k, 1) &
118 &, nelorbj5, winvjbarsz, nelorbjh, 0.d0, psip, nel)
119 call upjastrowallfat_sz(nel, nelup, jastrowall_ee(1, 1, k), psip)
120 end if
121 end do
122 end if
123
124 ! then update jastrowall_ee for the wave function
125 !call dgemm('T','N',nel,nelused,ipj*nelorbjh,1.d0,winvjbar &
126 ! &,ipj*nelorbjh,winvj,nelorbj5,0.d0,psip,nel)
127 call dgemm_('T', 'N', nel, nelused, nelorbjh, 1.d0, winvjbar &
128 &, ipj*nelorbjh, winvj, nelorbj5, 0.d0, psip, nel)
129 if (ipj == 2) then
130 call dgemm_('T', 'N', nel, neldo, nelorbjh, 1.d0, winvjbar(1 + nelorbjh) &
131 &, 2*nelorbjh, winvj(1, 0, 1 + nelup), nelorbj5, 0.d0, psip(1, 1 + nelup), nel)
132 end if
133
134 ! write(*,*) " psip in save_jastrowall:", psip(1:nel,1:nel)
135#ifdef _OFFLOAD
136!$omp target update from(psip)
137#endif
138 call upjastrowallpsi(nel, jastrowall_ee, psip)
139 if (iessz) then
140 call dgemm('T', 'N', nel, nel, nelorbjh, 1.d0, winvjbarsz &
141 &, nelorbjh, winvj, nelorbj5, 0.d0, psip, nel)
142 call upjastrowallpsi_sz(nel, nelup, jastrowall_ee, psip)
143 end if
144
145 return
146
147end subroutine save_jall
