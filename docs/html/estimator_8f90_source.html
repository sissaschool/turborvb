16!> @brief Statistical estimator module for TurboRVB
17!> @details This module provides data structures and procedures for computing
18!> statistical estimators including averages, variances, and correlation functions.
19!> It supports both scalar and array data types with different levels of
20!> statistical analysis.
21!>
22!> The module defines several derived types:
23!> - avg_scalar/avg_array: For simple averaging
24!> - esti_scalar/esti_array: For statistical analysis with variance
25!> - corr_basic/corr_advanced: For correlation functions
26!>
27!> Each type has associated procedures for allocation, reset, data pushing,
28!> calculation, and cleanup.
29!>
30!> @author TurboRVB group
31!> @date 2022
32module estimator
33 implicit none
34
35 !> @brief Scalar average estimator type
36 !> @details Contains fields for number of samples, summation, and average
37 type :: avg_scalar
38 integer :: num !< Number of samples
39 double precision :: summation !< Sum of all values
40 double precision :: average !< Computed average
41 end type avg_scalar
42
43 !> @brief Array average estimator type
44 !> @details Contains fields for number of samples, dimension, and arrays
45 !> for summation and average values
46 type :: avg_array
47 integer :: num !< Number of samples
48 integer :: dimen !< Dimension of the array
49 double precision, allocatable :: summation(:) !< Sum of all values
50 double precision, allocatable :: average(:) !< Computed averages
51 end type avg_array
52
53 !> @brief Scalar statistical estimator type
54 !> @details Contains fields for statistical analysis including variance
55 !> and standard deviation
56 type :: esti_scalar
57 integer :: num !< Number of samples
58 double precision :: summation !< Sum of all values
59 double precision :: sumsqr !< Sum of squared values
60 double precision :: average !< Computed average
61 double precision :: vari !< Computed variance
62 double precision :: deviation !< Computed standard deviation
63 end type esti_scalar
64
65 !> @brief Array statistical estimator type
66 !> @details Contains fields for statistical analysis of array data
67 type :: esti_array
68 integer :: num !< Number of samples
69 integer :: dimen !< Dimension of the array
70 double precision, allocatable :: summation(:) !< Sum of all values
71 double precision, allocatable :: sumsqr(:) !< Sum of squared values
72 double precision, allocatable :: average(:) !< Computed averages
73 double precision, allocatable :: vari(:) !< Computed variances
74 double precision, allocatable :: deviation(:) !< Computed standard deviations
75 end type esti_array
76
77 !> @brief Basic correlation function estimator type
78 !> @details Uses average estimators for correlation analysis
79 type :: corr_basic
80 integer :: num !< Number of samples
81 type(avg_array) :: estia !< Estimator for first variable
82 type(avg_array) :: estib !< Estimator for second variable
83 double precision, allocatable :: corrsum(:) !< Sum of correlations
84 double precision, allocatable :: corrfun(:) !< Computed correlation function
85 end type corr_basic
86
87 !> @brief Advanced correlation function estimator type
88 !> @details Uses statistical estimators for correlation analysis
89 type :: corr_advanced
90 integer :: num !< Number of samples
91 type(esti_array) :: estia !< Estimator for first variable
92 type(esti_array) :: estib !< Estimator for second variable
93 double precision, allocatable :: corrsum(:) !< Sum of correlations
94 double precision, allocatable :: corrfun(:) !< Computed correlation function
95 end type corr_advanced

335 !> @brief Calculate statistical analysis for scalar estimator
336 !> @param estimator The statistical estimator
337 subroutine calc_esti_scalar(estimator)
338 implicit none
339 type(esti_scalar) :: estimator
340 integer :: ii
341
342 if (estimator%num <= 0) return
343 estimator%average = estimator%summation/estimator%num
344 estimator%vari = estimator%sumsqr/estimator%num - estimator%average*estimator%average
345 estimator%deviation = dsqrt(estimator%vari)
346 if (estimator%vari < 0.d0) write (6, *) "Warning : negative variation due to machine precision!"
347 end subroutine calc_esti_scalar
348
349 !> @brief Calculate statistical analysis for array estimator
350 !> @param estimator The statistical estimator
351 subroutine calc_esti_array(estimator)
352 implicit none
353 type(esti_array) :: estimator
354 integer :: ii
355
356 if (estimator%num <= 0) return
357 estimator%average = estimator%summation/estimator%num
358 estimator%vari = estimator%sumsqr/estimator%num - estimator%average*estimator%average
359 estimator%deviation = dsqrt(estimator%vari)
360 do ii = 1, estimator%dimen
361 if (estimator%vari(ii) < 0.d0) write (6, *) "Warning : negative variation due to machine precision!"
362 end do
363 end subroutine calc_esti_array
