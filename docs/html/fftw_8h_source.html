79/*********************************************
80 * Codelets
81 *********************************************/
82/*
83 * There are two kinds of codelets:
84 *
85 * NO_TWIDDLE computes the FFT of a certain size, operating
86 * out-of-place (i.e., take an input and produce a
87 * separate output)
88 *
89 * TWIDDLE like no_twiddle, but operating in place. Moreover,
90 * multiplies the input by twiddle factors.
91 */
92
93typedef void (notw_codelet) (const FFTW_COMPLEX *, FFTW_COMPLEX *, int, int);
94typedef void (twiddle_codelet) (FFTW_COMPLEX *, const FFTW_COMPLEX *, int,
95 int, int);
96typedef void (generic_codelet) (FFTW_COMPLEX *, const FFTW_COMPLEX *, int,
97 int, int, int);

131/*****************************
132 * Plans
133 *****************************/
134/*
135 * A plan is a sequence of reductions to compute a FFT of
136 * a given size. At each step, the FFT algorithm can:
137 *
138 * 1) apply a notw codelet, or
139 * 2) recurse and apply a twiddle codelet, or
140 * 3) apply the generic codelet.
141 */
142
143enum fftw_node_type {
144 FFTW_NOTW, FFTW_TWIDDLE, FFTW_GENERIC
145};
146
147/* structure that contains twiddle factors */
148typedef struct fftw_twiddle_struct {
149 int n;
150 int r;
151 int m;
152 FFTW_COMPLEX *twarray;
153 struct fftw_twiddle_struct *next;
154 int refcnt;
155} fftw_twiddle;
156
157/* structure that holds all the data needed for a given step */
158typedef struct fftw_plan_node_struct {
159 enum fftw_node_type type;
160
161 union {
162 /* nodes of type FFTW_NOTW */
163 struct {
164 int size;
165 notw_codelet *codelet;
166 } notw;
167
168 /* nodes of type FFTW_TWIDDLE */
169 struct {
170 int size;
171 twiddle_codelet *codelet;
172 fftw_twiddle *tw;
173 struct fftw_plan_node_struct *recurse;
174 } twiddle;
175
176 /* nodes of type FFTW_GENERIC */
177 struct {
178 int size;
179 generic_codelet *codelet;
180 fftw_twiddle *tw;
181 struct fftw_plan_node_struct *recurse;
182 } generic;
183
184 } nodeu;
185
186 int refcnt;
187} fftw_plan_node;

301 * Here, you can use all the nice timers available in your machine.

303
304/*
305 *
306 Things you should define to include your own clock:
307
308 fftw_time -- the data type used to store a time
309
310 extern fftw_time fftw_get_time(void);
311 -- a function returning the current time. (We have
312 implemented this as a macro in most cases.)
313
314 extern fftw_time fftw_time_diff(fftw_time t1, fftw_time t2);
315 -- returns the time difference (t1 - t2).
316 If t1 < t2, it may simply return zero (although this
317 is not required). (We have implemented this as a macro
318 in most cases.)
319
320 extern double fftw_time_to_sec(fftw_time t);
321 -- returns the time t expressed in seconds, as a double.
322 (Implemented as a macro in most cases.)
323
324 FFTW_TIME_MIN -- a double-precision macro holding the minimum
325 time interval (in seconds) for accurate time measurements.
326 This should probably be at least 100 times the precision of
327 your clock (we use even longer intervals, to be conservative).
328 This will determine how long the planner takes to measure
329 the speeds of different possible plans.
330
331 Bracket all of your definitions with an appropriate #ifdef so that
332 they will be enabled on your machine. If you do add your own
333 high-precision timer code, let us know (at fftw@theory.lcs.mit.edu).
334
335 Only declarations should go in this file. Any function definitions
336 that you need should go into timer.c.
337*/
338
339/* define a symbol so that we know that we have the fftw_time_diff
340 function/macro (it did not exist prior to FFTW 1.2) */
341#define FFTW_HAS_TIME_DIFF
342
343#ifdef SOLARIS
344
345/* we use the nanosecond virtual timer */
346#include <sys/time.h>
347
348typedef hrtime_t fftw_time;
349
350#define fftw_get_time() gethrtime()
351#define fftw_time_diff(t1,t2) ((t1) - (t2))
352#define fftw_time_to_sec(t) ((double) t / 1.0e9)
353
354/*
355 * a measurement is valid if it runs for at least
356 * FFTW_TIME_MIN seconds.
357 */
358#define FFTW_TIME_MIN (1.0e-4) /* for Solaris nanosecond timer */
359
360#endif /* SOLARIS */
361
362#if defined(MAC) || defined(macintosh)
363
364/* Use Macintosh Time Manager routines (maximum resolution is about 20
365 microseconds). */
366
367typedef struct fftw_time_struct {
368 unsigned long hi,lo;
369} fftw_time;
370
371extern fftw_time get_Mac_microseconds(void);
372
373#define fftw_get_time() get_Mac_microseconds()
374
375/* define as a function instead of a macro: */
376extern fftw_time fftw_time_diff(fftw_time t1, fftw_time t2);
377
378#define fftw_time_to_sec(t) ((t).lo * 1.0e-6 + 4294967295.0e-6 * (t).hi)
379
380/* very conservative, since timer should be accurate to 20e-6: */
381/* (although this seems not to be the case in practice) */
382#define FFTW_TIME_MIN (5.0e-2) /* for MacOS Time Manager timer */
383
384#endif /* Macintosh */
385
386#ifdef __WIN32__
387
388#include <time.h>
389
390typedef unsigned long fftw_time;
391extern unsigned long GetPerfTime(void);
392extern double GetPerfSec(double ticks);
393
394#define fftw_get_time() GetPerfTime()
395#define fftw_time_diff(t1,t2) ((t1) - (t2))
396#define fftw_time_to_sec(t) GetPerfSec(t)
397
398#define FFTW_TIME_MIN (5.0e-2) /* for Win32 timer */
399#endif /* __WIN32__ */
400
401#if defined(_CRAYMPP) /* Cray MPP system */
402
403double SECONDR(void); /*
404 * I think you have to link with -lsci to

406 */

408typedef double fftw_time;

413#define FFTW_TIME_MIN (1.0e-1) /* for Cray MPP SECONDR timer */
414
415#endif /* _CRAYMPP */

417/***********************************************
418 * last resort: good old Unix clock()
419 ***********************************************/
420#ifndef FFTW_TIME_MIN
421#include <time.h>
422
423typedef clock_t fftw_time;
424
425#ifndef CLOCKS_PER_SEC
426#ifdef sun
427 /* stupid sunos4 prototypes */
428#define CLOCKS_PER_SEC 1000000
429extern long clock(void);
430#else /* not sun, we don't know CLOCKS_PER_SEC */
431#error Please define CLOCKS_PER_SEC
432#endif
433#endif
434
435#if defined(__QK_USER__)
436#define fftw_get_time() ((long) (dclock() * 1000000.0L))
437#else
438#define fftw_get_time() clock()
439#endif
440#define fftw_time_diff(t1,t2) ((t1) - (t2))
441#define fftw_time_to_sec(t) (((double) (t)) / CLOCKS_PER_SEC)

445 * measurement must run for 200ms in order to be valid.
446 * You had better check the manual of your machine
447 * to discover if it can do better than this
448 */
449#define FFTW_TIME_MIN (2.0e-1) /* for default clock() timer */

451#endif /* UNIX clock() */
