16!-----------------------
17!> @brief Performs post-processing tasks for non-self-consistent calculations
18!> @details This subroutine handles various post-processing tasks for non-self-consistent
19!> calculations including band structure plotting, density of states evaluation,
20!> and Fermi surface calculations (to be implemented).
21!> @param[in] task Task identifier:
22!> - 1: Band structure calculation
23!> - 2: DOS calculation
24!> - 5: Evaluate all tasks
25!> - 0: No tasks (error)
26subroutine pp_nsc(task)
27!-----------------------
28 use allio, only: rank, qp
29 implicit none
30 integer, intent(in) :: task
31
32 select case (task)
33
34 case (1) ! band structure calculation
35 if (rank .eq. 0) call plot_bands
36
37 case (2) ! DOS calculation
38 if (rank .eq. 0) call evaluate_dos
39
40 !case(3) ! Fermi surface calculation (TO BE IMPLEMENTED)
41 !if(rank.eq.0) call evaluate_fermis
42
43 case (5) ! evaluate all
44 if (rank .eq. 0) then
45 call plot_bands
46 call evaluate_dos
47 end if
48
49 case (0)
50 call error(' PP_nsc ', ' No tasks required!! ', -1, rank)
51
52 case default
53 call error(' PP_nsc ', ' Task type not recognized!! ', -1, rank)
54
55 end select
56
57 return
58
59end subroutine pp_nsc

61!---------------------
62!> @brief Plots band structure from non-self-consistent calculations
63!> @details This routine generates band structure plots from non-self-consistent
64!> calculations. It identifies high symmetry points, creates k-point paths,
65!> and writes band structure data to a file compatible with gnuplot.
66!> Supports both spin-up and spin-down bands for LSDA calculations.
67!> @note Requires kp_type=2 or 3 for k-points path calculations
68subroutine plot_bands
69!---------------------
70
71 ! Routine for plotting band structures from NSC calculations.
72 ! kp_type must be 2 or 3 in order to run this routine.
73 ! In the case of a kpoints path (kp_type=3), one already knows the number of k-points and the
74 ! number of lines in the path, which are "nk" and "nk1-1" respectively.
75
76 use setup, only: eigmol_sav, eigmoldo_sav, yeslsda, bands
77 use kpoints_mod, only: nk, xkp
78 use constants, only: ipc, energy_unit ! Ha to eV conversion
79 implicit none
80 logical, dimension(:), allocatable :: high_symmetry, in_range, in_rangedo
81 integer i, j, k
82 real(8), dimension(:), allocatable :: kp_path
83 real(8) tmp_norm, k1(3), k2(3), dkl, dkl_sav, emin, emax
84 character(len=80) :: filename
85
86 ! initialization
87 write (6, *) '---------------------------'
88 write (6, *) ' Evaluating band structure '
89 write (6, *) '---------------------------'
90
91 filename = 'band_structure.dat'
92 allocate (high_symmetry(nk), in_range(bands))
93 in_range = .true.
94 if (yeslsda .or. ipc .eq. 2) then
95 allocate (in_rangedo(bands))
96 in_rangedo = .true.
97 end if
98 allocate (kp_path(nk))
99 high_symmetry = .false.
100 kp_path = 0.d0
101 !
102 ! find high symmetry points in the 2D plot
103 !
104 do k = 1, nk
105 if (k .eq. 1 .or. k .eq. nk) then
106 high_symmetry(k) = .true.
107 else
108 k1(:) = xkp(:, k) - xkp(:, k - 1)
109 k2(:) = xkp(:, k + 1) - xkp(:, k)
110 tmp_norm = (k1(1)*k2(1) + k1(2)*k2(2) + k1(3)*k2(3))/ &
111 dsqrt(k1(1)*k1(1) + k1(2)*k1(2) + k1(3)*k1(3))/ &
112 dsqrt(k2(1)*k2(1) + k2(2)*k2(2) + k2(3)*k2(3))
113 ! if changing direction in the 1BZ, then it's a high symmetry point
114 high_symmetry(k) = (abs(tmp_norm - 1.d0) .gt. 1.0d-4)
115 end if
116 if (sum(xkp(:, k)**2) .lt. 1.0d-9) high_symmetry(k) = .true.
117 ! distance between 2 kps.
118 if (k .eq. 2) dkl_sav = dsqrt(k1(1)**2 + k1(2)**2 + k1(3)**2)
119 end do
120 !
121 ! create 2D path
122 !
123 do k = 2, nk
124
125 dkl = dsqrt((xkp(1, k) - xkp(1, k - 1))**2 + &
126 (xkp(2, k) - xkp(2, k - 1))**2 + &
127 (xkp(3, k) - xkp(3, k - 1))**2)
128
129 if (dkl .gt. 5.d0*dkl_sav) then
130 kp_path(k) = kp_path(k - 1)
131 elseif (dkl .gt. 1.0d-5) then
132 kp_path(k) = kp_path(k - 1) + dkl
133 dkl_sav = dkl
134 else
135 kp_path(k) = kp_path(k - 1) + dkl
136 end if
137
138 end do
139 ! find minimum and maximum values
140 emin = eigmol_sav(1, 1)
141 emax = eigmol_sav(1, 1)
142 do k = 1, nk
143 do j = 1, bands
144 emin = min(emin, eigmol_sav(j, k))
145 if (yeslsda .or. ipc .eq. 2) emin = min(emin, eigmoldo_sav(j, k))
146 emax = max(emax, eigmol_sav(j, k))
147 if (yeslsda .or. ipc .eq. 2) emax = max(emax, eigmoldo_sav(j, k))
148 end do
149 end do
150 ! find bands values which are in the specified energy range
151 in_range = .true.
152 in_rangedo = .true.
153 do i = 1, bands
154 in_range(i) = any(eigmol_sav(i, 1:nk) >= emin .and. eigmol_sav(i, 1:nk) <= emax)
155 in_rangedo(i) = any(eigmoldo_sav(i, 1:nk) >= emin .and. eigmoldo_sav(i, 1:nk) <= emax)
156 end do
157
158 write (6, *) in_range
159 write (6, *) kp_path
160
161 write (6, *) ' Band structures INFORMATION: '
162 write (6, *) 'Minimum/maximum bands values:', emin, emax
163 write (6, *) 'High symmetry points: '
164 do k = 1, nk
165 if (high_symmetry(k)) then
166 write (6, *) kp_path(k), xkp(:, k)
167 end if
168 end do
169 !
170 ! now draw bands
171 open (unit=80, file=trim(filename), form='formatted', status='unknown')
172 write (80, *) ' # Spin up DFT bands (eV) '
173 do i = 1, bands
174 if (in_range(i)) then
175 if (mod(i, 2) /= 0) then
176 write (80, 100) (kp_path(j), energy_unit*eigmol_sav(i, j), j=1, nk)
177 else
178 write (80, 100) (kp_path(j), energy_unit*eigmol_sav(i, j), j=nk, 1, -1)
179 end if
180 end if
181 end do
182 if (yeslsda .or. ipc .eq. 2) then
183 ! add two carriage return for gnuplot
184 write (80, *)
185 write (80, *)
186 write (80, *) ' # Spin down DFT bands (eV)'
187 do i = 1, bands
188 if (in_rangedo(i)) then
189 if (mod(i, 2) /= 0) then
190 write (80, 100) (kp_path(j), energy_unit*eigmoldo_sav(i, j), j=1, nk)
191 else
192 write (80, 100) (kp_path(j), energy_unit*eigmoldo_sav(i, j), j=nk, 1, -1)
193 end if
194 end if
195 end do
196 end if
197 !
198 close (80)
199 deallocate (high_symmetry, kp_path, in_range)
200 if (yeslsda .or. ipc .eq. 2) deallocate (in_rangedo)
201
202100 format(2f10.4)
203
204 return
205
206end subroutine plot_bands

208!-----------------------
209!> @brief Calculates density of states from band energies
210!> @details This routine calculates the density of states using smearing functions
211!> applied to the band energies. It supports different smearing types and
212!> handles both spin-up and spin-down contributions for LSDA calculations.
213!> @note Requires optocc>0 and non-zero epsshell for smearing calculations
214subroutine evaluate_dos
215!-----------------------
216
217 !
218 ! Routine to calculate the density of states.
219 ! Use optocc>0 and a non-zero value of the smearing (epsshell)
220 ! in order to compute the density of states.
221 !
222
223 use constants, only: ipc, energy_unit
224 use kpoints_mod, only: nk, xkp, wkp
225 use setup, only: eigmol_sav, eigmoldo_sav, epsshell, optocc, deltae, &
226 emin, emax, yeslsda, bands
227 use compute_efermi, only: smeard
228
229 implicit none
230 integer :: i, j, k, ibnd, indk, smear_type
231 integer :: ndos ! # of points in the DOS
232 real(8), dimension(:), allocatable :: tdos, dos_up, dos_down
233 real(qp) :: e, dt
234 character(len=80) :: filename
235
236 ! initialization
237 write (6, *) '------------------------------'
238 write (6, *) ' Evaluating density of states '
239 write (6, *) '------------------------------'
240 filename = 'density_of_states.dat'
241 if (optocc .eq. 0 .or. epsshell .eq. 0.d0) then
242 write (6, *) 'Warning: DOS must be updated with a smearing function! Setting default values.'
243 optocc = 1
244 epsshell = 0.01_qp
245 end if
246 smear_type = optocc
247 dt = epsshell
248 ! setting min/max values of the energies
249 ! if not chosen from the user
250 if (emin .eq. 0.d0 .and. emax .eq. 0.d0) then
251 emin = eigmol_sav(1, 1)
252 emax = eigmol_sav(1, 1)
253 do k = 1, nk
254 do j = 1, bands
255 emin = min(emin, eigmol_sav(j, k))
256 if (yeslsda .or. ipc .eq. 2) emin = min(emin, eigmoldo_sav(j, k))
257 emax = max(emax, eigmol_sav(j, k))
258 if (yeslsda .or. ipc .eq. 2) emax = max(emax, eigmoldo_sav(j, k))
259 end do
260 end do
261 end if
262 !
263 ndos = int((emax - emin)/deltae)
264 allocate (dos_up(ndos + 1), dos_down(ndos + 1), tdos(ndos + 1))
265
266 dos_up = 0.d0
267 dos_down = 0.d0
268 tdos = 0.d0
269 e = 0.d0
270
271 open (unit=80, file=trim(filename), form='formatted', status='unknown')
272 write (80, *) '# energy (eV) Density_of_states (spin up)'
273 do i = 0, ndos
274 e = emin + deltae*i
275 do indk = 1, nk
276 do ibnd = 1, bands
277 dos_up(i + 1) = dos_up(i + 1) + wkp(indk)*smeard(eigmol_sav(ibnd, nk), dt, e, smear_type)
278 end do
279 end do
280 write (80, 100) e*energy_unit, dos_up(i + 1)
281 end do
282
283 if (yeslsda .or. ipc .eq. 2) then
284 write (80, *)
285 write (80, *)
286 write (80, *) '# energy (eV) Density_of_states (spin down)'
287 do i = 0, ndos
288 e = emin + deltae*i
289 do indk = 1, nk
290 do ibnd = 1, bands
291 dos_down(i + 1) = dos_down(i + 1) + wkp(indk)*smeard(eigmoldo_sav(ibnd, indk), dt, e, smear_type)
292 end do
293 end do
294 write (80, 100) e*energy_unit, dos_down(i + 1)
295 end do
296 end if
297
298 deallocate (dos_up, dos_down, tdos)
299 write (6, *)
300 close (80)
301
302100 format(2f10.4)
303
304 return
305
306end subroutine evaluate_dos
