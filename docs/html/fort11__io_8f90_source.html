16!=======================================================================
17!> @file fort11_io.f90
18!> @brief Fort.11 file I/O operations for TurboRVB
19!> @details This module handles reading and writing of fort.11 files,
20!> which contain simulation state information for restart
21!> and continuation of quantum Monte Carlo calculations.
22!> @author TurboRVB group
23!> @date 2022
24!> @section Features
25!> - Write simulation state to fort.11 for restart capability
26!> - Read simulation state from fort.11 for continuation
27!> - Handle various simulation parameters and arrays
28!> - Support for different simulation modes (VMC, LRDMC, etc.)
29!> - Compatibility checking between read and current parameters
30!> @section Files
31!> fort.11 contains:
32!> - Simulation parameters (nw, np, nmatb, etc.)
33!> - Walker configurations (wconfn)
34!> - Correlation weights (wcorw)
35!> - Kinetic energy derivatives (dek, dekg)
36!> - Ion velocities (velion)
37!> - CG optimization data (reduce)
38!> - Contraction matrices (detmat_proj, projmat_c)
39!=======================================================================

72!-----------------------------------------------------------------------
73!> @brief Write simulation data arrays to fort.11
74!> @details Writes the main simulation data arrays to fort.11 file,
75!> including walker configurations, correlation weights,
76!> kinetic energy derivatives, and optimization data.
77!> @note Writes different data sets based on itestr value
78!> @note Handles contraction matrices if contraction > 0
79!> @note Includes ion velocities for dynamics simulations (itestr = -5)
80!> @note Writes CG optimization data (reduce array) for parameter optimization
81!> @note Conditionally writes detmat_proj and projmat_c if allocated
82!> @note Writes covariance matrix (cov_old) if allocated
83subroutine write_fort11_end
84 use allio
85 implicit none
86 integer indtel, l, i, j, k
87 indtel = nel*(indt + 1)
88 if (itestr .eq. -5) then
89 if (contraction .gt. 0) then
90 write (11) (wconfn(j), j=1, nw), (wcorw(j), j=1, nfat)&
91 &, (dek(k), k=1, ieskin)&
92 &, (dekg(k), k=1, iesking) &
93 &, ((velion(j, i), j=1, 3), i=1, ieskindim) &
94 &, ((reduce(i, j), j=1, np), i=1, ncgdim) &
95 &, (allowcontr(i, 1), allowcontr(i, 2), i=1, nelorb_c)

100 &, ((velion(j, i), j=1, 3), i=1, ieskindim) &
101 &, ((reduce(i, j), j=1, np), i=1, ncgdim)
102 end if
103 else
