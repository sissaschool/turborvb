87 !---------------------------------------------------------
88 !> @brief Initialize k-points for all kp_type methods
89 !> @details Initializes the k-point grid and related arrays for periodic boundary
90 !> condition calculations. Handles various k-point generation schemes:
91 !> Monkhorst-Pack, custom input, band structure paths, random, and flavor twist.
92 !> Also sets up symmetry operations and transforms coordinates as needed.
93 !> @param[inout] iflag Integer flag for error handling (0=success, 1=error)
94 !> @param[in] nel Number of electrons
95 !> @param[in] nion Number of ions
96 !> @param[in] rion Ion positions (3, nion)
97 !> @param[in] atom_number Atomic numbers (nion)
98 !> @param[in] rs Wigner-Seitz radius parameter
99 !> @note K-points are left in crystal coordinates and must be transformed to cartesian
100 !> coordinates (multiply by 2*pi/cell_scale) for use in other routines.
101 !> @note Handles symmetry reduction, time reversal, and double grid for up/down spins.
102 !> @note Writes symmetry information to stdout for user reference.
103 !> @see kpoint_grid, generate_kp_path, set_sym_bl, purge_isymm
104 subroutine get_kpoints(iflag, nel, nion, rion, atom_number, rs)
105 !---------------------------------------------------------
106 !
107 ! This routine initialize k-points for all kp_type
108 !!!!!!! NB !!!!!!!
109 ! k-points are left in crystal coordinates. Every subroutine
110 ! which needs k-points must transform them back to carthesian
111 ! coordinates (multiply by 2*pi/cell_scale)
112 !
113
114 use constants, only: pi
115
116 implicit none
117
118 integer :: iflag, nel, nion
119 real(8) :: rs, t_comp, alphap, betap, gammap
120 integer :: i, j, ikp, nks_up, nks_do
121 real(8), dimension(:), allocatable :: wkp_, wkp_down_
122 real(8), dimension(:, :), allocatable :: xkp_, xkp_down_
123 real(8), external :: ran
124 real*8, intent(in) :: rion(3, *), atom_number(*)
125 logical smerding
126
127 ! initialize cell in any case
128 celldm(1) = 1.d0
129 ! celldm(4:6) = 90d0*PI/180.d0
130
131 omega = celldm(2)*celldm(3)
132
133 ! omega=celldm(2)*celldm(3)
134 celldm(1) = (pi*nel*4.d0/3.d0/omega)**(1.d0/3.d0)*rs
135
136 ! omega=celldm(2)*celldm(3)
137 ! celldm(1)=(PI*nel*4.d0/3.d0/omega)**(1.d0/3.d0)*rs
138 ! setting up direct and reciprocal lattices
139 if (yes_tilted) then
140 givens2r = .true.
141 else
142 givens2r = .false.
143 end if
144 call initcell(nion, nel, .true.)
145 ! compute Bravais lattice symmetry operations
146 call set_sym_bl(s2r)
147 call purge_isymm(.true., nrot, isymm, sname, t_rev, rion, nion, atom_number, cellscale)
148 write (6, *) 'number of point symmetries found', nrot
149 write (6, *) ' Symmetries types found:'
150 do i = 1, nrot
151 write (6, *) sname(i)
152 write (6, *) (isymm(j, 1, i), j=1, 3)
153 write (6, *) (isymm(j, 2, i), j=1, 3)
154 write (6, *) (isymm(j, 3, i), j=1, 3)
155 end do
156 smerding = .false.
157 if (kp_type .lt. 0) then
158 smerding = .true.
159 kp_type = -kp_type
160 end if
161 !
162 if (kp_type .eq. 0) then
163 xkp(:, 1) = phase(:)
164 xkp_down(:, 1) = phase_down(:)
165 wkp(:) = 1.d0
166 wkp_down(:) = 1.d0
167
168 elseif (kp_type .eq. 1) then
169
170 call kpoint_grid(nrot, time_reversal, skip_equivalence, isymm, t_rev, &
171 nk, k1, k2, k3, nk1, nk2, nk3, nks, xkp, wkp)
172 nk = nks ! define the right number of k-points
173 !
174 ! in the case of an automatic grid, I restrict the choice of the
175 ! down spin electrons to the opposite of up spin ones. This
176 ! should probably be generalized.
177
178 !
179 if (opposite_phase) then
180 xkp_down = -xkp
181 else
182 xkp_down = xkp
183 end if
184 wkp_down = wkp
185
186 elseif (kp_type .eq. 2 .or. kp_type .eq. 3) then
187 if (found_sec(5, 'KPOINTS', iflag)) then
188 iflag = 1
189 do i = 1, nk1
190 read (5, *, err=121) xkp(1, i), xkp(2, i), xkp(3, i), wkp(i)
191 end do
192 iflag = 0
193 if (double_kpgrid) then
194 read (5, *) ! one blank
195 do i = 1, nk1
196 read (5, *, err=121) xkp_down(1, i), xkp_down(2, i), xkp_down(3, i), wkp_down(i)
197 end do
198 else
199 if (opposite_phase) then
200 xkp_down = -xkp
201 else
202 xkp_down = xkp
203 end if
204 wkp_down = wkp
205 end if
206 end if
207 !
208 if (kp_type .eq. 3) then
209 call generate_kp_path(xkp, wkp, nk1, nk2, iflag)
210 if (double_kpgrid) then
211 call generate_kp_path(xkp_down, wkp_down, nk1, nk2, iflag)
212 else
213 if (opposite_phase) then
214 xkp_down = -xkp
215 else
216 xkp_down = xkp
217 end if
218 end if
219 wkp(:) = 1.d0/nk
220 wkp_down(:) = wkp(:)
221 end if
222
223 elseif (kp_type .eq. 4) then
224 call init_random_seed
225 nk = nk1 ! # of k-points
226 wkp(:) = 1.d0/nk
227 wkp_down(:) = wkp(:)
228 do i = 1, nk
229 do j = 1, 3
230 call random_number(t_comp)
231 t_comp = t_comp - 0.5d0
232 xkp(j, i) = t_comp
233 if (opposite_phase) then
234 xkp_down(j, i) = -t_comp
235 else
236 xkp_down(j, i) = t_comp
237 end if
238 end do
239 end do
240
241 elseif (kp_type .eq. 5) then
242
243 nk = nint(sqrt(dble(nk)))
244
245 allocate (xkp_(3, nk), wkp_(nk))
246 allocate (xkp_down_(3, nk), wkp_down_(nk))
247 xkp_ = 0.d0
248 wkp_ = 0.d0
249 xkp_down_ = 0.d0
250 wkp_down_ = 0.d0
251
252 ! combine MP grids with and without offset to create
253 ! the k-points matrix needed by flavor twist
254
255 ! no offset
256 call kpoint_grid(nrot, time_reversal, skip_equivalence, isymm, t_rev, &
257 nk, 0, 0, 0, nk1, nk2, nk3, nks, xkp_, wkp_)
258 nks_up = nks
259
260 ! with offset
261 call kpoint_grid(nrot, time_reversal, skip_equivalence, isymm, t_rev, &
262 nk, 1, 1, 1, nk1, nk2, nk3, nks, xkp_down_, wkp_down_)
263 nks_do = nks
264
265 nk = nks_up*nks_do ! now I have a k-point matrix in the case of flavor twist
266
267 ikp = 1
268 do i = 1, nks_up
269 do j = 1, nks_do
270 xkp(:, ikp) = xkp_(:, i)
271 xkp_down(:, ikp) = xkp_down_(:, j)
272 wkp(ikp) = wkp_(i)*wkp_down_(j)
273 wkp_down(ikp) = wkp_(i)*wkp_down_(j)
274 ikp = ikp + 1
275 end do
276 end do
277
278 deallocate (xkp_, wkp_, xkp_down_, wkp_down_)
279
280 end if
281 if (smerding) then
282 write (6, *) ' Warning adding a small pseudo random twist &
283 & to remove degeneracies '
284 ! adding a pseudo random small contribution to remove all possible
285 ! degeneracy of the eigenvalues
286 xkp(1, :) = xkp(1, :) + 1.d-8/dsqrt(5.d0)
287 xkp(2, :) = xkp(2, :) - 1.d-8/dsqrt(7.d0)
288 xkp(3, :) = xkp(3, :) - 1.d-8/dsqrt(11.d0)
289 if (opposite_phase) then
290 xkp_down(1, :) = xkp_down(1, :) - 1.d-8/dsqrt(5.d0)
291 xkp_down(2, :) = xkp_down(2, :) + 1.d-8/dsqrt(7.d0)
292 xkp_down(3, :) = xkp_down(3, :) + 1.d-8/dsqrt(11.d0)
293 else
294 xkp_down(1, :) = xkp_down(1, :) + 1.d-8/dsqrt(5.d0)
295 xkp_down(2, :) = xkp_down(2, :) - 1.d-8/dsqrt(7.d0)
296 xkp_down(3, :) = xkp_down(3, :) - 1.d-8/dsqrt(11.d0)
297 end if
298 end if
299121 return
300
301 end subroutine get_kpoints

303 !----------------------------------------------------------------------------
304 !> @brief Generate uniform grid of k-points using Monkhorst-Pack algorithm
305 !> @details Implements the Monkhorst-Pack algorithm for generating uniform k-point grids.
306 !> Applies symmetry operations and time reversal to reduce the number of inequivalent
307 !> k-points. Handles equivalence checking and normalization of weights. Ported from
308 !> QuantumESPRESSO.
309 !> @param[in] nrot Number of rotation operations (symmetries)
310 !> @param[in] time_reversal Logical flag to use time reversal symmetry
311 !> @param[in] skip_equivalence Logical flag to skip k-point equivalence check
312 !> @param[in] s Symmetry operations (3,3,nrot)
313 !> @param[in] t_rev Time reversal flags for each symmetry operation (nrot)
314 !> @param[inout] nkp Total number of k-points (input: max, output: actual)
315 !> @param[in] k1,k2,k3 Offset parameters for Monkhorst-Pack grid
316 !> @param[in] nk1,nk2,nk3 Grid dimensions
317 !> @param[out] nks Number of inequivalent k-points
318 !> @param[out] xk K-point coordinates (3,nkp)
319 !> @param[out] wk K-point weights (nkp)
320 !> @note Uses symmetry operations to reduce the number of inequivalent k-points.
321 !> @note Normalizes weights to sum to one. Writes summary to stdout.
322 !> @see get_kpoints, set_sym_bl
323 subroutine kpoint_grid(nrot, time_reversal, skip_equivalence, s, t_rev, &
324 nkp, k1, k2, k3, nk1, nk2, nk3, nks, xk, wk)
325 !----------------------------------------------------------------------------
326 !
327 ! Automatic generation of a uniform grid of k-points with MP algorithm.
328 ! Ported from QuantumESPRESSO.
329 !
330 implicit none
331 !
332 integer, intent(in) :: nrot, k1, k2, k3, nk1, nk2, nk3, &
333 t_rev(48), s(3, 3, 48)
334 logical, intent(in) :: time_reversal, skip_equivalence
335 !
336 integer, intent(inout) :: nks, nkp
337 real(8), intent(out) :: xk(3, nkp)
338 real(8), intent(out) :: wk(nkp)
339 ! LOCAL:
340 real(8), parameter :: eps = 1.0d-5
341 real(8) :: xkr(3), fact, xx, yy, zz
342 real(8), allocatable :: xkg(:, :), wkk(:)
343 integer :: nkr, i, j, k, ns, n, nk
344 integer, allocatable :: equiv(:)
345 logical :: in_the_list
346 !
347 nkr = nk1*nk2*nk3
348 allocate (xkg(3, nkr), wkk(nkr))
349 allocate (equiv(nkr))
350 !
351 do i = 1, nk1
352 do j = 1, nk2
353 do k = 1, nk3
354 ! this is nothing but consecutive ordering
355 n = (k - 1) + (j - 1)*nk3 + (i - 1)*nk2*nk3 + 1
356 ! xkg are the components of the complete grid in crystal axis
357 xkg(1, n) = dble(i - 1)/nk1 + dble(k1)/2/nk1
358 xkg(2, n) = dble(j - 1)/nk2 + dble(k2)/2/nk2
359 xkg(3, n) = dble(k - 1)/nk3 + dble(k3)/2/nk3
360 end do
361 end do
362 end do
363
364 ! equiv(nk) =nk : k-point nk is not equivalent to any previous k-point
365 ! equiv(nk)!=nk : k-point nk is equivalent to k-point equiv(nk)
366
367 do nk = 1, nkr
368 equiv(nk) = nk
369 end do
370
371 if (skip_equivalence) then
372 write (6, *) ' Warning k-points grid: skip check of k-points equivalence '
373 wkk = 1.d0
374 else
375 do nk = 1, nkr
376 ! check if this k-point has already been found equivalent to another
377 if (equiv(nk) == nk) then
378 wkk(nk) = 1.0d0
379 ! check if there are equivalent k-point to this in the list
380 ! (excepted those previously found to be equivalent to another)
381 ! check both k and -k
382 do ns = 1, nrot
383 do i = 1, 3
384 xkr(i) = s(i, 1, ns)*xkg(1, nk) &
385 + s(i, 2, ns)*xkg(2, nk) &
386 + s(i, 3, ns)*xkg(3, nk)
387 xkr(i) = xkr(i) - nint(xkr(i))
388 end do
389 if (t_rev(ns) == 1) xkr = -xkr
390 xx = xkr(1)*nk1 - 0.5d0*k1
391 yy = xkr(2)*nk2 - 0.5d0*k2
392 zz = xkr(3)*nk3 - 0.5d0*k3
393 in_the_list = abs(xx - nint(xx)) <= eps .and. &
394 abs(yy - nint(yy)) <= eps .and. &
395 abs(zz - nint(zz)) <= eps
396 if (in_the_list) then
397 i = mod(nint(xkr(1)*nk1 - 0.5d0*k1 + 2*nk1), nk1) + 1
398 j = mod(nint(xkr(2)*nk2 - 0.5d0*k2 + 2*nk2), nk2) + 1
399 k = mod(nint(xkr(3)*nk3 - 0.5d0*k3 + 2*nk3), nk3) + 1
400 n = (k - 1) + (j - 1)*nk3 + (i - 1)*nk2*nk3 + 1
401 if (n > nk .and. equiv(n) == n) then
402 equiv(n) = nk
403 wkk(nk) = wkk(nk) + 1.0d0
404 end if
405 end if
406 if (time_reversal) then
407 xx = -xkr(1)*nk1 - 0.5d0*k1
408 yy = -xkr(2)*nk2 - 0.5d0*k2
409 zz = -xkr(3)*nk3 - 0.5d0*k3
410 in_the_list = abs(xx - nint(xx)) <= eps .and. abs(yy - nint(yy)) <= eps &
411 .and. abs(zz - nint(zz)) <= eps
412 if (in_the_list) then
413 i = mod(nint(-xkr(1)*nk1 - 0.5d0*k1 + 2*nk1), nk1) + 1
414 j = mod(nint(-xkr(2)*nk2 - 0.5d0*k2 + 2*nk2), nk2) + 1
415 k = mod(nint(-xkr(3)*nk3 - 0.5d0*k3 + 2*nk3), nk3) + 1
416 n = (k - 1) + (j - 1)*nk3 + (i - 1)*nk2*nk3 + 1
417 if (n > nk .and. equiv(n) == n) then
418 equiv(n) = nk
419 wkk(nk) = wkk(nk) + 1.0d0
420 end if
421 end if
422 end if
423 end do
424 end if
425 end do
426 end if
427
428 ! count irreducible points and order them
429 nks = 0
430 fact = 0.0d0
431 do nk = 1, nkr
432 if (equiv(nk) == nk) then
433 nks = nks + 1
434 wk(nks) = wkk(nk)
435 fact = fact + wk(nks)
436 ! bring back into to the first BZ
437 do i = 1, 3
438 xk(i, nks) = xkg(i, nk) - nint(xkg(i, nk))
439 end do
440 end if
441 end do
442 ! normalize weights to one
443 do nk = 1, nks
444 wk(nk) = wk(nk)/fact
445 end do
446
447 write (6, *) 'number of inequivalent/total k-points found', nks, nkp
448 ! set total number of k-points as the number of inequivalent ones
449 nkp = nks
450
451 deallocate (equiv)
452 deallocate (xkg, wkk)
453
454 return
455
456 end subroutine kpoint_grid

458 ! ----------------------------------------
459 !> @brief Generate k-points path for band structure calculations
460 !> @details Generates a continuous k-point path in the first Brillouin zone for band
461 !> structure calculations. Interpolates between user-specified endpoints to
462 !> create a path with specified resolution. Computes distances along the path.
463 !> @param[inout] xkp K-point coordinates (3,nk): input = path endpoints, output = full path
464 !> @param[inout] wkp K-point weights (nk): output = path distances
465 !> @param[in] nk1 Number of path segments (number of endpoints)
466 !> @param[in] nk2 Number of points per segment
467 !> @param[inout] iflag Error flag (0=success, 1=error)
468 !> @note Uses linear interpolation between endpoints. Adds last point explicitly.
469 !> @note Writes error to stdout if path generation fails.
470 !> @see get_kpoints
471 subroutine generate_kp_path(xkp, wkp, nk1, nk2, iflag)
472 ! ----------------------------------------
473 !
474 ! This subroutine generates a k-points path in the 1BZ according
475 ! to the points read in the section KPOINTS.
476
477 implicit none
478
479 integer, intent(in) :: nk1, nk2
480 integer, intent(inout) :: iflag
481 real(8), intent(out) :: xkp(3, nk), wkp(nk)
482 real(8), dimension(:, :), allocatable :: klines
483 integer :: i, j, k
484 real(8) :: dk, t, xpar(3), xold(3), wold, xmod
485
486 ! read lines of the path
487 allocate (klines(3, nk1))
488 do i = 1, nk1
489 klines(:, i) = xkp(:, i)
490 end do
491 xkp = 0.d0
492 wkp = 0.d0
493 dk = 1.d0/nk2
494 iflag = 0
495 ! generates k-points along the lines
496 ! using parametrization
497 do i = 1, nk1 - 1
498 t = 1.d0
499 do j = 1, nk2
500 if (t .lt. 0) then
501 iflag = 1
502 go to 123
503 end if
504
505 if (j .eq. 1) then
506 xold(:) = klines(:, i)
507 else
508 xold(:) = xkp(:, (i - 1)*nk2 + j - 1)
509 end if
510
511 wold = wkp((i - 1)*nk2 + j)
512 xpar(:) = t*(klines(:, i) - klines(:, i + 1)) + klines(:, i + 1)
513 xkp(:, (i - 1)*nk2 + j) = xpar(:)
514 xmod = dsqrt(sum((xpar(:) - xold(:))**2))
515 wkp((i - 1)*nk2 + j + 1) = wold + xmod
516 t = t - dk
517
518 end do
519 end do
520 ! add the last point
521 xkp(:, (nk1 - 1)*nk2 + 1) = klines(:, nk1)
522 !
523 !
524123 if (iflag .ne. 0) write (6, *) 'ERROR in generate k-points path'
525 deallocate (klines)
526
527 return
528
529 end subroutine generate_kp_path

531 !> @brief Check consistency between wavefunction phases and k-points
532 !> @details Verifies that the wavefunction phases in fort.10 files correspond to the
533 !> input k-points for all MPI processes. Ensures that the k-point grid and
534 !> the wavefunction phases are consistent across the pool. Calls error routine
535 !> if any mismatch is detected.
536 !> @param[in] id Process ID in the k-point pool
537 !> @param[in] comm MPI communicator
538 !> @param[in] rank MPI rank
539 !> @note Uses reduce_base_real for collective communication. Writes error and aborts
540 !> if phases do not match k-points.
541 !> @see error, reduce_base_real
542 subroutine check_kpoints(id, comm, rank)
543
544 implicit none
545 integer, intent(in) :: id, comm, rank
546 integer :: i, indk
547 real(8), dimension(:, :), allocatable :: all_phases, all_phases_down
548 real(8) :: phase_diff, phase_diff_down
549 logical :: is_same(3), is_same_do(3)
550
551 allocate (all_phases(3, nk), all_phases_down(3, nk))
552 all_phases = 0.d0
553 all_phases_down = 0.d0
554 indk = id + 1
555
556 all_phases(:, indk) = phase(:)
557 all_phases_down(:, indk) = phase_down(:)
558
559 call reduce_base_real(size(all_phases), all_phases, comm, -1)
560 call reduce_base_real(size(all_phases_down), all_phases_down, comm, -1)
561
562 do i = 1, nk
563
564 is_same(:) = ((abs(xkp(:, i) - all_phases(:, i))) < 1.0d-5)
565 is_same_do(:) = ((abs(xkp_down(:, i) - all_phases_down(:, i))) < 1.0d-5)
566
567 if ((.not. all(is_same)) .or. (.not. all(is_same_do))) then
568 phase_diff = sum(abs(xkp(:, i) - all_phases(:, i)))
569 phase_diff_down = sum(abs(xkp_down(:, i) - all_phases_down(:, i)))
570 call error('kpoints', ' Some wavefunction phases do not correspond to the &
571 & input k-points! Check &kpoints card of fort.10_000xxx ', 1, rank)
572 end if
