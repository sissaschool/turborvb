29!> @brief Report errors and warnings with MPI-aware handling
30!>
31!> This subroutine provides centralized error reporting for TurboRVB calculations.
32!> It handles both warnings (ierror < 0) and fatal errors (ierror > 0) with
33!> appropriate MPI coordination for parallel execution.
34!>
35!> @param[in] routin Name of the calling routine for error identification
36!> @param[in] messag Error or warning message to display
37!> @param[in] ierror Error code:
38!> - ierror < 0: Warning (program continues)
39!> - ierror = 0: No error (subroutine returns immediately)
40!> - ierror > 0: Fatal error (program terminates)
41!> @param[in] rank MPI rank of the calling process
42!>
43!> @details
44!> The subroutine performs the following operations:
45!> 1. For warnings (ierror < 0):
46!> - Displays warning message with routine name and error code
47!> - Only rank 0 outputs the message
48!> - Program continues execution
49!> 2. For fatal errors (ierror > 0):
50!> - Displays error message with routine name and error code
51!> - For ierror = 3, all ranks output their rank number
52!> - Otherwise, only rank 0 outputs the error
53!> - Terminates all MPI processes and stops the program
54!>
55!> @note The subroutine uses formatted output with separator lines for
56!> clear visual distinction of error messages
57!> @note For parallel execution, MPI_ABORT is called to ensure all processes
58!> terminate when a fatal error occurs
59!> @note Error code 3 is special and causes all ranks to output their rank number
60subroutine error(routin, messag, ierror, rank)
61 use allio, only: iflagerr
62 implicit none
63 ! the name of the calling routine
64 character(*) :: routin
65 ! the output message
66 character(*) :: messag
67 ! the error flag
68 integer :: ierror, ierr, info
69 integer rank, ierror_mpi
70
71#ifdef PARALLEL
72 include 'mpif.h'
73#endif
74
75 if (ierror == 0) return
76
77 if (ierror < 0) then
78 if (rank .eq. 0) then
79 write (*, *) ' '
80 write (*, '(1x,78(''%''))')
81 write (*, '(5x,''from '',a,'' : Warning #'',i10)') routin, ierror
82 write (*, '(5x,a)') 'Warning '//messag
83 write (*, '(1x,78(''%''))')
84 end if
85 else
86 if (rank .eq. 0 .or. ierror .eq. 3) then
87 if (ierror .eq. 3) then
88 write (*, *) ' From proc # ', rank
89 else
90 write (*, *) ' From Master '
91 end if
92 ! write(*,'(1x,78(''%''))')
93 write (*, '(5x,''from '',a,'' : ERROR #'',i10)') routin, ierror
94 write (*, '(5x,a)') 'ERROR '//messag
95 ! write(*,'(1x,78(''%''))')
96 end if
97 end if
98
99 if (ierror < 0) then
100 ! simple warning, the program goes on.
101 return
102
103 else
104 ! ! increment error counter
105 ! iflagerr=iflagerr+1
106
107 ! Fatal error, abort all MPI processes and stop the program.
108#ifdef PARALLEL
109 call mpi_abort(mpi_comm_world, 0, ierr)
110#endif
111 stop
112
113 end if
114
115end subroutine error

150!> @brief Global error checking and synchronization for parallel execution
151!>
152!> This subroutine provides global error checking across all MPI processes
153!> and ensures proper synchronization for error handling in parallel
154!> quantum Monte Carlo calculations.
155!>
156!> @param[in] iflagerr Local error flag for current process
157!> @param[in] rank MPI rank of the calling process
158!> @param[in] messag Error message to display if global error exists
159!>
160!> @details
161!> The subroutine performs the following operations:
162!> 1. Performs global reduction of error flags across all MPI processes
163!> 2. If any process has an error (iflagerrall != 0):
164!> - Rank 0 displays the error message
165!> - For unreliable networks, synchronizes with barrier
166!> - Frees MPI communicators if k-point averaging is active
167!> - Finalizes MPI and terminates all processes
168!> 3. If no global errors, subroutine returns normally
169!>
170!> @note This subroutine must be called by ALL MPI processes, not just rank 0
171!> @note Do not use this subroutine inside rank.eq.0 regions
172!> @note The subroutine handles both reliable and unreliable network configurations
173!> @note For k-point averaging calculations, it properly cleans up MPI communicators
174subroutine checkiflagerr(iflagerr, rank, messag)
175 use kpoints_mod, only: kaverage
176 use allio, only: commcolrep_mpi, commrep_mpi
177 implicit none
178 integer ierr, iflagerrall, iflagerr, rank
179 character(*) :: messag
180
181#ifdef PARALLEL
182 include 'mpif.h'
183#endif
184
185#ifdef PARALLEL
186 call mpi_allreduce(iflagerr, iflagerrall, 1, mpi_integer, mpi_sum, mpi_comm_world, ierr)
187#else
188 iflagerrall = iflagerr
189#endif
190
191 if (iflagerrall .ne. 0) then
192 if (rank .eq. 0) write (6, *) messag
193#ifdef PARALLEL
194#ifdef UNREL
195! For unreliable networks.
196 call mpi_barrier(mpi_comm_world, ierr)
197!$omp barrier
198#endif
199 if (kaverage) then
200 call mpi_comm_free(commrep_mpi, ierr)
201 call mpi_comm_free(commcolrep_mpi, ierr)
202 end if
203 call mpi_finalize(ierr)
204#endif
205 stop
206 end if
207
208 return
209
210end subroutine checkiflagerr
