29 SUBROUTINE zlasktrd( UPLO, MODE, N, NB, A, LDA, E, TAU, W, LDW )
30*
31* -- Written on 10/22/2010
32* Michael Wimmer, Universiteit Leiden
33* Derived from the LAPACK routine ZLATRD (www.netlib.org)
34* -- LAPACK auxiliary routine (version 3.2) --
35* -- LAPACK is a software package provided by Univ. of Tennessee, --
36* -- Univ. of California Berkeley, Univ. of Colorado Denver and NAG Ltd..--
37* November 2006
38*
39* .. Scalar Arguments ..
40 CHARACTER UPLO, MODE
41 INTEGER LDA, LDW, N, NB
42* ..
43* .. Array Arguments ..
44 DOUBLE PRECISION E( * )
45 DOUBLE COMPLEX A( LDA, * ), TAU( * ), W( LDW, * )
46* ..
47*
48* Purpose
49* =======
50*
51* ZLASKTRD reduces NB rows and columns of a complex skew-symmetric matrix A to
52* skew-symmetric tridiagonal form by a unitary congruence
53* transformation Q^H * A * Q^*, and returns the matrices V and W which are
54* needed to apply the transformation to the unreduced part of A.
55*
56* If UPLO = 'U', ZLASKTRD reduces the last NB rows and columns of a
57* matrix, of which the upper triangle is supplied;
58* if UPLO = 'L', ZLASKTRD reduces the first NB rows and columns of a
59* matrix, of which the lower triangle is supplied.
60*
61* Alternatively, the routine can also be used to compute a partial
62* tridiagonal form
63*
64* This is an auxiliary routine called by ZSKTRD.
65*
66* Arguments
67* =========
68*
69* UPLO (input) CHARACTER*1
70* Specifies whether the upper or lower triangular part of the
71* skew-symmetric matrix A is stored:
72* = 'U': Upper triangular
73* = 'L': Lower triangular
74*
75* MODE (input) CHARACTER*1
76* = 'N': A is fully tridiagonalized
77* = 'P': A is partially tridiagonalized for Pfaffian computation
78*
79* N (input) INTEGER
80* The order of the matrix A. N must be even if MODE = 'P'.
81*
82* NB (input) INTEGER
83* The number of rows and columns to be reduced.
84*
85* A (input/output) DOUBLE COMPLEX array, dimension (LDA,N)
86* On entry, the skew-symmetric matrix A.
87* If UPLO = 'U', the leading n-by-n upper triangular part
88* of A contains the upper triangular part of the matrix A,
89* and the strictly lower triangular part of A is not referenced.
90* If UPLO = 'L', the leading n-by-n lower triangular part
91* of A contains the lower triangular part of the matrix A,
92* and the strictly upper triangular part of A is not referenced.
93* On exit:
94* if UPLO = 'U', the last NB columns have been reduced to
95* tridiagonal form, with the diagonal elements overwriting
96* the diagonal elements of A; the elements above the diagonal
97* with the array TAU, represent the unitary matrix Q as a
98* product of elementary reflectors. If MODE = 'P' only the
99* even columns of the last NB columns contain meaningful values.
100* if UPLO = 'L', the first NB columns have been reduced to
101* tridiagonal form, with the diagonal elements overwriting
102* the diagonal elements of A; the elements below the diagonal
103* with the array TAU, represent the unitary matrix Q as a
104* product of elementary reflectors. If MODE = 'P' only the
105* odd columns of the first NB columns contain meaningful values.
106*
107* LDA (input) INTEGER
108* The leading dimension of the array A. LDA >= max(1,N).
109*
110* E (output) DOUBLE PRECISION array, dimension (N-1)
111* If UPLO = 'U', E(n-nb:n-1) contains the superdiagonal
112* elements of the last NB columns of the reduced matrix;
113* if UPLO = 'L', E(1:nb) contains the subdiagonal elements of
114* the first NB columns of the reduced matrix.
115*
116* TAU (output) DOUBLE COMPLEX array, dimension (N-1)
117* The scalar factors of the elementary reflectors, stored in
118* TAU(n-nb:n-1) if UPLO = 'U', and in TAU(1:nb) if UPLO = 'L'.
119* See Further Details.
120*
121* W (output) DOUBLE COMPLEX array, dimension (LDW,NB)
122* The n-by-nb matrix W required to update the unreduced part
123* of A.
124*
125* LDW (input) INTEGER
126* The leading dimension of the array W. LDW >= max(1,N).
127*
128* =====================================================================
129*
130* .. Parameters ..
131 DOUBLE COMPLEX ZERO, ONE
132 parameter( zero = ( 0.0d+0, 0.0d+0 ),
133 $ one = ( 1.0d+0, 0.0d+0 ))
134* ..
135* .. Local Scalars ..
136 INTEGER I, NW, NW2, STEP, NPANEL
137 DOUBLE COMPLEX ALPHA
138* ..
139* .. External Subroutines ..
140 EXTERNAL zgemv, zskmv, zlacgv, zlarfg
141* ..
142* .. External Functions ..
143 LOGICAL LSAME
144 EXTERNAL lsame
145* ..
146* .. Intrinsic Functions ..
147 INTRINSIC min, conjg
148* ..
149* .. Executable Statements ..
150*
151* Quick return if possible
152*
153 IF( n.LE.0 )
154 $ RETURN
155*
156
157 IF( lsame( mode, 'P' ) ) THEN
158 step = 2
159 ELSE
160 step = 1
161 END IF
162 npanel = nb * step
163
164 IF( lsame( uplo, 'U' ) ) THEN
165*
166* Reduce last NPANEL columns of upper triangle
167*
168 nw=0
169
170 DO 10 i = n, max(n - npanel + 1, 2), -1
171
172 nw2 = nw - mod(i,step)
173 IF( nw2 .GT. 0 ) THEN
174*
175* Update A(1:i,i)
176*
177 a( i, i ) = zero
178 CALL zgemv( 'No transpose', i, nw2, +one,
179 $ a( 1, n-(nw2-1)*step ), lda*step,
180 $ w( i, nb-nw2+1 ), ldw, one, a( 1, i ), 1 )
181 CALL zgemv( 'No transpose', i, nw2, -one,
182 $ w( 1, nb-nw2+1 ), ldw,
183 $ a( i, n-(nw2-1)*step ), lda*step,
184 $ one, a( 1, i ), 1 )
185 a( i, i ) = zero
186 END IF
187
188* In the Pfaffian mode, only zero all even columns
189 IF( step.EQ.2 .AND. mod( i, step ).EQ.1 ) THEN
190 tau( i-1 ) = zero
191 GOTO 10
192 END IF
193
194 IF( i.GT.1 ) THEN
195*
196* Generate elementary reflector H(i) to annihilate
197* A(1:i-2,i)
198*
199 alpha = a( i-1, i )
200 CALL zlarfg( i-1, alpha, a( 1, i ), 1, tau( i-1 ) )
201 e( i-1 ) = alpha
202 a( i-1, i ) = one
203*
204* Compute W(1:i-1,i)
205*
206* We need to multiply with v^*
207 CALL zlacgv( i-1, a( 1, i ), 1)
208
209 CALL zskmv( 'Upper', i-1, conjg(tau(i-1)), a, lda,
210 $ a( 1, i ), 1, zero, w( 1, nb-nw ), 1 )
211 IF( nw .GT. 0 ) THEN
212 CALL zgemv( 'Transpose', i-1, nw, one,
213 $ w( 1, nb-nw+1 ), ldw, a( 1, i ), 1, zero,
214 $ w( i+1, nb-nw ), 1 )
215 CALL zgemv( 'No transpose', i-1, nw, conjg(tau(i-1)),
216 $ a( 1, n-(nw-1)*step ), lda*step,
217 $ w( i+1, nb-nw ), 1, one,
218 $ w( 1, nb-nw ), 1 )
219 CALL zgemv( 'Transpose', i-1, nw, one,
220 $ a( 1, n-(nw-1)*step ), lda*step,
221 $ a( 1, i ), 1, zero,
222 $ w( i+1, nb-nw ), 1 )
223 CALL zgemv( 'No transpose', i-1, nw,
224 $ -conjg(tau(i-1)),
225 $ w( 1, nb-nw+1 ), ldw,
226 $ w( i+1, nb-nw ), 1,
227 $ one, w( 1, nb-nw ), 1 )
228 END IF
229
230* Undo complex conjugation
231 CALL zlacgv( i-1, a( 1, i ), 1)
232
233* One more complete entry in W
234 nw = nw + 1
235
236 END IF
237
238* Note: setting A(I-1,I) back to alpha happens in the calling routine
239
240 10 CONTINUE
241 ELSE
242*
243* Reduce first NPANEL columns of lower triangle
244*
245 nw = 0
246
247 DO 20 i = 1, min(npanel, n-1)
248*
249* Update A(i:n,i)
250*
251 nw2 = nw - mod(i+1,step)
252 IF( nw2 .GT. 0 ) THEN
253 a( i, i ) = zero
254 CALL zgemv( 'No transpose', n-i+1, nw2, +one, a( i, 1 ),
255 $ lda*step, w( i, 1 ), ldw, one, a( i, i ), 1 )
256 CALL zgemv( 'No transpose', n-i+1, nw2, -one, w( i, 1 ),
257 $ ldw, a( i, 1 ), lda*step, one, a( i, i ), 1 )
258 a( i, i ) = zero
259 END IF
260
261* In the Pfaffian mode, only zero all odd columns
262 IF( step.EQ.2 .AND. mod( i, step ).EQ.0 ) THEN
263 tau( i ) = zero
264 GOTO 20
265 END IF
266
267 IF( i.LT.n ) THEN
268*
269* Generate elementary reflector H(i) to annihilate
270* A(i+2:n,i)
271*
272 alpha = a( i+1, i )
273 CALL zlarfg( n-i, alpha, a( min( i+2, n ), i ), 1,
274 $ tau( i ) )
275 e( i ) = alpha
276 a( i+1, i ) = one
277*
278* Compute W(i+1:n,i)
279* This is given by tau A^(i)*v^*=tau(A*v^* + VW^T v^* - WV^T v^*)
280
281* complex conjugate of v
282 CALL zlacgv( n-i, a( i+1, i ), 1 )
283
284 CALL zskmv( 'Lower', n-i, conjg(tau( i )),
285 $ a( i+1, i+1 ), lda,
286 $ a( i+1, i ), 1, zero, w( i+1, nw+1 ), 1 )
287 IF( nw .GT. 0 ) THEN
288 CALL zgemv( 'Transpose', n-i, nw, one,
289 $ w( i+1, 1 ), ldw, a( i+1, i ), 1, zero,
290 $ w( 1, nw+1 ), 1 )
291 CALL zgemv( 'No transpose', n-i, nw, conjg(tau( i )),
292 $ a( i+1, 1 ), lda*step, w( 1, nw+1 ), 1,
293 $ one, w( i+1, nw+1 ), 1 )
294 CALL zgemv( 'Transpose', n-i, nw, one,
295 $ a( i+1, 1 ), lda*step, a( i+1, i ), 1, zero,
296 $ w( 1, nw+1 ), 1 )
297 CALL zgemv( 'No transpose', n-i, nw,
298 $ -conjg(tau( i )),
299 $ w( i+1, 1 ), ldw, w( 1, nw+1 ), 1,
300 $ one, w( i+1, nw+1 ), 1 )
301 END IF
302
303* Undo complex conjugation
304 CALL zlacgv( n-i, a( i+1, i ), 1 )
305
306* One more complete entry in W
307 nw = nw + 1
308 END IF
309*
310 20 CONTINUE
311 END IF
312*
313 RETURN
314*
315* End of ZLASKTRD
316*
317 END
