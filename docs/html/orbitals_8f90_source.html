77 !> @brief Read orbital definitions from input file
78 !> @details This subroutine reads orbital definitions for both determinant
79 !> and Jastrow orbitals from the input file. It processes each
80 !> atom's orbital shells, counts the total number of orbitals
81 !> and parameters, and allocates the necessary arrays.
82 !>
83 !> The subroutine handles:
84 !> - Determinant orbitals (detorb)
85 !> - Jastrow orbitals (jasorb)
86 !> - Shell counting and parameter estimation
87 !> - Atomic Jastrow matrices
88 !> - Hybrid orbital generation
89 !>
90 !> @param[in] natoms Number of atoms
91 !> @param[in] nel Number of electrons
92 !> @param[in] noonebody Logical flag to exclude one-body terms
93 !> @param[in] zeta Zeta parameters for atoms (2, natoms)
94 !> @param[out] J3_off Logical array disabling J3 terms for specific atoms
95 !> @param[out] cut_hybrid Array specifying hybrid orbital cuts per atom
96 !> @param[out] detorb Determinant orbitals array
97 !> @param[out] ndetorb Number of determinant orbitals
98 !> @param[out] totshelldet Total number of determinant shells
99 !> @param[out] ndetpar Number of determinant parameters
100 !> @param[out] jasorb Jastrow orbitals array
101 !> @param[out] njasorb Number of Jastrow orbitals
102 !> @param[out] totshelljas Total number of Jastrow shells
103 !> @param[out] njaspar Number of Jastrow parameters
104 !> @param[in] orbtype Orbital type for determinant
105 !> @param[in] jorbtype Orbital type for Jastrow
106 !> @param[in] funit Input file unit
107 !> @param[in] shiftbeta Beta shift parameter
108 !> @param[in] complexfort10 Logical flag for complex fort.10
109 !> @param[in] symmagp Logical flag for Hermitian constraints
110 !> @param[in] ncell Number of cells
111 !> @param[in] scale_jasfat Scaling factor for Jastrow fatness
112 subroutine read_orbitals(natoms, nel, noonebody, zeta, J3_off, cut_hybrid, detorb, ndetorb&
113 &, totshelldet, ndetpar, jasorb, njasorb, totshelljas, njaspar, orbtype, jorbtype&
114 &, funit, shiftbeta, complexfort10, symmagp, ncell, scale_jasfat)
115 implicit none
116 integer, intent(in) :: funit, natoms, nel, shiftbeta, ncell
117 real(8), intent(in) :: zeta(2, natoms)
118 logical, intent(out) :: J3_off(natoms)
119 integer, intent(out) :: cut_hybrid(natoms)
120 integer, intent(out) :: ndetorb, totshelldet, ndetpar, njasorb, totshelljas, njaspar
121 type(orbital), pointer :: detorb(:), jasorb(:)
122 character(20), intent(in) :: orbtype, jorbtype
123 integer :: i1, i, j, ii, ix, iy, ishelljas, ishelldet
124 integer :: iorbd, iorbj, iorbj_before, orbj_atom, nunc, nuncj, ncost
125 character(6) :: intc
126 logical :: complexfort10, symmagp
127 logical, intent(in) :: noonebody
128 real*8 jascost, jasfat, scale_jasfat
129 yesmolat = .false.
130 yesmolatj = .false.
131 ndetpar = 0
132 njaspar = 0
133 ndetorb = 0
134 njasorb = 0
135 totshelljas = 0
136 totshelldet = 0
137 yesalloc_jas = .false.
138
139 do i1 = 1, natoms
140 write (6, *) ' atom number =', nint(zeta(1, i1)), i1, orbtype
141 !
142 nshelldet = 0
143 nshelljas = 0
144 njas_hyb = 0
145 ndet_hyb = 0
146 ncut_hyb = -1
147 no_3body_jas = .false.
148 !
149 section_name = "ATOM_"//trim(intc(zeta(1, i1)))
150 call findsection(funit, section_name)
151 read (funit, nml=shells)
152 if (nshelljas .lt. 0) then
153 nshelljas = -nshelljas
154 yesalloc_jas = .true.
155 end if
156 totshelldet = totshelldet + nshelldet + ndet_hyb
157 totshelljas = totshelljas + nshelljas + njas_hyb
158 j3_off(i1) = no_3body_jas
159 cut_hybrid(i1) = ncut_hyb
160 ! if(njas_hyb.gt.0) call errore('read_orbitals',"the automatical &
161 ! &generation of hybrid jastrow orbitals hasn't been implemented!",1)
162
163 ! computing dimension of the basis for DET and JAS
164 call counting_orb(funit, nshelldet, ndet_hyb, orbtype, ndetorb, ndetpar, complexfort10, ipf)
165
166 if (nshelljas .gt. 0) then
167 read (funit, *) ! read a comment line between detorb and jasorb
168 write (6, *) ' nshelljas, njas_hyb: read =', nshelljas, njas_hyb
169 call counting_orb(funit, nshelljas, njas_hyb, jorbtype, njasorb, njaspar, .false., 1)
170 else
171 if (njas_hyb .gt. 0) call errore('read_orbitals', 'No jastrow shell, no jastrow hybrid!', 1)
172 end if
173 !
174 end do
175
176 write (*, *) ' # Det Shells : ', totshelldet
177 write (*, *) ' # Jas Shells : ', totshelljas
178 write (*, *) ' # Det Orbitals : ', ndetorb
179 write (*, *) ' # Jas Orbitals : ', njasorb
180 write (*, *) ' # Det Parameters : ', ndetpar
181 write (*, *) ' # Jas Parameters : ', njaspar
182
183 allocate (detorb(ndetorb))
184 if (totshelljas .gt. 0) allocate (jasorb(njasorb))
185 if (yesalloc_jas) then
186 if (noonebody) then
187 allocate (atomic_jasmat(njasorb, njasorb))
188 else
189 allocate (atomic_jasmat(njasorb + 1, njasorb + 1))
190 end if
191 atomic_jasmat = 0.d0
192 end if
193
194 ishelldet = 0
195 ishelljas = 0
196 iorbd = 1
197 iorbj = 1
198 nunc = 0
199 nuncj = 0
200 do i1 = 1, natoms
201 write (6, *) ' atom number =', i1
202 !
203 nshelldet = 0
204 nshelljas = 0
205 njas_hyb = 0
206 ndet_hyb = 0
207 no_3body_jas = .false.
208 yesjas_atom = .false.
209 !
210 section_name = "ATOM_"//trim(intc(zeta(1, i1)))
211 call findsection(funit, section_name)
212 read (funit, nml=shells, err=105, end=105)
213 if (nshelljas .lt. 0) then
214 yesjas_atom = .true.
215 yesalloc_jas = .true.
216 nshelljas = -nshelljas
217 end if
218 ! reading the basis set and filling detorb and jasorb
219 call read_shells(detorb, iorbd, nunc, ishelldet, nshelldet, ndet_hyb, orbtype, i1, funit&
220 &, shiftbeta, yesmolat, complexfort10, ipf, symmagp)
221 if (nshelljas .gt. 0) then
222 if (scale_jasfat .ne. 0.d0) then
223 jasfat = (scale_jasfat - 1.d0)/dble(nel*ncell - 1) ! For H we take the H2 Jastrow
224
225 else
226 jasfat = max((zeta(2, i1) - 1.d0), 1.d0)/dble(nel*ncell - 1) ! For H we take the H2 Jastrow
227 end if
228
229 iorbj_before = iorbj - 1
230 read (funit, *, err=105, end=105) ! read a comment line between detorb and jasorb
231 call read_shells(jasorb, iorbj, nuncj, ishelljas, nshelljas, njas_hyb, jorbtype, i1, funit&
232 &, shiftbeta, yesmolatj, .false., 1, symmagp)
233 if (yesjas_atom) then
234 read (funit, *, err=105, end=105) ! read a comment line
235 read (funit, *) nnozero, ncost
236 if (ncost .eq. -1) then ! all one body term
237 do ii = 1, nnozero
238 read (funit, *) ix, jascost
239 atomic_jasmat(ix + iorbj_before, njasorb + 1) = jascost*jasfat
240 atomic_jasmat(njasorb + 1, ix + iorbj_before) = jascost*jasfat
241 end do
242 elseif (ncost .eq. 0) then ! all two body
243 do ii = 1, nnozero
244 read (funit, *) ix, iy, jascost
245 atomic_jasmat(ix + iorbj_before, iy + iorbj_before) = jascost
246 end do
247 else ! one body and two body
248 do ii = 1, nnozero
249 read (funit, *) ix, iy, jascost
250 if (iy .ne. ncost) then
251 atomic_jasmat(ix + iorbj_before, iy + iorbj_before) = jascost
252 atomic_jasmat(iy + iorbj_before, ix + iorbj_before) = jascost
253 else
254 atomic_jasmat(ix + iorbj_before, njasorb + 1) = jascost*jasfat
255 atomic_jasmat(njasorb + 1, ix + iorbj_before) = jascost*jasfat
256 end if
257 end do
258 end if
259! HERE load atomic_jasmat with the right scaling
260 end if
261 end if
262 !
263 ishelldet = ishelldet + nshelldet + ndet_hyb
264 ishelljas = ishelljas + nshelljas + njas_hyb
265 !
266 end do
267 if (ipf .eq. 2) then
268 do i = 1, ndetorb
269 if (detorb(i)%ioptorb .eq. 900000) then
270 do j = detorb(i)%nparm/4 + 1, detorb(i)%nparm/2
271 detorb(i)%parms(j) = detorb(i)%parms(j) + nunc*ncell
272 end do
273 end if
274 end do
275 end if
276
277 return
278105 call errore("read_orbitals", " Error reading orbitals ! ", 1)
279 end subroutine read_orbitals

286 !> @brief Count orbitals and parameters from input file
287 !> @details This subroutine reads orbital definitions from the input file
288 !> and counts the total number of orbitals and parameters needed
289 !> for both determinant and Jastrow orbitals.
290 !>
291 !> The subroutine handles:
292 !> - Different orbital types (normal, tempered, mixed)
293 !> - Complex coefficients for contracted orbitals
294 !> - Hybrid orbital counting
295 !> - Parameter validation
296 !>
297 !> @param[in] funit Input file unit
298 !> @param[in] nshell Number of shells
299 !> @param[in] nhyb Number of hybrid orbitals
300 !> @param[in] orbtype Orbital type string
301 !> @param[in,out] norb Number of orbitals (output)
302 !> @param[in,out] npar Number of parameters (output)
303 !> @param[in] complexfort10 Logical flag for complex fort.10
304 !> @param[in] ipf Pfaffian flag
305 subroutine counting_orb(funit, nshell, nhyb, orbtype, norb, npar, complexfort10, ipf)
306 implicit none
307 integer, intent(in) :: funit, nshell, nhyb, ipf
308 integer, intent(inout) :: norb, npar
309 character(20), intent(in) :: orbtype
310 !
311 integer :: atomicnparm, nparm, nparm2, i2, i3, multi, ioptorb
312 character(200) :: linedata
313 !
314 integer :: tempint
315 double precision :: tempdb
316 integer, external :: check_multioptorb
317 logical complexfort10, cmplx_coeff
318
319 ! COMPLEX DEB
320 ! complex coefficients implemented only for 'normal' orbtype, so far...
321
322 atomicnparm = 0
323 cmplx_coeff = .false.
324 !
325 do i2 = 1, nshell
326 read (funit, "(a)", err=100) linedata
327 read (linedata, *, err=100) multi, nparm, ioptorb ! reading multiplicity and orbital type
328 nparm2 = 0
329 ! check to have a correct number of parameters
330 if (.not. (mod(nparm, 2) .eq. 0 .or. nparm .eq. 1)) then
331 call errore("counting_orb", " ERROR number of parameters (nparm) must be even ! ", 1)
332 end if
333 if (complexfort10 .and. nparm .gt. 1) cmplx_coeff = .true.
334 !
335 if (orbtype == "mixed" .and. ioptorb .ne. 900000) then
336 read (linedata, *, err=100) multi, nparm, ioptorb, nparm2
337 if (check_multioptorb(multi, ioptorb, 'W') .eq. 1)&
338 &write (6, *) ' Pay attention to the multiplicity!'
339 write (6, *) ' shell number =', i2, multi, nparm, nparm2
340 else
341 if (check_multioptorb(multi, ioptorb, 'W') .eq. 1)&
342 &write (6, *) ' Pay attention to the multiplicity!'
343 write (6, *) ' shell number =', i2, multi, nparm
344 end if
345
346 if (orbtype == "tempered" .and. ioptorb .ne. 900000) then
347 read (funit, *, err=100) tempint, tempdb, tempdb
348 elseif (orbtype == "mixed" .and. nparm2 .gt. 0 .and. ioptorb .ne. 900000) then
349 read (funit, *, err=100) tempint, (tempdb, i3=1, nparm + 2)
350 else
351 ! orbtype=='normal'
352 if (.not. cmplx_coeff .or. real_contracted) then
353 read (funit, *, err=100) tempint, (tempdb, i3=1, nparm)
354 else ! contracted coefficients are considered complex
355 ! while exponents remain real.
356 ! in the case of uncontracted basis only detmat is complex
357 read (funit, *, err=100) tempint, (tempdb, i3=1, nparm*3/2)
358 end if
359 end if
360 if (ioptorb .ne. 900000) atomicnparm = atomicnparm + (max(nparm/2, 1)*2 + nparm2)*multi
361 if (ioptorb .eq. 900000 .and. ipf .eq. 2) then
362 npar = npar + nparm
363 end if
364 nparm = nparm + nparm2
365 npar = npar + nparm
366 norb = norb + multi
367 end do
368 !
369 ! adding hybrid orbitals
370
371 npar = npar + atomicnparm*nhyb*ipf
372 norb = norb + nhyb
373 !
374 return
375100 call errore("counting_orb", " ERROR reading orbitals ! ", 1)
376 end subroutine counting_orb

378 ! -----------------------------------------------------------------------
379 ! This routine stores all information related to orbitals of determinant
380 ! and Jastrow, such as orbitals types, # of parameters and so on.
381 ! -----------------------------------------------------------------------
382
383 !> @brief Read orbital shells and store orbital information
384 !> @details This subroutine reads orbital shell definitions from the input file
385 !> and stores all information related to orbitals including types,
386 !> parameters, and transformation vectors.
387 !>
388 !> The subroutine handles:
389 !> - Different orbital types (s, p, d, f, g)
390 !> - Parameter reading and validation
391 !> - Hybrid orbital generation
392 !> - Transformation vector setup
393 !> - Complex coefficient handling
394 !>
395 !> @param[out] orbitals Array of orbital structures
396 !> @param[in,out] iorb Current orbital index
397 !> @param[in,out] nunc Number of uncontracted orbitals
398 !> @param[in] ishell Starting shell index
399 !> @param[in] nshell Number of shells
400 !> @param[in] nhyb Number of hybrid orbitals
401 !> @param[in] orbtype Orbital type string
402 !> @param[in] kion Atom index
403 !> @param[in] funit Input file unit
404 !> @param[in] shiftbeta Beta shift parameter
405 !> @param[in] yesmolat Logical flag for molecular orbitals
406 !> @param[in] complexfort10 Logical flag for complex fort.10
407 !> @param[in] ipf Pfaffian flag
408 !> @param[in] symmagp Logical flag for Hermitian constraints
409 subroutine read_shells(orbitals, iorb, nunc, ishell, nshell, nhyb, orbtype, kion, funit&
410 &, shiftbeta, yesmolat, complexfort10, ipf, symmagp)
411 implicit none
412 integer, intent(in) :: funit, nshell, nhyb, kion, ishell, shiftbeta, ipf
413 character(20), intent(in) :: orbtype
414 logical :: yesmolat, yeshyb
415 integer :: iorb, indocc, nuncr, nunc
416 type(orbital) :: orbitals(*)
417 integer :: i1, i2, nparm, c1
418 integer :: shellatom, inorbatom
419 integer :: nparm2
420 double precision :: alpha, beta
421 double precision, allocatable :: tpar(:)
422 character(200) :: linedata
423 integer, external :: check_multioptorb
424 logical complexfort10, cmplx_coeff, symmagp
425
426 nuncr = nunc
427 shellatom = 0
428 inorbatom = 0
429 cmplx_coeff = .false.
430 yeshyb = .false.
431 c1 = iorb
432
433 do i1 = 1, nshell + nhyb
434 if (i1 <= nshell) then
435 read (funit, "(a)", err=101) linedata
436 read (linedata, *, err=101) orbitals(c1)%itype, nparm, orbitals(c1)%ioptorb
437 ! first line of each orbital section:
438 ! itype = s,p,d... (multiplicity)
439 ! nparm = # of parameters: exponents + contracted coefficients
440 ! ioptorb = functional form of the orbital
441 nparm2 = 0
442 if (orbtype == "mixed" .and. orbitals(c1)%ioptorb .ne. 900000) then
443 read (linedata, *, err=101) orbitals(c1)%itype, nparm, orbitals(c1)%ioptorb, nparm2
444 end if
445
446 if (check_multioptorb(orbitals(c1)%itype, orbitals(c1)%ioptorb, 'W') .eq. 1)&
447 &write (6, *) ' Pay attention to the multiplicity!'
448
449 if (.not. (mod(nparm, 2) == 0 .or. nparm == 1)) &
450 & call errore("read_shells", " Error nparm must be even!", 1)
451 if (mod(nparm2, 2) .ne. 0) &
452 & call errore("read_shells", " Error when orbtype=mixed nparm2 must be even!", 1)
453
454 if (complexfort10 .and. nparm .gt. 1) cmplx_coeff = .true.
455 if (orbitals(c1)%ioptorb .ne. 900000 .or. ipf .eq. 1) then
456 if (.not. cmplx_coeff) then
457 allocate (orbitals(c1)%parms(nparm + nparm2))
458 else
459 allocate (orbitals(c1)%parms(nparm*3/2 + nparm2))
460 end if
461 else ! below ipf=2 & orbital=900000
462 if (complexfort10) then
463 allocate (orbitals(c1)%parms(3*nparm))
464 else
465 allocate (orbitals(c1)%parms(2*nparm))
466 end if
467 end if
468 !
469 !
470 orbitals(c1)%parms(:) = 0.d0
471 if (orbitals(c1)%ioptorb .ne. 900000) then
472 nunc = nunc + (max(nparm/2, 1) + nparm2/2)*orbitals(c1)%itype ! gaussians exponents
473 end if
474
475 !!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!
476 ! COMPLEX MODIF : complex coefficients implemented only for orbital type 'normal'
477 ! so far. Necessary extension to 'tempered' and 'mixed' types. Eventually to hybrids too!
478 !!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!
479
480 if (orbtype == "tempered" .and. orbitals(c1)%ioptorb .ne. 900000) then
481 !
482 read (funit, *, err=101) orbitals(c1)%kion, alpha, beta
483 call adjustbeta(nparm, alpha, beta, shiftbeta)
484 call defz(nparm/2, alpha, beta, orbitals(c1)%parms)
485
486 if (cmplx_coeff) then
487 indocc = mod(2*(i1 - 1), nparm/2) + 1
488 else
489 indocc = mod(i1 - 1, nparm/2) + 1
490 end if
491
492 orbitals(c1)%parms(nparm/2 + indocc) = 1.d0
493 elseif (orbtype == "mixed" .and. nparm2 .gt. 0 .and. orbitals(c1)%ioptorb .ne. 900000) then
494 allocate (tpar(nparm))
495 read (funit, *, err=101) orbitals(c1)%kion, (tpar(i2), i2=1, nparm), alpha, beta
496 call adjustbeta(nparm2, alpha, beta, shiftbeta)
497 call defz(nparm2/2, alpha, beta, orbitals(c1)%parms(nparm/2 + 1))
498
499 do i2 = 1, nparm/2
500 orbitals(c1)%parms(i2) = tpar(i2)
501 orbitals(c1)%parms((nparm + nparm2)/2 + i2) = tpar(nparm/2 + i2)
502 end do
503
504 if (nparm .eq. 0) then
505 indocc = mod(i1 - 1, nparm2/2) + 1
506 orbitals(c1)%parms(nparm2/2 + indocc) = 1.d0
507 elseif (sum(abs(tpar(nparm/2 + 1:nparm))) .eq. 0) then
508 indocc = mod(i1 - 1, nparm2/2) + 1
509 orbitals(c1)%parms(nparm2/2 + nparm + indocc) = 1.d0
510 end if
511
512 nparm = nparm + nparm2
513 deallocate (tpar)
514 else ! orbtype='normal'
515 if (orbitals(c1)%ioptorb .ne. 900000 .or. ipf .eq. 1) then
516 if (.not. cmplx_coeff) then
517 read (funit, *, err=101) orbitals(c1)%kion, (orbitals(c1)%parms(i2), i2=1, nparm)
518 elseif (real_contracted) then
519 read (funit, *, err=101) orbitals(c1)%kion, (orbitals(c1)%parms(i2), i2=1, nparm/2) &
520 , (orbitals(c1)%parms(i2), i2=nparm/2 + 1, nparm*3/2, 2)
521 do i2 = nparm/2 + 2, nparm*3/2, 2
522 orbitals(c1)%parms(i2) = 0.d0
523 end do
524 else
525 read (funit, *, err=101) orbitals(c1)%kion, (orbitals(c1)%parms(i2), i2=1, nparm*3/2)
526 end if
527 else ! below ipf=2 and hybrid orbital
528 if (.not. cmplx_coeff) then
529 read (funit, *, err=101) orbitals(c1)%kion, (orbitals(c1)%parms(i2), i2=1, nparm/2) &
530 , (orbitals(c1)%parms(i2), i2=nparm + 1, nparm + nparm/2)
531 do i2 = 1, nparm/2
532 orbitals(c1)%parms(nparm/2 + i2) = orbitals(c1)%parms(i2)
533 orbitals(c1)%parms(nparm + nparm/2 + i2) = orbitals(c1)%parms(i2 + nparm)
534 end do
535 else
536 if (real_contracted) then
537 read (funit, *, err=101) orbitals(c1)%kion, (orbitals(c1)%parms(i2), i2=1, nparm/2) &
538 , (orbitals(c1)%parms(i2), i2=nparm + 1, 2*nparm, 2)
539 do i2 = nparm + 2, 2*nparm, 2
540 orbitals(c1)%parms(i2) = 0.d0
541 end do
542 else
