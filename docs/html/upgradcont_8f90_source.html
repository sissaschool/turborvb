16!> @brief Update gradient with cutoff for DMC calculations
17!> @details Computes and updates gradients with cutoff for each electron.
18!> Implements the gradient cutoff scheme based on proximity to nuclei
19!> and applies scaling factors for DMC calculations. Uses Eq.36 from
20!> Umrigar J.Chem.Phys. vol.99 p.2866 for cutoff calculations.
21!> @param[out] gradpsibar Gradient with cutoff for each electron
22!> @param[out] gradpsi Raw gradient for each electron
23!> @param[in] indt Index offset for derivatives
24!> @param[in] nelup Number of up electrons
25!> @param[in] neldo Number of down electrons
26!> @param[in] winvup Winv array for up electrons
27!> @param[in] winvdo Winv array for down electrons
28!> @param[in] tabpip Table of pip values
29!> @param[in] ttry Trial parameter for scaling
30!> @param[out] gradtotbar Total gradient with cutoff
31!> @param[out] gradtot Total raw gradient
32!> @param[in] rcart Cartesian coordinates
33!> @param[in] dist Distance array
34!> @param[in] rion Ion positions
35!> @param[in] nion Number of ions
36!> @param[in] zeta Zeta array
37!> @param[in] LBox Box length for periodic boundary conditions
38subroutine upgradcont(gradpsibar, gradpsi, indt, nelup, neldo &
39 &, winvup, winvdo, tabpip, ttry, gradtotbar, gradtot, rcart, dist &
40 &, rion, nion, zeta, LBox)
41 use cell
42 use allio, only: zetamin, distmin
43 implicit none
44 integer i, jj, jn, iout, indt, nelup, neldo, ion, nion, nel
45 real*8 winvup(nelup, *), winvdo(neldo, *), tabpip(nelup + neldo, *) &
46 &, gradpsibar(3, *), gradpsi(3, *), ttry, dist(nion, nelup + neldo) &
47 &, gradtot, gradtotbar, gradpsimod, rion(3, *), rdiff(3) &
48 &, rdiffmod, cutoff, zeta(*), rdiffmod2, gradpsimod2, scale &
49 &, gradpsibarmod2
50 real*8 rcart(3, *), lbox, sdiff(3)
51
52 nel = nelup + neldo
53 gradtotbar = 0.d0
54 gradtot = 0.d0
55
56 do iout = 1, nel
57
58 ! compute gradient for the particle iout
59 if (iout .le. nelup) then
60 do jn = 1, 3
61 jj = indt + jn
62 gradpsi(jn, iout) = tabpip(iout, jj) + winvup(iout, jj)
63 end do
64 else
65 do jn = 1, 3
66 jj = indt + jn
67 gradpsi(jn, iout) = tabpip(iout, jj) + winvdo(iout - nelup, jj)
68 end do
69 end if
70
71 gradpsimod2 = 0.d0
72 do jn = 1, 3
73 gradpsimod2 = gradpsimod2 + gradpsi(jn, iout)**2
74 end do
75 ! updating the modulus square of total gradient
76 gradtot = gradtot + gradpsimod2
77 gradpsimod = sqrt(gradpsimod2)
78
79 ! find the closest nucleus ion to the iout electron
80 ion = 1
81 distmin(iout) = dist(ion, iout)
82 do jn = 2, nion
83 if (dist(jn, iout) .lt. distmin(iout)) then
84 ion = jn
85 distmin(iout) = dist(jn, iout)
86 end if
87 end do
88 zetamin(iout) = zeta(ion)
89
90 ! rdiff vector connecting ion nucleus to iout electron
91 rdiffmod2 = 0.d0
92 rdiff(:) = rcart(:, iout) - rion(:, ion)
93 if (lbox .gt. 0.d0) then
94 call applypbc(rdiff, 1)
95 end if
96 rdiffmod2 = sum(rdiff(:)**2)
97 rdiffmod = sqrt(rdiffmod2)
98
99 ! compute gradpsibar (with cutoff)
100 cutoff = 0.d0
101 do jn = 1, 3
102 cutoff = cutoff + gradpsi(jn, iout)*rdiff(jn)
103 end do
104 ! Eq.36 Umrigar J.Chem.Phys. vol.99 p.2866
105 cutoff = cutoff/rdiffmod/gradpsimod
106 cutoff = 0.5d0*(1.d0 + cutoff) + zeta(ion)**2*rdiffmod2/10.d0 &
107 & /(4.d0 + zeta(ion)**2*rdiffmod2)
108
109 scale = (sqrt(1.d0 + 4.d0*cutoff*gradpsimod2*ttry) - 1.d0) &
110 & /cutoff/2.d0/gradpsimod2/ttry
111
112 gradpsibarmod2 = 0.d0
113 do jn = 1, 3
114 gradpsibar(jn, iout) = scale*gradpsi(jn, iout)
115 gradpsibarmod2 = gradpsibarmod2 + gradpsibar(jn, iout)**2
116 end do
117
118 ! updating the modulus square of total cutoff gradient
119 gradtotbar = gradtotbar + gradpsibarmod2
120
121 end do
122
123 return
124end
