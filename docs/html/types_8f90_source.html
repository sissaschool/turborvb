16!> @brief Module defining derived data types for TurboRVB
17!>
18!> This module contains the definition of various derived data types used
19!> throughout the TurboRVB codebase for managing arrays, ion components,
20!> k-point grids, and wave function factors.
21!>
22!> Types
23!> -----
24!> ARRAY_INT : Dynamic integer array
25!> Contains an allocatable integer array for flexible storage.
26!>
27!> ION_COMP : Ion component information
28!> Stores multiplicity and arrays of ion indices and components.
29!>
30!> NKGRID : k-point grid structure
31!> Manages k-point grid dimensions and processing status.
32!>
33!> WF_FACTOR : Wave function factor object
34!> Contains parameters and matrices for Jastrow/determinant factors.
35!>
36!> Notes
37!> -----
38!> - All types use allocatable arrays for dynamic memory management.
39!> - The WF_FACTOR type is used for both Jastrow and determinant factors.
40!> - Provides a structured approach to data organization in TurboRVB.
41!>
42!> Example
43!> -------
44!> Used throughout TurboRVB for structured data management and wave function handling.
45module types
46 implicit none
47
48!> @brief Dynamic integer array type
49!>
50!> A simple wrapper for an allocatable integer array, providing
51!> a structured way to handle dynamic integer arrays.
52 type array_int
53 integer, allocatable :: col(:) !< Allocatable integer array
54 end type array_int
55
56!> @brief Ion component information type
57!>
58!> Stores information about ion components including multiplicity
59!> and arrays of ion indices and their corresponding components.
60 type ion_comp
61 integer :: mult !< Multiplicity of the component
62 integer, allocatable :: ion(:) !< Array of ion indices
63 integer, allocatable :: comp(:) !< Array of component values
64 end type ion_comp
65
66!> @brief k-point grid structure type
67!>
68!> Manages k-point grid information including grid dimensions,
69!> k-point indices, and processing status flags.
70 type nkgrid
71 integer :: dimshell !< Dimension of the shell
72 integer, allocatable :: kpip(:, :) !< k-point indices (2D array)
73 logical, allocatable :: tobedone(:) !< Processing status flags
74 end type nkgrid
75
76!> @brief Wave function factor object type
77!>
78!> Comprehensive data structure for storing wave function factor
79!> parameters and matrices. Used for both Jastrow and determinant factors.
80!> Contains two-body parameters, exponents, and various matrices for
81!> different spin configurations.
82 type wf_factor
83 real(8), allocatable :: twobody_par(:) !< Two-body parameters
84 real(8), allocatable :: exps(:) !< Exponent parameters
85 real(8), allocatable :: bas_mat(:) !< Basis matrix
86 real(8), allocatable :: exp_mat(:) !< Exponent matrix
87 real(8), allocatable :: exp_mat_sz(:) !< Exponent matrix for Sz
88 real(8), allocatable :: exp_mat_c(:) !< Complex exponent matrix
89 real(8), allocatable :: exp_mat_sz_c(:) !< Complex exponent matrix for Sz
90 end type
91
92end module types
