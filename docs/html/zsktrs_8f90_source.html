16!> @brief Solve complex skew-symmetric tridiagonal system A*X = B
17!>
18!> This subroutine solves the linear system A*X = B where A is a
19!> non-singular complex skew-symmetric tridiagonal matrix of even dimension n.
20!> The matrix A is stored in packed format with A(i+1,i) = a(i) and
21!> A(i,i+1) = -a(i) for i = 1, ..., n-1.
22!>
23!> Parameters
24!> ----------
25!> UPLO : character, in
26!> Specifies which part of the matrix A is stored:
27!> 'U' or 'u': Upper triangular part is stored (A(i,i+1) = a(i)).
28!> 'L' or 'l': Lower triangular part is stored (A(i+1,i) = a(i)).
29!> N : integer, in
30!> Order of the matrix A (must be even).
31!> NHRS : integer, in
32!> Number of right-hand sides.
33!> A : complex*16 array, in
34!> The complex skew-symmetric tridiagonal matrix stored in packed format.
35!> For UPLO = 'U': A(i,i+1) = a(i), i = 1, ..., n-1.
36!> For UPLO = 'L': A(i+1,i) = a(i), i = 1, ..., n-1.
37!> B : complex*16 array, inout
38!> Right-hand side matrix (N × NHRS). On exit, contains the solution X.
39!> LDB : integer, in
40!> Leading dimension of array B.
41!> X : complex*16 array, work
42!> Workspace array for storing intermediate results.
43!> INFO : integer, out
44!> = 0: successful exit.
45!> < 0: if INFO = -i, the i-th argument had an illegal value.
46!> > 0: if INFO = i, A(i,i+1) is exactly zero; the matrix is singular.
47!>
48!> Notes
49!> -----
50!> - The matrix A must be of even dimension n.
51!> - The solution is computed using forward and backward substitution.
52!> - Even-indexed variables are solved forward, odd-indexed backward.
53!> - The algorithm exploits the special structure of complex skew-symmetric matrices.
54!> - For UPLO = 'L', the signs of the matrix elements are temporarily flipped.
55!> - Complex arithmetic is used throughout the computation.
56!>
57!> Algorithm
58!> ---------
59!> 1. For UPLO = 'L', flip signs of matrix elements
60!> 2. Solve even-indexed variables forward: x(2), x(4), ..., x(n)
61!> 3. Solve odd-indexed variables backward: x(n-1), x(n-3), ..., x(1)
62!> 4. Copy solution back to B
63!> 5. Restore original matrix signs if needed
64!>
65!> Example
66!> -------
67!> Used in complex skew-symmetric matrix inversion and linear system solving.
68subroutine zsktrs(uplo, n, nhrs, a, b, ldb, x, info)
69 implicit none
70
71 ! argument parameters
72 integer, intent(in) :: n, nhrs, ldb
73 integer, intent(out) :: info
74 complex*16, intent(in out) :: a(*), b(ldb, nhrs)
75 complex*16, intent(out) :: x(nhrs, *)
76 character, intent(in) :: uplo
77
78 ! local variables
79 integer i, j
80
81!> @brief Initialize INFO and flip signs for lower triangular storage
82 info = 0
83 if (uplo .eq. 'l' .or. uplo .eq. 'L') then
84 a(1:n - 1) = -a(1:n - 1)
85 end if
86!> @brief Solve A*X = B for complex skew-symmetric tridiagonal matrix A
87!> A is stored as A(i+1,i) = a(i), A(i,i+1) = -a(i), i = 1, ..., n-1
88!> Matrix dimension n must be even
89!> @brief Solve even-indexed variables forward
90 x(:, 2) = b(1, :)/a(1)
91 do i = 4, n, 2
92 x(:, i) = (b(i - 1, :) + a(i - 2)*x(:, i - 2))/a(i - 1)
93 end do
94!> @brief Solve odd-indexed variables backward
95 x(:, n - 1) = -b(n, :)/a(n - 1)
96 do i = n - 3, 1, -2
97 x(:, i) = -(b(i + 1, :) - a(i + 1)*x(:, i + 2))/a(i)
98 end do
99!> @brief Copy solution back to B
100 do i = 1, n
101 b(i, :) = x(:, i)
102 end do
103!> @brief Restore original matrix signs
104 if (uplo .eq. 'l' .or. uplo .eq. 'L') then
105 a(1:n - 1) = -a(1:n - 1)
106 end if
107 return
108end
