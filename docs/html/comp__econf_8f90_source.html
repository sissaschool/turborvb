20! Main variables:
21! econf = array containing the values of energy and correlations functions

26!> @brief Compute energy and correlation functions for walkers
27!> @details This subroutine computes the energy and other correlation functions
28!> after one loop over the walkers. It stores different components
29!> of the total energy based on the iese parameter, including kinetic
30!> energy, diffusion, and pseudo-potential contributions.
31!> @param[in] j Walker index
32!> @param[in] js Secondary walker index
33!> @param[out] econf Array containing energy and correlation function values
34!> @param[in] enert Energy arrays
35!> @param[in] diffkin Kinetic energy differences
36!> @param[in] vpot Potential energy
37!> @param[in] vcut Cutoff potential
38!> @param[in] voffpseudo Pseudo-potential offset
39!> @param[in] table Table arrays
40subroutine comp_econf(j, js, econf, enert, diffkin, vpot, vcut, voffpseudo, table)
41
42 use constants, only: ipc
43 use allio, only: itest, iese, in1, yes_complex, nel, nelup, neldo, indt, &
44 psip, indtj, istart, gamma, ivic, indksj, isfix, rank
45
46 implicit none
47
48 integer j, js, kk
49 real(8) econf(*), diffkin(3, *), enert(ipc, *), vpot(*), &
50 voffpseudo, table(*), vcut(*)
51
52 if (iese .ne. 0) then
53 ! energy
54 econf(j) = enert(1, j)
55 if (ipc .eq. 2) econf(j + in1) = enert(2, j)
56 if (iese .ge. 1 + ipc) then
57 econf(j + in1*ipc) = diffkin(2, j)*0.5d0 ! unit Ha
58 end if
59 if (iese .ge. 2 + ipc) then
60 call diffus(nel, indt, gamma, ivic(1, 1, indksj), table(indtj), psip, istart)
61 econf(j + (1 + ipc)*in1) = psip(1)
62 end if
63 if (iese .eq. 3 + ipc) then
64 econf(j + (2 + ipc)*in1) = voffpseudo*0.5d0
65 end if
66 end if
67 ! correlation functions
68 if (iese .ne. 3 + ipc) then
69 if (iese .ge. 3 + ipc) econf(j + (2 + ipc)*in1) = vcut(j)
70 if (iese .ge. 4 + ipc) econf(j + (3 + ipc)*in1) = diffkin(1, j)*0.5d0 !unit Ha
71 if (iese .ge. 5 + ipc) econf(j + (4 + ipc)*in1) = diffkin(2, j)*0.5d0 !unit Ha
72 end if
73
74 return
75end subroutine comp_econf
