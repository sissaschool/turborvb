88 !> @brief Initialize cell transformation matrices
89 !> @details This subroutine initializes the Cartesian to crystal coordinate
90 !> transformation matrix by computing the inverse of the cell
91 !> matrix using LAPACK routines.
92 !>
93 !> The subroutine:
94 !> - Uses DGETRF for LU decomposition
95 !> - Uses DGETRI for matrix inversion
96 !> - Stores the transformation matrix in car2cry
97 subroutine init_cell
98 implicit none
99 integer ipiv(3), i, info
100 real*8 matscra(9)
101 car2cry = at_qe
102 call dgetrf(3, 3, car2cry, 3, ipiv, info)
103 if (info .ne. 0) then
104 write (6, *) ' ERROR in initialization cell (dgetrf) !!! ', info
105 do i = 1, 3
106 write (6, *) i, ipiv(i), car2cry(i, i)
107 end do
108 else
109 call dgetri(3, car2cry, 3, ipiv, matscra, 9, info)
110 if (info .ne. 0) write (6, *) ' ERROR in initialization cell (dgetri) !!! '
111 end if
112 end subroutine init_cell
113
114 !> @brief Transform coordinates from Cartesian to crystal coordinates
115 !> @details This subroutine transforms a set of vectors from Cartesian
116 !> coordinates to crystal coordinates using the transformation
117 !> matrix computed by init_cell.
118 !>
119 !> The transformation is applied using BLAS DGEMM:
120 !> r_crystal = car2cry * r_cartesian
121 !>
122 !> @param[in,out] r Coordinate vectors to transform (3, howmany)
123 !> @param[in] howmany Number of vectors to transform
124 subroutine cartesiantocrystal(r, howmany)
125 integer, intent(in) :: howmany
126 double precision, dimension(3, howmany), intent(inout) :: r
127 double precision s(3)
128 integer i
129 do i = 1, howmany
130 s(:) = r(:, i)
131 call dgemv('N', 3, 3, 1.d0, car2cry, 3, s, 1, 0.d0, r(1, i), 1)
132 end do
133 end subroutine cartesiantocrystal
