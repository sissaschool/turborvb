29!> @brief Generate symmetry-equivalent matrix elements for skew-symmetric part of Pfaffian
30!> @details This subroutine generates symmetry-equivalent matrix elements
31!> for the skew-symmetric part of the Pfaffian matrix. It applies
32!> rotation and translation symmetries to identify equivalent orbital
33!> pairs and groups them into records for efficient computation.
34!>
35!> The subroutine handles:
36!> - Rotation symmetries (symrot)
37!> - Translation symmetries (symtra)
38!> - Skew-symmetric constraints (i < j)
39!> - Hermitian treatment for boundary conditions
40!> - Phase locking for long bonds
41!>
42!> @param[in] norb Number of orbitals
43!> @param[in] symrot Rotation symmetry matrix (norb, nrot)
44!> @param[in] nrot Number of rotation symmetries
45!> @param[in] symtra Translation symmetry matrix (norb, ntra)
46!> @param[in] ntra Number of translation symmetries
47!> @param[out] occupied Logical matrix indicating allowed matrix elements
48!> @param[out] recordsym Symmetry records for matrix elements
49!> @param[out] lenrec Length of each symmetry record
50!> @param[in,out] nrec Number of records (input: max, output: actual)
51!> @param[in] ipsip Temporary array for symmetry operations
52!> @param[in] rion Atomic positions (3, ntotatoms)
53!> @param[in] ntotatoms Total number of atoms
54!> @param[in] orb2atom Mapping from orbitals to atoms
55!> @param[in] cellscale Cell dimensions
56!> @param[in] deps Numerical tolerance for comparisons
57!> @param[in] zeta Zeta parameters for atoms (2, ntotatoms)
58!> @param[in] yes_hermite Logical flag for Hermitian treatment
59subroutine makeskew(norb, symrot, nrot, symtra, ntra, occupied, recordsym, &
60 lenrec, nrec, ipsip, rion, ntotatoms, orb2atom, cellscale, deps, zeta, yes_hermite)
61 implicit none
62
63 integer norb, nrot, ntra, symrot(norb, nrot), symtra(norb, ntra)
64 logical occupied(norb, norb), accept, sameorb, yes_hermite
65 integer recordsym(2, ntra*nrot, *), lenrec(*), i, j, k, l, jfirst, imap, jmap&
66 &, signi, signj, imapr, jmapr, signir, signjr, ipsip(2, ntra*nrot), indsym, countrec&
67 &, nrec, count, kk, ntotatoms, check
68 real*8 rion(3, ntotatoms), rdiff(3), rdiff_new(3), rdiff_rot(3), rdiff_roti(3), cellscale(3), cost, deps, zeta(2, ntotatoms)
69 integer orb2atom(*), posi, posj, posi_new, posj_new
70 integer, dimension(:), allocatable :: atbasis, refat
71
72 ! j=symrot(i,is) is=1,nrot are the possible rotation symmetries
73 ! |j|>0 is the orbital obtained after application of the symmetry is
74 ! to the orbital i. if symrot(i,is)=0 it is not possible to apply this
75 ! symmetry to the orbital i (generates more than a single orbital).
76 ! the sign of j is the sign obtained after this transformation to the
77 ! orbital including the sign of the boundary conditions.
78
79 ! symtra(norb,ntra) as above for the translation symmetries.
80 ! assumed that the translation symmetries are always allowed (never symtra(i,j)=0.
81 ! Output occupied=.true. for an allowed matrix element of the
82 ! lambda_ij
83 ! output record(2,1:lenrec(k),k) k<=nrec, if nrec large enough all matrix
84 ! elements are classified otherwise a Warning appears and rec is the number
85 ! of records that are actually required.
86 ! lenrec(k) is the length of the record #k
87
88 occupied = .false.
89 countrec = 0
90
91 allocate (atbasis(ntotatoms), refat(ntotatoms))
92 atbasis = 0
93 do i = 1, norb
94 atbasis(orb2atom(i)) = atbasis(orb2atom(i)) + 1
95 end do
96 ! computation cumulative index
97 refat = atbasis
98 do i = 2, ntotatoms
99 refat(i) = refat(i) + refat(i - 1)
100 end do
101 do i = 1, ntotatoms
102 refat(i) = refat(i) - atbasis(i)
103 end do
104
105 do i = 1, norb
106 jfirst = i + 1
107 do j = jfirst, norb
108 posi = orb2atom(i)
109 posj = orb2atom(j)
110 rdiff(:) = rion(:, posi) - rion(:, posj)
111 call makeimage(rdiff, cellscale, deps)
112 accept = .false.
113 if (.not. occupied(i, j)) then
114 accept = .true.
115 indsym = 0
116 do k = 1, ntra
117 signi = 1
118 signj = 1
119 imap = symtra(i, k)
120 jmap = symtra(j, k)
121 !Check with Sandro
122 if (imap .lt. 0) then
123 imap = -imap
124 signi = -signi
125 end if
126 if (jmap .lt. 0) then
127 jmap = -jmap
128 signj = -signj
129 end if
130
131 if (imap .eq. 0 .or. jmap .eq. 0) then
132 write (6, *) ' Error in symtra !!! ', imap, jmap
133
134 end if
135 do l = 1, nrot
136 imapr = symrot(imap, l)
137 jmapr = symrot(jmap, l)
138
139 if (imapr .lt. 0) then
140 imapr = -imapr
141 signir = -signi
142 else
143 signir = signi
144 end if
145 if (jmapr .lt. 0) then
146 jmapr = -jmapr
147 signjr = -signj
148 else
149 signjr = signj
150 end if
151
152 !Only for the same type of orbital and the same type of ion
153 !Checking if the two atoms are at distance cellscale/2 and with the same orbital
154 if (imapr .ne. 0 .and. jmapr .ne. 0) then
155 if (yes_hermite) then ! if it is an allowed symmetry
156 posi = orb2atom(abs(imapr))
157 posj = orb2atom(abs(jmapr))
158 rdiff(:) = rion(:, posi) - rion(:, posj)
159 sameorb = .false.
160 call makeimage(rdiff, cellscale, deps)
161
162 if ((abs(imapr) - refat(posi)) .eq. (abs(jmapr) - refat(posj)) &
163 .and. zeta(1, orb2atom(i)) .eq. zeta(1, orb2atom(j)) .and. posi .ne. posj) sameorb = .true.
164
165 !If the distance of the atoms has only component L/2 and 0 (at least one has to be L/2)
166 if (sameorb) then
167 check = 0
168 do kk = 1, 3
169 if (abs(2*abs(rdiff(kk)) - cellscale(kk)) .lt. deps) then
170 check = check - 1
171 else if (abs(2*abs(rdiff(kk))) .gt. deps) then
172 check = check + 4
173 end if
174 end do
175 if (check .lt. 0) accept = .false.
176 end if
177 end if
178
179 if (accept) then
180 indsym = indsym + 1
181 ipsip(1, indsym) = imapr*signjr*signir
182 ipsip(2, indsym) = jmapr
183 end if
184 end if
185 end do
186 end do
187
188 ! check if it is a compatible record
189 if (indsym .gt. 0 .and. accept) then
190 accept = .true.
191 do k = 1, indsym
192 occupied(abs(ipsip(1, k)), abs(ipsip(2, k))) = .true.
193 occupied(abs(ipsip(2, k)), abs(ipsip(1, k))) = .true.
194 end do
195 else
196 accept = .false.
197 end if
198
199 !Checking if some elements cannot be accepted because they are zero by definition
200 do k = 1, indsym
201 if ((abs(ipsip(2, k)) .eq. i .and. ipsip(1, k) .eq. j) .or. &
202 (ipsip(1, k) .eq. -i .and. abs(ipsip(2, k)) .eq. j)) then
203 accept = .false.
204 end if
205 end do
206
207 !Checking if it is already occupied
208 if (accept) then
209 countrec = countrec + 1
210 if (countrec .le. nrec) then
211 ! group toghether the various pairs
212
213 lenrec(countrec) = 1
214 recordsym(1:2, 1, countrec) = ipsip(1:2, 1)
215 do k = 2, indsym
216 accept = .true.
217
218 do l = 1, k - 1
219 if ((abs(ipsip(1, k)) .eq. abs(ipsip(1, l)) .and. abs(ipsip(2, k)) .eq. abs(ipsip(2, l))) .or.&
220 &(abs(ipsip(1, k)) .eq. abs(ipsip(2, l)) .and. abs(ipsip(2, k)) .eq. abs(ipsip(1, l)))) &
221 & accept = .false.
222 end do
223
224 if (accept) then
225 lenrec(countrec) = lenrec(countrec) + 1
226 recordsym(1:2, lenrec(countrec), countrec) = ipsip(1:2, k)
227 end if
228 end do
229
230 end if ! if countrec<=nrec
231 end if ! endif accepted
232 end if
233 end do
234
235 end do
236 deallocate (atbasis, refat)
237
238 write (6, *) "SKEW"
239 do i = 1, nrec
240 write (6, *) "step", i, "lenrec(step)", lenrec(i)
241 do j = 1, lenrec(i)
242 write (6, *) recordsym(1, j, i), recordsym(2, j, i)
243 end do
244 end do
245
246 nrec = countrec
247
248end subroutine makeskew

254!> @brief Assemble the complete Pfaffian matrix from AGP and skew-symmetric parts
255!> @details This subroutine combines the AGP (antisymmetrized geminal power)
256!> and skew-symmetric parts to create the complete Pfaffian matrix.
257!> It handles different configurations based on the presence of
258!> up and down Pfaffian components.
259!>
260!> The subroutine handles:
261!> - AGP matrix elements (recordsymagp)
262!> - Skew-symmetric matrix elements (recordsymskw)
263!> - Up/down Pfaffian configurations
264!> - Orbital indexing for spin-up and spin-down components
265!>
266!> @param[in] norb Number of orbitals
267!> @param[in] nrot Number of rotation symmetries
268!> @param[in] ntra Number of translation symmetries
269!> @param[out] occupied Logical matrix indicating allowed matrix elements
270!> @param[in] recordsymagp Symmetry records for AGP matrix elements
271!> @param[in] lenrecagp Length of each AGP symmetry record
272!> @param[in] recordsymskw Symmetry records for skew-symmetric matrix elements
273!> @param[in] lenrecskw Length of each skew-symmetric symmetry record
274!> @param[out] recordsym Combined symmetry records for Pfaffian matrix
275!> @param[out] lenrec Length of each combined symmetry record
276!> @param[in,out] nrec Number of combined records (input: max, output: actual)
277!> @param[in] nrecagp Number of AGP records
278!> @param[in] nrecskw Number of skew-symmetric records
279!> @param[in] nouppfaff Logical flag for up Pfaffian
280!> @param[in] nodownpfaff Logical flag for down Pfaffian
281!> @param[in] opposite_phase Logical flag for opposite phase treatment
282subroutine makepfaff(norb, nrot, ntra, occupied, recordsymagp, lenrecagp, recordsymskw, lenrecskw, recordsym, &
283 lenrec, nrec, nrecagp, nrecskw, nouppfaff, nodownpfaff, opposite_phase)
284 implicit none
285 integer :: norb, nrot, ntra, nrec, nrecagp, nrecskw, recordymmagp, count, i, j, k, kk
286 logical :: occupied(2*norb, 2*norb), nodownpfaff, nouppfaff, opposite_phase
287 integer :: recordsym(2, ntra*nrot, nrec), recordsymagp(2, ntra*nrot, nrecagp), recordsymskw(2, ntra*nrot, nrecskw), &
288 lenrec(nrec), lenrecagp(nrecagp), lenrecskw(nrecskw)

290 recordsym = 0

292 !Building a new and larger recordsym and lenrec
293 if (nodownpfaff .and. .not. nouppfaff) then
294 recordsym(:, 1:nrot*ntra, 1:nrecskw) = recordsymskw(:, 1:nrot*ntra, 1:nrecskw)
295 recordsym(1, 1:nrot*ntra, nrecskw + 1:nrecskw + nrecagp) = recordsymagp(1, 1:nrot*ntra, 1:nrecagp)
296 recordsym(2, 1:nrot*ntra, nrecskw + 1:nrecskw + nrecagp) = recordsymagp(2, 1:nrot*ntra, 1:nrecagp) &
297 + sign(1, recordsymagp(2, 1:nrot*ntra, 1:nrecagp))*norb
298 lenrec(1:nrecskw) = lenrecskw(1:nrecskw)
299 lenrec(1 + nrecskw:nrecagp + nrecskw) = lenrecagp(1:nrecagp)
300 nrec = nrecagp + nrecskw
301
302 else if (nouppfaff .and. .not. nodownpfaff) then
303 recordsym(:, 1:nrot*ntra, 1:nrecskw) = recordsymskw(:, 1:nrot*ntra, 1:nrecskw) &
304 + norb*sign(1, recordsymskw(:, 1:nrot*ntra, 1:nrecskw))
305 recordsym(1, 1:nrot*ntra, nrecskw + 1:nrecskw + nrecagp) = recordsymagp(1, 1:nrot*ntra, 1:nrecagp)
306 recordsym(2, 1:nrot*ntra, nrecskw + 1:nrecskw + nrecagp) = recordsymagp(2, 1:nrot*ntra, 1:nrecagp) &
307 + norb*sign(1, recordsymagp(2, 1:nrot*ntra, 1:nrecagp))
308 lenrec(1:nrecskw) = lenrecskw(1:nrecskw)
309 lenrec(1 + nrecskw:nrecagp + nrecskw) = lenrecagp(1:nrecagp)
310 nrec = nrecagp + nrecskw
311 ! else if (symmagp) then
312 ! write (6,*) "Symmagp is active with same/opposite phase"
313 ! do i=1, nrecskw
314 ! recordsym(:, 1:lenrecskw(i) , i )=recordsymskw(:, 1:lenrecskw(i), i)
315 ! recordsym(:, 1+lenrecskw(i):2*lenrecskw(i), i)= recordsymskw(:, 1:lenrecskw(i), i)&
316 ! +norb*sign(1,recordsymskw(:, 1:lenrecskw(i), i))
317 ! end do
318 ! recordsym(1,1:nrot*ntra ,nrecskw+1:nrecskw+nrecagp)=recordsymagp(1,1:nrot*ntra,1:nrecagp)
319 ! recordsym(2,1:nrot*ntra ,nrecskw+1:nrecskw+nrecagp)=recordsymagp(2,1:nrot*ntra,1:nrecagp)&
320 ! + norb*sign(1,recordsymagp(2,1:nrot*ntra,1:nrecagp))
321 ! lenrec(1:nrecskw)=lenrecskw(1:nrecskw)
322 ! lenrec(1+nrecskw:nrecagp+nrecskw)=lenrecagp(1:nrecagp)
323 ! nrec=nrecagp+nrecskw
324 elseif (nouppfaff .and. nodownpfaff) then
325 recordsym(1, 1:nrot*ntra, 1:nrecagp) = recordsymagp(1, 1:nrot*ntra, 1:nrecagp)
326 recordsym(2, 1:nrot*ntra, 1:nrecagp) = recordsymagp(2, 1:nrot*ntra, 1:nrecagp) &
327 + norb*sign(1, recordsymagp(2, 1:nrot*ntra, 1:nrecagp))
328 lenrec(1:nrecagp) = lenrecagp(1:nrecagp)
329 nrec = nrecagp
330 else
331 recordsym(:, 1:nrot*ntra, 1:nrecskw) = recordsymskw(:, 1:nrot*ntra, 1:nrecskw)
332 recordsym(:, 1:nrot*ntra, nrecskw + 1:nrecskw*2) = recordsymskw(:, 1:nrot*ntra, 1:nrecskw) &
333 + norb*sign(1, recordsymskw(:, 1:nrot*ntra, 1:nrecskw))
334 recordsym(1, 1:nrot*ntra, nrecskw*2 + 1:nrecskw*2 + nrecagp) = recordsymagp(1, 1:nrot*ntra, 1:nrecagp)
335 recordsym(2, 1:nrot*ntra, nrecskw*2 + 1:nrecskw*2 + nrecagp) = recordsymagp(2, 1:nrot*ntra, 1:nrecagp) &
336 + norb*sign(1, recordsymagp(2, 1:nrot*ntra, 1:nrecagp))
337 lenrec(1:nrecskw) = lenrecskw(1:nrecskw)
338 lenrec(1 + nrecskw:nrecskw*2) = lenrecskw(1:nrecskw)
339 lenrec(1 + nrecskw*2:nrecagp + nrecskw*2) = lenrecagp(1:nrecagp)
340 nrec = nrecagp + 2*nrecskw
341 end if

343 ! write (6,*) "Inside makepfaff"
344
345 ! do i=1, nrec
346 ! write (6,*) "step", i ,"lenrec(step)",lenrec(i)
347 ! do j=1, lenrec(i)

350 ! end do

353 count = 0
354 do i = 1, nrec
355 ! write (6,*) i,"lenrec", lenrec(i)
356 do j = 1, lenrec(i)
357 occupied(abs(recordsym(1, j, i)), abs(recordsym(2, j, i))) = .true.
358 ! occupied(abs(recordsym(2,j,i)),abs(recordsym(1,j,i)))=.true.
359
360 do k = 1, lenrec(i)

362 &abs(recordsym(2, k, i)) .and. k .ne. j) then

369 end if
370 end do

376 do i = 1, 2*norb
