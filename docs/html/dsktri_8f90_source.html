16!> @brief Invert a skew-symmetric matrix using LU decomposition
17!>
18!> This subroutine computes the inverse of a skew-symmetric matrix A
19!> using the LU decomposition A = L*U*P, where P is a permutation matrix.
20!> The inverse is computed as A^(-1) = P^T * U^(-1) * L^(-1).
21!>
22!> Parameters
23!> ----------
24!> UPLO : character, in
25!> Specifies which part of the matrix A is stored:
26!> 'U' or 'u': Upper triangular part is stored.
27!> 'L' or 'l': Lower triangular part is stored.
28!> N : integer, in
29!> Order of the matrix A.
30!> A : real*8 array, in
31!> The skew-symmetric matrix A, stored in packed format.
32!> On entry, contains the LU factors from DSKTRF.
33!> LDA : integer, in
34!> Leading dimension of array A.
35!> AINV : real*8 array, out
36!> The inverse of matrix A.
37!> LDINV : integer, in
38!> Leading dimension of array AINV.
39!> IPIV : integer array, in
40!> Pivot indices from DSKTRF; dimension at least N.
41!> WORK : real*8 array, work
42!> Workspace array; dimension at least N^2 + 12*N - 2.
43!> INFO : integer, out
44!> = 0: successful exit.
45!> < 0: if INFO = -i, the i-th argument had an illegal value.
46!> > 0: if INFO = i, U(i,i) is exactly zero; the matrix is singular.
47!>
48!> Notes
49!> -----
50!> - Assumes that DSKTRF has been called previously to factorize A.
51!> - Uses a three-step process: permutation, triangular solve, permutation.
52!> - Supports both upper and lower triangular storage formats.
53!> - The algorithm handles the special structure of skew-symmetric matrices.
54!>
55!> Algorithm
56!> ---------
57!> 1. Initialize AINV as identity matrix
58!> 2. Apply first permutation (reverse order)
59!> 3. Extract skew-symmetric elements and prepare for triangular solve
60!> 4. Solve triangular system using DTRTRS
61!> 5. Solve tridiagonal system using DSKTRS or DGTSVX
62!> 6. Solve triangular system with transpose
63!> 7. Apply final permutation
64!>
65!> Example
66!> -------
67!> Used in quantum Monte Carlo calculations for matrix operations.
68subroutine dsktri(uplo, n, a, lda, ainv, ldinv, ipiv, work, info)
69 implicit none
70 character uplo
71 integer n, i, j, lda, ldinv, info, lwork, ipiv(*)
72 real*8 a(lda, *), ainv(ldinv, *), work(*)
73 real*8 rcond
74 logical yeslap
75!> @brief Flag to choose between LAPACK and homemade algorithm
76 yeslap = .false. ! if .false. the homemade algorithm is done.
77!> @brief Initialize AINV as identity matrix
78 do i = 1, n
79 do j = 1, i - 1
80 ainv(j, i) = 0.d0
81 end do
82 ainv(i, i) = 1.d0
83 do j = i + 1, n
84 ainv(j, i) = 0.d0
85 end do
86 end do
87!> @brief Apply first permutation in reverse order
88 do i = n, 1, -1
89 work(1:n) = ainv(i, 1:n)
90 ainv(i, 1:n) = ainv(ipiv(i), 1:n)
91 ainv(ipiv(i), 1:n) = work(1:n)
92 end do
93!> @brief Extract skew-symmetric elements and prepare triangular solve
94 if (uplo .eq. 'u' .or. uplo .eq. 'U') then
95!> @brief Upper triangular case: extract superdiagonal elements
96 do i = 1, n - 1
97 work(i) = a(i, i + 1)
98 work(n + i - 1) = -a(i, i + 1)
99 end do
100 do j = 2, n - 1
101 a(1:j - 1, j) = a(1:j - 1, j + 1)
102 end do
103 a(1:n - 1, n) = 0.d0
104 else
105!> @brief Lower triangular case: extract subdiagonal elements
106 do i = 1, n - 1
107 work(i) = -a(i + 1, i)
108 work(n + i - 1) = a(i + 1, i)
109 end do
110 do j = n - 1, 2, -1
111 a(j + 1:n, j) = a(j + 1:n, j - 1)
112 end do
113 a(2:n, 1) = 0.d0
114 end if
115!> @brief Set diagonal elements to zero (skew-symmetric property)
116 work(2*n - 1:3*n - 2) = 0.d0 ! diagonal elements of skew matrix , obviously set to zero.
117!> @brief First triangular solve: U * X = B
118 call dtrtrs(uplo, 'N', 'U', n, n, a, lda, ainv, ldinv, info)
119!> @brief Solve tridiagonal system using LAPACK or homemade algorithm
120 if (yeslap) then
121!> @brief Use LAPACK DGTSVX for tridiagonal solve
122 call dgtsvx('N', 'N', n, n, work(n), work(2*n - 1), work, work(3*n)&
123 &, work(4*n), work(5*n), work(6*n), ipiv(n + 1), ainv, lda, work(7*n), n&
124 &, rcond, work(7*n + n*n - 1), work(8*n + n*n - 1)&
125 &, work(9*n + n*n - 1), ipiv(2*n + 1), info)
126 do i = 1, n
127 ainv(1:n, i) = work(7*n + (i - 1)*n:7*n + i*n - 1)
128 end do
129 else
130!> @brief Use homemade DSKTRS for skew-symmetric tridiagonal solve
131 call dsktrs(uplo, n, n, work, ainv, ldinv, work(n), info)
132 end if
133!> @brief Second triangular solve: L^T * X = B
134 call dtrtrs(uplo, 'T', 'U', n, n, a, lda, ainv, ldinv, info)
135!> @brief Apply final permutation to complete A^(-1) = P^T * U^(-1) * L^(-1)
136 do i = 1, n
137 work(1:n) = ainv(i, 1:n)
138 ainv(i, 1:n) = ainv(ipiv(i), 1:n)
139 ainv(ipiv(i), 1:n) = work(1:n)
140 end do
141
142 return
143end
