16!> @brief Automatically adjust learning rate (tpar) during optimization
17!> @details This subroutine automatically adjusts the learning rate (tpar) during
18!> optimization based on the stability of energy and error measurements.
19!> It monitors running averages and standard deviations of energy and error,
20!> and increases or decreases tpar accordingly to maintain stable optimization.
21!> The algorithm includes mechanisms to handle unstable behavior and
22!> maintain a list of stable tpar values.
23!> @param[in] i_main Main iteration counter
24!> @param[in] nweight Number of weights
25!> @param[in] energy Current energy value
26!> @param[in] error_energy Current energy error
27!> @param[in,out] tpar Learning rate (adjusted by this subroutine)
28!> @param[in] ngentry Number of entries
29!> @param[in] itestr4 Test parameter
30subroutine adjust_tpar(i_main, nweight, energy, error_energy, tpar, ngentry, itestr4)

32 use allio, only: energy_list, error_energy_list, tpar_buffer_filled, &
33 tpar_increased, inc_tpar_frequency, stop_increasing_tpar, &
34 tpar_unstble_stop, counter_unstable_tpar, &
35 counter_unstable_energy, min_running_ave, min_running_std, &
36 min_running_ave_energy, min_running_std_energy, tpar_max, counter_unstable_err, divide_tpar, &
37 multiply_tpar, len_shorter_buffer, cut_sigma, tpar_buffer_len, times_tpar_decreased, &
38 use_stable_tpar, tpar_stable_list, len_tpar_stable_list, n_sigmas_tpar

44 integer index_ene, count_list, index_stable_list, half_stable_tpars

46 half_stable_tpars = int(len_tpar_stable_list/2)
47
48 if (use_stable_tpar) then
49 !check if we have filled the list of "stable" values of tpar
50 do index_stable_list = 1, size(tpar_stable_list)
51 if (tpar_stable_list(index_stable_list) .eq. -1) exit
52 end do
53
54 write (*, *) tpar_stable_list
55
56 ! if we have all the stable values of tpar we just use the mean of the last ten values
57 if (index_stable_list .gt. size(tpar_stable_list)) then
58 write (*, *) 'adjust_tpar: warning, using average of stable value for tpar', int(i_main/nweight)
59 tpar = sum(tpar_stable_list(size(tpar_stable_list) - (half_stable_tpars - 1):size(tpar_stable_list)))/half_stable_tpars
60 return
61 end if
62 end if
63
64 count_list = 0
65 do index_ene = 1, size(energy_list)
66 if (tpar_buffer_filled(index_ene)) count_list = count_list + 1
67 end do
68
69 do index_ene = 1, size(energy_list)
70 if (.not. tpar_buffer_filled(index_ene)) then
71 energy_list(index_ene) = energy
72 error_energy_list(index_ene) = error_energy
73 tpar_buffer_filled(index_ene) = .true.
74 exit
75 end if
76 end do
77
78 ! If return_tpar just stores them, this subroutine skips the following part.
79 if (count_list .lt. tpar_buffer_len - 1) then
80 ! The buffer is filled even when count_list=size(energy_list)-1
81 ! because the previous loop will certainly add one element in the list
82 return
83 end if
84
85 running_ave_energy = sum(energy_list(1:count_list))/count_list
86 running_std_energy = std(energy_list, running_ave_energy, count_list)
87
88 if ((running_std_energy/error_energy)**2 .ge. 3) then !.and. count_list .ge. len_shorter_buffer) then
89 write (*, *) 'adjust_tpar: warning, using only last elements of the buffer', int(i_main/nweight)
90 running_ave_energy = sum(energy_list(tpar_buffer_len - len_shorter_buffer + 1:tpar_buffer_len))/len_shorter_buffer
91 running_std_energy = std(energy_list(tpar_buffer_len - len_shorter_buffer + 1), running_ave_energy, len_shorter_buffer)
92 running_ave = sum(error_energy_list(tpar_buffer_len - len_shorter_buffer + 1:tpar_buffer_len))/len_shorter_buffer
93 running_std = std(error_energy_list(tpar_buffer_len - len_shorter_buffer + 1), running_ave, len_shorter_buffer)
94 else
95 running_ave = sum(error_energy_list)/size(error_energy_list)
96 running_std = std(error_energy_list, running_ave, count_list)
97
98 end if
99
100 if (running_ave .lt. min_running_ave) then
101 min_running_ave = running_ave
102 min_running_std = running_std
103 end if
104
105 if (running_ave_energy .lt. min_running_ave_energy) then
106 min_running_ave_energy = running_ave_energy
107 min_running_std_energy = running_std_energy
108 end if
109
110 write (*, *) "adjust_tpar: minimum running average and std var/ene", &
111 min_running_ave, min_running_std, min_running_ave_energy, min_running_std_energy
112
113 iter_step = mod(i_main/nweight, inc_tpar_frequency) + 1
114
115 ! When the counter is just filled begin to increase tpar too
116 if (count_list .eq. size(energy_list) - 1 .or. mod(int(i_main/nweight), inc_tpar_frequency) .eq. 0) then
117 stop_increasing_tpar = .false.
118 times_tpar_decreased = 0
119 end if
120
121 if (energy .gt. min_running_ave_energy + 3*min_running_std_energy) then
122 counter_unstable_energy = counter_unstable_energy + 1
123 ! if(energy.gt.min_running_ave_energy+6*min_running_std_energy) &
124 ! counter_unstable_energy=counter_unstable_energy+3
125 else
126 counter_unstable_energy = 0
127 end if
128
129 if (error_energy .gt. min_running_ave + n_sigmas_tpar*min_running_std) then
130 counter_unstable_err = counter_unstable_err + 1
131 ! if (error_energy .gt. min_running_ave+6*min_running_std) &
132 ! counter_unstable_err=counter_unstable_err+3
133 else
134 counter_unstable_err = 0
135 end if
136
137 if (times_tpar_decreased .eq. 0) then
138 stop_tpar = tpar_unstble_stop
139 else
140 stop_tpar = tpar_unstble_stop*2
141 end if
142
143 if (counter_unstable_energy .ge. stop_tpar .or. counter_unstable_err .ge. stop_tpar) then
144
145 if (tpar_increased) then
146
147 tpar = tpar/divide_tpar
148
149 if (use_stable_tpar) then
150 ! we fill the list of stable values of tpar until it is filled
151 if (tpar_stable_list(index_stable_list) .eq. -1) then
152 tpar_stable_list(index_stable_list) = tpar
153 end if
154
155 end if
156
157 if (counter_unstable_energy .ge. stop_tpar .and. counter_unstable_err .lt. stop_tpar) then
158 write (*, *) " adjust_tpar: energy too big, tpar decreased", tpar
159 elseif (counter_unstable_err .ge. stop_tpar .and. counter_unstable_energy .lt. stop_tpar) then
160 write (*, *) " adjust_tpar: variance too big, tpar decreased", tpar
161 else
162 write (*, *) " adjust_tpar: variance & energy too big, tpar decreased", tpar
163 end if
164
165 counter_unstable_energy = 0
166 counter_unstable_err = 0
167 stop_increasing_tpar = .true.
168 times_tpar_decreased = times_tpar_decreased + 1
169 if (times_tpar_decreased .gt. 1) then
170 write (*, *) 'adjust_tpar: warning, reinitializing references at iteration ', int(i_main/nweight)
171 tpar_increased = .false.
172 min_running_ave_energy = (running_ave_energy + min_running_ave_energy)/2
173 min_running_std_energy = running_std_energy
174 min_running_ave = (running_ave + min_running_ave)/2
175 min_running_std = running_std
176 end if
177 end if
178
179 elseif (.not. stop_increasing_tpar) then
180 times_tpar_decreased = 0
181 tpar = tpar*multiply_tpar
182 if (itestr4 .eq. -4) tpar = min(tpar, tpar_max)
183 stop_increasing_tpar = .false.
184 tpar_increased = .true.
185 end if
186
187 if (count_list .eq. size(energy_list)) then
188 do index_ene = 1, size(energy_list) - 1
189 energy_list(index_ene) = energy_list(index_ene + 1)
190 error_energy_list(index_ene) = error_energy_list(index_ene + 1)
191 end do
192 energy_list(index_ene) = energy
193 error_energy_list(index_ene) = error_energy
194 end if

197 function std(vector, mean, dim)
198 integer dim
199 real*8 vector(dim)
200 real*8 mean, sum, std
201 integer i
202
203 sum = 0
204 do i = 1, dim
205 sum = sum + (vector(i) - mean)**2
206 end do
207 std = sqrt(sum/(dim - 1))
208 end function std
