18 ! This program take in input a real wave function fort.10 and
19 ! build the correspondent complex fort.10_complex setting to zero (or
20 ! to a random number) the imaginary parts of the complex coefficients.

64 nmol = molecular - ndiff

78 ioccup_c(1:occ_c - nmol) = ioccup_cread(1:occ_c - nmol)
79 ioccup_c(occ_c - nmol + 1:occ_c) = 1
80 mult_c(1:nshell_o) = mult_cread(1:nshell_o)
81 nparam_c(1:nshell_o) = nparam_cread(1:nshell_o)
82 ioptorb_c(1:nshell_o) = ioptorb_cread(1:nshell_o)
83 kion_c(1:nshell_o) = kion_cread(1:nshell_o)

85 deallocate (ioptorb_cread, mult_cread, nparam_cread, ioccup_cread, kion_cread)
86 nelorb_cnew = nelorb_c + nmol
87 allocate (dup_c(2*ndupcr + (nmol + ndiff)*ipf*nelorbh*4), nozero_cnew(nnozero_c))
88
89 do i = 1, nnozero_c - molecular
90 iy = (nozero_c(i) - 1)/nelorb_c + 1
91 ix = nozero_c(i) - (iy - 1)*nelorb_c
92 if (iy .gt. nelorb_c) iy = iy + nmol
93 nozero_cnew(i) = (iy - 1)*nelorb_cnew + ix
94 end do
95 ! First molecular orbital ME
96 iy1 = (nozero_c(nnozero_c - molecular + 1) - 1)/nelorb_c + 1
97 ix1 = nozero_c(nnozero_c - molecular + 1) - (iy1 - 1)*nelorb_c
98 indnn = 1
99 match = .true.
100 do while (match)
101 ii = jbradetn(indnn)
102 if (ii .ne. -1) then
103 indnn = indnn - 2*ii + 1
104 else
105 ix = jbradetn(indnn + 1)
106 iy = jbradetn(indnn + 2)
107 if (ix .eq. ix1 .and. iy .eq. iy1) then
108 match = .false.
109 else
110 indnn = indnn + 3
111 end if
112 end if
113 end do
114 indnn = indnn - 1
115
116 do i = nshell_o + 1, nshell_c
117 mult_c(i) = 1
118 ioptorb_c(i) = 1000000
119 nparam_c(i) = nparam_c(nshell_o)
120 end do
121 ind = nnozero_c - molecular
122 do i = 1, nmol
123 ix = nelorb_cnew - 2*nmol - ndiff + 2*i
124 iy = ix - 1
125 ind = ind + 1
126 nozero_cnew(ind) = nelorb_cnew*(iy - 1) + ix
127 indnn = indnn + 1
128 jbradetn(indnn) = -1
129 jbradetn(indnn + 1) = ix
130 jbradetn(indnn + 2) = iy
131 indnn = indnn + 2
132 end do
133 do i = nmol + 1, nmol + ndiff
134 ind = ind + 1
135 j = nelorb_cnew - ndiff - nmol + i
136 nozero_cnew(ind) = nelorb_cnew*(nelorb_cnew + i - 1 - nmol) + j
137 indnn = indnn + 1
138 jbradetn(indnn) = -1
139 jbradetn(indnn + 1) = j
140 jbradetn(indnn + 2) = nelorb_cnew + i - nmol
141 indnn = indnn + 2
142 end do
143
144 else
145 nelorb_cnew = nelorb_c
146 allocate (dup_c(2*ndupcr))
147 end if
148 count_mol = 0
149 dup_c = 0.d0
150 indpar = 0
151 indpar_read = 0
152 ! molyes is defined only in read_fort10_fast and not in read_fort10
153 if (molecular .gt. 0) then
154 molyesread = .true.
155 else
156 molyesread = .false.
157 end if
158 molyes = .false.
159 do is = 1, nshell_o
160 if (ioptorb_c(is) .lt. 900000) then
161 if (nparam_c(is) .eq. 1) then
162 dup_c(2*indpar + 1) = dupcread(indpar_read + 1)
163 dup_c(2*indpar + 2) = 0.d0
164 else
165 do j = 1, 2*nparam_c(is), 2
166 dup_c(2*indpar + j) = dupcread(indpar_read + j/2 + 1)
167 dup_c(2*indpar + j + 1) = 0.d0
168 end do
169 end if
170 elseif (ioptorb_c(is) .eq. 900000) then
171 do j = 1, nparam_c(is), 2
172 dup_c(2*indpar + j) = dupcread(indpar_read + j/2 + 1)
173 dup_c(2*indpar + j + 1) = 0.d0
174 end do
175 do j = nparam_c(is) + 1, 2*nparam_c(is), 2
176 dup_c(2*indpar + j) = dupcread(indpar_read + j/2 + 1)
177 dup_c(2*indpar + j + 1) = 0.d0
178 end do
179 elseif (ioptorb_c(is) .eq. 1000000) then
180 count_mol = count_mol + 1
181 do j = 1, nparam_c(is), 2
182 dup_c(2*indpar + j) = dupcread(indpar_read + j/2 + 1)
183 dup_c(2*indpar + j + 1) = 0.d0
184 end do
185 do j = nparam_c(is) + 1, 2*nparam_c(is), 2
186 dup_c(2*indpar + j) = dupcread(indpar_read + j/2 + 1)
187 dup_c(2*indpar + j + 1) = 0.d0
188 end do
189 if (symmagp .and. count_mol .le. nmol) then
190 indpar = indpar + nparam_c(is)
191 do j = 1, nparam_c(is), 2
192 dup_c(2*indpar + j) = dupcread(indpar_read + j/2 + 1)
193 dup_c(2*indpar + j + 1) = 0.d0
194 end do
195 do j = nparam_c(is) + 1, 2*nparam_c(is), 2
196 dup_c(2*indpar + j) = dupcread(indpar_read + j/2 + 1)
197 dup_c(2*indpar + j + 1) = 0.d0
198 end do
199 end if
200 end if
201 ! check if shells are correctly ordered
202 if (molyes .and. ioptorb_c(is) .lt. 900000) go to 103
203 if (ioptorb_c(is) .ge. 1000000) molyes = .true.
204 indpar = indpar + nparam_c(is)
205 indpar_read = indpar_read + nparam_c(is)
206 end do
207
208 if (molyesread .neqv. molyes) go to 104
209 !
210 ! transform geminal matrix to complex
211
212 !
213 call set_new_detmat(contraction)
214 do is = 1, nnozeror
215 if (contraction .ne. 0) then
216 if (molecular .ne. 0 .and. symmagp) then
217 if (2*nozero_cnew(is) .gt. size(detmat_c)) then
218 write (6, *) ' ERROR ', 2*nozero_cnew(is), size(detmat_c), 2*ndetmatr
219 stop
220 end if
221 detmat_c(2*nozero_cnew(is) - 1) = detmatread(nozero_c(is))
222 detmat_c(2*nozero_cnew(is)) = 0.d0
223 else
224 detmat_c(2*nozero_c(is) - 1) = detmatread(nozero_c(is))
225 detmat_c(2*nozero_c(is)) = 0.d0
226 end if
227 else
228 detmat(2*nozero(is) - 1) = detmatread(nozero(is))
229 detmat(2*nozero(is)) = 0.d0
230 end if
231 end do
232 if (molecular .ne. 0 .and. symmagp) then
233 nozero_c = nozero_cnew
234 nelorb_c = nelorb_cnew
235 deallocate (nozero_cnew)
236 end if
237
238 do i = nnozero_c - molecular + 1, nnozero_c
239 iy = (nozero_c(i) - 1)/nelorb_c + 1
240 ix = nozero_c(i) - (iy - 1)*nelorb_c
241 end do
242#if defined DEBUG
243 write (6, *) ' Check basis set '
244 do i = 1, iesupr_c
245 write (6, *) dupcread(i), dup_c(2*(i - 1) + 1), dup_c(2*(i - 1) + 2)
246 end do
247 write (6, *) ' Check detmat '
248 if (contraction .ne. 0) then
249 do i = 1, nnozero_c
250 write (6, *) detmatread(nozero_c(i)), detmat_c(2*nozero_c(i) - 1), detmat_c(2*nozero_c(i))
251 end do
252 else
253 do i = 1, nnozero
254 write (6, *) detmatread(nozero(i)), detmat(2*nozero(i) - 1), detmat(2*nozero(i))
255 end do
256 end if
257#endif
258 !
259 ! write the output complex wave function in fort.10_complex
260 !
261 close (ufort10)
262 open (unit=ufort10, file='fort.10_complex', status='unknown', form='formatted', err=105)
263 nelcol_c = nelorb_cnew
264 ipc = 2
265 call write_fort10(ufort10)
266 close (ufort10)
267
268 deallocate (dupcread, detmatread)
269 call deallocate_all
270
271 stop

288 !> @brief Sets up new determinant matrix for complex wavefunction conversion
289 !> @details This subroutine prepares the determinant matrix for conversion
290 !> from real to complex wavefunctions. It allocates new arrays
291 !> with double the size to accommodate complex numbers and
292 !> handles both contracted and uncontracted cases.
293 !> @param[in] contraction Flag indicating if contraction is used
294 subroutine set_new_detmat(contraction)
295 implicit none
296 integer, intent(in) :: contraction
297 if (contraction .ne. 0) then
298 ndetmatr = nelorb_c*max(nelcol_c, nel)
299 allocate (detmatread(ndetmatr))
300 detmatread(:) = detmat_c(:)
301 deallocate (detmat_c)
302 if (molecular .ne. 0 .and. symmagp) then
303 ndetmatr = nelorb_cnew*max(nelorb_cnew + ndiff, nel)
304 end if
305 allocate (detmat_c(2*ndetmatr))
306 detmat_c = 0.d0
307 nnozeror = nnozero_c
308 else
309 ndetmatr = ipf*nelorbh*max(nelcolh, nel)
310 allocate (detmatread(ndetmatr))
311 detmatread(:) = detmat(:)
312 deallocate (detmat)
313 allocate (detmat(2*ndetmatr))
314 detmat = 0.d0
315 nnozeror = nnozero
316 end if
317 return
318 end subroutine set_new_detmat
