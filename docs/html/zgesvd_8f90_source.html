16subroutine zgesvd_my(JOBZ, N, M, A, LDA, ZR, W, nproc, rank, comm_mpi)
17 use dspev_module
18 use allio, only: sub_comm_diag
19 use constants, only: zzero, zone

29 ! INPUT A common to all processors
30 ! OUTPUT A contains the left eigenvectors in SVD stored culumnwise
31 ! ZR contains the right eigenvectors in SVD stored rowwise
32 ! (as in standard dgesvd in lapack).
33 !
34 ! -- LAPACK driver routine (version 2.0) --
35 ! Univ. of Tennessee, Univ. of California Berkeley, NAG Ltd.,
36 ! Courant Institute, Argonne National Lab, and Rice University
37 ! September 30, 1994
38 ! NB At variance of lapack the singular values W are sorted in ascending order W(i+1)>=W(i).
39 ! Also the right eigenvectors of A are not conjugated as in lapack
40 ! A= (Output A) lambda ZR^* in this notations.
41 !
42 ! .. Scalar Arguments ..
43 character JOBZ
44 integer LDA, N, M, I
45 ! ..
46 ! .. Array Arguments ..
47 complex*16 A(LDA, N), ZR(LDA, *)
48 real*8 w(*)
49

51 complex*16, dimension(:), allocatable :: workp
52 integer, dimension(:), allocatable :: iwork, ifail
53 if (n .le. 0) return
54
55 if (sub_comm_diag%parent .ne. comm_mpi) then
56 ! write(6,*) "Warning! dsyev communicator mismatch",sub_comm_diag%parent,comm_mpi
57 ! INFO=1000
58 ! return
59 yesbcast = .true.
60 else
61 yesbcast = .false.
62 end if
63 if (sub_comm_diag%yesin) then
64
65 ! NB the parallel input is passed to the wrapper through the
66 ! WORK space ad follows:
67 allocate (b(lda, n))
68 b = zzero
69 call zgemm_my('N', 'C', n, n, m, zone, a, lda, a, lda, zzero, b, lda&
70 &, sub_comm_diag%nproc, sub_comm_diag%rank, sub_comm_diag%comm)
71 !&,nproc,rank,comm_mpi)
72
73 machp = dlamch('E')
74 ! Now standard 1D distribution.
75 ! compute the left eigenvectors
76 nrl = n/sub_comm_diag%nproc
77 if (nrl*sub_comm_diag%nproc .ne. n) nrl = nrl + 1
78 allocate (ap(nrl, n), z(nrl, n))
79 firsti = sub_comm_diag%rank + 1
80 i = 0
81 ap = zzero
82 z = zzero
83 ind = firsti
84 do while (ind .le. n)
85 do j = 1, n
86 ap(i + 1, j) = b(ind, j)
87 end do
88 i = i + 1
89 ind = sub_comm_diag%nproc*i + firsti
90 end do
91
92 call zdspev_drv_ss(jobz, ap, nrl, w, z, nrl, &
93 &nrl, n, sub_comm_diag%nproc, sub_comm_diag%rank, sub_comm_diag%comm, machp, 0)
94 ! & nrl, n, nproc,rank, comm_mpi, machp , 0 )
95
96 ! save conjugate original a in b
97
98 b = dconjg(a)
99
100 ! now gather a
101 ! remove all the part not used in the processor
102 if (sub_comm_diag%rank .ne. 0) then
103 do i = 1, n - 1
104 a(1:lda, i) = zzero
105 end do
106 a(1:n, n) = zzero
107 else
108 do i = 1, n
109 a(1:n, i) = zzero
110 end do
111 end if
112 i = 0
113 ind = firsti
114 do while (ind .le. n)
115 do j = 1, n
116 a(ind, j) = z(i + 1, j)
117 end do
118 i = i + 1
119 ind = sub_comm_diag%nproc*i + firsti
120 end do
121
122 deallocate (ap, z)
123 else
124
125 do i = 1, n - 1
126 a(1:lda, i) = zzero
127 end do
128 a(1:n, n) = zzero
129
130 end if
131
132#ifdef PARALLEL
133 dima = 2*(lda*(n - 1) + n)
134 if (yesbcast) then
135 if (sub_comm_diag%yesin) then
136 call reduce_base_real(dima, a, sub_comm_diag%comm, -1)
137 end if
138 call bcast_real(a, dima, 0, comm_mpi)
139 else
140 call reduce_base_real(dima, a, comm_mpi, -1)
141 end if
142#endif
143
144 if (sub_comm_diag%yesin) then
145 ! Now compute the right eigenvectors and put in zr stored rowwise
146 call zgemm_my('T', 'N', n, m, n, zone, a, lda, b, lda, zzero, zr, lda&
147 &, sub_comm_diag%nproc, sub_comm_diag%rank, sub_comm_diag%comm)
148 ! To make the same output of zgesvd
149 ! call conjmat(n,m,zr,lda)
150 !&,nproc,rank,comm_mpi)
151 deallocate (b)
152 end if
153
154#ifdef PARALLEL
155 dima = 2*(lda*(m - 1) + n)
156 call bcast_real(zr, dima, 0, comm_mpi)
157 call bcast_real(w, n, 0, comm_mpi)
158#endif
159 do i = 1, n
160 if (w(i) .gt. 0.d0) then
161 w(i) = dsqrt(w(i))
162 else
163 w(i) = dznrm2(m, zr(i, 1), lda)
164 end if
165 if (w(i) .gt. 0.d0) then
166 do j = 1, m
167 zr(i, j) = zr(i, j)/w(i)
168 end do
169 end if
170 end do
171 return
172end
