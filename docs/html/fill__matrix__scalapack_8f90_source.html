21!> @brief Fills SCALAPACK block matrix with real data from grid calculations
22!> @details This subroutine updates a block matrix in SCALAPACK format using
23!> real data from molecular orbital grid calculations. It performs
24!> matrix multiplication using BLAS routines and handles parallel
25!> reduction across MPI processes.
26!> @param[in] mat_in Input matrix to be updated
27!> @param[in] buffer Buffer array containing molecular orbital values
28!> @param[in] nelorb Leading dimension of the buffer array
29!> @param[in] buffer_weight Weight buffer array
30!> @param[in] nelorb_w Leading dimension of the weight buffer
31!> @param[in] bufbuf Number of buffered grid points
32!> @param[in] irc_ip Row indices for processor grid
33!> @param[in] nrc_ip Number of rows per processor
34!> @param[in] rank_ip Rank mapping for processor grid
35!> @param[in] nlax Leading dimension of the block matrix
36!> @param[in] adr_buf Address offset for buffer access
37subroutine fill_matrix_scalapack(mat_in, buffer, nelorb, buffer_weight, nelorb_w, bufbuf, &
38 irc_ip, nrc_ip, rank_ip, nlax, adr_buf)
39
40#ifdef __SCALAPACK
41
42 use constants, only: zzero
43 use descriptors
44 use allio, only: ortho_cntx, me_blacs, np_ortho, me_ortho, &
45 ortho_comm, ortho_comm_id, rank, commrep_mpi
46 use setup, only: volmesh
47 implicit none
48 ! input
49 integer, intent(in) :: nelorb, bufbuf, irc_ip(np_ortho(1)), nrc_ip(np_ortho(1)), &
50 rank_ip(np_ortho(1), np_ortho(2)), adr_buf, nlax, nelorb_w
51
52 ! REAL VERSION
53 ! nelorb = leading dimension of the buffer as it has been allocated in the calling routine
54 ! descla( nlax_ ) = leading dimension of the block matrix
55 ! adr_buf = index to be added to the address of the buffers buffer/buf_conj
56 ! in order to access its different parts
57 ! This trick must be done since the BLAS on the block matrix needs to know
58 ! the exact leading dimension of the buffer.
59
60 real(8), intent(in) :: buffer(nelorb, *), buffer_weight(nelorb_w, *)
61 real(8), intent(out) :: mat_in(nlax, nlax)
62 ! local
63 integer :: irow, icol, nr, ir, root, ierr, ic, nc, ii
64 real(8), dimension(:, :), allocatable :: work, work2
65 !
66#ifdef PARALLEL
67 include "mpif.h"
68#endif
69 !
70 ! allocate workspace
71 allocate (work(nlax, nlax), work2(nlax, nlax))
72 work = 0.d0
73 work2 = 0.d0
74 !
75#ifdef _OFFLOAD
76!$omp target data map(to:buffer(:,1:bufbuf)&
77!$omp& ,buffer_weight(:,1:bufbuf)) map(to:work)
78#endif
79 do icol = 1, descla(la_npc_) ! loop on column procs
80 nc = nrc_ip(icol)
81 ic = irc_ip(icol) + adr_buf
82 do irow = 1, descla(la_npr_)
83 nr = nrc_ip(irow)
84 ir = irc_ip(irow)
85 ! rank of the processor for which this block (irow,icol) is destinated
86 root = rank_ip(irow, icol)
87 ! use blas subs. on the matrix block
88 if (nr .gt. 0 .and. nc .gt. 0) then
89 call dgemm_('N', 'T', nr, nc, bufbuf, volmesh, &
90 buffer_weight(ir, 1), nelorb_w, buffer(ic, 1), nelorb, 0.d0, work, nlax)
91#ifdef _OFFLOAD
92!$omp target update from(work)
93#endif
94 end if
95 ! accumulate result on dm of root proc.
96#ifdef UNREL
97 call reduce_base_real_to(size(work), work, work2, commrep_mpi, -1)
98 if (rank .eq. root) mat_in = work2 + mat_in
99#else
100 call reduce_base_real_to(size(work), work, work2, commrep_mpi, root)
101#endif
102 end do
103 end do
104#ifdef _OFFLOAD
105!$omp end target data
106#endif
107
108#ifndef UNREL
109 if (descla(lambda_node_) > 0) then
110 mat_in = work2 + mat_in
111 end if
112#endif
113 deallocate (work, work2)
114
115 return
116
117#else
118 return
119#endif
120
121end subroutine fill_matrix_scalapack
