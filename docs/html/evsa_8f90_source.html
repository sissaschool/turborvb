203!> @brief Performs linear fitting with weighted least squares
204!> @details This subroutine performs linear fitting using weighted
205!> least squares method. It computes the slope and intercept
206!> of a linear function that best fits the data points.
207!> @param[in] z Independent variable array
208!> @param[in] y Dependent variable array
209!> @param[in] er Error weights array
210!> @param[in] n Number of data points
211!> @param[out] stot Total sum of weighted squares
212!> @param[out] a Slope of the linear fit
213!> @param[out] c Intercept of the linear fit
214subroutine fun(z, y, er, n, stot, a, c)
215 implicit none
216 real(8) y(*), z(*), er(*), somz, somy, somzy, somzq, wtot, stot, yp, a, c&
217 &, Szy, Szz
218 integer i, n
219 somz = 0.0
220 somy = 0.0
221 somzy = 0.0
222 somzq = 0.0
223
224 wtot = 0.d0
225
226 do i = 1, n
227 somy = somy + y(i)*er(i)
228 somz = somz + z(i)*er(i)
229 somzy = somzy + z(i)*y(i)*er(i)
230 somzq = somzq + z(i)**2*er(i)
231 wtot = wtot + er(i)
232 end do
233 somy = somy/wtot
234 somz = somz/wtot
235 somzy = somzy/wtot
236 somzq = somzq/wtot
237
238 szy = somzy - somz*somy
239 szz = somzq - somz**2
240
241 ! ONLY TWO PARAMETER FIT ....
242 a = szy/szz
243 c = somy - a*somz
244 stot = 0.0
245 do i = 1, n
246 yp = a*z(i) + c
247 stot = stot + (yp - y(i))**2*er(i)
248 end do
249
250 return
251end
