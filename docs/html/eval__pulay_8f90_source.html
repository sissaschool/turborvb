16!> @brief Evaluate Pulay forces for orbital derivatives in quantum Monte Carlo
17!> @details This subroutine computes the Pulay forces (orbital derivatives) needed for
18!> wave function optimization in quantum Monte Carlo calculations. It evaluates the
19!> derivatives of the orbitals with respect to nuclear positions, taking into account
20!> the inverse matrices for up and down electrons, and handles both regular and
21!> spin-dependent Jastrow factors. The calculation includes contributions from
22!> orbital derivatives and Jastrow factor derivatives.
23!> @param[in] winv Inverse of orbital matrix for all electrons
24!> @param[in] nelorb Number of orbitals
25!> @param[in] nel Total number of electrons
26!> @param[in] nelup Number of up electrons
27!> @param[in] neldo Number of down electrons
28!> @param[in] indt Number of derivatives
29!> @param[in] ainvup Inverse of orbital matrix for up electrons
30!> @param[in] ainvdo Inverse of orbital matrix for down electrons
31!> @param[in] winvj Inverse of Jastrow matrix
32!> @param[in] nelorbj Number of Jastrow orbitals
33!> @param[in] winvjbar Inverse of Jastrow matrix (bar)
34!> @param[out] orbderiv Orbital derivatives (3 components per orbital)
35!> @param[in] winvjbarsz Inverse of spin-dependent Jastrow matrix
36!> @param[in] iessz Flag for spin-dependent Jastrow
37!> @param[in] add_pulay Flag for additional Pulay terms
38!> @param[in,out] psip Work array for temporary calculations
39!> @param[in] nelorbh Number of orbitals (half)
40!> @param[in] nelorbjh Number of Jastrow orbitals (half)
41subroutine eval_pulay(winv, nelorb, nel, nelup, neldo, indt &
42 &, ainvup, ainvdo, winvj, nelorbj, winvjbar, orbderiv &
43 &, winvjbarsz, iessz, add_pulay, psip, nelorbh, nelorbjh)
44 use constants, only: ip4
45 implicit none
46 integer nel, nelorb, nelup, indt, neldo, i, l, nelorbj, j, m, d &
47 &, add_pulay, nelorbj5, nelorbh, nelorbjh, indt5, nelorb5
48 real(8) winv(nelorb, 0:indt + ip4, *), orbderiv(3, *) &
49 &, ainvup(nelup, *), ainvdo(neldo, *), ddot, winvjbar(nelorbjh, *) &
50 &, winvjbarsz(*), winvj(nelorbj, 0:indt + ip4, *), tmp1, tmp2 &
51 &, psip(nelorbjh, *)
52 logical iessz
53
54 call dscalzero(3*(nelorb + nelorbj), 0.d0, orbderiv, 1)
55
56 indt5 = indt + ip4 + 1
57 nelorbj5 = nelorbj*indt5
58 nelorb5 = nelorb*indt5
59
60 if (add_pulay .gt. 1) then
61 do l = 1, nelorbh
62 do d = 1, 3
63 orbderiv(d, l) = orbderiv(d, l) &
64 & - ddot(nelup, winv(l, indt + d, 1), nelorb5, ainvup(1, l), 1)
65 orbderiv(d, l) = orbderiv(d, l) &
66 & - ddot(neldo, winv(l, indt + d, 1 + nelup), nelorb5, ainvdo(1, l), 1)
67 end do
68 end do
69 end if
70
71 do i = 1, nelorbjh
72 psip(i, 1) = 0.d0
73 psip(i, 2) = 0.d0
74 psip(i, 3) = 0.d0
75 psip(i, 4) = 0.d0
76 do j = 1, nel
77 psip(i, 1) = psip(i, 1) + winvj(i, indt + 1, j)
78 psip(i, 2) = psip(i, 2) + winvj(i, indt + 2, j)
79 psip(i, 3) = psip(i, 3) + winvj(i, indt + 3, j)
80 psip(i, 4) = psip(i, 4) + winvjbar(i, j)
81 end do
82 end do
83
84 do l = 1, nelorbjh
85 orbderiv(1, nelorb + l) = orbderiv(1, nelorb + l) - psip(l, 1)*psip(l, 4) &
86 & + ddot(nel, winvj(l, indt + 1, 1), nelorbj5, winvjbar(l, 1), nelorbjh)
87 orbderiv(2, nelorb + l) = orbderiv(2, nelorb + l) - psip(l, 2)*psip(l, 4) &
88 & + ddot(nel, winvj(l, indt + 2, 1), nelorbj5, winvjbar(l, 1), nelorbjh)
89 orbderiv(3, nelorb + l) = orbderiv(3, nelorb + l) - psip(l, 3)*psip(l, 4) &
90 & + ddot(nel, winvj(l, indt + 3, 1), nelorbj5, winvjbar(l, 1), nelorbjh)
91 end do
92
93 if (iessz) then
94 do i = 1, nelorbjh
95 psip(i, 1) = 0.d0
96 psip(i, 2) = 0.d0
97 psip(i, 3) = 0.d0
98 psip(i, 4) = 0.d0
99 do j = 1, nelup
100 psip(i, 1) = psip(i, 1) + winvj(i, indt + 1, j)
101 psip(i, 2) = psip(i, 2) + winvj(i, indt + 2, j)
102 psip(i, 3) = psip(i, 3) + winvj(i, indt + 3, j)
103 psip(i, 4) = psip(i, 4) + winvjbarsz(i + (j - 1)*nelorbjh)
104 end do
105 do j = nelup + 1, nel
106 psip(i, 1) = psip(i, 1) - winvj(i, indt + 1, j)
107 psip(i, 2) = psip(i, 2) - winvj(i, indt + 2, j)
108 psip(i, 3) = psip(i, 3) - winvj(i, indt + 3, j)
109 psip(i, 4) = psip(i, 4) - winvjbarsz(i + (j - 1)*nelorbjh)
110 end do
111 end do
112 do l = 1, nelorbjh
113 ! tmp1=ddot(nelorbj,jasmatsz(l),nelorbj,winvj(1,0,j),1)
114 ! tmp2=ddot(nelorbj,jasmatsz(l),nelorbj,winvj(1,0,i),1)
115 orbderiv(1, nelorb + l) = orbderiv(1, nelorb + l) - psip(l, 1)*psip(l, 4) &
116 & + ddot(nel, winvj(l, indt + 1, 1), nelorbj5, winvjbarsz(l), nelorbjh)
117 orbderiv(2, nelorb + l) = orbderiv(2, nelorb + l) - psip(l, 2)*psip(l, 4) &
118 & + ddot(nel, winvj(l, indt + 2, 1), nelorbj5, winvjbarsz(l), nelorbjh)
119 orbderiv(3, nelorb + l) = orbderiv(3, nelorb + l) - psip(l, 3)*psip(l, 4) &
120 & + ddot(nel, winvj(l, indt + 3, 1), nelorbj5, winvjbarsz(l), nelorbjh)
121 end do
122 end if
123
124 return
125end
