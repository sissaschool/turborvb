63!-----------------------------------------------------------------------
64!> @brief Setup 3D spline interpolation for external potential forces
65!> @details Performs 3-dimensional (x, y, z) spline interpolation of
66!> the external potential for force calculations. Sets up B-spline
67!> coefficients and knots for efficient force evaluation.
68!> @note Uses 5th order B-splines for smooth force interpolation
69!> @note Allocates force arrays for ions and electrons
70!> @note Handles QM/MM link atom forces if enabled
71!> @note Calls dbsnak for knot generation and dbs3in for interpolation
72!> @note Sets up van der Waals forces if vdw is enabled
73subroutine forces_interpolate()
74
75 use ext_forces
76 use bspline
77 use allio, only: rank, nion, ieskin, nel
78 use extpot, only: n_x, n_y, n_z, pot, xdata, ydata, zdata, link_atom
79 use splines, only: nxcoef, nycoef, nzcoef
80 use van_der_waals, only: vdw
81
82 implicit none
83
84 f_kx = 5
85 f_ky = 5
86 f_kz = 5
87
88 f_nx = n_x + f_kx
89 f_ny = n_y + f_ky
90 f_nz = n_z + f_kz
91
92 allocate (f_coef(n_x, n_y, n_z))
93 allocate (f_xknot(f_nx), f_yknot(f_ny), f_zknot(f_nz))
94 allocate (forcext(3, nion), forcext_el(3, nel))
95 if (vdw) allocate (force_vdw(3, nion))
96
97 !generate knots
98 call dbsnak(n_x, xdata, f_kx, f_xknot)
99 call dbsnak(n_y, ydata, f_ky, f_yknot)
100 call dbsnak(n_z, zdata, f_kz, f_zknot)
101
102 !interpolate
103 if (rank .eq. 0) then
104 write (*, *)
105 write (*, *) '|******************************************|'
106 write (*, *) '| FORCES FROM THE QMC/MM POTENTIAL |'
107 write (*, *) '|******************************************|'
108 write (*, *)
109 write (*, '(a50,3i4)') &
110 & ' Interpolating with 3D spline of order (x,y,z) : ' &
111 &, f_kx, f_ky, f_kz
112 write (*, *) ''
113 end if
114
115 ! 3D interpolation
116 call dbs3in(n_x, xdata, n_y, ydata, n_z, zdata, pot, &
117 & n_x, n_y, f_kx, f_ky, f_kz, f_xknot, f_yknot, f_zknot, &
118 & f_coef)
119
120 if (rank .eq. 0) then
121 write (*, *) ''
122 write (*, *) ' Interpolation done.'
123 write (*, *) ''
124 write (*, *) ''
125 write (*, *) '|******************************************|'
126 write (*, *) '| FORCES FROM THE QMC/MM POTENTIAL |'
127 write (*, *) '|******************************************|'
128 write (*, *)
129 end if
130 nxcoef = n_x
131 nycoef = n_y
132 nzcoef = n_z
133
134 ! by E. Coccia (17/5/11): forces from QM/MM region
135 if (link_atom) then
136 allocate (mm_f_theta(3, nion), mm_f_dihed(3, nion), mm_f_impr(3, nion))
137 end if
138
139 return
140
141end subroutine forces_interpolate

143!-----------------------------------------------------------------------
144!> @brief Evaluate derivatives of external potential for force calculation
145!> @details Calculates x-, y-, and z-derivatives of the interpolated
146!> external potential at specified grid points. Uses B-spline
147!> derivatives for smooth and accurate force evaluation.
148!> @param[in] nxvec Number of x-coordinates
149!> @param[in] nyvec Number of y-coordinates
150!> @param[in] nzvec Number of z-coordinates
151!> @param[in] xvec Array of x-coordinates for evaluation
152!> @param[in] yvec Array of y-coordinates for evaluation
153!> @param[in] zvec Array of z-coordinates for evaluation
154!> @param[out] dedx x-derivative of potential at evaluation points
155!> @param[out] dedy y-derivative of potential at evaluation points
156!> @param[out] dedz z-derivative of potential at evaluation points
157!> @note Uses dbs3gd for B-spline derivative evaluation
158!> @note Derivatives are computed using pre-computed spline coefficients
159!> @note Supports vectorized evaluation at multiple points
160subroutine forces_evaluate(nxvec, nyvec, nzvec, xvec, yvec, zvec, dedx, dedy, dedz)
161
162 use ext_forces
163 use splines, only: nxcoef, nycoef, nzcoef
164 use bspline
165
166 implicit none
167
168 integer :: nxvec, nyvec, nzvec
169 real*8 :: xvec(nxvec), yvec(nyvec), zvec(nzvec)
170 real*8 :: dedx(nxvec, nyvec, nzvec), dedy(nxvec, nyvec, nzvec), dedz(nxvec, nyvec, nzvec)
171
172 ! derivative with respect to x
173 call dbs3gd(1, 0, 0, nxvec, xvec, nyvec, yvec, nzvec, zvec, &
174 & f_kx, f_ky, f_kz, f_xknot, f_yknot, f_zknot, nxcoef, &
175 & nycoef, nzcoef, f_coef, dedx, nxvec, nyvec)
176 ! derivative with respect to y
177 call dbs3gd(0, 1, 0, nxvec, xvec, nyvec, yvec, nzvec, zvec, &
178 & f_kx, f_ky, f_kz, f_xknot, f_yknot, f_zknot, nxcoef, &
179 & nycoef, nzcoef, f_coef, dedy, nxvec, nyvec)
180 ! derivative with respect to z
181 call dbs3gd(0, 0, 1, nxvec, xvec, nyvec, yvec, nzvec, zvec, &
182 & f_kx, f_ky, f_kz, f_xknot, f_yknot, f_zknot, nxcoef, &
183 & nycoef, nzcoef, f_coef, dedz, nxvec, nyvec)
184
185 return
186
187end subroutine forces_evaluate

189!-----------------------------------------------------------------------
190!> @brief Calculate forces due to external potential
191!> @details Evaluates forces on ions and electrons from the external
192!> potential using interpolated spline derivatives. Handles
193!> both nuclear and electronic forces with proper scaling.
194!> @param[in] nion Number of ions
195!> @param[in] rion Ion positions (3, nion)
196!> @param[in] zetar Ion charges
197!> @param[in] nel Number of electrons
198!> @param[in] rel Electron positions (3, nel)
199!> @note Nuclear forces are scaled by ion charges (zetar)
200!> @note Electronic forces have opposite sign (negative gradient)
201!> @note Forces are zero outside the potential box boundaries
202!> @note Uses forces_evaluate for derivative calculation
203!> @note Handles boundary conditions for potential box
204subroutine ext_force(nion, rion, zetar, nel, rel)
205
206 use ext_forces, only: forcext, forcext_el
207 use extpot, only: x0, delta, n_x, n_y, n_z
208
209 implicit none
210
211 integer, intent(in) :: nion, nel
212 real*8, intent(in) :: rion(3, nion), rel(3, nel)
213 real*8, intent(in) :: zetar(nion)
214
215 integer :: i, j
216 real*8 :: point(3), x1(3)
217 real*8 :: dedx(1, 1, 1), dedy(1, 1, 1), dedz(1, 1, 1)
218
219 ! nuclear forces
220 do i = 1, nion
221 point(:) = rion(:, i)
222 call forces_evaluate(1, 1, 1, point(1), point(2), point(3), dedx, dedy, dedz)
223
224 forcext(1, i) = dedx(1, 1, 1)*zetar(i)
225 forcext(2, i) = dedy(1, 1, 1)*zetar(i)
226 forcext(3, i) = dedz(1, 1, 1)*zetar(i)
227
228 end do
229
230 ! electronic forces
231 x1(1) = x0(1) + (n_x - 1)*delta(1)
232 x1(2) = x0(2) + (n_y - 1)*delta(2)
233 x1(3) = x0(3) + (n_z - 1)*delta(3)
234 do i = 1, nel
235 point(:) = rel(:, i)
236 ! check if the point is out of the potential box
237 if (point(1) .gt. x0(1) .and. point(1) .lt. x1(1) .and. &
238 & point(2) .gt. x0(2) .and. point(2) .lt. x1(2) .and. &
239 & point(3) .gt. x0(3) .and. point(3) .lt. x1(3)) then
240 call forces_evaluate(1, 1, 1, point(1), point(2), point(3), dedx, dedy, dedz)
241
242 forcext_el(1, i) = -dedx(1, 1, 1)
243 forcext_el(2, i) = -dedy(1, 1, 1)
244 forcext_el(3, i) = -dedz(1, 1, 1)
245 else
246 forcext_el(1, i) = 0.d0
247 forcext_el(2, i) = 0.d0
248 forcext_el(3, i) = 0.d0
249 end if
250 end do
251
252 return
253
254end subroutine ext_force

296!-----------------------------------------------------------------------
297!> @brief Calculate van der Waals forces between QM and MM atoms
298!> @details Computes Lennard-Jones forces between quantum mechanical
299!> atoms and classical molecular mechanics atoms. Handles
300!> exclusion lists and 1-4 interactions for link atoms.
301!> @param[in] nion Number of ions (QM atoms)
302!> @param[in] rion Ion positions (3, nion)
303!> @note Uses 12-6 Lennard-Jones potential: V(r) = C12/r^12 - C6/r^6
304!> @note Handles QM/MM link atom exclusions
305!> @note Supports 1-4 interactions with different parameters
306!> @note Forces are accumulated in force_vdw array
307!> @note Handles both QM->MM and MM->QM force calculations
308subroutine vdw_force(nion, rion)
309
310 use van_der_waals
311 use ext_forces, only: force_vdw
312 use exc_list
313 use extpot, only: link_atom
314
315 implicit none
316
317 integer, intent(in) :: nion
318 real*8, intent(in) :: rion(3, nion)
319
320 integer :: i, ii, jj, j, iii, jjj
321 real*8 :: xr, yr, zr, r13, r7, flj, tmp, r1, rr
322 logical :: compute_vdw, compute_14
323
324 force_vdw = 0.d0
325 ! Exclusion list in the case of link atoms
326 if (link_atom) then
327 do i = 1, nat_tot
328 do j = i + 1, nat_tot
329 if (grom(i)%qm .and. .not. grom(j)%qm) then
330 if (compute_vdw(i, j)) then
331 iii = grom(i)%it
332 jjj = grom(j)%it
333 xr = rion(1, iii) - coord_nn(1, jjj)
334 yr = rion(2, iii) - coord_nn(2, jjj)
335 zr = rion(3, iii) - coord_nn(3, jjj)
336 rr = sqrt(xr**2 + yr**2 + zr**2)
337 r1 = rr**(-1)
338 r7 = rr**(-7)
339 r13 = rr**(-13)
340 ii = qmc_vdw(iii)
341 jj = nn_vdw(jjj)
342 if (compute_14(i, j)) then
343 tmp = (12.d0*r13*cs12(ii, jj) - 6.d0*r7*cs6(ii, jj))*r1
344 else
345 tmp = (12.d0*r13*c12(ii, jj) - 6.d0*r7*c6(ii, jj))*r1
346 end if
347 flj = tmp*xr
348 force_vdw(1, iii) = force_vdw(1, iii) + flj
349 flj = tmp*yr
350 force_vdw(2, iii) = force_vdw(2, iii) + flj
351 flj = tmp*zr
352 force_vdw(3, iii) = force_vdw(3, iii) + flj
353 end if
354 elseif (grom(j)%qm .and. .not. grom(i)%qm) then
355 if (compute_vdw(i, j)) then
356 iii = grom(i)%it
357 jjj = grom(j)%it
358 xr = rion(1, jjj) - coord_nn(1, iii)
359 yr = rion(2, jjj) - coord_nn(2, iii)
360 zr = rion(3, jjj) - coord_nn(3, iii)
361 rr = sqrt(xr**2 + yr**2 + zr**2)
362 r1 = rr**(-1)
363 r7 = rr**(-7)
364 r13 = rr**(-13)
365 ii = qmc_vdw(jjj)
366 jj = nn_vdw(iii)
367 if (compute_14(i, j)) then
368 tmp = (12.d0*r13*cs12(ii, jj) - 6.d0*r7*cs6(ii, jj))*r1
369 else
370 tmp = (12.d0*r13*c12(ii, jj) - 6.d0*r7*c6(ii, jj))*r1
371 end if
372 flj = tmp*xr
373 force_vdw(1, jjj) = force_vdw(1, jjj) + flj
374 flj = tmp*yr
375 force_vdw(2, jjj) = force_vdw(2, jjj) + flj
376 flj = tmp*zr
377 force_vdw(3, jjj) = force_vdw(3, jjj) + flj
378 end if
379 end if
380 end do
381 end do
382 else
383 do j = 1, nat_nn
384 do i = 1, nion
385 xr = rion(1, i) - coord_nn(1, j)
386 yr = rion(2, i) - coord_nn(2, j)
387 zr = rion(3, i) - coord_nn(3, j)
388 rr = sqrt(xr**2 + yr**2 + zr**2)
389 r1 = rr**(-1)
390 r7 = rr**(-7)
391 r13 = rr**(-13)
392 ii = qmc_vdw(i)
393 jj = nn_vdw(j)
394 tmp = (12.d0*r13*c12(ii, jj) - 6.d0*r7*c6(ii, jj))*r1
395 flj = tmp*xr
396 force_vdw(1, i) = force_vdw(1, i) + flj
397 flj = tmp*yr
398 force_vdw(2, i) = force_vdw(2, i) + flj
399 flj = tmp*zr
400 force_vdw(3, i) = force_vdw(3, i) + flj
401 end do
402 end do
403 end if
404
405 return
406
407end subroutine vdw_force

409!-----------------------------------------------------------------------
410!> @brief Apply force capping for link atoms in QM/MM simulations
411!> @details Modifies forces on capping atoms based on the relationship
412!> between QM and capping atom positions. Uses the capping
413!> factor to scale forces appropriately.
414!> @param[in,out] rionb Force array (3, nion) - contains energy gradients
415!> @note Force on capping atom: dE/dX_capping = 1/(1-alpha) * dE/dX_QM
416!> @note Uses calpha parameter from link atom definition
417!> @note Modifies forces in-place for all link atoms
418!> @note Handles multiple link atoms (latoms)
419subroutine force_capping(rionb)
420
421 use extpot, only: latoms
422 use link_atoms, only: capping, calpha, qm, prt, cap
423 use allio, only: nion
424 use van_der_waals, only: grom
425
426 implicit none
427
428 real*8, intent(inout), dimension(3, nion) :: rionb
429 integer :: i, j, k, m, jj, kk
430 real*8 :: factor
431
432 ! rionb(:,:) contains the gradients of the energy
433 ! with respect to the atomic positions.
434 ! With the definition of R_capping (subroutine r_capping)
435 ! the gradient for the capping atom is:
436 ! dE/dX_capping = 1/(1-alpha) * dE/dX_QM
437
438 do i = 1, latoms
439 !j = capping(i)%cap
440 !k = capping(i)%qm
441 !jj = grom(j)%is
442 !kk = grom(k)%is
443 !m = capping(i)%mm
444
445 jj = cap(i)
446 kk = qm(i)
447 factor = 1.d0 - calpha(i)
448 factor = 1.d0/factor
449 rionb(:, jj) = factor*rionb(:, kk)
450 end do
451
452 return
453
454end subroutine force_capping

456!-----------------------------------------------------------------------
457!> @brief Calculate angle forces for QM/MM link atoms
458!> @details Computes forces due to bond angle restraints in QM/MM
459!> simulations. Handles mixed QM/MM angle interactions.
460!> @note Uses harmonic potential: V(theta) = k_theta * (theta - theta_eq)^2
461!> @note Forces are computed for QM atoms only
462!> @note Handles I-J-K angle geometry (I and K atoms, J central atom)
463!> @note Accumulates forces in mm_f_theta array
464!> @note Follows GROMOS force field conventions
465subroutine force_angle()
466
467 use allio, only: rion, nion
468 use extpot, only: latoms
469 use van_der_waals, only: grom, coord_nn, nat_tot
470 use link_angle, only: linangle, ntheta
471 use ext_forces, only: mm_f_theta
472
473 implicit none
474
475 integer :: i, j, k, ii, jj, kk, iatom
476 real*8 :: rijsq, rkjsq
477 real*8 :: costh, dvdcos, dvdx, dvdy, dvdz
478 real*8 :: dot, ix, iy, iz, kx, ky, kz, tmp
479 real*8, dimension(3) :: ai, aj, ak, rij, rkj
480 logical :: qm_atom(3)
481
482 ! Derivative of V(theta) with respect to
483 ! the QM atoms
484 ! See the GROMOS manual (II-17)
485
486 mm_f_theta = 0.d0
487 do i = 1, latoms
488 do j = 1, ntheta(i)
489 iatom = 0
490 qm_atom = .false.
491 ! BOND ANGLE for I J and K
492 !
493 ! I K
494 ! \ /
495 ! \ /
496 ! J
497 do k = 1, nat_tot
498 ! I ATOM
499 if (grom(k)%ind .eq. linangle(i, j)%i) then
500 if (grom(k)%qm) then
501 ! QM atom
502 ai(:) = rion(:, grom(k)%it)
503 qm_atom(1) = .true.
504 ii = grom(k)%it
505 else
506 ! MM atom
507 ai(:) = coord_nn(:, grom(k)%it)
508 end if
509 iatom = iatom + 1
510 ! J ATOM
511 elseif (grom(k)%ind .eq. linangle(i, j)%j) then
512 if (grom(k)%qm) then
513 ! QM atom
514 aj(:) = rion(:, grom(k)%it)
515 qm_atom(2) = .true.
516 jj = grom(k)%it
517 else
518 ! MM atom
519 aj(:) = coord_nn(:, grom(k)%it)
520 end if
521 iatom = iatom + 1
522 ! K ATOM
523 elseif (grom(k)%ind .eq. linangle(i, j)%k) then
524 if (grom(k)%qm) then
525 ! QM atom
526 ak(:) = rion(:, grom(k)%it)
527 qm_atom(3) = .true.
528 kk = grom(k)%it
529 else
530 ! MM atom
531 ak(:) = coord_nn(:, grom(k)%it)
532 end if
533 iatom = iatom + 1
534 end if
535 if (iatom .eq. 3) exit
536 end do
537
538 ! I-J distance
539 rij(:) = ai(:) - aj(:)
540 rijsq = sqrt(dot_product(rij, rij))
541
542 ! K-J distance
543 rkj(:) = ak(:) - aj(:)
544 rkjsq = sqrt(dot_product(rkj, rkj))
545
546 ! COS(THETA)
547 dot = dot_product(rij, rkj)
548 tmp = rijsq*rkjsq
549 costh = dot/tmp
550 ! Derivative of V with respect to COS(THETA)
551 dvdcos = -linangle(i, j)%ktheta*(costh - linangle(i, j)%thetaeq)
552
553 ix = (rkj(1)/rkjsq - (rij(1)/rijsq)*costh)/rijsq
554 iy = (rkj(2)/rkjsq - (rij(2)/rijsq)*costh)/rijsq
555 iz = (rkj(3)/rkjsq - (rij(3)/rijsq)*costh)/rijsq
556 kx = (rij(1)/rijsq - (rkj(1)/rkjsq)*costh)/rkjsq
557 ky = (rij(2)/rijsq - (rkj(2)/rkjsq)*costh)/rkjsq
558 kz = (rij(3)/rijsq - (rkj(3)/rkjsq)*costh)/rkjsq
559 ! ATOM I is QM
560 if (qm_atom(1)) then
561 mm_f_theta(1, ii) = mm_f_theta(1, ii) + dvdcos*ix
562 mm_f_theta(2, ii) = mm_f_theta(2, ii) + dvdcos*iy
563 mm_f_theta(3, ii) = mm_f_theta(3, ii) + dvdcos*iz
564 end if
565 ! ATOM J IS QM
566 if (qm_atom(2)) then
567 dvdx = ix + kx
568 dvdy = iy + ky
569 dvdz = iz + kz
570 mm_f_theta(1, jj) = mm_f_theta(1, jj) - dvdcos*dvdx
571 mm_f_theta(2, jj) = mm_f_theta(2, jj) - dvdcos*dvdy
572 mm_f_theta(3, jj) = mm_f_theta(3, jj) - dvdcos*dvdz
573 end if
574 ! ATOM K IS QM
575 if (qm_atom(3)) then
576 mm_f_theta(1, kk) = mm_f_theta(1, kk) + dvdcos*kx
577 mm_f_theta(2, kk) = mm_f_theta(2, kk) + dvdcos*ky
578 mm_f_theta(3, kk) = mm_f_theta(3, kk) + dvdcos*kz
579 end if
580 end do
581 end do
582
583 return

587!-----------------------------------------------------------------------
588!> @brief Calculate dihedral angle forces for QM/MM link atoms
589!> @details Computes forces due to dihedral angle restraints in QM/MM
590!> simulations. Handles mixed QM/MM dihedral interactions.
591!> @note Uses harmonic potential for dihedral angles
592!> @note Forces are computed for QM atoms only
593!> @note Handles I-J-K-L dihedral geometry
594!> @note Accumulates forces in mm_f_dihed array
595!> @note Follows GROMOS force field conventions
596subroutine force_dihed()
597
598 use allio, only: rion, nion
599 use extpot, only: latoms
600 use van_der_waals, only: grom, coord_nn, nat_tot
601 use link_angle, only: lindhd, nphi, pi
602 use ext_forces, only: mm_f_dihed
603
604 implicit none
605
606 integer :: i, j, k, ii, jj, kk, ll, iatom
607 real*8 :: phi, cosphi, dvdphi, dvdx, dvdy, dvdz, der
608 real*8 :: dum, dot, dot1, app, rkj2, rim(3), rln(3), app1(3), app4
609 real*8 :: ix, iy, iz, lx, ly, lz
610 real*8 :: rij(3), rkj(3), rkl(3), rmj2, rnk2, rnk(3), rmj(3), rlnsq, rimsq, app2, app3
611 real*8, dimension(3) :: ai, aj, ak, al
612 logical :: qm_atom(4)
613
614 ! Derivative of V(phi) with respect to
615 ! the QM atoms
616
617 mm_f_dihed = 0.d0
618 do i = 1, latoms
619 do j = 1, nphi(i)
620 iatom = 0
621 qm_atom = .false.
622 do k = 1, nat_tot
623 ! I ATOM
624 if (grom(k)%ind .eq. lindhd(i, j)%i) then
625 if (grom(k)%qm) then
626 ! QM atom
627 ai(:) = rion(:, grom(k)%it)
628 qm_atom(1) = .true.
629 ii = grom(k)%it
630 else
631 ! MM atom
632 ai(:) = coord_nn(:, grom(k)%it)
633 end if
634 iatom = iatom + 1
635 ! J ATOM
636 elseif (grom(k)%ind .eq. lindhd(i, j)%j) then
637 if (grom(k)%qm) then
638 ! QM atom
639 aj(:) = rion(:, grom(k)%it)
640 qm_atom(2) = .true.
641 jj = grom(k)%it
642 else
643 ! MM atom
644 aj(:) = coord_nn(:, grom(k)%it)
645 end if
646 iatom = iatom + 1
647 ! K ATOM
648 elseif (grom(k)%ind .eq. lindhd(i, j)%k) then
649 if (grom(k)%qm) then
650 ! QM atom
651 ak(:) = rion(:, grom(k)%it)
652 qm_atom(3) = .true.
653 kk = grom(k)%it
654 else
655 ! MM atom
656 ak(:) = coord_nn(:, grom(k)%it)
657 end if
658 iatom = iatom + 1
659 ! L ATOM
660 elseif (grom(k)%ind .eq. lindhd(i, j)%l) then
661 if (grom(k)%qm) then
662 ! QM atom
663 al(:) = rion(:, grom(k)%it)
664 qm_atom(4) = .true.
665 ll = grom(k)%it
666 else
667 ! MM atom
668 al(:) = coord_nn(:, grom(k)%it)
669 end if
670 iatom = iatom + 1
671 end if
672 if (iatom .eq. 4) exit
673 end do
674
675 call compute_dihed(ai, aj, ak, al, dum, dum, rij, rkj, rkl, rmj2, rnk2, rnk, rmj)
676 !See the GROMOS manual (II-26)
677 rkj2 = dot_product(rkj, rkj)
678 dot = dot_product(rij, rkj)
679 dot1 = dot_product(rkl, rkj)
680 app2 = dot/rkj2 - 1.d0
681 app3 = dot1/rkj2
682 app4 = app2 + 1.d0
683 rim(:) = rij(:) - (dot/rkj2)*rkj(:)
684 rln(:) = -rkl(:) + (dot1/rkj2)*rkj(:)
685
686 app = dot_product(rim, rln)
687 app = app/sqrt(dot_product(rim, rim)*dot_product(rln, rln))
688 phi = dacos(app)
689 call cross(rkj, rkl, app1)
690 phi = sign(1.d0, dot_product(rij, app1))*phi
691 phi = modulo(phi, 2.d0*pi)
692
693 ! dV/dcos(m*phi) * dcos(m*phi)/dcos(phi)
694 call cos_mult_der(phi, lindhd(i, j)%mult, der)
695 dvdphi = -lindhd(i, j)%kphi*lindhd(i, j)%pcos*der
696
697 cosphi = cos(phi)
698 rlnsq = sqrt(dot_product(rln, rln))
699 rimsq = sqrt(dot_product(rim, rim))
700 rlnsq = 1.d0/rlnsq
701 rimsq = 1.d0/rimsq
702
703 ix = (rln(1)*rlnsq - rim(1)*rimsq*cosphi)*rimsq
704 iy = (rln(2)*rlnsq - rim(2)*rimsq*cosphi)*rimsq
705 iz = (rln(3)*rlnsq - rim(3)*rimsq*cosphi)*rimsq
706 lx = (rim(1)*rimsq - rln(1)*rlnsq*cosphi)*rlnsq
707 ly = (rim(2)*rimsq - rln(2)*rlnsq*cosphi)*rlnsq
708 lz = (rim(3)*rimsq - rln(3)*rlnsq*cosphi)*rlnsq
709
710 ! ATOM I is QM
711 if (qm_atom(1)) then
712
713 mm_f_dihed(1, ii) = mm_f_dihed(1, ii) + dvdphi*ix
714 mm_f_dihed(2, ii) = mm_f_dihed(2, ii) + dvdphi*iy
715 mm_f_dihed(3, ii) = mm_f_dihed(3, ii) + dvdphi*iz
716 end if
717 ! ATOM J IS QM
718 if (qm_atom(2)) then
719
720 dvdx = app2*ix - app3*lx
721 dvdy = app2*iy - app3*ly
722 dvdz = app2*iz - app3*lz
723
724 mm_f_dihed(1, jj) = mm_f_dihed(1, jj) + dvdphi*dvdx
725 mm_f_dihed(2, jj) = mm_f_dihed(2, jj) + dvdphi*dvdy
726 mm_f_dihed(3, jj) = mm_f_dihed(3, jj) + dvdphi*dvdz
727 end if
728 ! ATOM K IS QM
729 if (qm_atom(3)) then
730
731 dvdx = app4*ix - (app3 - 1.d0)*lx
732 dvdy = app4*iy - (app3 - 1.d0)*ly
733 dvdz = app4*iz - (app3 - 1.d0)*lz
734
735 mm_f_dihed(1, kk) = mm_f_dihed(1, kk) - dvdphi*dvdx
736 mm_f_dihed(2, kk) = mm_f_dihed(2, kk) - dvdphi*dvdy
737 mm_f_dihed(3, kk) = mm_f_dihed(3, kk) - dvdphi*dvdz
738 end if
739 ! ATOM L IS QM
740 if (qm_atom(4)) then
741
742 mm_f_dihed(1, ll) = mm_f_dihed(1, ll) + dvdphi*lx
743 mm_f_dihed(2, ll) = mm_f_dihed(2, ll) + dvdphi*ly
744 mm_f_dihed(3, ll) = mm_f_dihed(3, ll) + dvdphi*lz
745 end if
746 end do
747 end do
748
749 return
750
751end subroutine force_dihed

771
772 integer :: i, j, k, ii, jj, kk, ll, iatom
773 real*8 :: phi, cosphi, dvdphi, dvdx, dvdy, dvdz, rkj2sq
774 real*8 :: rkj2, rmj2, tmp, dot, rnk2, tmp1, dot1
775 real*8 :: ix, iy, iz, lx, ly, lz

799 end if
800 iatom = iatom + 1
801 ! J ATOM
802 elseif (grom(k)%ind .eq. linimp(i, j)%j) then
803 if (grom(k)%qm) then
804 ! QM atom
805 aj(:) = rion(:, grom(k)%it)
806 qm_atom(2) = .true.
807 jj = grom(k)%it
808 else
809 ! MM atom
810 aj(:) = coord_nn(:, grom(k)%it)
811 end if
812 iatom = iatom + 1
813 ! K ATOM
814 elseif (grom(k)%ind .eq. linimp(i, j)%k) then
815 if (grom(k)%qm) then
816 ! QM atom
817 ak(:) = rion(:, grom(k)%it)
818 qm_atom(3) = .true.
819 kk = grom(k)%it
820 else
821 ! MM atom
822 ak(:) = coord_nn(:, grom(k)%it)
823 end if
824 iatom = iatom + 1
825 ! L ATOM
826 elseif (grom(k)%ind .eq. linimp(i, j)%l) then
827 if (grom(k)%qm) then
828 ! QM atom
829 al(:) = rion(:, grom(k)%it)
830 qm_atom(4) = .true.
831 ll = grom(k)%it
832 else
833 ! MM atom

835 end if
836 iatom = iatom + 1
837 end if

839 end do

854 lz = tmp1*rnk(3)

865 end if

868

870 dvdy = (dot/rkj2 - 1.d0)*iy - (dot1/rkj2)*ly
871 dvdz = (dot/rkj2 - 1.d0)*iz - (dot1/rkj2)*lz
872
873 mm_f_impr(1, jj) = mm_f_impr(1, jj) + dvdphi*dvdx
874 mm_f_impr(2, jj) = mm_f_impr(2, jj) + dvdphi*dvdy

876 end if
877 ! ATOM K IS QM
878 if (qm_atom(3)) then
879
880 dvdx = (dot/rkj2)*ix - (dot1/rkj2 - 1.d0)*lx
881 dvdy = (dot/rkj2)*iy - (dot1/rkj2 - 1.d0)*ly
882 dvdz = (dot/rkj2)*iz - (dot1/rkj2 - 1.d0)*lz
883
884 mm_f_impr(1, kk) = mm_f_impr(1, kk) - dvdphi*dvdx
885 mm_f_impr(2, kk) = mm_f_impr(2, kk) - dvdphi*dvdy
886 mm_f_impr(3, kk) = mm_f_impr(3, kk) - dvdphi*dvdz

888 ! ATOM L IS QM
889 if (qm_atom(4)) then
890
891 mm_f_impr(1, ll) = mm_f_impr(1, ll) + dvdphi*lx
892 mm_f_impr(2, ll) = mm_f_impr(2, ll) + dvdphi*ly
893 mm_f_impr(3, ll) = mm_f_impr(3, ll) + dvdphi*lz

896 end do

904!> @brief Calculate derivative of cos(mult*phi) with respect to cos(phi)

906!> for use in dihedral force calculations.
907!> @param[in] phi Dihedral angle in radians
908!> @param[in] mult Multiplicity factor (0-6)
909!> @param[out] der Derivative value
910!> @note Supports multiplicities 0-6
911!> @note Returns 0 for mult=0, 1 for mult=1
912!> @note Higher multiplicities use Chebyshev polynomial forms
913subroutine cos_mult_der(phi, mult, der)
914 !*****************************************************************
915
916 implicit none
917
918 real*8, intent(in) :: phi
919 integer, intent(in) :: mult
920 real*8, intent(out) :: der
921
922 ! Calculation of dcos(mult*phi)/dcos(phi)
923 if (mult .eq. 0) then
924 der = 0.d0
925 elseif (mult .eq. 1) then
926 der = 1.d0
927 elseif (mult .eq. 2) then
928 der = 4.d0*cos(phi)
929 elseif (mult .eq. 3) then
930 der = 12.d0*cos(phi)**2 - 3.d0
931 elseif (mult .eq. 4) then
932 der = 32.d0*cos(phi)**3 - 16.d0*cos(phi)
933 elseif (mult .eq. 5) then
934 der = 80.d0*cos(phi)**4 - 60.d0*cos(phi)**2 + 5.d0
935 elseif (mult .eq. 6) then
936 der = 192.d0*cos(phi)**5 - 192.d0*cos(phi)**3 + 36.d0*cos(phi)
937 end if

943!-----------------------------------------------------------------------
944!> @brief Calculate forces for harmonic distance restraints
945!> @details Computes forces due to harmonic bond length restraints
946!> between pairs of atoms. Uses harmonic potential.
947!> @note Uses potential: V(r) = k_bond * (r - r_eq)^2
948!> @note Forces are scaled by mm_fact factor
949!> @note Forces on atom pairs are equal and opposite
950!> @note Accumulates forces in restr_f_bond array
951subroutine rf_bond()
952 !******************************************************************
953
954 ! Forces for the harmonic distance restraint
955
956 use allio, only: rion
957 use cl_restr, only: mm_fact, nbonds, restr_f_bond
958 use link_angle, only: cl_bond
959
960 implicit none
961
962 integer :: i, j, kk
963 real*8 :: x, y, z, rr, dvdr
964
965 restr_f_bond = 0.d0
966 do kk = 1, nbonds
967
968 i = cl_bond(kk)%i
969 j = cl_bond(kk)%j
970
971 x = rion(1, i) - rion(1, j)
972 y = rion(2, i) - rion(2, j)
973 z = rion(3, i) - rion(3, j)
974
975 rr = sqrt(x**2 + y**2 + z**2)
976
977 dvdr = -cl_bond(kk)%kbond*(rr - cl_bond(kk)%req)

981 restr_f_bond(2, i) = restr_f_bond(2, i) + dvdr*y/rr*mm_fact
982 restr_f_bond(3, i) = restr_f_bond(3, i) + dvdr*z/rr*mm_fact
983
984 !ATOM J
985 restr_f_bond(1, j) = -restr_f_bond(1, i)
986 restr_f_bond(2, j) = -restr_f_bond(2, i)

989 end do

995!-----------------------------------------------------------------------
996!> @brief Calculate forces for harmonic bond angle restraints
997!> @details Computes forces due to harmonic bond angle restraints between
998!> triplets of atoms (I-J-K). Uses harmonic potential:
999!> V(theta) = k_theta * (cos(theta) - cos(theta_eq))^2
1000!> @note Forces are scaled by mm_fact factor
1001!> @note Forces are distributed among the three atoms in the angle
1002!> @note Accumulates forces in restr_f_angle array
1003!> @note Follows GROMOS force field conventions
1004subroutine rf_angle()
1005
1006 use allio, only: rion
1007 use cl_restr, only: mm_fact, nth, restr_f_angle
1008 use link_angle, only: cl_angle
1009
1010 implicit none
1011
1012 integer :: i, j, k, kk
1013 real*8 :: rijsq, rkjsq
1014 real*8 :: costh, dvdcos, dvdx, dvdy, dvdz
1015 real*8 :: dot, ix, iy, iz, kx, ky, kz, tmp
1016 real*8, dimension(3) :: rij, rkj
1017
1018 restr_f_angle = 0.d0
1019 do kk = 1, nth
1020
1021 i = cl_angle(kk)%i
1022 j = cl_angle(kk)%j
1023 k = cl_angle(kk)%k
1024
1025 ! I-J distance
1026 rij(:) = rion(:, i) - rion(:, j)
1027 rijsq = sqrt(dot_product(rij, rij))
1028
1029 ! K-J distance
1030 rkj(:) = rion(:, k) - rion(:, j)
1031 rkjsq = sqrt(dot_product(rkj, rkj))
1032
1033 ! COS(THETA)
1034 dot = dot_product(rij, rkj)
1035 tmp = rijsq*rkjsq
1036 costh = dot/tmp
1037 ! Derivative of V with respect to COS(THETA)
1038 dvdcos = -cl_angle(kk)%ktheta*(costh - cl_angle(kk)%thetaeq)
1039
1040 ix = (rkj(1)/rkjsq - (rij(1)/rijsq)*costh)/rijsq
1041 iy = (rkj(2)/rkjsq - (rij(2)/rijsq)*costh)/rijsq
1042 iz = (rkj(3)/rkjsq - (rij(3)/rijsq)*costh)/rijsq
1043 kx = (rij(1)/rijsq - (rkj(1)/rkjsq)*costh)/rkjsq
1044 ky = (rij(2)/rijsq - (rkj(2)/rkjsq)*costh)/rkjsq
1045 kz = (rij(3)/rijsq - (rkj(3)/rkjsq)*costh)/rkjsq
1046 ! ATOM I
1047 restr_f_angle(1, i) = restr_f_angle(1, i) + dvdcos*ix*mm_fact
1048 restr_f_angle(2, i) = restr_f_angle(2, i) + dvdcos*iy*mm_fact
1049 restr_f_angle(3, i) = restr_f_angle(3, i) + dvdcos*iz*mm_fact
1050 ! ATOM J
1051 dvdx = ix + kx
1052 dvdy = iy + ky
1053 dvdz = iz + kz
1054 restr_f_angle(1, j) = restr_f_angle(1, j) - dvdcos*dvdx*mm_fact
1055 restr_f_angle(2, j) = restr_f_angle(2, j) - dvdcos*dvdy*mm_fact
1056 restr_f_angle(3, j) = restr_f_angle(3, j) - dvdcos*dvdz*mm_fact
1057 ! ATOM K
1058 restr_f_angle(1, k) = restr_f_angle(1, k) + dvdcos*kx*mm_fact
1059 restr_f_angle(2, k) = restr_f_angle(2, k) + dvdcos*ky*mm_fact
1060 restr_f_angle(3, k) = restr_f_angle(3, k) + dvdcos*kz*mm_fact
1061 end do
1062
1063 return
1064
1065end subroutine rf_angle

1067!-----------------------------------------------------------------------
1068!> @brief Calculate forces for harmonic dihedral angle restraints
1069!> @details Computes forces due to harmonic dihedral angle restraints
1070!> between quadruplets of atoms. Uses harmonic potential for dihedrals.
1071!> @note Uses potential: V(phi) = k_qhi * (cos(phi) - cos(phi_eq))^2
1072!> @note Forces are scaled by mm_fact factor
1073!> @note Forces are distributed among the four atoms in the dihedral
1074!> @note Accumulates forces in restr_f_dihe array
1075subroutine rf_dihe()
1076 !******************************************************************
1077
1078 ! Forces for the harmonic restraint around a proper dihedral
1079
1080 use allio, only: rion, rank
1081 use cl_restr, only: mm_fact, ndihe, restr_f_dihe
1082 use link_angle, only: cl_dihe, pi
1083
1084 implicit none
1085
1086 integer :: i, j, k, l, mm
1087 real*8 :: phi, cosphi, dvdphi, dvdx, dvdy, dvdz, der
1088 real*8 :: dum, dot, dot1, app, rkj2, rim(3), rln(3), app1(3), app4
1089 real*8 :: ix, iy, iz, lx, ly, lz
1090 real*8 :: rij(3), rkj(3), rkl(3), rmj2, rnk2, rnk(3), rmj(3), rlnsq, rimsq, app2, app3
1091 real*8, dimension(3) :: ai, aj, ak, al
1092
1093 restr_f_dihe = 0.d0
1094
1095 do mm = 1, ndihe
1096
1097 i = cl_dihe(mm)%i
1098 j = cl_dihe(mm)%j
1099 k = cl_dihe(mm)%k
1100 l = cl_dihe(mm)%l
1101
1102 call compute_dihed(rion(:, i), rion(:, j), rion(:, k), rion(:, l), dum, dum, rij, rkj, rkl, rmj2, rnk2, rnk, rmj)
1103 !See the GROMOS manual (II-26)
1104 rkj2 = dot_product(rkj, rkj)
1105 dot = dot_product(rij, rkj)
1106 dot1 = dot_product(rkl, rkj)
1107 app2 = dot/rkj2 - 1.d0
1108 app3 = dot1/rkj2
1109 app4 = app2 + 1.d0
1110 rim(:) = rij(:) - (dot/rkj2)*rkj(:)
1111 rln(:) = -rkl(:) + (dot1/rkj2)*rkj(:)
1112
1113 app = dot_product(rim, rln)
1114 app = app/sqrt(dot_product(rim, rim)*dot_product(rln, rln))
1115 phi = dacos(app)
1116 call cross(rkj, rkl, app1)
