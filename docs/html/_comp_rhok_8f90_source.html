16!> @brief Computes density in k-space from plane wave matrix
17!> @details This subroutine computes the density in k-space (rhok) by
18!> summing the plane wave matrix contributions from all particles.
19!> It separates contributions from different particle types based
20!> on the nppss array.
21!> @param[out] rhoknew Total density in k-space
22!> @param[out] rhoknew2 Density in k-space separated by particle type (2*nvects, 2)
23!> @param[in] nparts Number of particles
24!> @param[in] nvects Number of wave vectors
25!> @param[in] pwmat Plane wave matrix (2*nvects, nparts)
26!> @param[in] nppss Number of particles per species
27subroutine comprhok(rhoknew, rhoknew2, nparts, nvects, pwmat, nppss)
28 ! compute rhok
29 implicit none
30
31 integer j, k, nparts, nvects, nvects2, nppss(*)
32 real*8 rhoknew(*), rhoknew2(2*nvects, *)
33 real*8 pwmat(2*nvects, *)
34
35 nvects2 = 2*nvects
36
37 do k = 1, nvects2
38 rhoknew(k) = 0.d0
39 rhoknew2(k, 1) = 0.d0
40 rhoknew2(k, 2) = 0.d0
41 end do
42
43 ! electron contribution
44 do j = 1, nparts
45 do k = 1, nvects2
46 rhoknew(k) = rhoknew(k) + pwmat(k, j)
47 if (j .le. nppss(1)) then
48 rhoknew2(k, 1) = rhoknew2(k, 1) + pwmat(k, j)
49 else
50 rhoknew2(k, 2) = rhoknew2(k, 2) + pwmat(k, j)
51 end if
52 end do
53 end do
54
55 return
56end
