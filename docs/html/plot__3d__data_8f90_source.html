16!> @brief 3D data visualization output in XCrySDen format
17!>
18!> This subroutine generates 3D data visualization files in XCrySDen (.xsf) format
19!> for quantum Monte Carlo calculations. It outputs crystal structure information
20!> and 3D data grids that can be visualized using XCrySDen or similar software.
21!>
22!> The subroutine handles unit conversion from atomic units to Angstroms and
23!> supports both periodic and non-periodic systems with proper file naming
24!> conventions.
25!>
26!> @param[in] ipc Number of components per grid point (1 for scalar, 3 for vector)
27!> @param[in] cell Cell dimensions in atomic units (3)
28!> @param[in] cellscale Cell scaling factors in atomic units (3)
29!> @param[in] natoms Number of atoms in the system
30!> @param[in] pos Atomic positions in atomic units (3, natoms)
31!> @param[in] zeta Atomic numbers or charges (natoms)
32!> @param[in] periodic Flag indicating periodic boundary conditions
33!> @param[in] mesh Grid dimensions (3)
34!> @param[in] origin Origin of the grid in atomic units (3)
35!> @param[in] datagrid 3D data grid to be plotted (ipc*mesh(1), mesh(2), mesh(3))
36!> @param[in] indmol Molecular index for file naming
37!> @param[in] word Identifier string for file naming
38!>
39!> @details
40!> The subroutine performs the following operations:
41!> 1. Checks file unit availability and opens output file
42!> 2. Converts all units from atomic units to Angstroms:
43!> - Cell dimensions, scaling factors, origin, and positions
44!> - Data grid values (divided by length_unit^3)
45!> 3. Writes XCrySDen format header:
46!> - CRYSTAL section with primitive vectors
47!> - PRIMCOORD section with atomic positions and charges
48!> 4. Writes 3D data grid section:
49!> - BEGIN_BLOCK_DATAGRID_3D header
50!> - Grid dimensions and origin
51!> - Primitive vectors for grid orientation
52!> - Data values in optimized format (3 values per line)
53!> 5. Closes the output file
54!>
55!> @note Output filename format: output_<word><indmol>.xsf
56!> @note The subroutine handles both scalar (ipc=1) and vector (ipc=3) data
57!> @note Data is written in chunks of 3 values per line for efficiency
58!> @note Used for visualizing electron density, wave functions, and other
59!> quantum mechanical properties
60!> @note Compatible with XCrySDen, VESTA, and other visualization software
61subroutine plot_3d_data(ipc, cell, cellscale, natoms, pos, zeta, periodic, mesh, origin, datagrid, indmol, word)
62
63 use constants, only: length_unit
64
65 implicit none
66 integer, parameter :: ofile = 55555555
67 !
68 integer, intent(in) :: mesh(3), ipc
69 double precision :: datagrid(ipc*mesh(1), mesh(2), mesh(3))
70 double precision :: cell(3), cell_rescaled(3), cellscale(3), cellscale_rescaled(3)
71 double precision :: origin(3), origin_rescaled(3)
72 double precision :: pos(3, natoms), pos_rescaled(3, natoms)
73 double precision, parameter :: eps = 1.d-6
74 real*8, intent(in) :: zeta(natoms)
75 integer, intent(in) :: natoms
76 logical, intent(in) :: periodic
77 character(14), intent(in) :: word
78 !
79 integer :: i1, i2, indmol, idigit(6), ix, iy, iz, i3
80 logical :: control
81 logical :: UNITOK, UNITOP
82
83 inquire (unit=ofile, exist=unitok, opened=unitop)
84
85 if (unitok .and. .not. unitop) then
86
87 call convertdec(indmol, idigit)
88 open (unit=ofile, file="output_"//trim(word)//char(idigit(1))//char(idigit(2))// &
89 char(idigit(3))//char(idigit(4))//char(idigit(5))//char(idigit(6))//".xsf", form="formatted", status="unknown")
90
91 else
92
93 write (6, *) 'error from open unit in plot_3d_data, unit already opened'
94 return
95
96 end if
97
98 ! rescale units from au to Angstrom
99 cell_rescaled = cell*length_unit
100 cellscale_rescaled = cellscale*length_unit
101 origin_rescaled = origin*length_unit
102 pos_rescaled = pos*length_unit
103 datagrid = datagrid/length_unit**3
104
105 !
106
107 write (ofile, *) 'CRYSTAL '
108 write (ofile, *) 'PRIMVEC '
109 write (ofile, *) cellscale_rescaled(1), 0.0, 0.0
110 write (ofile, *) 0.0, cellscale_rescaled(2), 0.0
111 write (ofile, *) 0.0, 0.0, cellscale_rescaled(3)
112 write (ofile, *) 'PRIMCOORD '
113 write (ofile, *) natoms, 1
114 do i1 = 1, natoms
115 write (ofile, '(i3,6(F16.8,4X))') nint(zeta(i1)), pos_rescaled(:, i1)
116 end do
117
118 if (any(mesh(:) .ne. 0)) then
119 write (ofile, *)
120 write (ofile, *) 'BEGIN_BLOCK_DATAGRID_3D'
121 write (ofile, *) ' my_datagrid'
122 write (ofile, *) ' BEGIN_DATAGRID_3D_datagrid'
123 write (ofile, '(3i5)') mesh(:)
124 write (ofile, '(3f14.8)') origin_rescaled(:)
125 write (ofile, '(3f14.8)') cell_rescaled(1), 0.0, 0.0
126 write (ofile, '(3f14.8)') 0.0, cell_rescaled(2), 0.0
127 write (ofile, '(3f14.8)') 0.0, 0.0, cell_rescaled(3)
128 !
129 ! do ix=1,mesh(1)
130 ! do iy=1,mesh(2)
131 ! write(ofile,*) (datagrid(ix,iy,iz),iz=1,mesh(3))
132 ! enddo
133 ! enddo
134 !
135 !write(ofile,*) (((datagrid(ix,iy,iz),iz=1,mesh(3)),iy=1,mesh(2)),ix=1,mesh(1))
136 do i1 = 1, mesh(3)
137 do i2 = 1, mesh(2)
138 ! write(ofile,'(1000(e14.6,4x))') (datagrid(i3,i2,i1),i3=1,mesh(1))
139 do i3 = 1, mesh(1), 3
140 if (i3 + 2 .le. mesh(1)) then
141 write (ofile, *) datagrid(ipc*(i3 - 1) + 1, i2, i1), datagrid(ipc*i3 + 1, i2, i1)&
142 &, datagrid(ipc*(i3 + 1) + 1, i2, i1)
143 elseif (i3 + 1 .le. mesh(1)) then
144 write (ofile, *) datagrid(ipc*(i3 - 1) + 1, i2, i1), datagrid(ipc*i3 + 1, i2, i1)
145 else
146 write (ofile, *) datagrid(ipc*(i3 - 1) + 1, i2, i1)
147 end if
148 end do
149 end do
150 end do
151 !
152 write (ofile, *) 'END_DATAGRID_3D'
153 write (ofile, *) 'END_BLOCK_DATAGRID_3D'
154
155 end if
156
157 close (ofile)
158
159end subroutine plot_3d_data
