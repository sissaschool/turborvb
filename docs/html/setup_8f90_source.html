203 !> scratch files, reading pseudopotentials, initializing random

208
209 implicit none

228 iflagerrall = 0

230 !

252 if ((yeswrite10 .or. yesread10) .and. rankrep .eq. 0) then
253
254#ifdef __KCOMP
255 charascratch = trim(scratchpath)//'fort.10_'//trim(chara)
256#else

258#endif

260 end if
261 !
262 ! open the formatted files to save LDA quantities needed by post processing tools
263 ! and nonSC run:
264 ! total_densities.sav -> charge/spin densities
265 ! wavefunction.sav -> eigenvectors
266 !
267 if (rank .eq. 0 .and. writescratch .eq. 0) then
268 ! save total charge/spin densities
269 charascratch = trim(scratchpath)//'total_densities.sav'
270 open (unit=unit_scratch_densities, file=charascratch, form='unformatted', position='rewind', status='unknown')
271 charascratch = trim(scratchpath)//'wavefunction.sav'
272 open (unit=unit_scratch_eigenvects, file=charascratch, form='unformatted', position='rewind', status='unknown')
273 end if
274 !
275 ! reading pseudo potential if any
276 !
277 mindist = 1d-8 ! Initialization minimo e-i distance.

280 pseudologic = pseudorandom
281 !
282 ! these indices are required by the "write_fort10" at the
283 ! end of a SC calculation.
284 !
285 iseed = abs(iseedr)
286 nmat = np + 1
287 ndim = np + 1
288 npm = np + 1
289 npmn = nbinmax + 1
290 nmats = nmat - iesm - iesd
291 ndimp = np
292 ndimpdim = max(ndimp, 1)
293 nwm = nw + 1
294 nws = in1 + 1
295 nwm = nw
296 nws = in1
297 npp = np + 1
298 npf = np
299 nprest = np
300 nindt = 0
301 ! initialize random number generator anyway
302 irstart = iseed
303 do i = 1, rank
304 irstart = ran(iseed)*2**29
305 end do
306 iseed = 2*irstart + 1
307 call rand_init(iseed)
308 !
309 ! reading the wave function fort.10/fort.10_new
310 ! Get all the parameters related to the system.
311 !
312 ! with the new convention for read_fort10, we have to open
313 ! the file for each processor pool.
314 if (rank .eq. 0) then
315 if (iopt .eq. 1) then
316 open (unit=ufort10, file='fort.10', form='formatted', status='unknown')
317 else
318 open (unit=ufort10, file='fort.10_new', form='formatted', status='unknown')
319 end if
320 end if
321
322 if ((iopt .ne. 1 .and. kaverage .and. yesread10) .or. (manyfort10 .and. yesread10 .and. nbead .gt. 1)) then
323 !
324 ! In case of yeswrite10=.true. and continuation
325 ! read molecular orbitals from each turborvb.scratch/fort.10_00xxxx and
326 ! fill the matrix dup_c for all k-points separately.
327 !
328 indk = rankcolrep + 1
329 call read_fort10(unit_scratch_fort10)
330 ! if k-point read from input is not correct, substitute it with the w.f. phase
331 if (all(abs(phase(:) - xkp(:, indk)) .gt. 1d-10) .or. &
332 all(abs(phase_down(:) - xkp_down(:, indk)) .gt. 1d-10)) then
333 xkp(:, indk) = phase(:)
334 xkp_down(:, indk) = phase_down(:)
335 call error(' prep ', ' Some input k-points are wrong, substituted with WF phase! ', -1, rank)
336 end if
337 else
338 ! in the other cases (start from scratch, or wave functions for each k-point not
339 ! saved to disk, then just read the same fort.10 for all.
340 commrep_mpi_sav = commrep_mpi
341 rankrep_sav = rankrep
342#ifdef PARALLEL
343 commrep_mpi = mpi_comm_world
344#else
345 commrep_mpi = 0
346#endif
347 call read_fort10(ufort10)
348 commrep_mpi = commrep_mpi_sav
349 rankrep = rankrep_sav
350 end if
351
352 !
353 ! printout parameters related to Jastrow 1body
354 if (rank .eq. 0) then
355 write (6, *) ' symmagp =', symmagp
356 write (6, *) 'Number 1body Jastrow parameters: ', niesd
357 if (niesd .ge. 1) then
358 do i = 1, niesd
359 write (6, *) i, vj(i)
360 end do
361 else
362 write (6, *) '1body Jastrow not present'
363 end if
364 if (iesdr .le. -5) then
365 write (6, *) ' initial costz, costz3, zeta_Q_Caffarel '
366 do i = 1, nion
367 write (6, *) i, costz(i), costz3(i), zetaq(i)
368 end do
369 end if
370 end if
371 do j = 1, nw
372 jbra(j) = j
373 end do
374 ! main check after reading the wave function
375 if (contraction .eq. 0) &
376 call error(' setup_calculation ', ' DFT works only with molecular orbitals ', 1, rank)
377 !
378 ! read card MOLECULAR from std input
379 ! and set default values

382 ay = 0.d0
383 az = 0.d0
384 nx = 0
385 ny = 0
386 nz = 0
387 nmolmax = 0
388 nmolmin = 0
389 nmol = 0
390 yesmin = 0
391 weight_loc = -1.d0
392 power = 1.d0
393 call read_datasmin_mol()
394 !
395 ! read DFT cards from std input
396 ! and set default values
397 !
398 ! &DFT card
399 maxit = 100
400 mixing = -1.d0
401 typedft = 1
402 epsdft = 1.d-7
403 optocc = -1 ! Do not optimize the occupation

412 neloccdo = 0 ! Used for LSDA
413 bands = 2*neldo + nelup - neldo ! Assuming at most 8 fold degeneracy last shell
414 mixingder = -1.d0
415 maxcg = 40
416 weightcorr = 1.d0 ! the amount of correlation, weightcorr=0--> fake HF.
417 epsshell = 0.005d0 ! energy level difference to consider it degenerate
418 weightxc = 1.d0 ! the amount of exchange
419 weightvh = 1.d0 ! the weight for the Hartree potential
420 if (iopt .eq. 0) then
421 randspin = 0.d0 ! do nothing in continuation
422 else
423 randspin = -1.d0 ! if>0 add random component to magnetization on the defined grid.

425 end if
426 h_field = 0.d0 ! external magnetic field
427 h_charge = 0.d0 ! external charge field
428 maxold = 3 ! maximum number of previous iterations for which density is saved

430 nys = -1
431 nxs = -1
432 jaccond = 0.d0
433 nc = -1
434 rmax = 0.d0

436 contracted_on = .false.
437 optimize_overs = .true.

441 setj_onebody = -1.d0

443 task = 0
444 emin = 0.d0
445 emax = 0.d0
446 deltae = 0.d0

479 else
480 scale_hartree = -1.d0
481 corr_hartree = .false.
482 end if
483
484 zero_jas = .false.
485 write_matrix = .false.
486 rion_from(1) = 1.d23
487 vmax0_in = -100.d0
488 nproc_fft = -1
489 !
490 if (rank .eq. 0) then
491
492 iflagerr = 1
493 read (5, nml=dft, err=200)
494 iflagerr = 0
495200 if (iflagerr .ne. 0) call error(' setup_calculation ', ' error in reading DFT card ', 1, rank)
496
497 if (compute_bands) then
498 iflagerr = 1
499 read (5, nml=band_structure, err=201)
500 iflagerr = 0
501201 if (iflagerr .ne. 0) call error(' setup_calculation ', ' error in reading band_structure card ', 1, rank)
502 end if
503 if (nx_at .gt. 0 .and. ny_at .lt. 0) then

512 write (6, *) ' Input scale for finer mesh =', scale_z

516 'around EACH atom'
517 else
518 write (6, *) ' Finer mesh on a ortho cell of side ', nx_at, ny_at, nz_at &
519 , ' unit ax,ay,az original mesh around EACH atom '
520 end if
521 else
522 if (l0_at .gt. 0 .and. nx_at .lt. 0) then
523 write (6, *) ' Finer mesh on a cube of side ', 2*l0_at&
524 &, ' referenced from the center'
525 else
526 write (6, *) ' Finer mesh on a ortho cell of side ', nx_at, ny_at, nz_at &
527 , ' unit ax,ay,az original mesh, referenced from the center '

529 end if
530 end if
531 ! The integration mesh includes the points at the boarder
532 if (nx_at .gt. 0) nx_at = nx_at + 1

534 if (nz_at .gt. 0) nz_at = nz_at + 1

539
540 end if

547#ifdef PARALLEL
548 ! DFT self-consistent calculation

555 call mpi_bcast(weightxc, 1, mpi_double_precision, 0, mpi_comm_world, ierr)
