16!> @brief Apply complex constraints for AGP and Pfaffian wave functions
17!> @details This subroutine applies complex constraints for AGP (Antisymmetrized
18!> Geminal Power) and Pfaffian wave functions. It handles both real and
19!> complex parameters with proper Hermitian relations and Cauchy relations.
20!> The subroutine manages symmetry constraints for AGP matrices and
21!> handles ghost variables for Pfaffian calculations.
22!> @param[in] iessw Number of constraints
23!> @param[in] derl Derivatives array
24!> @param[in] n Number of parameters
25!> @param[in] nozero Non-zero indices
26!> @param[in,out] psip Work array
27!> @param[in,out] econf Configuration energy array
28!> @param[in] nw Number of walkers
29!> @param[in] jbradet Branching determinant array
30subroutine constraint_complex(iessw, derl, n, nozero&
31 &, psip, econf, nw, jbradet)
32 use constants, only: ipc, ipf, deps
33 use cell, only: cellscale, phase2pi
34 use allio, only: rank, kiontot, rion, yes_crystal, ndim_detmat&
35 &, symmagp, yes_hermite, yes_correct, nelorb_at, pfaffup, nnozero_eagp&
36 &, eagp_pfaffb, ndiff
37 implicit none
38 integer ind_transp, ind, ndim, iessw, nozero(*), i, ii, j, ix, iy, n, nw, jbradet(*)
39 real*8 econf(nw, *), psip(*), derl(*)
40
41 if (iessw .eq. 0) return
42
43 do i = 1, iessw
44 econf(1, i) = 0.d0
45 end do
46 if (ipc .eq. 1) then
47 do i = 1, n
48 psip(i) = derl(nozero(i))
49 end do
50 do i = 1, nnozero_eagp
51 ind = n + i
52 iy = (nozero(ind) - 1)/ndiff + 1
53 ix = nozero(ind) - (iy - 1)*ndiff
54 psip(ind) = eagp_pfaffb(ix, iy)
55 end do
56 do j = 1, n + nnozero_eagp
57 i = jbradet(j)
58 if (i .gt. 0) then
59 econf(1, i) = econf(1, i) + psip(j)
60 elseif (i .lt. 0) then
61 econf(1, -i) = econf(1, -i) - psip(j)
62 end if
63 end do
64 else
65
66 if (symmagp .and. yes_correct) then
67 do i = 1, n
68 iy = (nozero(i) - 1)/ndim_detmat + 1
69 ix = nozero(i) - (iy - 1)*ndim_detmat
70 if (ipf .eq. 2) then
71 ndim = nelorb_at/2
72 if (ix .le. ndim .and. iy .le. ndim .and. .not. pfaffup) then
73 ind_transp = (iy + ndim - 1)*ndim_detmat + ix + ndim
74 elseif (iy .gt. ndim .and. ix .le. ndim .and. iy .le. ndim_detmat) then
75 ind_transp = (ix + ndim - 1)*ndim_detmat + iy - ndim
76 else
77 ind_transp = -nozero(i) ! this case should be treated as normal
78 end if
79 else
80 ind_transp = (ix - 1)*ndim_detmat + iy
81 end if
82 if (iy .le. ndim_detmat .and. ind_transp .ne. -nozero(i)) then
83 psip(4*i - 3) = derl(2*nozero(i) - 1)
84 psip(4*i - 2) = derl(2*nozero(i))
85 psip(4*i - 1) = derl(2*ind_transp - 1)
86 psip(4*i) = derl(2*ind_transp)
87 else
88 psip(4*i - 3) = derl(2*nozero(i) - 1)
89 psip(4*i - 2) = derl(2*nozero(i))
90 psip(4*i - 1) = 0.d0
91 psip(4*i) = 0.d0
92 end if
93 if (ipf .eq. 2) then
94 if (iy .gt. ndim .and. ix .le. ndim .and. iy - ndim .eq. ix) psip(4*i - 3:4*i) = psip(4*i - 3:4*i)/2.d0
95 else
96 if (ix .eq. iy) psip(4*i - 3:4*i) = psip(4*i - 3:4*i)/2.d0 ! to avoid double counting
97 end if
98 end do
99 do i = 1, nnozero_eagp
100 ind = n + i
101 iy = (nozero(ind) - 1)/ndiff + 1
102 ix = nozero(ind) - (iy - 1)*ndiff
103 ! The ghost are always considered normal variables with no Hermitian relation.
104 psip(4*ind - 3) = eagp_pfaffb(2*ix - 1, iy)
105 psip(4*ind - 2) = eagp_pfaffb(2*ix, iy)
106 psip(4*ind - 1) = 0.d0
107 psip(4*ind) = 0.d0
108 end do
109 else
110 do i = 1, n
111 psip(2*i - 1) = derl(2*nozero(i) - 1)
112 psip(2*i) = derl(2*nozero(i))
113 end do
114 do i = 1, nnozero_eagp
115 ind = n + i
116 iy = (nozero(ind) - 1)/ndiff + 1
117 ix = nozero(ind) - (iy - 1)*ndiff
118 psip(2*ind - 1) = eagp_pfaffb(2*ix - 1, iy)
119 psip(2*ind) = eagp_pfaffb(2*ix, iy)
120 end do
121 end if
122
123 do j = 1, n
124 i = jbradet(j)
125 iy = (nozero(j) - 1)/ndim_detmat + 1
126 ix = nozero(j) - (iy - 1)*ndim_detmat
127 if (i .gt. 0) then
128 if (symmagp .and. yes_correct) then
129 if (yes_hermite) then
130 ! old (wrong? cauchy relation is not satisfied)
131 !econf(1, 4*i - 3) = econf(1, 4*i - 3) + psip(4*j - 3) + psip(4*j - 1)
132 !econf(1, 4*i - 2) = econf(1, 4*i - 2) + psip(4*j - 2) + psip(4*j)
133 !econf(1, 4*i - 1) = econf(1, 4*i - 1) + psip(4*j - 2) - psip(4*j)
134 !econf(1, 4*i) = econf(1, 4*i) - psip(4*j - 3) + psip(4*j - 1)
135
136 ! fixed
137 econf(1, 4*i - 3) = econf(1, 4*i - 3) + psip(4*j - 3) + psip(4*j - 1)
138 econf(1, 4*i - 2) = econf(1, 4*i - 2) + psip(4*j - 2) - psip(4*j)
139 econf(1, 4*i - 1) = econf(1, 4*i - 1) + psip(4*j - 2) - psip(4*j)
140 econf(1, 4*i) = econf(1, 4*i) - psip(4*j - 3) - psip(4*j - 1)
141
142 else ! cauchy relation is ok
143 econf(1, 4*i - 3) = econf(1, 4*i - 3) + psip(4*j - 3) + psip(4*j - 1)
144 econf(1, 4*i - 2) = econf(1, 4*i - 2) + psip(4*j - 2) + psip(4*j)
145 econf(1, 4*i - 1) = econf(1, 4*i - 1) + psip(4*j - 2) + psip(4*j)
146 econf(1, 4*i) = econf(1, 4*i) - psip(4*j - 3) - psip(4*j - 1)
147 end if
148 end if
149 elseif (i .lt. 0) then
150 if (symmagp .and. yes_correct) then
151 if (yes_hermite) then
152 econf(1, -4*i - 3) = econf(1, -4*i - 3) - psip(4*j - 3) - psip(4*j - 1)
153 econf(1, -4*i - 2) = econf(1, -4*i - 2) - psip(4*j - 2) + psip(4*j)
154 econf(1, -4*i - 1) = econf(1, -4*i - 1) - psip(4*j - 2) + psip(4*j)
155 econf(1, -4*i) = econf(1, -4*i) + psip(4*j - 3) + psip(4*j - 1)
156 else
157 econf(1, -4*i - 3) = econf(1, -4*i - 3) - psip(4*j - 3) - psip(4*j - 1)
158 econf(1, -4*i - 2) = econf(1, -4*i - 2) - psip(4*j - 2) - psip(4*j)
159 econf(1, -4*i - 1) = econf(1, -4*i - 1) - psip(4*j - 2) - psip(4*j)
160 econf(1, -4*i) = econf(1, -4*i) + psip(4*j - 3) + psip(4*j - 1)
161 end if
162 else
163 econf(1, -2*i - 1) = econf(1, -2*i - 1) - psip(2*j - 1)
164 econf(1, -2*i) = econf(1, -2*i) - psip(2*j)
165 end if
166 end if
167 end do
168 do ii = 1, nnozero_eagp
169 j = n + ii
170 i = jbradet(j)
171 if (i .gt. 0) then
172 if (symmagp .and. yes_correct) then
173 econf(1, 4*i - 3) = econf(1, 4*i - 3) + psip(4*j - 3)
174 econf(1, 4*i - 2) = econf(1, 4*i - 2) + psip(4*j - 2)
175 econf(1, 4*i - 1) = 0.d0
176 econf(1, 4*i) = 0.d0
177 else
178 econf(1, 2*i - 1) = econf(1, 2*i - 1) + psip(2*j - 1)
179 econf(1, 2*i) = econf(1, 2*i) + psip(2*j)
180 end if
181 elseif (i .lt. 0) then
182 if (symmagp .and. yes_correct) then
183 econf(1, -4*i - 3) = econf(1, -4*i - 3) - psip(4*j - 3)
184 econf(1, -4*i - 2) = econf(1, -4*i - 2) - psip(4*j - 2)
185 econf(1, -4*i - 1) = 0.d0
186 econf(1, -4*i) = 0.d0
187 else
188 econf(1, -2*i - 1) = econf(1, -2*i - 1) - psip(2*j - 1)
189 econf(1, -2*i) = econf(1, -2*i) - psip(2*j)
190 end if
191 end if
192 end do
193 end if ! endif ipc
194 return
195end
