36!-------------------------------------------------------------------------------
37!> @brief Evaluate invariants between two orbitals for Jastrow or symmetry purposes.
38!>
39!> This subroutine computes a set of invariants (e.g., s-s, s-p, p-p) between two orbitals
40!> based on their types and the relative position vector. These invariants are used in the
41!> construction of Jastrow factors or symmetry-adapted wave functions.
42!>
43!> @param[in] vec_r Relative position vector between the two orbitals (length >= 3)
44!> @param[in] ix Index of the first orbital
45!> @param[in] iy Index of the second orbital
46!> @param[in] typeorb Array specifying the type of each orbital (0=s, 1-3=p_x/p_y/p_z)
47!> @param[out] jas_invariant Array of computed invariants (length 4)
48!> @param[out] orbps Logical flag, true if the order of orbitals is s-p (for symmetry)
49!>
50!> The invariants are:
51!> - jas_invariant(1): s-s overlap (always 1 if both are s)
52!> - jas_invariant(2): s-p or p-s overlap (component of vec_r)
53!> - jas_invariant(3): p-p overlap (product of components)
54!> - jas_invariant(4): 1 if both p orbitals are the same, 0 otherwise
55!>
56!> @note The routine is designed for use in QMC codes where orbital symmetry and
57!> Jastrow factor construction require such invariants.
58!>
59!> @section Examples
60!> @code{.f90}
61!> real(8) :: vec_r(3), jas_invariant(4)
62!> integer :: typeorb(2)
63!> logical :: orbps
64!> typeorb = [0, 1] ! s and p_x
65!> call evaluate_invariant(vec_r, 1, 2, typeorb, jas_invariant, orbps)
66!> @endcode
67subroutine evaluate_invariant(vec_r, ix, iy, typeorb, jas_invariant, orbps)

70 real(8) jas_invariant(4)
71 integer i, ix, iy, typeorb(*)
72 real(8) vec_r(*)
73 logical orbps

75 ! vec_phi1=0.d0
76 ! vec_phi2=0.d0

78 jas_invariant = 0.d0
79 orbps = .false.

81 !############################################################
82 ! s s
83 if (typeorb(ix) .eq. 0 .and. typeorb(iy) .eq. 0) then
84 jas_invariant(1) = 1.d0
85 end if
86
87 !#############################################################
88 ! s p case
89 if ((typeorb(ix) .gt. 0 .and. typeorb(ix) .le. 3) .and. typeorb(iy) .eq. 0) &
90 & then
91
92 ! vec_phi1(typeorb(ix))=1
93 ! phi1_dot_r=0.d0
94
95 ! do i=1,3
96 ! phi1_dot_r=phi1_dot_r+vec_phi1(i)*vec_r(i)
97 ! enddo
98
99 jas_invariant(2) = vec_r(typeorb(ix))
100
101 ! if(abs(jas_invariant(2)).gt.2)&
102 ! & write(6,*) ' ERROR in jas invariant '
103
104 ! jas_invariant(2)=jas_invariant(2)+10.d0 ! I pass in this way the information
105 ! that typeorb(ix)>typeorb(iy)
106 orbps = .true.
107
108 elseif &
109 & ((typeorb(iy) .gt. 0 .and. typeorb(iy) .le. 3) .and. typeorb(ix) .eq. 0) &
110 &then
111
112 ! vec_phi1(typeorb(iy))=1
113 ! phi1_dot_r=0.d0
114
115 ! do i=1,3
116 ! phi1_dot_r=phi1_dot_r+vec_phi1(i)*vec_r(i)
117 ! enddo
118 jas_invariant(2) = vec_r(typeorb(iy))
119
120 end if
121
122 !###################################################################
123 ! p p case
124 if ((typeorb(ix) .gt. 0 .and. typeorb(ix) .le. 3) .and. &
125 & (typeorb(iy) .gt. 0 .and. typeorb(iy) .le. 3)) then
126
127 ! vec_phi1(typeorb(ix))=1
128 ! vec_phi2(typeorb(iy))=1
129
130 ! phi1_dot_r=0.d0
131 ! phi2_dot_r=0.d0
132 !
133 ! do i=1,3
134 ! phi1_dot_r =phi1_dot_r+ vec_phi1(i)*vec_r(i)
135 ! phi2_dot_r =phi2_dot_r+ vec_phi2(i)*vec_r(i)
136 ! enddo
137
138 jas_invariant(3) = vec_r(typeorb(ix))*vec_r(typeorb(iy))
139
140 if (typeorb(ix) .eq. typeorb(iy)) jas_invariant(4) = 1.d0
141
142 end if
143
144 return
145end
