141 subroutine set_sym_bl(at)
142 !> @brief Set up Bravais lattice symmetries
143 !> @details Provides symmetry operations for all Bravais lattices. Tests first
144 !> the 24 proper rotations for the cubic lattice, then the 8 rotations
145 !> specific for the hexagonal axis, then adds inversion. All Bravais
146 !> lattices have inversion symmetry.
147 !> @param[in] at Direct lattice vectors (3, 3)
148 !> @note Based on Quantum ESPRESSO symmetry routines adapted for QMC
149 implicit none
150 ! sin3 = sin(pi/3), cos3 = cos(pi/3), msin3 = -sin(pi/3), mcos3 = -cos(pi/3)
151 !
152 real(8), dimension(3, 3), intent(in) :: at
153 real(8), parameter :: sin3 = 0.866025403784438597d0, cos3 = 0.5d0, &
154 msin3 = -0.866025403784438597d0, mcos3 = -0.5d0
155 real(8) :: s0(3, 3, 32), overlap(3, 3), rat(3), rot(3, 3), value
156 ! s0: the s matrices in cartesian axis
157 ! overlap: inverse overlap matrix between direct lattice
158 ! rat: the rotated of a direct vector ( cartesian )
159 ! rot: the rotated of a direct vector ( crystal axis )
160 ! value: component of the s matrix in axis basis
161 integer :: jpol, kpol, mpol, irot, i
162 ! counters over the polarizations and the rotations
163
164 data s0/1.d0, 0.d0, 0.d0, 0.d0, 1.d0, 0.d0, 0.d0, 0.d0, 1.d0, &
165 -1.d0, 0.d0, 0.d0, 0.d0, -1.d0, 0.d0, 0.d0, 0.d0, 1.d0, &
166 -1.d0, 0.d0, 0.d0, 0.d0, 1.d0, 0.d0, 0.d0, 0.d0, -1.d0, &
167 1.d0, 0.d0, 0.d0, 0.d0, -1.d0, 0.d0, 0.d0, 0.d0, -1.d0, &
168 0.d0, 1.d0, 0.d0, 1.d0, 0.d0, 0.d0, 0.d0, 0.d0, -1.d0, &
169 0.d0, -1.d0, 0.d0, -1.d0, 0.d0, 0.d0, 0.d0, 0.d0, -1.d0, &
170 0.d0, -1.d0, 0.d0, 1.d0, 0.d0, 0.d0, 0.d0, 0.d0, 1.d0, &
171 0.d0, 1.d0, 0.d0, -1.d0, 0.d0, 0.d0, 0.d0, 0.d0, 1.d0, &
172 0.d0, 0.d0, 1.d0, 0.d0, -1.d0, 0.d0, 1.d0, 0.d0, 0.d0, &
173 0.d0, 0.d0, -1.d0, 0.d0, -1.d0, 0.d0, -1.d0, 0.d0, 0.d0, &
174 0.d0, 0.d0, -1.d0, 0.d0, 1.d0, 0.d0, 1.d0, 0.d0, 0.d0, &
175 0.d0, 0.d0, 1.d0, 0.d0, 1.d0, 0.d0, -1.d0, 0.d0, 0.d0, &
176 -1.d0, 0.d0, 0.d0, 0.d0, 0.d0, 1.d0, 0.d0, 1.d0, 0.d0, &
177 -1.d0, 0.d0, 0.d0, 0.d0, 0.d0, -1.d0, 0.d0, -1.d0, 0.d0, &
178 1.d0, 0.d0, 0.d0, 0.d0, 0.d0, -1.d0, 0.d0, 1.d0, 0.d0, &
179 1.d0, 0.d0, 0.d0, 0.d0, 0.d0, 1.d0, 0.d0, -1.d0, 0.d0, &
180 0.d0, 0.d0, 1.d0, 1.d0, 0.d0, 0.d0, 0.d0, 1.d0, 0.d0, &
181 0.d0, 0.d0, -1.d0, -1.d0, 0.d0, 0.d0, 0.d0, 1.d0, 0.d0, &
182 0.d0, 0.d0, -1.d0, 1.d0, 0.d0, 0.d0, 0.d0, -1.d0, 0.d0, &
183 0.d0, 0.d0, 1.d0, -1.d0, 0.d0, 0.d0, 0.d0, -1.d0, 0.d0, &
184 0.d0, 1.d0, 0.d0, 0.d0, 0.d0, 1.d0, 1.d0, 0.d0, 0.d0, &
185 0.d0, -1.d0, 0.d0, 0.d0, 0.d0, -1.d0, 1.d0, 0.d0, 0.d0, &
186 0.d0, -1.d0, 0.d0, 0.d0, 0.d0, 1.d0, -1.d0, 0.d0, 0.d0, &
187 0.d0, 1.d0, 0.d0, 0.d0, 0.d0, -1.d0, -1.d0, 0.d0, 0.d0, &
188 cos3, sin3, 0.d0, msin3, cos3, 0.d0, 0.d0, 0.d0, 1.d0, &
189 cos3, msin3, 0.d0, sin3, cos3, 0.d0, 0.d0, 0.d0, 1.d0, &
190 mcos3, sin3, 0.d0, msin3, mcos3, 0.d0, 0.d0, 0.d0, 1.d0, &
191 mcos3, msin3, 0.d0, sin3, mcos3, 0.d0, 0.d0, 0.d0, 1.d0, &
192 cos3, msin3, 0.d0, msin3, mcos3, 0.d0, 0.d0, 0.d0, -1.d0, &
193 cos3, sin3, 0.d0, sin3, mcos3, 0.d0, 0.d0, 0.d0, -1.d0, &
194 mcos3, msin3, 0.d0, msin3, cos3, 0.d0, 0.d0, 0.d0, -1.d0, &
195 mcos3, sin3, 0.d0, sin3, cos3, 0.d0, 0.d0, 0.d0, -1.d0/
196
197 !
198 ! the routine computes the BL symmetries using
199 ! the lattice vectors "s2r(3,3)". Since so far the cell
200 ! is always orthorombic, the versors "at(3,3)" are always the same.
201 !
202 ! compute the overlap matrix for crystal axis
203 do jpol = 1, 3
204 do kpol = 1, 3
205 rot(kpol, jpol) = at(1, kpol)*at(1, jpol) + &
206 at(2, kpol)*at(2, jpol) + &
207 at(3, kpol)*at(3, jpol)
208 end do
209 end do
210 !
211 ! then its inverse (rot is used as work space)
212 !
213 call invmat(3, rot, overlap, value)
214 nrot = 1
215 do irot = 1, 32
216 !
217 ! for each possible symmetry
218 !
219 do jpol = 1, 3
220 do mpol = 1, 3
221 !
222 ! compute, in cartesian coordinates the rotated vector
223 !
224 rat(mpol) = s0(mpol, 1, irot)*at(1, jpol) + &
225 s0(mpol, 2, irot)*at(2, jpol) + &
226 s0(mpol, 3, irot)*at(3, jpol)
227 end do
228
229 do kpol = 1, 3
230 !
231 ! the rotated vector is projected on the direct lattice
232 !
233 rot(kpol, jpol) = at(1, kpol)*rat(1) + &
234 at(2, kpol)*rat(2) + &
235 at(3, kpol)*rat(3)
236 end do
237 end do
238 !
239 ! and the inverse of the overlap matrix is applied
240 !
241 do jpol = 1, 3
242 do kpol = 1, 3
243 value = overlap(jpol, 1)*rot(1, kpol) + &
244 overlap(jpol, 2)*rot(2, kpol) + &
245 overlap(jpol, 3)*rot(3, kpol)
246 if (abs(dble(nint(value)) - value) > 1.0d-6) then
247 !
248 ! if a noninteger is obtained, this implies that this operation
249 ! is not a symmetry operation for the given lattice
250 !
251 go to 10
252 end if
253 isymm(kpol, jpol, nrot) = nint(value)
254 sname(nrot) = s0name(irot)
255 end do
256 end do
257 nrot = nrot + 1
25810 continue
259 end do
260 nrot = nrot - 1
261 !
262 ! set the inversion symmetry ( Bravais lattices have always inversion
263 ! symmetry )
264 !
265 do irot = 1, nrot
266 do kpol = 1, 3
267 do jpol = 1, 3
268 isymm(kpol, jpol, irot + nrot) = -isymm(kpol, jpol, irot)
269 sname(irot + nrot) = s0name(irot + 32)
270 end do
271 end do
272 end do
273
274 nrot = 2*nrot
275
276 return
277 !
278 end subroutine set_sym_bl

280 ! subroutine set_sym_cell ( nion, rion, atom_number, iespbc, nx, ny, nz )
281 !
282 ! implicit none
283 ! !
284 ! ! input variables
285 ! !
286 ! integer,intent(in) :: nion
287 ! real(8),intent(in) :: rion(3,nion),atom_number(nion)
288 ! logical,intent(in) :: iespbc
289 ! integer,optional :: nx,ny,nz
290 ! !
291 ! ! local variables
292 ! !
293 ! integer :: irot,nrot_,na,nb,ind,i,j,isymm_(3,3,48),ierr
294 ! logical,dimension(:,:),allocatable :: sym_map
295 ! logical :: found_sym
296 ! real(8) :: pos(3), pos_rot(3), diff(3)
297 ! character :: sname_(64)*45
298 !
299 ! allowed_sym(1:nrot) = .false.
300 !
301 ! allocate(sym_map(nion,nrot))
302 ! sym_map(:,:) = .false.
303 !
304 ! do na = 1,nion
305 ! do irot = 1,nrot
306 ! pos_rot(:) = matmul( isymm(:,:,irot),rion(:,na) )
307 ! found_sym = .true.
308 ! nb = 1
309 ! do while( nb.le.nion .and. found_sym )
310 ! pos(:) = rion(:,nb)
311 ! diff(:) = pos_rot(:) - pos(:)
312 ! if(iespbc) call applyPBC(diff,1)
313 ! if( (sum(abs(diff(:))) .lt. 1.d-5) .and. &
314 ! (nint(atom_number(na)) .eq. nint(atom_number(nb))) ) then
315 ! sym_map(na,irot) = .true.
316 ! found_sym = .false.
317 ! endif
318 ! nb = nb + 1
319 ! enddo
320 ! enddo
321 ! enddo
322 ! ! identity
323 ! do irot = 1,nrot
324 ! sym_map(1,irot) = .true.
325 ! enddo
326 ! !
327 ! ! fill vector of symmetries allowed by the supercell,
328 ! ! in general less than the ones allowed by the Bravais lattice.
329 ! !
330 ! do irot = 1,nrot
331 ! if( all(sym_map(:,irot)) ) allowed_sym(irot) = .true.
332 ! enddo
333 ! !
334 ! ! check that the grid is compatible with the S rotation
335 ! !
336 ! if( present(nx) .and. present(ny) .and. present(nz) ) then
337 ! do irot = 1, nrot
338 ! if ( mod (isymm(2, 1, irot) * nx, ny) /= 0 .or. &
339 ! mod (isymm(3, 1, irot) * nx, nz) /= 0 .or. &
340 ! mod (isymm(1, 2, irot) * ny, nx) /= 0 .or. &
341 ! mod (isymm(3, 2, irot) * ny, nz) /= 0 .or. &
342 ! mod (isymm(1, 3, irot) * nz, nx) /= 0 .or. &
343 ! mod (isymm(2, 3, irot) * nz, ny) /= 0 ) then
344 ! allowed_sym(irot) = .false.
345 ! endif
346 ! enddo
347 ! endif
348 ! !
349 ! ! fill symmetry vector isymm with new allowed symmetries
350 ! ! update the number of allowed symmetries
351 ! !
352 ! nrot_ = nrot
353 ! nrot = 0
354 ! do irot = 1,nrot_
355 ! if(allowed_sym(irot)) then
356 ! nrot = nrot + 1
357 ! isymm_(:,:,nrot) = isymm(:,:,irot)
358 ! sname_(nrot) = sname(irot)
359 ! endif
360 ! enddo
361 ! isymm(:,:,:) = 0
362 ! isymm = isymm_
363 ! sname(:) = " "
364 ! sname(:) = sname_(:)
365 !
366 ! deallocate(sym_map)
367 !
368 ! return
369 !
370 ! end subroutine set_sym_cell

372 subroutine purge_isymm(iespbc, nsym, isymm, sname, t_rev, rion, nion, zeta, cellscale, nx, ny, nz)
373 !> @brief Purge symmetry operations based on atomic positions and cell geometry
374 !> @details Removes symmetry operations that are not compatible with the actual
375 !> atomic positions, cell geometry, and periodic boundary conditions.
376 !> Checks that each symmetry operation maps atoms to equivalent positions.
377 !> @param[in] iespbc Logical: periodic boundary conditions
378 !> @param[in] nsym Number of input symmetries
379 !> @param[inout] isymm Symmetry matrices (3, 3, nsym)
380 !> @param[inout] sname Symmetry names
381 !> @param[inout] t_rev Time reversal flags
382 !> @param[in] rion Atomic positions (3, nion)
383 !> @param[in] nion Number of ions
384 !> @param[in] zeta Atomic numbers/charges
385 !> @param[in] cellscale Cell dimensions
386 !> @param[in] nx, ny, nz Grid dimensions (optional)
387 !> @note Updates nrot to the number of allowed symmetries
388 use constants, only: deps
389 implicit none
390
391 integer :: j, nion, nsym, nsym_acc, is, i2, npip(3), isymm(3, 3, *), t_rev(*)
392 integer, optional :: nx, ny, nz
393 logical :: found, accept, iespbc
394 real(8) :: rionmap(3), rion(3, nion), dist(3), zeta(*), cellscale(3)
395 character :: sname(64)*45
396 logical, dimension(:), allocatable :: yessite
397
398 npip = 0
399 allocate (yessite(nion))
400 nsym_acc = 0
401
402 nrot_full = nrot
403
404 do is = 1, nrot_full
405 accept = .true.
406 yessite = .false.
407 do j = 1, nion
408 rionmap(1:3) = matmul(isymm(:, :, is), rion(:, j))
409 i2 = 1
410 found = .true.
411 do while (i2 .le. nion .and. found)
412 dist(1:3) = rionmap(1:3) - rion(1:3, i2)
413 if (iespbc) then
414 call makeimage(dist, cellscale, deps)
415 end if
416 if (sum(abs(dist(:))) .lt. deps .and. nint(zeta(i2)) .eq. nint(zeta(j))) then
417 found = .false.
418 yessite(i2) = .true.
419 end if
420 i2 = i2 + 1
421 end do
422 end do
423 if (found) then
424 accept = .false.
425 else
426 accept = .true.
427 do j = 1, nion
428 if (.not. yessite(j)) accept = .false.
429 end do
430 end if
431 !
432 ! check that the grid is compatible with the S rotation
433 !
434 if (present(nx) .and. present(ny) .and. present(nz)) then
435 if (mod(isymm(2, 1, is)*nx, ny) /= 0 .or. &
436 mod(isymm(3, 1, is)*nx, nz) /= 0 .or. &
437 mod(isymm(1, 2, is)*ny, nx) /= 0 .or. &
438 mod(isymm(3, 2, is)*ny, nz) /= 0 .or. &
439 mod(isymm(1, 3, is)*nz, nx) /= 0 .or. &
440 mod(isymm(2, 3, is)*nz, ny) /= 0) then
441 accept = .false.
442 end if
443 end if
444
445 if (accept) then
446 nsym_acc = nsym_acc + 1
447 isymm(:, :, nsym_acc) = isymm(:, :, is)
448 sname(nsym_acc) = sname(is)
449 t_rev(nsym_acc) = t_rev(is)
450 end if
451 end do
452
453 nrot = nsym_acc
454 deallocate (yessite)
455
456 return
457
458 end subroutine purge_isymm

460 subroutine transform_point(s, i, j, k, nx, ny, nz, ri, rj, rk)
461 !> @brief Transform a point through a symmetry operation
462 !> @details Computes the rotated point (ri, rj, rk) from the input point (i, j, k)
463 !> through the symmetry operation s. Then computes the equivalent point
464 !> on the original real space mesh. Used for charge density symmetrization.
465 !> @param[in] s Symmetry matrix (3, 3)
466 !> @param[in] i, j, k Input point coordinates
467 !> @param[in] nx, ny, nz Grid dimensions
468 !> @param[out] ri, rj, rk Transformed point coordinates
469 !> @note Used in DFT code for charge density symmetrization with k-point sampling
470 implicit none
471 !
472 ! first the dummy variables
473 !
474 integer, intent(in) :: s(3, 3), i, j, k, nx, ny, nz
475 integer, intent(inout) :: ri, rj, rk
476 ! input: the rotation matrix
477 ! input: the fractionary translation
478 ! \
479 ! input: the point to rotate
480 ! /
481 ! \
482 ! input: the dimension of the mesh
483 ! /
484 !\
485 ! output: the rotated point
486 !/
487 !
488 ! local variable
489 !
490 ! the rotation matrix in scaled crystallographic coordinates
491
492 ri = s(1, 1)*(i - 1) + s(2, 1)*(j - 1) + s(3, 1)*(k - 1)
493 ri = mod(ri, nx) + 1
494 if (ri .lt. 1) ri = ri + nx
495 rj = s(1, 2)*(i - 1) + s(2, 2)*(j - 1) + s(3, 2)*(k - 1)
496 rj = mod(rj, ny) + 1
497 if (rj .lt. 1) rj = rj + ny
498 rk = s(1, 3)*(i - 1) + s(2, 3)*(j - 1) + s(3, 3)*(k - 1)
499 rk = mod(rk, nz) + 1
500 if (rk .lt. 1) rk = rk + nz
501
502 return
503
504 end subroutine transform_point

507 subroutine invmat(n, a, a_inv, da)
508 !> @brief Compute the inverse of a matrix using LAPACK
509 !> @details Computes the inverse "a_inv" of matrix "a", both dimensioned (n,n).
510 !> Matrix "a" is unchanged on output. Uses LAPACK DGETRF and DGETRI.
511 !> @param[in] n Matrix dimension
512 !> @param[in] a Input matrix (n, n)
513 !> @param[out] a_inv Inverse matrix (n, n)
514 !> @param[out] da Determinant of matrix a
515 !> @note Adapted from Quantum ESPRESSO matrix inversion routines
516 !
517 ! computes the inverse "a_inv" of matrix "a", both dimensioned (n,n)
518 ! matrix "a" is unchanged on output - LAPACK
519 !
520 implicit none
521 integer :: n
522 real(8), dimension(n, n) :: a, a_inv
523 real(8) :: da
524 integer :: info, lda, lwork, ipiv(n)
525 ! info=0: inversion was successful
526 ! lda : leading dimension (the same as n)
527 ! ipiv : work space for pivoting (assumed of length lwork=n)
528 real(8) :: work(n)
529 ! more work space
530 lda = n
531 lwork = n
532 a_inv(:, :) = a(:, :)
533 !
534 call dgetrf(n, n, a_inv, lda, ipiv, info)
535 if (info .ne. 0) call errore('invmat', 'error in DGETRF', abs(info))
536 call dgetri(n, a_inv, lda, ipiv, work, lwork, info)
537 if (info .ne. 0) call errore('invmat', 'error in DGETRI', abs(info))
538 !
539 if (n == 3) then
540 da = a(1, 1)*(a(2, 2)*a(3, 3) - a(2, 3)*a(3, 2)) + &
541 a(1, 2)*(a(2, 3)*a(3, 1) - a(2, 1)*a(3, 3)) + &
542 a(1, 3)*(a(2, 1)*a(3, 2) - a(3, 1)*a(2, 2))
543 if (abs(da) < 1.d-10) call errore(' invmat ', ' singular matrix ', 1)
544 else
545 da = 0.d0
546 end if
547 return
548 end subroutine invmat
