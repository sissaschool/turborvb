16!> @brief Assign orbital types for Jastrow orbitals based on shell structure
17!>
18!> This subroutine assigns orbital type indices to Jastrow orbitals based on
19!> their shell multiplicity and occupation. The orbital types follow a specific
20!> numbering scheme for different angular momentum shells (s, p, d, f, g).
21!>
22!> Parameters
23!> ----------
24!> NSHELLJ : integer, in
25!> Number of shells in the Jastrow basis.
26!> MULTIJ : integer array, in
27!> Multiplicity of each shell (1=s, 3=p, 5=d, 7=f, 9=g).
28!> IOCCJ : integer array, in
29!> Occupation numbers for each orbital (0=unoccupied, 1=occupied).
30!> TYPEORB : integer array, out
31!> Assigned orbital type indices for occupied orbitals.
32!>
33!> Notes
34!> -----
35!> - Only occupied orbitals (IOCCJ ≠ 0) are assigned type indices.
36!> - The orbital type numbering follows the scheme:
37!> * s orbitals (multiplicity 1): type = 0
38!> * p orbitals (multiplicity 3): type = 1, 2, 3
39!> * d orbitals (multiplicity 5): type = 4, 5, 6, 7, 8
40!> * f orbitals (multiplicity 7): type = 9, 10, 11, 12, 13, 14, 15
41!> * g orbitals (multiplicity 9): type = 16, 17, 18, 19, 20, 21, 22, 23, 24
42!> - The subroutine processes shells sequentially and assigns types incrementally.
43!> - Used in Jastrow factor construction for quantum Monte Carlo.
44!>
45!> Algorithm
46!> ---------
47!> For each shell i with multiplicity multij(i):
48!> For each orbital j in the shell:
49!> If ioccj(ii) ≠ 0 (occupied):
50!> Assign typeorb based on multiplicity and orbital index
51!>
52!> Example
53!> -------
54!> Used in Jastrow factor parameterization and orbital basis construction.
55subroutine write_type_orb(nshellj, multij, ioccj, typeorb)
56 implicit none
57
58 ! argument parameters
59 integer, intent(in) :: nshellj, multij(*), ioccj(*)
60 integer, intent(out) :: typeorb(*)
61
62 ! local variables
63 integer i, ii, j, ind_type
64
65!> @brief Initialize counters
66 ii = 0
67 ind_type = 1
68
69!> @brief Process each shell and assign orbital types
70 do i = 1, nshellj
71 ! write(*,*)'xxx',i,multij(i)
72 do j = 1, multij(i)
73 ii = ii + 1
74 if (ioccj(ii) .ne. 0) then
75!> @brief Assign orbital type based on shell multiplicity
76 select case (multij(i))
77 case (1)
78!> @brief s orbital: type = 0
79 typeorb(ind_type) = 0
80 ind_type = ind_type + 1
81 case (3)
82!> @brief p orbitals: type = 1, 2, 3
83 typeorb(ind_type) = j
84 ind_type = ind_type + 1
85 case (5)
86!> @brief d orbitals: type = 4, 5, 6, 7, 8
87 typeorb(ind_type) = 3 + j
88 ind_type = ind_type + 1
89 case (7)
90!> @brief f orbitals: type = 9, 10, 11, 12, 13, 14, 15
91 typeorb(ind_type) = 8 + j
92 ind_type = ind_type + 1
93 case (9)
94!> @brief g orbitals: type = 16, 17, 18, 19, 20, 21, 22, 23, 24
95 typeorb(ind_type) = 15 + j
96 ind_type = ind_type + 1
97 case default
98!> @brief Error for unsupported orbital multiplicity
99 write (6, *) 'ERROR non existing orbital in Jastrow '
100 end select
101 end if
102 end do
103 end do
104 ! write(*,*)'XXXX',ind_type-1
105 ! do i=1,ind_type-1
106 ! write(*,*)i,typeorb(i)
107 ! enddo
108 return
109end
