63 !> @brief Calculates quadrupole moment tensor of the system
64 !> @details This subroutine calculates the quadrupole moment tensor
65 !> using the formula Q_ij = sum_n 0.5*(q_k*3*x_in*x_jn - r_n^2*delta_ij).
66 !> It computes both nuclear and electronic contributions relative
67 !> to the center of charge.
68 !> @param[in] r_el Electron positions (3, nel)
69 !> @param[in] vdim Vector dimensions
70 !> @param[in] ell System dimensions
71 subroutine calc_quadrupole(r_el, vdim, ell)
72
73 ! to be modified!!!
74
75 use allio, only: rion, zetar, nion, nel
76 implicit none
77
78 real(8), intent(in) :: r_el(3, *)
79 integer, intent(in) :: vdim(*)
80 real(8), intent(in) :: ell(*)
81 integer :: i, j, n
82 real(8) :: qtot, dist, cion(3, nion), cel(3, nel), sumw, nuc_charge
83
84 !!Calculation of the center of the nuclear charge
85 !!(use this until the problem of the asymmetry of the electronic charge is solved)
86 !!Warning: this way to calculate the quadrupole is correct only if the dipole of the system is zero
87 cc_chg = 0.d0
88 do i = 1, nion
89 cc_chg(:) = cc_chg(:) + zetar(i)*rion(:, i)
90 end do
91 cc_chg(:) = cc_chg(:)/sum(zetar(:))
92
93 ! Reference frame of center of charge
94 do i = 1, nion
95 cion(:, i) = rion(:, i) - cc_chg(:)
96 end do
97 do i = 1, nel
98 cel(:, i) = r_el(:, i) - cc_chg(:)
99 end do
100
101 !Formula of quadrupole moment is as following by E.Coccia
102 !Q_ij = \sum_n^nel 0.5*(q_k*3*x_in*x_jn - r_n^2d_ij)
103 !Nuclear contribution to the quadrupole moment
104 quad = 0.d0
105 do n = 1, nion
106 dist = cion(1, n)**2 + cion(2, n)**2 + cion(3, n)**2
107 do i = 1, 3
108 do j = i, 3
109 if (i .eq. j) then
110 quad(i, j) = quad(i, j) + zetar(n)*(3.d0*cion(i, n)*cion(j, n) - dist)
111 else
112 quad(i, j) = quad(i, j) + zetar(n)*3.d0*cion(i, n)*cion(j, n)
113 quad(j, i) = quad(i, j)
114 end if
115 end do
116 end do
117 end do
118
119 !Electronic contribution to the quadrupole moment
120 do n = 1, nel
121 dist = cel(1, n)**2 + cel(2, n)**2 + cel(3, n)**2
122 do i = 1, 3
123 do j = i, 3
124 if (i .eq. j) then
125 quad(i, j) = quad(i, j) - (3.d0*cel(i, n)*cel(j, n) - dist)
126 else
127 quad(i, j) = quad(i, j) - 3.d0*cel(i, n)*cel(j, n)
128 quad(j, i) = quad(i, j)
129 end if
130 end do
131 end do
132 end do
133
134 quad = 0.5d0*quad
135 return
136 end subroutine

139 !> @details This subroutine diagonalizes the quadrupole moment tensor
140 !> using LAPACK routines DSYTRD and DSTERF. The eigenvalues
141 !> are stored in quad_diag array.
142 subroutine diag_quad
143
144 integer :: i, j, n, INFO
145 real*8 :: e(2), tau(2), work(10)
146
147 call dsytrd('U', nquad, quad, nquad, quad_diag, e, tau, work, 10, info)
148 call dsterf(nquad, quad_diag, e, info)
149
150 end subroutine diag_quad
