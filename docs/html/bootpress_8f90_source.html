16!> @brief Bootstrap method to evaluate pressure and its error bars
17!> @details This subroutine implements the bootstrap method to evaluate pressure
18!> and its error bars for each bin. It calculates pressure contributions
19!> including Pulay corrections and Hartree-Fock contributions.
20!> The method uses jackknife resampling to estimate statistical errors
21!> for pressure calculations in VMC/DMC simulations.
22!> @param[in] nbin Number of bins
23!> @param[in] efenergy Energy and weight arrays
24!> @param[in] e Energy arrays for pressure calculation
25!> @param[out] eta Average pressure
26!> @param[out] derr Pressure error
27!> @param[out] etanp Average pressure without Pulay contributions
28!> @param[out] derrnp Pressure error without Pulay contributions
29!> @param[in] LBox Box length
30!> @param[in] omega Frequency parameter
31!> @param[in] ris Scaling factors
32!> @param[in] rank Process rank
33!> @param[in] nproc Number of processes
34subroutine bootpress(nbin, efenergy, e, eta, derr &
35 &, etanp, derrnp, LBox, omega, ris, rank, nproc)
36 use constants, only: ipc
37 implicit none
38
39 ! *** Does BootStrap to evalute force and its error for each bin
40 ! energy ! derrivative of the ener! log derivative wf! energy x log de
41
42 !******* etanp= Pressure without pulay contributes
43 !
44 integer k, j, jj, kk, ieskin, nbin, kmain, rank, nproc, nbins, ierr
45 real*8 e(3, *), et(4), eall(4), eta, givenmeas, derr, omega, ris(4) &
46 &, efenergy(1 + ipc, *), weight, weightall, lbox, etanp, derrnp &
47 &, summpi(5), outmpi(5)
48
49#ifdef PARALLEL
50 include 'mpif.h'
51#endif
52
53 ! initialization opara e opars
54
55 eta = 0.d0
56 derr = 0.d0
57 etanp = 0.d0
58 derrnp = 0.d0
59 call dscalzero(4, 0.d0, eall, 1)
60
61 weightall = 0.d0
62
63 do k = 1, nbin
64 do j = 2, 4
65 eall(j) = eall(j) + e(j - 1, k)
66 end do
67 ! BUG in scaling pressure, the Pulay was transfomed in Hartree while the HF contr. was in
68 ! Ry. Then both contributions were divided by the same factor. Thus the
69 ! Pulay in pressure before 3 Sept. 2014 was underestimated by a factor 2
70 ! always. That's why there were always almost the same.
71 ! The calculation was instead OK when computing pressure at fixed conf.
72 ! with forcevmc.sh or forcefn.sh
73 ! eall(1)=eall(1)+efenergy(1,k)*ris(2)
74 eall(1) = eall(1) + efenergy(1, k)
75 weightall = weightall + efenergy(1 + ipc, k)
76 end do
77
78#ifdef PARALLEL
79 summpi(1:4) = eall(1:4)
80 summpi(5) = weightall
81 call mpi_allreduce(summpi, outmpi, 5, mpi_double_precision &
82& , mpi_sum, mpi_comm_world, ierr)
83 eall(1:4) = outmpi(1:4)
84 weightall = outmpi(5)
85#endif
86
87 do j = 1, 4
88 eall(j) = eall(j)/weightall
89 end do
90
91 ! write(*,*) 'eall = ',eall(1),eall(2),eall(3),eall(4)
92
93 ! if(LBox.ne.1.d0) write(6,*) ' eall 2 scaled =',eall(2)*LBox
94
95 ! Jet Lag evaluation of error bar with bin length lbin=100
96
97 do kmain = 1, nbin
98
99 do jj = 2, 4
100 !w(jj,j)*e(jj,j)
101 et(jj) = e(jj - 1, kmain)
102 end do
103 ! BUG ??? Why the following scaling from Ry-H while everything
104 ! is done in Ry?
105 et(1) = efenergy(1, kmain)
106 weight = efenergy(1 + ipc, kmain)
107
108 ! write(6,*) ' et read ',et(1)/weight,et(2)/weight,et(3)/weight
109 ! 1,et(4)/weight
110
111 do jj = 1, 4
112 !wt(jj)
113 et(jj) = (eall(jj)*weightall - et(jj))/(weightall - weight)
114 end do
115
116 ! now average correlation function on the given bootstrap
117
118 givenmeas = -(et(2) + 2.d0*(et(4) - et(1)*et(3)))
119
120 ! write(6,*) ' givenmeas =',givenmeas,-et(2)
121
122 eta = eta + givenmeas
123 derr = derr + givenmeas**2
124
125 etanp = etanp - et(2)
126 derrnp = derrnp + et(2)**2
127
128 end do
129
130#ifdef PARALLEL
131 summpi(1) = eta
132 summpi(2) = derr
133 summpi(3) = etanp
134 summpi(4) = derrnp
135 call mpi_allreduce(summpi, outmpi, 4, mpi_double_precision &
136& , mpi_sum, mpi_comm_world, ierr)
137 eta = outmpi(1)
138 derr = outmpi(2)
139 etanp = outmpi(3)
140 derrnp = outmpi(4)
141#endif
142 nbins = nbin*nproc
143
144 eta = eta/nbins
145 derr = derr/nbins
146
147 etanp = etanp/nbins
148 derrnp = derrnp/nbins
149
150 ! write(6,*) 'etanp derrnp =',eta,derr,etanp,derrnp
151
152 derrnp = dsqrt(dble(nbins - 1)**2/dble(nbins) &
153 & *(derrnp - etanp**2))
154
155 derr = dsqrt(dble(nbins - 1)**2/dble(nbins)*(derr - eta**2))
156 eta = -(eall(2) + 2.d0*(eall(4) - eall(1)*eall(3)))
157 etanp = -eall(2)
158
159 ! scaling for unit Hartree at the end for all (after 3 Sept. 2014).
160 eta = eta*ris(2)
161 etanp = etanp*ris(2)
162 derr = derr*ris(2)
163 derrnp = derrnp*ris(2)
164
165 return
166end
