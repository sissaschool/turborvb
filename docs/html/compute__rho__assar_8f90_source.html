35 !> @brief Initializes Assaraf density calculation parameters
36 !> @details This subroutine initializes the parameters needed for
37 !> Assaraf density calculations. It computes distances
38 !> between grid points and ions, sets up switching functions,

108 !> @brief Computes electron density using the Assaraf method
109 !> @details This subroutine computes the electron density using
110 !> the Assaraf method, which provides a way to extract
111 !> density from quantum Monte Carlo wave functions.
112 !> It handles both up and down electrons separately and
113 !> supports both external and regular grid calculations.
114 !> @param[in] dist Electron-ion distances (nelup+neldo, nion)
115 !> @param[in] rion Ion positions (3, nion)
116 !> @param[in] nion Number of ions
117 !> @param[in] kel Electron positions (3, *)
118 !> @param[in] nelup Number of up-spin electrons
119 !> @param[in] neldo Number of down-spin electrons
120 !> @param[in] indt Index for derivatives
121 !> @param[in] tabpip Tabulated pip values
122 !> @param[in] winvup Up-spin inverse matrix
123 !> @param[in] winvdo Down-spin inverse matrix
124 !> @param[out] density Electron density values
125 !> @param[in] zetaq Atomic numbers
126 subroutine compute_rho_assar(dist, rion, nion, kel, nelup, neldo, indt, tabpip, winvup, winvdo, density, zetaq)
127 use grid_module
128 use constants, only: pi
129 implicit none
130 integer i, j, k, jp, indt, nelup, neldo, nion !nelup, neldo - number spinup spindown, nion - # of atoms
131 !Coordinates and distances------------------------------------------------------------------------------------
132 real*8 rion(3, nion) !coord'tes of ions, x,y,z and nion - # of atoms
133 real*8 kel(3, *) !all the electron cooord computed in VMC run
134 real*8 dist(nelup + neldo, nion) !distance between an ion and an electron (VMC)
135 real*8 :: comp_grid_ele(3), dist_grid_ele !coordinates and distance between a grid point and an electron (VMC)
136 !Functions-----------------------------------------------------------------------------------------------------
137 real*8 density(*) !vector of density
138 real*8 :: fx
139 real*8 :: gradf(3), nablaf !Operators on F grad laplas
140 real*8 :: wave_laplacian, wave_gradient(3)

142 !Parameters----------------------------------------------------------------------------------------------------
143 real*8 :: zetaq(nion) !, assar_cut !Zq-the atomic number,assar_cut - cutoff
144 real*8 :: exp_grid !exp(-lambda|ri-r|), distance |ri-r| cut off
145 integer ix, iy, iz, ax, ay, az, bx, by, bz, jx, jy, jz
146 !--------------------------------------------------------------------------------------------------------------
147 do i = 1, grid_points
148 density(i) = 0.d0
149 end do
150
151 if (ext_grid) then
152 !Per gli elettroni UP
153 do j = 1, nelup
154 do i = 1, grid_points
155 comp_grid_ele(:) = -out_grid(:, i) + kel(:, j)
156 dist_grid_ele = sqrt(sum(comp_grid_ele(:)**2))
157 exp_grid = exp(-assar_parr*dist_grid_ele)
158
159 !Calculating f, grad(f) and nabla(f)------------------------------------------
160 fx = 1
161 gradf(:) = 0.d0
162 nablaf = 0.d0
163 do k = 1, nion
164 fx = fx + am(i, k)*2*zetaq(k)*(dist(j, k) - dist_grid_ion(i, k))
165 gradf(:) = gradf(:) + am(i, k)*2*zetaq(k)*(kel(:, j) - rion(:, k))/dist(j, k)
166 nablaf = nablaf + am(i, k)*4*zetaq(k)/dist(j, k)
167 end do
168 fx = fx - bm(i) + bm(i)*(1 + assar_parr*dist_grid_ele)*exp_grid
169 gradf(:) = gradf(:) - bm(i)*assar_parr2*comp_grid_ele(:)*exp_grid
170 nablaf = nablaf + bm(i)*(assar_parr3*dist_grid_ele - 3*assar_parr2)*exp_grid
171 !------------------------------------------------------------------------------
172 wave_gradient(:) = (winvup(j, indt + 1:indt + 3) + tabpip(j, indt + 1:indt + 3))
173 wave_laplacian = winvup(j, indt + 4) + tabpip(j, indt + 4) &
174 + 2*sum(winvup(j, indt + 1:indt + 3)*tabpip(j, indt + 1:indt + 3)) &
175 + sum(tabpip(j, indt + 1:indt + 3)*tabpip(j, indt + 1:indt + 3))
176 density(i) = density(i) &
177 - 1/(4*pi)*(1/dist_grid_ele - bm(i)*gx(i)) &
178 &*(nablaf + 2*fx*(wave_laplacian + sum(wave_gradient(:)**2)) &
179 &+ 4*sum(gradf(:)*wave_gradient(:)))
180 end do
181 end do
182 !Per gli elettroni DOWN
183 do j = 1, neldo
184 jp = j + nelup
185 do i = 1, grid_points
186 comp_grid_ele(:) = -out_grid(:, i) + kel(:, jp)
187 dist_grid_ele = sqrt(sum(comp_grid_ele(:)**2))
188 exp_grid = exp(-assar_parr*dist_grid_ele)
189 !Calculating f, grad(f) and nabla(f)------------------------------------------

192 nablaf = 0.d0
193 do k = 1, nion
194 fx = fx + am(i, k)*2*zetaq(k)*(dist(jp, k) - dist_grid_ion(i, k))
195 gradf(:) = gradf(:) + am(i, k)*2*zetaq(k)*(kel(:, jp) - rion(:, k))/dist(jp, k)
196 nablaf = nablaf + am(i, k)*4*zetaq(k)/dist(jp, k)
197 end do
198 fx = fx - bm(i) + bm(i)*(1 + assar_parr*dist_grid_ele)*exp_grid
199 gradf(:) = gradf(:) - bm(i)*assar_parr2*comp_grid_ele(:)*exp_grid
200 nablaf = nablaf + bm(i)*(assar_parr3*dist_grid_ele - 3*assar_parr2)*exp_grid
201 !------------------------------------------------------------------------------
202 wave_gradient(:) = winvdo(j, indt + 1:indt + 3) + tabpip(jp, indt + 1:indt + 3)
203 wave_laplacian = winvdo(j, indt + 4) + tabpip(jp, indt + 4) &
204 & + 2*sum(winvdo(j, indt + 1:indt + 3)*tabpip(jp, indt + 1:indt + 3))&
205 & + sum(tabpip(jp, indt + 1:indt + 3)*tabpip(jp, indt + 1:indt + 3))
206 density(i) = density(i) &
207 - 1/(4*pi)*(1/dist_grid_ele - bm(i)*gx(i)) &
208 *(nablaf + 2*fx*(wave_laplacian + sum(wave_gradient(:)**2)) &
209 + 4*sum(gradf(:)*wave_gradient(:)))
210 end do
211 end do
212 else
213 do j = 1, nelup
214 !------------------------------------------------------
215 ix = int(kel(1, j)/dxas) + coefx
216 ax = max(1, ix - bins_cut(1))
217 bx = min(nx, ix + bins_cut(1))
218 !-----------------------------------------------------
219 iy = int(kel(2, j)/dyas) + coefy
220 ay = max(1, iy - bins_cut(2))
221 by = min(ny, iy + bins_cut(2))
222 !-----------------------------------------------------
223 iz = int(kel(3, j)/dzas) + coefz
224 az = max(1, iz - bins_cut(3))
225 bz = min(nz, iz + bins_cut(3))

228 do jy = ay, by
229 do jx = ax, bx

242 end do

244 gradf(:) = gradf(:) - bm(i)*assar_parr2*comp_grid_ele(:)*exp_grid
245 nablaf = nablaf + bm(i)*(assar_parr3*dist_grid_ele - 3*assar_parr2)*exp_grid
246 !------------------------------------------------------------------------------

248 wave_laplacian = winvup(j, indt + 4) + tabpip(j, indt + 4) &
249 + 2*sum(winvup(j, indt + 1:indt + 3)*tabpip(j, indt + 1:indt + 3)) &
250 + sum(tabpip(j, indt + 1:indt + 3)*tabpip(j, indt + 1:indt + 3))
251 density(i) = density(i) &
252 - 1/(4*pi)*(1/dist_grid_ele - bm(i)*gx(i)) &
253 *(nablaf + 2*fx*(wave_laplacian + &
254 sum(wave_gradient(:)**2)) + 4*sum(gradf(:)*wave_gradient(:)))
255 end do
256 end do
257 end do
258 end do
259 do j = 1, neldo
260 jp = j + nelup
261 !------------------------------------------------------
262 ix = int(kel(1, jp)/dxas) + coefx
263 ax = max(1, ix - bins_cut(1))
264 bx = min(nx, ix + bins_cut(1))
265 !-----------------------------------------------------
266 iy = int(kel(2, jp)/dyas) + coefy
267 ay = max(1, iy - bins_cut(2))
268 by = min(ny, iy + bins_cut(2))
269 !------------------------------------------------------

271 az = max(1, iz - bins_cut(3))
272 bz = min(nz, iz + bins_cut(3))
273 !-----------------------------------------------------
274 do jz = az, bz
275 do jy = ay, by
276 do jx = ax, bx
277 i = jx + (jy - 1)*nx + (jz - 1)*nx*ny
278
279 comp_grid_ele(:) = -out_grid(:, i) + kel(:, jp)
280 dist_grid_ele = sqrt(sum(comp_grid_ele(:)**2))
281 exp_grid = exp(-assar_parr*dist_grid_ele)
282 !Calculating f, grad(f) and nabla(f)------------------------------------------
283 fx = 1
284 gradf(:) = 0.d0
285 nablaf = 0.d0
286 do k = 1, nion
287 fx = fx + am(i, k)*2*zetaq(k)*(dist(jp, k) - dist_grid_ion(i, k))
288 gradf(:) = gradf(:) + am(i, k)*2*zetaq(k)*(kel(:, jp) - rion(:, k))/dist(jp, k)
289 nablaf = nablaf + am(i, k)*4*zetaq(k)/dist(jp, k)
290 end do

292 gradf(:) = gradf(:) - bm(i)*assar_parr2*comp_grid_ele(:)*exp_grid
