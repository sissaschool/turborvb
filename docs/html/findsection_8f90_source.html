16!> @brief Find a specific section in a file
17!> @details This subroutine searches for a specific section name in a file
18!> by reading through the file line by line. It rewinds the file
19!> to the beginning and searches for an exact match with the
20!> provided section name. If the section is found, the file pointer
21!> is positioned at the beginning of that section. If not found,
22!> an error is raised.
23!> @param[in] funit File unit number to search in
24!> @param[in] section_name Name of the section to find
25subroutine findsection(funit, section_name)
26 implicit none
27 logical done
28 integer, intent(in) :: funit
29 character(len=*), intent(in) :: section_name
30 character(20) :: line
31 character(50) :: empty_string
32
33 done = .false.
34
35 rewind(funit)
36 do while (.not. done)
37 read (funit, '(a)', end=100) line
38 !
39 if (trim(line) == trim(section_name)) done = .true.
40 !
41 end do
42
43 return
44
45100 write (empty_string, *) " Section ", section_name, " not found "
46 call errore("findsection", empty_string, 1)
47end subroutine findsection
