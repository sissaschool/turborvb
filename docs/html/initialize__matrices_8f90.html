subroutine initialize_mats_new

Initializes main matrices for self-consistent DFT calculations.

This subroutine initializes the overlap and Hamiltonian matrices used in self-consistent field calculations. It handles both real and complex wavefunctions, supports parallel execution with SCALAPACK, and includes double mesh capabilities for improved accuracy around atomic positions. The matrices are computed by integrating over the real-space grid.

969 !> @brief Computes wavefunction values and matrix elements for a single grid point

981 !> @param volmesh Volume element for this grid point

983 , memlarge, volmesh)
984 ! This routine computes the WF on a single grid point.
985 ! it is thread-safe and intended to be called from threaded region.
986 use allio, only: kappa, iespbc, nion, rion, zetar, neigh, x_neigh, iflagerr, metric, applypbc
987 use dielectric
988 implicit none
989 ! arguments
990 ! single electron positions
991 real(8), intent(inout) :: x(3, 0:indt)
992 ! index to the scratch space
993 integer, intent(in) :: tid
994 ! index in the grid buffer
995 integer, intent(in) :: ind
996 ! global index of the current grid point
997 integer, intent(in) :: indmesh

1002 ! local variables
1003 real(8) :: pseudolocalnew

1018 dist_kel(3) = x(3, 0) - rion(3, kk)
1019 call applypbc(dist_kel, 1)
1020 dist_scratch(kk, tid) = max(dsqrt(dist_kel(1)**2 + dist_kel(2)**2 + dist_kel(3)**2), mindist)
1021 dist_shift_scratch(1, tid) = -2.d0*zetar(kk)*rep_erfc(dist_scratch(kk, tid), kappa)
1022#ifdef _SIMD
1023!$omp simd
1024#endif
1025 do ii = 2, neigh
1026! dist_kel(1) = dist_kel(1)+x_neigh(ii,1)
1027! dist_kel(2) = dist_kel(2)+x_neigh(ii,2)
1028! dist_kel(3) = dist_kel(3)+x_neigh(ii,3)
1029
1030 dist_shift_scratch(ii, tid) = max(dsqrt((dist_kel(1) + x_neigh(ii, 1))**2 &
1031 + (dist_kel(2) + x_neigh(ii, 2))**2 &
1032 + (dist_kel(3) + x_neigh(ii, 3))**2), mindist)
1033
1034! if(dist_scratch(kk, tid).lt.mindist) dist_scratch(kk, tid) = mindist
1035! const = -2.d0 * zetar(kk) * derfc(kappa * dist_scratch(kk, tid)) / dist_scratch(kk, tid)

1046 do kk = 1, nion

1048 (x(2, 0) - rion(2, kk))**2 + &

1051! const = -2.d0 * zetar(kk) / dist_scratch(kk, tid)
1052 const = -2.d0*zetar(kk)*veps(dist_scratch(kk, tid))

1054 end do
1055 end if
1056 !
1057 ! evaluate pseudopotential contribution
1058 !
1059 if (npsa .gt. 0) then
1060
1061 indtm = istart - 1
1062

1064 &, thread_active, dist_scratch(1, tid), rion, wpseudo_and_psip_scratch(nparshellmax + 1, tid) &
1065 &, npsa, indt - istart + 1, indtm, .false., angle_buffer(1, ind) &
1066 &, wpseudo_and_psip_scratch(1, tid), lbox, mindist, iflagerr, vec)
1067

1069

1071
1072 !if(rank.eq.0) write(6,'(A, L2, 2I10, 3F15.10)') "ivic_debug "&
1073 !, pseudologic, indmesh, tid, ivic(1,istart+1,tid), ivic(2,istart+1,tid), ivic(3,istart+1,tid)
1074 ! definition mesh for pseudo
