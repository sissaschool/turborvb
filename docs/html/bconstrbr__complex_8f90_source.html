16!> @file bconstrbr_complex.f90
17!> @brief Module containing complex constraint handling for Jastrow parameters
18!> @author TurboRVB group
19!> @date 2022
20!> @version 1.0
21
22!> @brief Translates constrained Jastrow parameter changes to unconstrained representation
23!> @details This subroutine handles the transformation of constrained Jastrow parameter
24!> changes back to their original unconstrained representation. It is specifically
25!> designed for handling complex wave functions and 3-body Jastrow terms.
26!>
27!> The subroutine performs the following operations:
28!> 1. Maps constrained Jastrow parameters to their original positions using jbraj array
29!> 2. Handles both real (ipc=1) and complex (ipc=2) wave functions
30!> 3. Supports sign-flipped parameter mapping for constraint handling
31!> 4. Specifically designed for 3-body Jastrow correlation functions
32!>
33!> The algorithm uses a mapping array jbraj to translate between constrained
34!> and unconstrained parameter spaces, where:
35!> - Positive values: direct mapping from ddw to derjas
36!> - Negative values: sign-flipped mapping from ddw to derjas
37!>
38!> @param[in] iesfreer Switch flag for constraint handling (0: no constraints, >0: apply constraints)
39!> @param[in] n3body Number of 3-body Jastrow parameters
40!> @param[in] jbraj Mapping array from constrained to unconstrained Jastrow parameters
41!> @param[out] derjas Output array for unconstrained Jastrow derivatives
42!> @param[in] ddw Input array of constrained parameter changes
43!>
44!> @note If iesfreer = 0, the subroutine returns immediately without any operations
45!>
46!> @warning The subroutine assumes that ddw array contains valid constrained parameter values
47!>
48!> @par Algorithm Details:
49!> The subroutine implements parameter mapping based on the wave function type:
50!> 1. <b>Complex Wave Functions (ipc=2):</b> Handles real and imaginary parts separately
51!> 2. <b>Real Wave Functions (ipc=1):</b> Direct parameter mapping with sign handling
52!>
53!> @par Parameter Mapping:
54!> - <b>Positive jbraj values:</b> Direct copy from ddw to derjas
55!> - <b>Negative jbraj values:</b> Sign-flipped copy from ddw to derjas
56!>
57!> @par Complex Wave Function Handling:
58!> For complex wave functions (ipc=2), the subroutine handles:
59!> - Real parts: derjas(2*i-1) = ±ddw(2*j-1)
60!> - Imaginary parts: derjas(2*i) = ±ddw(2*j)
61!> where the sign depends on whether jbraj(i) is positive or negative.
62!>
63!> @par 3-Body Jastrow Terms:
64!> This subroutine is specifically designed for 3-body Jastrow correlation functions,
65!> which describe electron-electron-electron correlations in quantum Monte Carlo
66!> calculations. The n3body parameter specifies the number of such terms.
67!>
68!> @see Constants module for ipc parameter
69!> @see bconstraint subroutine for similar constraint handling in determinant matrices
70subroutine bconstrbr_complex(iesfreer, n3body, jbraj, derjas &
71 &, ddw)
72 use constants, only: ipc
73 implicit none
74
75 !> @param[in] iesfreer Switch flag for constraint handling (0: no constraints, >0: apply constraints)
76 integer, intent(in) :: iesfreer
77
78 !> @param[in] n3body Number of 3-body Jastrow parameters
79 integer, intent(in) :: n3body
80
81 !> @param[in] jbraj Mapping array from constrained to unconstrained Jastrow parameters
82 integer, intent(in) :: jbraj(*)
83
84 !> @param[out] derjas Output array for unconstrained Jastrow derivatives
85 real*8, intent(out) :: derjas(*)
86
87 !> @param[in] ddw Input array of constrained parameter changes
88 real*8, intent(in) :: ddw(*)
89
90 ! Local variables
91 integer :: i, j
92
93 ! Early return if no constraints are applied
94 if (iesfreer .eq. 0) return
95
96 ! Handle complex wave functions (ipc = 2)
97 if (ipc .eq. 2) then
98 ! Map complex Jastrow parameters with real and imaginary parts
99 do i = 1, n3body
100 j = jbraj(i)
101 if (j .gt. 0) then
102 ! Direct mapping for positive indices
103 derjas(2*i - 1) = ddw(2*j - 1) ! Real part
104 derjas(2*i) = ddw(2*j) ! Imaginary part
105 elseif (j .lt. 0) then
106 ! Sign-flipped mapping for negative indices
107 derjas(2*i - 1) = -ddw(-2*j - 1) ! Sign-flipped real part
108 derjas(2*i) = -ddw(-2*j) ! Sign-flipped imaginary part
109 end if
110 end do
111
112 ! Handle real wave functions (ipc = 1)
113 else
114 ! Map real Jastrow parameters
115 do i = 1, n3body
116 j = jbraj(i)
117 if (j .gt. 0) then
118 ! Direct mapping for positive indices
119 derjas(i) = ddw(j)
120 elseif (j .lt. 0) then
121 ! Sign-flipped mapping for negative indices
122 derjas(i) = -ddw(-j)
123 end if
124 end do
125 end if
126
127 return
128end
