16!> @brief Module for handling atomic orbital parameters and configurations
17!>
18!> This module defines data structures and parameters for representing atomic orbitals,
19!> their configurations, and related properties in quantum chemistry calculations.
20module atomsorb
21 implicit none
22
23 !> Maximum number of orbitals per atom
24 integer, parameter :: maxorb = 20
25
26 !> Maximum number of parameters for each orbital
27 integer, parameter :: maxpar = 10
28
29 !> Maximum number of shells per atom
30 integer, parameter :: max_shell = 5
31
32 !> Dimension for lambda arrays (maxorb * max_shell)
33 integer, parameter :: lambda_dim = maxorb*max_shell
34
35 !> @brief Type definition for orbital parameters and properties
36 !>
37 !> Contains optimization flags, number of parameters, multiplicity and parameter values
38 type orbital
39 integer ioptorb !< Orbital optimization flag
40 integer npar !< Number of parameters
41 integer mult !< Multiplicity
42 double precision parm(maxpar) !< Array of orbital parameters
43 logical fixparm(maxpar) !< Flags for fixed (non-optimizable) parameters
44 end type orbital
45
46 !> @brief Type definition for atomic configuration and orbitals
47 !>
48 !> Contains atomic properties, orbital configurations, and interaction parameters
49 type single_atom
50 integer kion !< Ion index
51 integer nconf !< Number of configurations
52 integer norb !< Number of pairing orbitals
53 integer norbj !< Number of three-body orbitals
54
55 integer shell_pos(maxorb) !< Orbital positions in shell
56 integer jshell_pos(maxorb) !< J-orbital positions in shell
57 integer nlambda, njlambda !< Lambda and J-lambda dimensions
58
59 integer map_lambda(lambda_dim) !< Lambda mapping array
60 integer map_jlambda(lambda_dim) !< J-lambda mapping array
61
62 double precision jonebody(lambda_dim) !< One-body J parameters
63
64 !> Onsite interaction parameters for lambda orbitals
65 double precision onsite_lambda(lambda_dim, lambda_dim)
66 !> Onsite interaction parameters for j-lambda orbitals
67 double precision onsite_jlambda(lambda_dim, lambda_dim)
68
69 type(orbital) orb_list(maxorb) !< List of pairing orbitals
70 type(orbital) jorb_list(maxorb) !< List of three-body orbitals
71 end type single_atom
72
73 !> Array of atomic configurations
74 type(single_atom), allocatable :: atom_list(:)
75end module atomsorb
