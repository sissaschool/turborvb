16!> @brief Prints electron density data to XCrySDen format
17!> @details This subroutine processes electron density data and writes it to a file
18!> compatible with XCrySDen visualization software. It handles both periodic
19!> and non-periodic systems, applies appropriate coordinate transformations,
20!> and manages parallel data collection from multiple MPI processes.
21!> @param[in] rion_ref Reference ion positions for coordinate system
22!> @param[in] rion_shift Shift vector for ion positions
23!> @param[in] nxl Local grid dimension for parallel calculations
24!> @param[in] dent Local density array from each MPI process
25subroutine printden(rion_ref, rion_shift, nxl, dent)
26
27 use allio
28
29 integer :: mesh(3), ind, i, j, k, proc
30 real(8) :: cell_dens(3), mesh_origin(3), rion_ref(*), rion_shift(*), dent(*)
31 real(8), allocatable, dimension(:, :, :) :: datagrid
32 real(8), allocatable, dimension(:, :) :: rion_plot
33
34#ifdef PARALLEL
35 real(8), allocatable, dimension(:, :) :: buffer_grid
36 include 'mpif.h'
37#endif
38
39 allocate (rion_plot(3, nion))
40 rion_plot = 0.d0
41
42 if (.not. iespbc) then
43
44 ! for non period systems translate both mesh_origin and nuclear geometry (rion) by L(i)/2
45 ! in order to place the fragment at the center of the cell drawn by Xcrysden
46 ! note: rion_ref(i) = (n(i)-1)/n(i) * L(i)/2 + mesh_offset(i) (the mesh_offset is rion_shift)
47 rion_plot(1, :) = rion(1, :) + rion_ref(1) - rion_shift(1)
48 rion_plot(2, :) = rion(2, :) + rion_ref(2) - rion_shift(2)
49 rion_plot(3, :) = rion(3, :) + rion_ref(3) - rion_shift(3)
50
51 cell_dens(1) = (nx - 1)*ax
52 cell_dens(2) = (ny - 1)*ay
53 cell_dens(3) = (nz - 1)*az
54
55 mesh(1) = nx
56 mesh(2) = ny
57 mesh(3) = nz
58
59 mesh_origin(1) = rion_ref(1)
60 mesh_origin(2) = rion_ref(2)
61 mesh_origin(3) = rion_ref(3)
62
63 else
64

74 mesh(2) = ny + 1
75 mesh(3) = nz + 1
76
77 mesh_origin(1) = rion_shift(1)

80
81 end if
82
83 if (rank .eq. 0) then

86 end if
87
88#ifdef PARALLEL
89 if (rank .eq. 0) then
90 allocate (buffer_grid(nxl*ny*nz, nproc))

92 end if
93
94 call mpi_gather(dent, nxl*ny*nz, mpi_double_precision, buffer_grid&
95 &, nxl*ny*nz, mpi_double_precision, 0, mpi_comm_world, ierr)
96
97 if (rank .eq. 0) then
98 do proc = 0, nproc - 1
99 ind = 0
100 do k = 1, nz
101 do j = 1, ny
102 do i = proc + 1, nx, nproc
103 ind = ind + 1
104 datagrid(i, j, k) = buffer_grid(ind, proc + 1)
105 end do
106 end do
107 end do
108 end do
109 deallocate (buffer_grid)
110 end if
111
112#else
113
114 ind = 0
115 do k = 1, nz
116 do j = 1, ny
117 do i = 1, nx
118 ind = ind + 1
119 datagrid(i, j, k) = dent(ind)
120 end do
121 end do
122 end do
123
124#endif
125
126 if (rank .eq. 0) then
127
128 if (iespbc) then
129 ! periodic condition on the grid
130 do k = 1, nz
131 do j = 1, ny
132 datagrid(mesh(1), j, k) = datagrid(1, j, k)
133 end do
134 end do
135
136 do k = 1, nz
137 do j = 1, mesh(1)
138 datagrid(j, mesh(2), k) = datagrid(j, 1, k)
139 end do
140 end do
141
142 do k = 1, mesh(2)
143 do j = 1, mesh(1)
144 datagrid(j, k, mesh(3)) = datagrid(j, k, 1)
145 end do
146 end do
147
148 end if
149
150 ! nion, rion, atom_number, iespbc read from read_fort10 called in the main driver
151
152 write (6, *) 'writing dft density on xcrysden file'
153 call plot_3d_data(1, cell_dens, cell_dens, nion, rion_plot &
154 , atom_number, iespbc, mesh, mesh_origin, datagrid, 1, 'dft_density')
155 deallocate (datagrid)
156 end if
157
158 deallocate (rion_plot)
159
160end subroutine printden
