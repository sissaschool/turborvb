16!> @file corrvar.f90
17!> @brief Program for computing energy variance and correlation analysis using bootstrap resampling
18!> @details This program reads block averages of energy and variance from fort.21 and fort.22 files
19!> and computes global averages and standard deviations using bootstrap resampling technique.
20!> It estimates energy error bars, correlation times, and variance statistics for quantum Monte Carlo
21!> calculations.

23!> @brief Main program for energy variance and correlation analysis
24!> @details Reads energy and variance block averages, performs bootstrap resampling to estimate
25!> statistical errors, computes energy variance, error bars, and correlation times. Supports
26!> k-points calculations and provides comprehensive statistical analysis of Monte Carlo data.
27program bootback
28
29 implicit none
30
31 integer :: iseed, nbin, n, nbinm, k, j, i, nmis, kmain, nwalk, bin_length
32 real(8), dimension(:), allocatable :: w, e, es
33 real(8) :: wpip, drand1
34 real(8) :: eps, wt, et, ets, etsv, variance
35 real(8) :: errn, givenmeas, givenmeas2
36 real(8) :: eta, eav, esav, eav0, err
37 real(8) :: eta_, eav_, esav_, err_
38 logical file_exists
39 iseed = 12375731
40 call rand_init(iseed)
41
42 ! open files with block-average of the energy fort.21
43 ! In case of k-points calculations it already contains
44 ! the block-average of the variance.
45 open (unit=11, form='formatted', status='old', file='fort.21', err=101)
46 open (unit=12, form='formatted', status='old', file='fort.22', err=102)
47 open (unit=13, form='formatted', status='old', file='parminimized.d', err=100)
48
49 ! inquire(file='kp_info.dat',exist=file_exists)
50 ! if(file_exists) then
51 ! open(unit=14,form='formatted',status='old',file='kp_info.dat')
52 ! read(14,*) nk
53 ! else
54 ! nk=1
55 ! endif
56
57 read (13, *)
58 read (13, *) nwalk
59 close (13)
60
61 ! nwalk=nwalk/nk
62
63 read (5, *) bin_length
64
65 ! inquire if k-points are present
66
67 n = 1
68 eps = 1d-6
69
70 ! reading datas
71 nbin = 0
72 do while (nbin .ge. 0)
73 read (11, *, err=101, end=500)
74 read (12, *, err=102, end=500)
75 nbin = nbin + 1
76 end do
77500 continue
78 rewind(11)
79 rewind(12)
80 allocate (w(nbin), e(nbin), es(nbin))
81 do i = 1, nbin
82 read (11, *, err=101, end=500) e(i), w(i)
83 read (12, *, err=102, end=500) es(i), wpip
84 if (wpip .ne. w(i)) then
85 write (6, *) ' warning the measures are not correlated !!!! '
86 end if
87 end do
88 close (11)
89 close (12)
90
91 ! write(6,*) ' Measures read '
92 ! do i=1,nbin
93 ! write(6,*) i,e(i),es(i),sqrt(es(i)-e(i)**2)
94 ! enddo
95
96 eav0 = sum(e(:)*w(:))/sum(w(:))
97 write (6, *) ' number of bins read =', nbin
98
99 nmis = 1000
100 err = 0.d0
101 eta = 0.d0
102 eav = 0.d0
103 esav = 0.d0
104 err_ = 0.d0
105 eta_ = 0.d0
106 eav_ = 0.d0
107 esav_ = 0.d0
108
109 do kmain = 1, nmis
110 !
111 wt = 0.d0
112 et = 0.d0
113 ets = 0.d0
114 etsv = 0.d0
115 ! bootstrap estimate sample
116 do k = 1, nbin
117 j = drand1()*nbin + 1
118 wt = wt + w(j)
119 et = et + w(j)*e(j)
120 ets = ets + w(j)*es(j) ! correlated estimate of <E**2>
121 etsv = etsv + w(j)*e(j)**2 ! uncorrelated estimate of <E**2>
122 end do
123
124 ! now average correlation functions on the given bootstrap
125 et = et/wt
126 ets = ets/wt
127 etsv = etsv/wt
128 errn = dsqrt(abs(etsv - et**2)/nbin)
129 givenmeas = ets - et**2
130 eta = eta + givenmeas
131 err = err + givenmeas**2
132 eav = eav + et
133 esav = esav + et**2
134 variance = ets - et**2
135 givenmeas2 = errn
136 eta_ = eta_ + givenmeas2
137 err_ = err_ + givenmeas2**2
138 givenmeas2 = nbin*bin_length*errn**2/variance
139 eav_ = eav_ + givenmeas2
140 esav_ = esav_ + givenmeas2**2
141
142 end do
143
144 ! normalize measures
145 eta = eta/nmis
146 err = err/nmis
147 esav = esav/nmis
148 eav = eav/nmis
149 err = sqrt(err - eta**2)
150 esav = sqrt(esav - eav**2)
151
152 eta_ = eta_/nmis
153 err_ = err_/nmis
154 esav_ = esav_/nmis
155 eav_ = eav_/nmis
156 err_ = dsqrt(err_ - eta_**2)
157 esav_ = dsqrt(esav_ - eav_**2)
158
159 write (6, *) ' Energy =', dble(eav0), dble(esav)
160 write (6, *) ' Variance square =', dble(eta), dble(err)
161 write (6, *) ' Est. energy error bar =', eta_, err_
162 write (6, *) ' Est. corr. time =', eav_*nwalk, esav_*nwalk
163
164 deallocate (e, es, w)
165 stop
166
167 !!!!!!!!!!! ERRORS !!!!!!!!!!!
168100 write (6, *) 'ERROR: file parminimized.d not found or wrong!'
169 stop
170101 write (6, *) 'ERROR: file fort.21 not found or wrong!'
171 stop
172102 write (6, *) 'ERROR: file fort.22 not found or wrong!'
173 stop
174103 write (6, *) 'ERROR: file kp_info.dat empty or wrong!'
175 stop
176
177end program bootback
