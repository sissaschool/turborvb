16!-----------------------------
17!> @brief Performs non-self-consistent band structure calculations
18!> @details This subroutine performs non-self-consistent calculations to compute
19!> band structures and electronic properties. It reads the density from
20!> a self-consistent calculation, diagonalizes the Hamiltonian at each
21!> k-point, and saves eigenvalues and eigenvectors for post-processing.
22!> Supports both real and complex wavefunctions with LSDA calculations.
23subroutine non_self_consistent_run()
24 !-----------------------------
25
26 use constants, only: two_pi, qp
27 use kpoints_mod
28 use setup
29 use freeelmod_complex, only: diagonalize_hamiltonian, checksum
30 use fourier_module
31 implicit none
32
33 integer :: indkp, mpi_err, i
34 logical :: occs_file_e
35 real(8) :: lworkr
36#ifdef PARALLEL
37 include "mpif.h"
38#endif
39 !
40 ! initialize vhartreeq in reciprocal space
41 ! Ewald correction to Hartree potential.
42 !
43 call evalvhartreeq()
44 if (rank .eq. 0) then
45 write (6, '(a,2F15.8)') ' Initial Hartree potential: ', sum(vhartree(:)), sum(vhartreeq(:))
46 if (iespbc) then
47 write (6, '(a,F15.8)') ' Kappa found: ', kappa
48 write (6, '(a,F15.8)') ' Ewald constant found:', ewaldion1b + (1.d0 - weightvh)*ewaldel1b
49 write (6, '(a,F15.8)') ' eselfion = ', ewaldion1b
50 write (6, '(a,F15.8)') ' eself1bel = ', ewaldel1b
51 end if
52 end if
53 !
54 ! read electronic density from self-consistent run
55 !
56 call read_density_from_file()
57
58 ! just to be sure initialize all to zero (again)
59 molecorb = 0.d0
60 if (yeslsda .or. ipc .eq. 2) molecorbdo = 0.d0
61 eigmol_sav = 0.d0
62 molecorb_sav = 0.d0
63 if (yeslsda .or. ipc .eq. 2) then
64 eigmoldo_sav = 0.d0
65 molecorbdo_sav = 0.d0
66 end if
67
68 k_loop: do indkp = 1, nk
69 !
70 ! initialize k-point
71 !
72 phase(:) = xkp(:, indkp)
73 phase_down(:) = xkp_down(:, indkp)
74 rphase(:) = 0.d0
75 phase2pi(:) = phase(:)*two_pi
76 phase2pi_down(:) = phase_down(:)*two_pi
77 indk = indkp
78 !
79 ! compute basis and Hamiltonian overlaps:
80 ! S_ij = < \phi_i | \phi_j >
81 ! H_ij = < \phi_i | H | \phi_j >
82 ! With the new basis set (yes_crystal=.true.) they are both dependent
83 ! from the chosen k-point.
84 !
85 call initialize_mats_new()
86 !
87 ! update density-dependent part of the hamiltonian
88 ! starting from the density read from file
89 !
90 call uphamilt_new()
91 !
92 ! diagonalize new hamiltonian
93 !
94 lworkr = 0
95 call diagonalize_hamiltonian(1, lworkr)
96 if (orthodiag) call improvediag()
97 !
98 ! check matrix elements sum in debug mode
99 !
100#ifdef DEBUG
101 call checksum()
102#endif
103
104 if (rank .eq. 0) then
105 write (6, *)
106 write (6, '(A)') ' Eigenvalues up/down: '
107 do i = 1, bands
108 write (6, '(I2,2X,2F10.6)') i, eigmol(i), eigmoldo(i)
109 end do
110 end if
111
112 ! save eigenvalues/eigenvectors
113 ! for postprocessing tools
114 eigmol_sav(1:bands, indkp) = eigmol(1:bands)
115 molecorb_sav(:, 1:bands, indkp) = molecorb(:, 1:bands)
116 occupations_sav(1:bands, indkp) = occupations(1:bands)
117 if (yeslsda .or. ipc .eq. 2) then
118 eigmoldo_sav(1:bands, indkp) = eigmoldo(1:bands)
119 molecorbdo_sav(:, 1:bands, indkp) = molecorbdo(:, 1:bands)
120 occupationsdo_sav(1:bands, indkp) = occupationdo(1:bands)
121 end if
122 if (rank .eq. 0) write (6, '(A,2X,I4,A,3F10.6)') ' k-point # ', indkp, ' completed: ', xkp(:, indkp)
123 ! put to zero variables related to hamiltonian
124 ! which will be updated next iteration
125 eigmol = 0.d0
126 molecorb = 0.d0
127 if (yeslsda .or. ipc .eq. 2) then
128 eigmoldo = 0.d0
129 molecorbdo = 0.d0
130 end if
131#ifdef __SCALAPACK
132 molecorbl = 0.d0
133 if (yeslsda .or. ipc .eq. 2) molecorbldo = 0.d0
134#endif
135
136 end do k_loop
137 !
138 ! compute relevant quantities such as band structure
139 ! or DOS. The type of computation is selected with the
140 ! input variable "task":
141 ! task=1 --> band structure calculation
142 ! task=2 --> Density of states
143 ! task=3 --> Fermi surface calculation (to be implemented)
144 !
145 call postprocessing(task)
146
147 return
148
149end subroutine non_self_consistent_run

151!----------------------------------
152!> @brief Reads electronic density from self-consistent calculation files
153!> @details This subroutine reads the electronic density from files generated
154!> during a self-consistent calculation. It handles both charge and
155!> spin density, supports parallel I/O, and validates the total charge
156!> conservation.
157subroutine read_density_from_file()
158 !----------------------------------
159
160 use constants, only: ipc
161 use allio, only: rank, nproc, nk, nprocrep, nx, ny, nz
162 use setup, only: nproco, nko, yeslsda, spint, dent, unit_scratch_distributed, meshproc, unit_scratch_densities
163 implicit none
164
165 integer :: i, j, k, proc, indmesh, mpi_err, ierr
166 real(8), dimension(:, :, :), allocatable :: dens_grid, spint_grid
167 real(8), dimension(:, :), allocatable :: buffer_grid
168#ifdef PARALLEL
169 include "mpif.h"
170#endif
171
172 if (nproc/nk .ne. nproco/nko) then
173 call error(' non_self_consistent_run ', ' Number of processor per k-point &
174 &different with respect to self-consistent run ', -1, rank)
175 end if
176
177 if (nproc/nk .eq. nproco/nko) then
178 ! if same number of processors (per k-point) as SC, then read
179 ! usual scratch files "tmp0xxxxx"
180 rewind(unit_scratch_distributed)
181 ! first line(s) contains the overlap matrices
182 read (unit_scratch_distributed)
183 if (ipc .eq. 2) read (unit_scratch_distributed)
184 if (yeslsda) then
185 read (unit_scratch_distributed) dent, spint
186 else
187 read (unit_scratch_distributed) dent
188 end if
189 close (unit_scratch_distributed)
190 else
191 if (rank .eq. 0) write (6, *) ' ERROR run with the same number of proc.! '
192#ifdef PARALLEL
193 call mpi_finalize(ierr)
194#endif
195 stop
196
197 end if
198
199 !
200 ! check if initial total charge read from file is OK
201 call cutdens()
202
203 return
204end subroutine read_density_from_file

207subroutine postprocessing(task)
208 !----------------------------------
209 use allio, only: rank
210 implicit none
211 integer, intent(in) :: task
212
213 select case (task)
214
215 case (1) ! band structure calculation
216 if (rank .eq. 0) call plot_bands()
217
218 case (2) ! DOS calculation
219 if (rank .eq. 0) call evaluate_dos()
220
221 !case(3) ! Fermi surface calculation (TO BE IMPLEMENTED)
222 !if(rank.eq.0) call evaluate_fermisurf
223
224 case (5) ! evaluate all
225 if (rank .eq. 0) then
226 call plot_bands()
227 call evaluate_dos()
228 end if
229
230 case (0)
231 call error(' non_sc_postprocessing ', ' No tasks required ', -1, rank)
232
233 case default
234 call error(' non_sc_postprocessing ', ' Task type not recognized ', -1, rank)
235
236 end select
237
238 return
239
240end subroutine postprocessing

242!---------------------
243subroutine plot_bands
244 !---------------------
245
246 ! Routine for plotting band structures from NSC calculations. Adapted from the
247 ! QuantumESPRESSO code.
248 ! kp_type must be 2 or 3 in order to run this routine.
249 ! In the case of a kpoints path (kp_type=3), one already knows the number of k-points and the
250 ! number of lines in the path, which are "nk" and "nk1-1" respectively.
251
252 use allio, only: rank
253 use kpoints_mod, only: nk, xkp, kp_type
254 use setup, only: eigmol_sav, eigmoldo_sav, yeslsda, bands
255 use constants, only: ipc, energy_unit ! Ha to eV conversion
256
257 implicit none
258 logical, dimension(:), allocatable :: high_symmetry, in_range, in_rangedo
259 integer i, j, k
260 real(8), dimension(:), allocatable :: kp_path
261 real(8) tmp_norm, k1(3), k2(3), dkl, dkl_sav, emin, emax
262 character(len=80) :: filename
263
264 if (kp_type .lt. 2 .or. kp_type .gt. 3) then
265 call error(' postprocessing ', ' Use kp_type=2 or kp_type=3 for plotting &
266 & the DFT band structure. ', -1, rank)
267 return
268 end if
269
270 ! initialization
271 write (6, *) '---------------------------'
272 write (6, *) ' Evaluating band structure '
273 write (6, *) '---------------------------'
274
275 filename = 'band_structure.dat'
276 allocate (high_symmetry(nk), in_range(bands))
277 in_range = .true.
278 if (yeslsda .or. ipc .eq. 2) then
279 allocate (in_rangedo(bands))
280 in_rangedo = .true.
281 end if
282 allocate (kp_path(nk))
283 high_symmetry = .false.
284 kp_path(1) = 0.d0
285 !
286 ! find high symmetry points in the 2D plot
287 !
288 do k = 1, nk
289 if (k .eq. 1 .or. k .eq. nk) then
290 high_symmetry(k) = .true.
291 else
292 k1(:) = xkp(:, k) - xkp(:, k - 1)
293 k2(:) = xkp(:, k + 1) - xkp(:, k)
294 tmp_norm = (k1(1)*k2(1) + k1(2)*k2(2) + k1(3)*k2(3))/ &
295 dsqrt(k1(1)*k1(1) + k1(2)*k1(2) + k1(3)*k1(3))/ &
296 dsqrt(k2(1)*k2(1) + k2(2)*k2(2) + k2(3)*k2(3))
297 ! if changing direction in the 1BZ, then it's a high symmetry point
298 high_symmetry(k) = (abs(tmp_norm - 1.d0) .gt. 1.0d-4)
299 end if
300 if (sum(xkp(:, k)**2) .lt. 1.0d-9) high_symmetry(k) = .true.
301 ! distance between 2 kps.
302 if (k .eq. 2) dkl_sav = dsqrt(k1(1)**2 + k1(2)**2 + k1(3)**2)
303 end do
304 !
305 ! create 2D path
306 !
307 do k = 2, nk
308
309 dkl = dsqrt((xkp(1, k) - xkp(1, k - 1))**2 + &
310 (xkp(2, k) - xkp(2, k - 1))**2 + &
311 (xkp(3, k) - xkp(3, k - 1))**2)
312
313 if (dkl .gt. 5.d0*dkl_sav) then
314 kp_path(k) = kp_path(k - 1)
315 elseif (dkl .gt. 1.0d-5) then
316 kp_path(k) = kp_path(k - 1) + dkl
317 dkl_sav = dkl
318 else
319 kp_path(k) = kp_path(k - 1) + dkl
320 end if
321
322 end do
323 ! find minimum and maximum values
324 emin = eigmol_sav(1, 1)
325 emax = eigmol_sav(1, 1)
326 do k = 1, nk
327 do j = 1, bands
328 emin = min(emin, eigmol_sav(j, k))
329 if (yeslsda .or. ipc .eq. 2) emin = min(emin, eigmoldo_sav(j, k))
330 emax = max(emax, eigmol_sav(j, k))
331 if (yeslsda .or. ipc .eq. 2) emax = max(emax, eigmoldo_sav(j, k))
332 end do
333 end do
334 ! find bands values which are in the specified energy range
335 in_range = .true.
336 in_rangedo = .true.
337 do i = 1, bands
338 in_range(i) = any(eigmol_sav(i, 1:nk) >= emin .and. eigmol_sav(i, 1:nk) <= emax)
339 in_rangedo(i) = any(eigmoldo_sav(i, 1:nk) >= emin .and. eigmoldo_sav(i, 1:nk) <= emax)
340 end do
341
342 write (6, *) ' Band structures INFORMATION: '
343 write (6, *) 'Minimum/maximum bands values:', emin, emax
344 write (6, *) 'High symmetry points: '
345 do k = 1, nk
346 if (high_symmetry(k)) then
347 write (6, '(F10.6,3F10.6)') kp_path(k), xkp(:, k)
348 end if
349 end do
350 !
351 ! now draw bands
352 open (unit=80, file=trim(filename), form='formatted', status='unknown')
353 write (80, *) ' # Spin up DFT bands (eV) '
354 do i = 1, bands
355 if (in_range(i)) then
356 if (mod(i, 2) /= 0) then
357 write (80, 100) (kp_path(j), energy_unit*eigmol_sav(i, j), j=1, nk)
358 else
359 write (80, 100) (kp_path(j), energy_unit*eigmol_sav(i, j), j=nk, 1, -1)
360 end if
361 end if
362 end do
363 if (yeslsda .or. ipc .eq. 2) then
364 ! add two carriage return for gnuplot
365 write (80, *)
366 write (80, *)
367 write (80, *) ' # Spin down DFT bands (eV)'
368 do i = 1, bands
369 if (in_rangedo(i)) then
370 if (mod(i, 2) /= 0) then
371 write (80, 100) (kp_path(j), energy_unit*eigmoldo_sav(i, j), j=1, nk)
372 else
373 write (80, 100) (kp_path(j), energy_unit*eigmoldo_sav(i, j), j=nk, 1, -1)
374 end if
375 end if
376 end do
377 end if
378 !
379 close (80)
380 deallocate (high_symmetry, kp_path, in_range)
381 if (yeslsda .or. ipc .eq. 2) deallocate (in_rangedo)
382
383100 format(2f10.4)
384
385 return
386
387end subroutine plot_bands

389!-----------------------
390subroutine evaluate_dos
391 !-----------------------
392
393 !
394 ! Routine to calculate the density of states.
395 ! Use optocc>0 and a non-zero value of the smearing (epsshell)
396 ! in order to compute the density of states.
397 !
398
399 use constants, only: ipc, energy_unit, qp
400 use kpoints_mod, only: nk, xkp, wkp
401 use setup, only: eigmol_sav, eigmoldo_sav, epsshell, optocc, deltae, &
402 emin, emax, yeslsda, bands
403 use compute_efermi, only: smeard
404
405 implicit none
406 integer :: i, j, k, ibnd, indk, smear_type
407 integer :: ndos ! # of points in the DOS
408 real(8), dimension(:), allocatable :: tdos, dos_up, dos_down
409 real(qp) :: e, dt
410 character(len=80) :: filename
411
412 ! initialization
413 write (6, *) '------------------------------'
414 write (6, *) ' Evaluating density of states '
415 write (6, *) '------------------------------'
416 filename = 'density_of_states.dat'
417 if (optocc .eq. 0 .or. epsshell .eq. 0.d0) then
418 write (6, *) 'Warning: DOS must be updated with a smearing function! Setting default values.'
419 optocc = 1
420 epsshell = 0.01d0
421 end if
422 smear_type = optocc
423 dt = epsshell
424 ! setting min/max values of the energies
425 ! if not chosen from the user
426 if (emin .eq. 0.d0 .and. emax .eq. 0.d0) then
427 emin = eigmol_sav(1, 1)
428 emax = eigmol_sav(1, 1)
429 do k = 1, nk
430 do j = 1, bands
431 emin = min(emin, eigmol_sav(j, k))
432 if (yeslsda .or. ipc .eq. 2) emin = min(emin, eigmoldo_sav(j, k))
433 emax = max(emax, eigmol_sav(j, k))
434 if (yeslsda .or. ipc .eq. 2) emax = max(emax, eigmoldo_sav(j, k))
435 end do
436 end do
437 end if
438 !
439 ndos = int((emax - emin)/deltae)
440 allocate (dos_up(ndos + 1), dos_down(ndos + 1), tdos(ndos + 1))
441
442 dos_up = 0.d0
443 dos_down = 0.d0
444 tdos = 0.d0
445 e = 0.d0
446
447 open (unit=80, file=trim(filename), form='formatted', status='unknown')
448 write (80, *) '# energy (eV) Density_of_states (spin up)'
449 do i = 0, ndos
450 e = emin + deltae*i
451 do indk = 1, nk
452 do ibnd = 1, bands
453 dos_up(i + 1) = dos_up(i + 1) + wkp(indk)*smeard(eigmol_sav(ibnd, nk), dt, e, smear_type)
454 end do
455 end do
456 write (80, 100) e*energy_unit, dos_up(i + 1)
457 end do
458
459 if (yeslsda .or. ipc .eq. 2) then
460 write (80, *)
461 write (80, *)
462 write (80, *) '# energy (eV) Density_of_states (spin down)'
463 do i = 0, ndos
464 e = emin + deltae*i
465 do indk = 1, nk
466 do ibnd = 1, bands
467 dos_down(i + 1) = dos_down(i + 1) + wkp(indk)*smeard(eigmoldo_sav(ibnd, indk), dt, e, smear_type)
468 end do
469 end do
470 write (80, 100) e*energy_unit, dos_down(i + 1)
471 end do
472 end if
473
474 deallocate (dos_up, dos_down, tdos)
475 write (6, *)
476 close (80)
477
478100 format(2f10.4)
479
480 return
481
482end subroutine evaluate_dos
