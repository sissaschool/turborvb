29 SUBROUTINE zlasktrf( UPLO, MODE, N, NB, A, LDA, IPIV, W, LDW,
30 $ INFO)
31*
32* .. Scalar Arguments ..
33 CHARACTER UPLO, MODE
34 INTEGER INFO, LDA, LDW, N, NB, STEP
35* ..
36* .. Array Arguments ..
37 INTEGER IPIV( * )
38 DOUBLE COMPLEX A( LDA, * )
39 DOUBLE COMPLEX W( LDW, * )
40*
41* Purpose
42* =======
43*
44* ZLASKTRF computes NB steps of the factorization of a skew-symmetric
45* matrix A using the Parlett-Reid algorithm:
46*
47* P*A*P^T = U*T*U^T or P*A*P^T = L*T*L^T
48*
49* If UPLO = 'U', ZLASKTRF reduces the last NB rows and columns of a
50* matrix, of which the upper triangle is supplied;
51* if UPLO = 'L', ZLASKTRF reduces the first NB rows and columns of a
52* matrix, of which the lower triangle is supplied.
53*
54* Alternatively, the routine can also be used to compute a partial
55* tridiagonal form
56*
57* This is an auxiliary routine called by ZSKTRF.
58*
59* Arguments
60* =========
61*
62* UPLO (input) CHARACTER*1
63* Specifies whether the upper or lower triangular part of the
64* symmetric matrix A is stored:
65* = 'U': Upper triangular
66* = 'L': Lower triangular
67*
68* MODE (input) CHARACTER*1
69* = 'N': A is fully tridiagonalized
70* = 'P': A is partially tridiagonalized for Pfaffian computation
71* (details see below)
72*
73* N (input) INTEGER
74* The order of the matrix A. N >= 0. N must be even if MODE = 'P'.
75*
76* NB (input) INTEGER
77* The number of rows and columns to be reduced.
78*
79* A (input/output) DOUBLE COMPLEX array, dimension (LDA,N)
80* On entry, the skew-symmetric matrix A.
81* If UPLO = 'U', the leading n-by-n upper triangular part
82* of A contains the upper triangular part of the matrix A,
83* and the strictly lower triangular part of A is not referenced.
84* If UPLO = 'L', the leading n-by-n lower triangular part
85* of A contains the lower triangular part of the matrix A,
86* and the strictly upper triangular part of A is not referenced.
87* On exit:
88* if UPLO = 'U', the last NB columns have been reduced to
89* tridiagonal form, with the diagonal elements overwriting
90* the diagonal elements of A; the elements above the diagonal
91* represent the upper unit triangular matrix U. If MODE = 'P',
92* only the even columns of the last NB columns contain
93* meaningful values.
94* if UPLO = 'L', the first NB columns have been reduced to
95* tridiagonal form, with the diagonal elements overwriting
96* the diagonal elements of A; the elements below the diagonal
97* represent the lower unit triangular matrix L. If MODE = 'P',
98* only the odd columns of the first NB columns contain
99* meaningful values.
100*
101* LDA (input) INTEGER
102* The leading dimension of the array A. LDA >= max(1,N).
103*
104* IPIV (output) INTEGER array, dimension (N)
105* Information about the permutation matrix P: row and column
106* i are interchanged with IPIV(i). If UPLO = 'U', those
107* interchanges are done in the order i = N ... 1, if UPLO = 'L'
108* in the order i = 1 ... N.
109*
110* W (workspace) DOUBLE COMPLEX array, dimension (LDW,NB)
111* The n-by-nb matrix W required to update the unreduced part
112* of A. (The update is performed in this routine)
113*
114* LDW (input) INTEGER
115* The leading dimension of the array W. LDW >= max(1,N).
116*
117* =====================================================================
118*
119* .. Parameters ..
120 DOUBLE COMPLEX ZERO, ONE
121 parameter( zero = (0.0d+0, 0.0d+0),
122 $ one = (1.0d+0,0.0d+0) )
123
124* .. Local Scalars ..
125 INTEGER K, KK, KP, NPANEL, WK
126 DOUBLE PRECISION COLMAX
127 DOUBLE COMPLEX T
128* ..
129* .. External Functions ..
130 LOGICAL LSAME
131 INTEGER IZAMAX
132 EXTERNAL lsame, izamax
133* ..
134* .. External Subroutines ..
135* ..
136* .. Intrinsic Functions ..
137 INTRINSIC abs, max
138* ..
139* .. Executable Statements ..
140*
141* No safety checks, it's an internal function
142
143 info = 0
144
145 IF( lsame( mode, 'P' ) ) THEN
146 step = 2
147 ELSE
148 step = 1
149 END IF
150
151* double the amount of panels before the block update, if STEP == 2
152 npanel = nb * step
153
154 IF( lsame( uplo, 'U' ) ) THEN
155
156* Factorize A as U * T * U^T using the upper triangle of A
157
158* Consider the last NB columns of the matrix:
159* First compute the transformations and apply them to the last NB
160* columns (only), then apply the accumulated transformations to the rest
161* of the matrix
162
163 wk = 0
164 DO 10 k=n, max(n-npanel+1, 2), -1
165*
166* Update A(1:K,K) with all the accumulated transformations
167* (if K<N: K=N,N-1 is updated when computing the Gauss vector,
168* K=N-1 is not affected by the K=N transform)
169*
170 kk = k-1
171
172 IF( k .LT. n) THEN
173
174 IF( wk .GT. 0 ) THEN
175 a( k, k ) = zero
176 CALL zgemv( 'N', k, wk, +one, a( 1, n-(wk-1)*step ),
177 $ lda*step, w( k, nb-wk+1 ), ldw, one,
178 $ a( 1, k ), 1 )
179 CALL zgemv( 'N', k, wk, -one, w( 1, nb-wk+1 ),
180 $ ldw, a( k, n-(wk-1)*step ), lda*step, one,
181 $ a( 1, k ), 1 )
182 a( k, k ) = zero
183 END IF
184
185* Store the (updated) column K in W(:,NB-WK+1)
186 IF( mod(k, step).EQ.1 .OR. step.EQ.1 ) THEN
187 wk = wk + 1
188 CALL zcopy(k, a(1, k), 1, w(1, nb-wk+1 ), 1)
189 END IF
190 END IF
191
192 IF( mod(k, step) .EQ. 0) THEN
193* For STEP == 1, process every column, but if
194* STEP == 2, do only things for the even columns
195
196* Find the pivot
197 kp = izamax(k-1, a( 1, k ), 1)
198 colmax = abs( a( kp, k ) )
199
200 IF( colmax.EQ.zero ) THEN
201* The column is completely zero - do nothing
202 IF( info.EQ.0 ) THEN
203 info = k - 1
204 END IF
205 kp = kk
206 END IF
207
208* swap rows and columns K+1 and KP in the
209* full matrix A(1:N,1:N)
210* Also, swap the first K-1 columns of W
211
212 IF( kp .NE. kk ) THEN
213 CALL zswap( kp-1, a( 1, kk ), 1, a( 1, kp ),1)
214 CALL zswap( kk-kp-1, a( kp+1, kk ), 1,
215 $ a( kp, kp+1 ), lda )
216
217 CALL zswap( n-k+1, a( kk, k), lda, a( kp, k), lda)
218
219 CALL zscal(kk-kp, -one, a(kp, kk), 1)
220 CALL zscal(kk-kp-1, -one, a(kp, kp+1), lda)
221
222* Swap in the last columns of W
223 IF( wk .GT. 0 ) THEN
224 CALL zswap( wk, w( kk, nb-wk+1 ), ldw,
225 $ w( kp, nb-wk+1 ), ldw)
226 END IF
227 END IF
228
229* (The column/row K+1 is not affected by the update)
230 IF( colmax .NE. zero ) THEN
231* Store L(k+1) in A(k)
232 CALL zscal(k-2, one/a( k-1, k ), a(1, k), 1)
233 END IF
234
235* Delay the update of the trailing submatrix (done at the beginning
236* of each loop for every column of the first NB columns, and then
237* finally in a rank-3 update for the last N-NB block)
238
239* Store Pivot
240 ipiv( k-1 ) = kp
241
242
243 ELSE
244* STEP == 2 and an even column, do nothing
245 ipiv( k-1 ) = k-1
246 END IF
247 10 CONTINUE
248
249* Now update the leading A(1:N-NB, 1:N-NB) submatrix in a level 3 update,
250* if necessary
251 IF( n-npanel+1 .GT. 2 ) THEN
252
253* For this we have to set the N-NB,N-NB+1 entry to zero,
254* but restore it later
255 t = a( n-npanel, n-npanel+1 )
256 a( n-npanel, n-npanel+1 ) = zero
257 IF( wk .LT. nb) THEN
258* Store the column N-NB in W, and update it in W
259 CALL zcopy(n-npanel, a(1, n-npanel), 1,
260 $ w(1, 1), 1)
261 w( n-npanel, 1 ) = zero
262 CALL zgemv( 'N', n-npanel, wk, +one,
263 $ a( 1, n-(wk-1)*step ), lda*step,
264 $ w( n-npanel, nb-wk+1 ), ldw, one, w( 1, 1 ), 1 )
265 CALL zgemv( 'N', n-npanel, wk, -one, w( 1, nb-wk+1 ),
266 $ ldw, a( n-npanel, n-(wk-1)*step ), lda*step, one,
267 $ w( 1, 1 ), 1 )
268 w( n-npanel, 1 ) = zero
269 wk = wk + 1
270 END IF
271
272* Now do the rank-3 update
273 CALL zskr2k( uplo, "N", n-npanel, nb, one,
274 $ a(1, n-(wk-1)*step), lda*step,
275 $ w(1,1), ldw, one, a(1, 1), lda)
276
277 a( n-npanel, n-npanel+1 ) = t
278 END IF
279
280 ELSE
281
282* Factorize A as L * T * L^T using the lower triangle of A
283
284* Consider the first NB columns of the matrix:
285* First compute the transformations and apply them to the first NB
286* columns (only), then apply the accumulated transformations to the rest
287* of the matrix
288
289 wk = 0
290 DO 30 k=1, min(npanel, n-1)
291
292*
293* Update A(K+1:n,K+1) with all the accumulated transformations
294* (if K>2: K=1,2 is updated when computing the Gauss vector,
295* K=2 is not affected by the K=1 transform)
296*
297 kk = k+1
298
299 IF( k .GT. 1) THEN
300
301 IF( wk .GT. 0) THEN
302 a( k, k ) = zero
303 CALL zgemv( 'N', n-k+1, wk, +one, a( k, 1 ),
304 $ lda*step, w( k, 1 ), ldw, one, a( k, k ), 1 )
305 CALL zgemv( 'N', n-k+1, wk, -one, w( k, 1 ),
306 $ ldw, a( k, 1 ), lda*step, one, a( k, k ), 1 )
307 a( k, k ) = zero
308 END IF
309
310* Store the (updated) column K in W(:,WK)
311 IF( mod(k, step) . eq. 0 ) THEN
312 wk = wk + 1
313 CALL zcopy(n-k+1, a(k, k), 1, w(k, wk), 1)
314 END IF
315 END IF
316
317 IF( mod(k, step) .EQ. 1 .OR. step .EQ. 1) THEN
318* For STEP == 1, process every column, but if
319* STEP == 2, do only things for the odd columns
320
321* Find the pivot
322 kp = k + izamax(n-k, a( k+1, k ), 1)
323 colmax = abs( a( kp, k ) )
324
325 IF( colmax.EQ.zero ) THEN
326* The column is completely zero - do nothing
327 IF( info.EQ.0 ) THEN
328 info = k
329 END IF
330 kp = kk
331 END IF
332
333* swap rows and columns K+1 and KP in the
334* full matrix A(1:N,1:N)
335* Also, swap the first K-1 columns of W
336
337 IF( kp .NE. kk ) THEN
338 IF( kp.LT.n ) THEN
339 CALL zswap( n-kp, a( kp+1, kk ), 1,
340 $ a( kp+1, kp ),1 )
341 END IF
342
343 CALL zswap( kp-kk-1, a( kk+1, kk ), 1,
344 $ a( kp, kk+1 ), lda )
345
346 CALL zswap( k, a( kk, 1), lda, a( kp, 1), lda)
347
348* The matrix is anti-symmetric, hence swaps beyond the diagonal need a -1
349
350 CALL zscal( kp-kk, -one, a(kk+1, kk), 1 )
351 CALL zscal( kp-kk-1, -one, a(kp, kk+1), lda )
352
353* Swap in the first columns of W
354 CALL zswap( wk, w( kk, 1 ), ldw, w( kp, 1 ), ldw)
355 END IF
356
357* (The column/row K+1 is not affected by the update)
358 IF( colmax .NE. zero .AND. k .LE. n-2) THEN
359* Store L(k+1) in A(k)
360 CALL zscal( n-k-1, one/a( k+1, k ), a(k+2, k), 1 )
361 END IF
362
363* Delay the update of the trailing submatrix (done at the beginning
364* of each loop for every column of the first NB columns, and then
365* finally in a rank-3 update for the last N-NB block)
366
367* Store Pivot
368 ipiv( k+1 ) = kp
369
370 ELSE
371* STEP == 2 and an even column, do nothing
372 ipiv(k+1) = k+1
373 END IF
374 30 CONTINUE
375
376* Now update the trailing A(NB+1:N, NB+1:N) submatrix in a level 3 update,
377* if necessary
378
379 IF( npanel .LT. n-1) THEN
380* For this we have to set the NB+1,NB entry to zero, but restore it later
381 t = a( npanel+1, npanel )
382 a( npanel+1, npanel ) = zero
383 IF( wk .LT. nb) THEN
384* Store the column NB+1 in W, and update it in W
385 CALL zcopy(n-npanel, a(npanel+1, npanel+1), 1,
386 $ w(npanel+1, nb), 1)
387
388 w( npanel+1, nb ) = zero
389 CALL zgemv( 'N', n-npanel, nb-1, +one, a( npanel+1, 1 ),
390 $ lda*step, w( npanel+1, 1 ), ldw,
391 $ one, w( npanel+1, nb ), 1 )
392 CALL zgemv( 'N', n-npanel, nb-1, -one, w( npanel+1, 1 ),
393 $ ldw, a( npanel+1, 1 ), lda*step, one,
394 $ w( npanel+1, nb ), 1 )
395 w( npanel+1, nb ) = zero
396 END IF
397
398* Now do the rank-3 update
399 CALL zskr2k( uplo, "N", n-npanel, nb, one,
400 $ a(npanel+1,1), lda*step,
401 $ w(npanel+1,1), ldw, one, a(npanel+1, npanel+1), lda)
402 a(npanel+1, npanel)=t
403 END IF
404
405 END IF
406
407 END
