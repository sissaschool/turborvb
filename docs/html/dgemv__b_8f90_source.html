21subroutine zgemv_b(TRANS, M, N, ALPHA, A, LDA, AB, LDAB, X, LDX, XB &
22 &, LDXB, BETA, YB, LDYB)
23 implicit none
24 character*1 trans
25 integer M, N, LDA, LDAB, LDX, LDXB, LDYB
26 complex*16, parameter :: one=(1.d0,0.d0)
27 complex*16 alpha, beta,alphac,betac
28 complex*16 A(LDA, *), AB(LDAB, *), X(*), XB(*), YB(*)
29 alphac=conjg(alpha)
30 betac=conjg(beta)
31 if(trans.eq.'n'.or.trans.eq.'N') then
32 call zgemv('C', m, n, alphac, a, lda, yb, ldyb, one, xb, ldxb)
33 call zgerc(m, n, alphac, yb, ldyb, x, ldx, ab, ldab)
34 call zscal(m,betac, yb, ldyb)
35 else
36! to avoid conjugate of matrix
37 call conjmat(1,m,xb,ldxb)
38 call conjmat(1,n,yb,ldyb)
39 call zgemv('N', m, n, alpha, a, lda, yb, ldyb, one, xb, ldxb)
40 call conjmat(1,m,xb,ldxb)
41 call conjmat(1,n,yb,ldyb)
42
43 call conjmat(1,m,x,ldx)
44 call zgeru(m,n,alphac, x, ldx, yb, ldyb, ab, ldab)
45 call conjmat(1,m,x,ldx)
46 call zscal(n,betac,yb,ldyb)
47 endif
48 return
49end
