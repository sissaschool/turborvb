16/**
17 * @brief nvtx profiling module for performance analysis
18 *
19 * this module provides an interface to nvidia tools extension(nvtx) for
20 * performance profiling and analysis. it allows marking code regions and
21 * events for visualization in nvidia profiling tools like nsight systems
22 * and nsight compute.
23 *
24 * @details
25 * the module provides the following functionality:
26 * - range markers for profiling code sections
27 * - custom colors for visual distinction
28 * - event attributes for detailed profiling
29 * - c-compatible string handling for nvtx interface
30 * - conditional compilation for nvtx support
31 *
32 * key features:
33 * - automatic color cycling for different regions
34 * - support for custom event attributes
35 * - integration with nvidia profiling tools
36 * - no performance impact when nvtx is disabled
37 * - thread-safe profiling markers
38 *
39 * @note
40 * - only available when _nvtx is defined
41 * - requires nvidia gpu and compatible drivers
42 * - used for performance analysis and optimization
43 * - no functionality when compiled without nvtx support
44 *
45 * @see nvtxstartrange(), nvtxendrange()
46 *
47 * @author turborvb group
48 * @date 2022
49 */
50module nvtx
51
52 use iso_c_binding
53 implicit none
54
55#ifdef _NVTX
56 /**
57 * @brief color array for nvtx range markers
58 *
59 * array of predefined colors for nvtx range markers, providing
60 * visual distinction between different code regions in profiling tools.
61 * colors are specified in argb format.
62 */
63 integer, private :: col(7) = [z'0000ff00', z'000000ff', z'00ffff00', z'00ff00ff', z'0000ffff', z'00ff0000', z'00ffffff']
64
65 /**
66 * @brief temporary character array for nvtx string handling
67 *
68 * Character array used for converting fortran strings to c-style
69 * null-terminated strings for nvtx interface calls.
70 */
71 character, private, target :: tempname(256)
72
73 /**
74 * @brief nvtx event attributes structure
75 *
76 * c-compatible structure for nvtx event attributes, used to specify
77 * custom properties for range markers including color, message, and
78 * payload information.
79 */
80 type, bind(C) :: nvtxEventAttributes
81 integer(c_int16_t) :: version = 1
82 integer(c_int16_t) :: size = 48 !
83 integer(c_int) :: category = 0
84 integer(c_int) :: colortype = 1 ! NVTX_COLOR_ARGB = 1
85 integer(c_int) :: color
86 integer(c_int) :: payloadtype = 0 ! NVTX_PAYLOAD_UNKNOWN = 0
87 integer(c_int) :: reserved0
88 integer(c_int64_t) :: payload ! union uint,int,double
89 integer(c_int) :: messagetype = 1 ! NVTX_MESSAGE_TYPE_ASCII = 1
90 type(c_ptr) :: message ! ascii char
91 end type
92
93 /**
94 * @brief Interface for nvtx range push operations
95 *
96 * Interface providing two methods for pushing nvtx range markers:
97 * - nvtxrangepusha: simple range with custom label and standard color
98 * - nvtxrangepushex: range with custom label and custom color/attributes
99 */
100 interface nvtxrangepush
101 ! push range with custom label and standard color
102 subroutine nvtxrangepusha(name) bind(C, name='nvtxRangePushA')
103 use iso_c_binding
104 character(kind=c_char) :: name(256)
105 end subroutine
106
107 ! push range with custom label and custom color
108 subroutine nvtxrangepushex(event) bind(C, name='nvtxRangePushEx')
109 use iso_c_binding
110 import :: nvtxeventattributes
111 type(nvtxeventattributes) :: event
112 end subroutine
113 end interface
114
115 /**
116 * @brief Interface for nvtx range pop operations
117 *
118 * Interface for popping nvtx range markers, ending the current
119 * profiling range.
120 */
121 interface nvtxrangepop
122 subroutine nvtxrangepop() bind(C, name='nvtxRangePop')
123 end subroutine
124 end interface
125
126contains
127
128 /**
129 * @brief start an nvtx profiling range
130 *
131 * this subroutine starts an nvtx profiling range with the specified name.
132 * it can optionally use a custom color based on an id parameter.
133 *
134 * @param[in] name name of the profiling range(will be displayed in profiling tools)
135 * @param[in] id Optional id for color selection(if not provided, uses standard color)
136 *
137 * @details
138 * the subroutine:
139 * 1. converts the fortran string to c-style null-terminated string
140 * 2. If id is provided, creates custom event attributes with color
141 * 3. calls appropriate nvtx function to start the range
142 * 4. colors are cycled through a predefined array based on id
143 *
144 * @note
145 * - only available when _nvtx is defined
146 * - colors cycle through 7 predefined colors
147 * - string is automatically null-terminated
148 * - no effect when nvtx is not available
149 *
150 * @see nvtxendrange()
151 */
152 subroutine nvtxstartrange(name, id)
153 character(kind=c_char, len=*) :: name
154 integer, optional :: id
155 type(nvtxeventattributes) :: event
156 character(kind=c_char, len=256) :: trimmed_name
157 integer :: i
158
159 trimmed_name = trim(name)//c_null_char
160
161 ! move scalar trimmed_name into character array tempName
162 do i = 1, len(trim(name)) + 1
163 tempname(i) = trimmed_name(i:i)
164 end do
165
166 if (.not. present(id)) then
167 call nvtxrangepush(tempname)
168 else
169 event%color = col(mod(id, 7) + 1)
170 event%message = c_loc(tempname)
171 call nvtxrangepushex(event)
172 end if
173 end subroutine
174
175 /**
176 * @brief End the current NVTX profiling range
177 *
178 * this subroutine ends the current nvtx profiling range, marking
179 * the end of a code section for profiling analysis.
180 *
181 * @details
182 * the subroutine calls nvtxrangepop to end the current range
183 * and return to the previous profiling context.
184 *
185 * @note
186 * - only available when _nvtx is defined
187 * - must be called to match each nvtxstartrange call
188 * - no effect when nvtx is not available
189 * - no parameters required
190 *
191 * @see nvtxstartrange()
192 */
193 subroutine nvtxendrange
194 call nvtxrangepop
195 end subroutine
196
197#endif
198
199end module nvtx
